26void SilhouetteMesh::updateAttributes() {
27 // Generate multiple LOD levels of the torus
28 const uint32_t numLODs = silhouetteCfg_.silhouette.tileCounts.size();
29 uint32_t numVertices = 0;
30 uint32_t numIndices = 0;
31 for (uint32_t lodIdx=0; lodIdx<numLODs; lodIdx++) {
32 const auto &uv_rects = silhouetteUVRects_[lodIdx];
33 auto &x = meshLODs_.emplace_back();
34 x.d->numVertices = uv_rects.size() * 4; // each rect has 4 vertices
35 x.d->numIndices = uv_rects.size() * 6; // each rect has 2 triangles, 3 indices each
36 x.d->vertexOffset = numVertices;
37 x.d->indexOffset = numIndices;
38 numVertices += x.d->numVertices;
39 numIndices += x.d->numIndices;
40 }
41
42 pos_->setVertexData(numVertices);
43 if (silhouetteCfg_.hasNormal) {
44 nor_->setVertexData(numVertices);
45 }
46 if (silhouetteCfg_.hasTangent) {
47 tan_->setVertexData(numVertices);
48 }
49 if (silhouetteCfg_.hasUV) {
50 uv_->setVertexData(numVertices);
51 }
52 indices_ = createIndexInput(numIndices, numVertices);
53
54 for (auto i = 0u; i < meshLODs_.size(); ++i) {
55 generateLODLevel(i, meshLODs_[i].d->vertexOffset, meshLODs_[i].d->indexOffset);
56 }
57
58 // Set up the vertex attributes
59 auto indexRef = setIndices(indices_, numVertices);
60 setInput(pos_);
61 if (silhouetteCfg_.hasNormal) {
62 setInput(nor_);
63 }
64 if (silhouetteCfg_.hasUV) {
65 setInput(uv_);
66 }
67 if (silhouetteCfg_.hasTangent) {
68 setInput(tan_);
69 }
70 updateVertexData();
71
72 for (auto &x: meshLODs_) {
73 // add the index buffer offset (in number of bytes)
74 x.d->indexOffset = indexRef->address() + x.d->indexOffset * indices_->dataTypeBytes();
75 }
76 activateLOD(0);
77
78 minPosition_ = Vec3f::create(-0.5f) * silhouetteCfg_.posScale;
79 maxPosition_ = Vec3f::create(0.5f) * silhouetteCfg_.posScale;
80}

82void SilhouetteMesh::generateLODLevel(
83 uint32_t lodLevel, uint32_t vertexOffset, uint32_t indexOffset) {
84 // map client data for writing
85 auto v_pos = (Vec3f*) pos_->clientBuffer()->clientData(0);
86 auto v_nor = (silhouetteCfg_.hasNormal ? (Vec3f*) nor_->clientBuffer()->clientData(0) : nullptr);
87 auto v_tan = (silhouetteCfg_.hasTangent ? (Vec4f*) tan_->clientBuffer()->clientData(0) : nullptr);
88 auto v_uv = (silhouetteCfg_.hasUV ? (Vec2f*) uv_->clientBuffer()->clientData(0) : nullptr);
89 auto indices = (byte*)indices_->clientBuffer()->clientData(0);
90 auto indexType = indices_->baseType();
91 // get a handle on the silhouette data for this LOD
92 const auto &uv_rects = silhouetteUVRects_[lodLevel];
93 // current offset into vertex and index arrays
94 uint32_t vi = vertexOffset;
95 uint32_t ii = indexOffset;
96 Mat4f rotMat = Mat4f::rotationMatrix(
97 silhouetteCfg_.rotation.x,
98 silhouetteCfg_.rotation.y,
99 silhouetteCfg_.rotation.z);
100
101 float areaSum = 0.0f;
102
103 for (const auto &uvRect : uv_rects) {
104 // uvRect = (u0, v0, u1, v1) in normalized 0..1 space
105 float u0 = uvRect.x, v0 = uvRect.y, u1 = uvRect.z, v1 = uvRect.w;
106
107 // positions in local sprite plane.
108 Vec2f local0( (u0+u1)*0.5f - 0.5f, (v0+v1)*0.5f - 0.5f );
109 Vec2f localSize( u1 - u0, v1 - v0 );
110
111 // We'll create geometry in UV-space scaled to [-0.5..0.5] so that instancing can position it in world.
112 Vec3f p0(local0.x - localSize.x*0.5f, local0.y - localSize.y*0.5f, 0.0f);
113 Vec3f p1(local0.x + localSize.x*0.5f, local0.y - localSize.y*0.5f, 0.0f);
114 Vec3f p2(local0.x + localSize.x*0.5f, local0.y + localSize.y*0.5f, 0.0f);
115 Vec3f p3(local0.x - localSize.x*0.5f, local0.y + localSize.y*0.5f, 0.0f);
116
117 // compute the area of the quad
118 float l1 = (p1 - p0).length();
119 float l2 = (p3 - p0).length();
120 areaSum += l1 * l2;
121
122 v_pos[vi+0] = rotMat.transformVector(silhouetteCfg_.posScale * p0);
123 v_pos[vi+1] = rotMat.transformVector(silhouetteCfg_.posScale * p1);
124 v_pos[vi+2] = rotMat.transformVector(silhouetteCfg_.posScale * p2);
125 v_pos[vi+3] = rotMat.transformVector(silhouetteCfg_.posScale * p3);
126
127 if (v_uv) {
128 v_uv[vi+0] = Vec2f(u0, v0);
129 v_uv[vi+1] = Vec2f(u1, v0);
130 v_uv[vi+2] = Vec2f(u1, v1);
131 v_uv[vi+3] = Vec2f(u0, v1);
132 }
133 if (v_nor) {
134 v_nor[vi+0] = Vec3f::front();
135 v_nor[vi+1] = Vec3f::front();
136 v_nor[vi+2] = Vec3f::front();
137 v_nor[vi+3] = Vec3f::front();
138 }
139 if (v_tan) {
140 // Tangent space: X is right, Y is up, Z is forward
141 v_tan[vi+0] = Vec4f(1.0f, 0.0f, 0.0f, 1.0f); // tangent
142 v_tan[vi+1] = Vec4f(1.0f, 0.0f, 0.0f, 1.0f);
143 v_tan[vi+2] = Vec4f(1.0f, 0.0f, 0.0f, 1.0f);
144 v_tan[vi+3] = Vec4f(1.0f, 0.0f, 0.0f, 1.0f);
145 }
146 // indices (two triangles)
147 setIndexValue(indices, indexType, ii++, vi + 0);
148 setIndexValue(indices, indexType, ii++, vi + 1);
149 setIndexValue(indices, indexType, ii++, vi + 2);
150 setIndexValue(indices, indexType, ii++, vi + 2);
151 setIndexValue(indices, indexType, ii++, vi + 3);
152 setIndexValue(indices, indexType, ii++, vi + 0);
153
154 vi += 4;
155 }
156
157 REGEN_INFO("Silhouette LOD " << lodLevel
158 << " with " << uv_rects.size() << " UV rectangles,"
159 << " area=" << areaSum
160 << " vertexOffset=" << vertexOffset
161 << ", indexOffset=" << indexOffset << " num vertices=" << uv_rects.size() * 4
162 << ", num indices=" << uv_rects.size() * 6);
163}

165void SilhouetteMesh::updateSilhouette() {
166 if (!silhouetteGenerator_.get()) {
