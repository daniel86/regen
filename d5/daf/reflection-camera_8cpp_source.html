115bool ReflectionCamera::updateReflection() {
116 if (isHidden() || !isReflectorValid_) {
117 return false;
118 }
119
120 bool reflectorChanged = false;
121 if (hasMesh_) {
122 if (transform_.get() != nullptr) {
123 const uint32_t tfStamp = transform_->stampOfReadData();
124 if (tfStamp != transformStamp_) {
125 reflectorChanged = true;
126 transformStamp_ = tfStamp;
127 }
128 }
129
130 const uint32_t norStamp = nor_->stampOfReadData();
131 const uint32_t posStamp = pos_->stampOfReadData();
132 if (norStamp != norStamp_) {
133 reflectorChanged = true;
134 norStamp_ = norStamp;
135 }
136 if (posStamp != posStamp_) {
137 reflectorChanged = true;
138 posStamp_ = posStamp;
139 }
140
141 // Compute plane parameters...
142 if (reflectorChanged) {
143 if (!pos_->hasClientData()) pos_->readServerData();
144 if (!nor_->hasClientData()) nor_->readServerData();
145 posWorld_ = pos_->mapClientData<Vec3f>(BUFFER_GPU_READ).r[vertexIndex_];
146 norWorld_ = nor_->mapClientData<Vec3f>(BUFFER_GPU_READ).r[vertexIndex_];
147
148 if (transform_.get() != nullptr) {
149 if (!transform_->hasClientData()) transform_->readServerData();
150 auto transform = transform_->mapClientData<Mat4f>(BUFFER_GPU_READ);
151 const Mat4f &M = transform.r[0];
152 posWorld_ = M.mul_t31(posWorld_);
153 norWorld_ = M.mul_t30(norWorld_);
154 norWorld_.normalize();
155 }
156
157 // update clip plane
158 clipPlane_[0].xyz() = norWorld_;
159 clipPlane_[0].w = norWorld_.dot(posWorld_);
160 sh_clipPlane_->setVertex(0, clipPlane_[0]);
161 }
162 }
163
164 // Switch normal if viewer is behind reflector.
165 bool isFront = norWorld_.dot(userCamera_->position(0) - posWorld_) > 0.0;
166 if (isFront != isFront_) {
167 isFront_ = isFront;
168 reflectorChanged = true;
169 }
170 // Skip back faces
171 if (!isFront && !hasBackFace_) return false;
172
173 // Compute reflection matrix...
174 if (reflectorChanged) {
175 if (isFront_) {
176 setClipPlane(0, Vec4f(
177 norWorld_.x, norWorld_.y, norWorld_.z,
178 norWorld_.dot(posWorld_)));
179 reflectionMatrix_ = Mat4f::reflectionMatrix(posWorld_, norWorld_);
180 } else {
181 // flip reflector normal
182 Vec3f n = -norWorld_;
183 setClipPlane(0, Vec4f(n.x, n.y, n.z, n.dot(posWorld_)));
184 reflectionMatrix_ = Mat4f::reflectionMatrix(posWorld_, n);
185 }
186 }
187
188 // Compute reflection camera direction
189 if (reflectorChanged || userCamera_->directionStamp() != camDirStamp_) {
190 camDirStamp_ = userCamera_->directionStamp();
191 Vec3f dir = reflectionMatrix_.rotateVector(userCamera_->direction(0));
192 dir.normalize();
193 setDirection(0, dir);
194
195 reflectorChanged = true;
196 }
197 // Compute reflection camera position
198 if (reflectorChanged || userCamera_->positionStamp() != camPosStamp_) {
199 camPosStamp_ = userCamera_->positionStamp();
200 setPosition(0, reflectionMatrix_.transformVector(userCamera_->position(0)));
201
202 reflectorChanged = true;
203 }
204
205 // Compute view matrix
206 if (reflectorChanged) {
207 updateView();
208 cameraChanged_ = true;
209 }
210
211 // Compute projection matrix
212 if (userCamera_->projectionStamp() != projStamp_) {
213 projStamp_ = userCamera_->projectionStamp();
214 setProjection(0, userCamera_->projection(0));
215 setProjectionInverse(0, userCamera_->projectionInverse(0));
216 cameraChanged_ = true;
217 }
218
219 // Compute view-projection matrix
220 if (cameraChanged_) {
221 updateViewProjection(0u,0u);
222 cameraChanged_ = false;
223 camStamp_ += 1u;
224 return true;
225 } else {
226 return false;
227 }
