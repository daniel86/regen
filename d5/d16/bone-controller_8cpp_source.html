32BoneController::BoneController(uint32_t instanceIdx, const ref_ptr<BoneAnimationItem> &animItem)
33 : instanceIdx_(instanceIdx), animItem_(animItem) {
34 // Fill in the motion clips for each motion type.
35 motionToData_.resize(static_cast<size_t>(MotionType::MOTION_LAST));
36 for (uint32_t clipIdx = 0; clipIdx < animItem_->clips.size(); ++clipIdx) {
37 auto &clip = animItem_->clips[clipIdx];
38 auto &motionData = motionToData_[static_cast<int>(clip.motion)];
39 motionData.clips.push_back(clipIdx);
40 }
41 // Initialize bone weights for motions with clips.
42 for (size_t i = 0; i < motionToData_.size(); ++i) {
43 auto &motionData = motionToData_[i];
44 if (!motionData.clips.empty()) {
45 initializeBoneWeights(motionData, static_cast<MotionType>(i));
46 }
47 }
48
49 // Extract some special motion parameters that are useful for navigation.
50 if (!motionToData_[static_cast<int>(MotionType::WALK)].clips.empty()) {
51 auto walkClip = animItem_->clips[motionToData_[static_cast<int>(MotionType::WALK)].clips[0]];
52 auto walkTPS = animItem_->boneTree->ticksPerSecond(walkClip.loop->trackIndex);
53 walkTime_ = 1000.0f * (walkClip.loop->range.y - walkClip.loop->range.x) / walkTPS;
54 }
55 if (!motionToData_[static_cast<int>(MotionType::RUN)].clips.empty()) {
56 auto runClip = animItem_->clips[motionToData_[static_cast<int>(MotionType::RUN)].clips[0]];
57 auto runTPS = animItem_->boneTree->ticksPerSecond(runClip.loop->trackIndex);
58 runTime_ = 1000.0f * (runClip.loop->range.y - runClip.loop->range.x) / runTPS;
59 }
60
61 // initialize the action-to-motion mapping.
62 // note: behavior tree selects actions, we map here to motions, and have
63 // a set of animation clips for each motion type.
64 actionToMotion_.resize(static_cast<size_t>(ActionType::LAST_ACTION));
65 for (size_t i = 0; i < actionToMotion_.size(); ++i) {
66 actionToMotion_[i] = getMotionTypesForAction(static_cast<ActionType>(i));
67 }
68
69 // Connect to animation stopped event such that we can restart the animation
70 // in case the corresponding motion is still desired.
71 const auto eventHandler = ref_ptr<LambdaEventHandler>::alloc(
72 [&](EventObject *, EventData *data) {
73 handleAnimationStopped(*static_cast<BoneTree::BoneEvent *>(data));
74 });
75 animItem_->boneTree->connect(Animation::ANIMATION_STOPPED, eventHandler);
76}

86void BoneController::handleAnimationStopped(BoneTree::BoneEvent &eventData) {
87 const BoneTree::AnimationHandle animHandle = eventData.animHandle;
88 const MotionData *stoppedMotion = nullptr;
89 // Find which motion this animation handle belongs to.
90 for (uint32_t i=0; i < numActiveMotions_; i++) {
91 auto &motionData = motionToData_[static_cast<int>(activeMotions_[i])];
92 if (motionData.handle == animHandle) {
93 stoppedMotion = &motionData;
94 break;
95 }
96 }

106void BoneController::initializeBoneWeights(MotionData &motionData, MotionType motionType) const {
107 Stack<std::pair<const BoneNode*,BodyPart>> traversalStack;
108 const BoneNode *rootNode = animItem_->boneTree->rootNode();
109 traversalStack.push(std::make_pair(rootNode, getBodyPartType(rootNode->name)));
110
111 // Get the body part weights for this motion type.
112 const float *partWeights = motionBodyPartWeights(motionType);
113 float baseWeight = partWeights[0];
114 partWeights += 1; // skip base weight
115 // Initialize the bone weights array.
116 motionData.boneWeights.resize(animItem_->boneTree->numNodes());
117
118 // Here the idea is that we keep track of the current body part as we traverse
119 // such that we can find the correct weight per bone which can later be indexed by node index.
120 while (!traversalStack.isEmpty()) {
121 auto [node,currentBodyPart] = traversalStack.top();
122 traversalStack.pop();
123 if (currentBodyPart == BodyPart::LAST) {
124 motionData.boneWeights[node->nodeIdx] = baseWeight;
125 } else {
126 motionData.boneWeights[node->nodeIdx] = partWeights[static_cast<size_t>(currentBodyPart)];
127 }
128
129 for (const auto &child : node->children) {
130 BodyPart childPartType = getBodyPartType(child->name);
131 if (childPartType == BodyPart::LAST) childPartType = currentBodyPart;
132 traversalStack.push(std::make_pair(child.get(), childPartType));
133 }
134 }

159void BoneController::updateDesiredMotion(
160 MotionType desiredMotion, ActionType desiredAction, bool selectNewClip) {
161 auto &motionData = motionToData_[static_cast<int>(desiredMotion)];
162 motionData.desired = true;
163 motionData.action = desiredAction;
164 if (selectNewClip && motionData.status != MOTION_INACTIVE) {
165 // Select a new clip for this motion type.

171 // note: we can keep the current weight and fade time.
172 BONE_CTRL_DEBUG(desiredMotion, "fading in");
173 motionData.status = MOTION_FADING_IN;
174 } else if (motionData.status == MOTION_INACTIVE) {
175 // was inactive, so we can start it right away.
176 BONE_CTRL_DEBUG(desiredMotion, "is now starting");
177 motionData.status = MOTION_STARTING;

182void BoneController::updateDesiredMotions(const ActionType *desiredActions, uint32_t numDesiredActions) {
183 auto anim = animItem_->boneTree;
184
185 for (uint32_t i = 0; i < numDesiredActions; ++i) {
186 MotionType desiredMotion = MotionType::MOTION_LAST;
187 ActionType desiredAction = desiredActions[i];
188 bool selectNewClip = false;
189
190 // Check if the action is currently being performed.
191 for (uint32_t j = 0; j < numActiveMotions_; ++j) {
192 MotionType currentMotion = activeMotions_[j];
193 auto &motionData = motionToData_[static_cast<int>(currentMotion)];
194 if (motionData.action == desiredAction) {
195 // For some action types, allow rapid changing of motion types.
196 // For now just allow that for ATTACK actions.
197 // Later, this should be defined in an ontology!
198 bool stickToCurrent = ((
199 desiredAction != ActionType::ATTACKING &&
200 desiredAction != ActionType::CONVERSING) ||
201 anim->isBoneAnimationActive(instanceIdx_, motionData.handle));
202 if (stickToCurrent) {
203 // The current animation clip segment is still active, or should be
204 // repeated.
205 desiredMotion = currentMotion;
206 } else {
207 // Allow selecting a new motion clip for this action.
208 // The last clip has already finished as !isBoneAnimationActive.
209 selectNewClip = true;
210 break;
211 }
212
213 }
214 }
215 // If not active, then select a motion for the action.
216 if (desiredMotion == MotionType::MOTION_LAST) {
217 auto &possibleMotions = actionToMotion_[static_cast<size_t>(desiredAction)];
218 if (!possibleMotions.empty()) {
219 if (possibleMotions.size() == 1) {
220 desiredMotion = possibleMotions[0];
221 } else {
222 desiredMotion = possibleMotions[math::randomInt() % possibleMotions.size()];
223 }
224 } else {
225 REGEN_WARN("No motion defined for action " << desiredAction << ".");
226 continue;
227 }
228 }
229 if (desiredMotion == MotionType::MOTION_LAST) {
230 continue;
231 }
232 updateDesiredMotion(desiredMotion, desiredAction, selectNewClip);
233 }

236bool BoneController::startMotionClip(MotionData &motion, float initialWeight) {
237 if (motion.clips.empty()) return false;
238
239 // Select a clip for this motion type.
240 uint32_t clipIndex;
241 if (motion.clips.size() == 1) {
242 clipIndex = 0;
243 } else {
244 clipIndex = math::randomInt() % motion.clips.size();
245 }
246 auto &clip = animItem_->clips[motion.clips[clipIndex]];
247 // begin range is optional, if not present we loop the clip directly.
248 bool hasBeginRange = (clip.begin != nullptr);
249
250 motion.lastClip = clipIndex;
251 motion.clipStatus = (hasBeginRange ? CLIP_BEGINNING : CLIP_LOOPING);
252
253 if (initialWeight + 1e-5f >= 1.0f) {
254 initialWeight = 1.0f;
255 motion.status = MOTION_ACTIVE;
256 motion.fadeTime = motionFadeDuration_;
257 } else {
258 motion.status = MOTION_FADING_IN;
259 motion.fadeTime = 0.0f;
260 }
261 if (hasBeginRange) {
262 motion.handle = startBoneAnimation(motion, clip.begin);
263 BONE_CTRL_DEBUG(clip.motion, REGEN_STRING(
264 "is starting clip '" << clip.begin->name << "' with begin range " <<
265 " in track " << clip.begin->trackIndex));
266 } else {
267 motion.handle = startBoneAnimation(motion, clip.loop);
268 BONE_CTRL_DEBUG(clip.motion, REGEN_STRING(
269 "is starting clip '" << clip.loop->name << "' with no begin range" <<
270 " in track " << clip.loop->trackIndex));
271 }
272 setMotionWeight(motion, initialWeight);
273 animItem_->boneTree->startAnimation();
274
275 return true;
276}

290void BoneController::updateMotionClip(MotionData &motion, MotionClip &clip) {
291 // animation-status is ACTIVE | FADING_OUT | FADING_IN
292 // clip-status is BEGINNING | LOOPING | ENDING
293 auto &anim = animItem_->boneTree;
294 // Check if the current segment is still active, if so we are done.
295 bool isActive = anim->isBoneAnimationActive(instanceIdx_, motion.handle);
296 if (isActive) return;
297
298 bool hasEndRange = (clip.end != nullptr);
299 bool isFadingOut = (motion.status == MOTION_FADING_OUT);
300
301 if (motion.clipStatus == CLIP_BEGINNING) {
302 // Beginning segment is done, now start looping segment if any...
303 if (clip.loop && (!isFadingOut || !hasEndRange)) {
304 // Activate the looping segment, unless we are fading out and have an end segment.
305 // In that case we skip the looping segment and go directly to the end segment.
306 motion.handle = startBoneAnimation(motion, clip.loop);
307 motion.clipStatus = CLIP_LOOPING;
308 BONE_CTRL_DEBUG(clip.motion, "is now looping");
309 } else if (clip.end) {
310 // Start the ending segment directly.
311 if (motion.nextHandle != -1) {
312 motion.handle = motion.nextHandle;
313 motion.nextHandle = -1;
314 } else {
315 motion.handle = startBoneAnimation(motion, clip.end);
316 }
317 motion.clipStatus = CLIP_ENDING;
318 BONE_CTRL_DEBUG(clip.motion, "is now ending");
319 } else if (!isFadingOut) {
320 // Switch back to beginning.
321 motion.handle = startBoneAnimation(motion, clip.begin);
322 motion.clipStatus = CLIP_BEGINNING;
323 } else {
324 if (motion.fadeTime > 1e-5f && clip.loop) {
325 // If we are still fading out, activate looping segment such that we
326 // can fade it out properly.
327 motion.handle = startBoneAnimation(motion, clip.loop);
328 motion.clipStatus = CLIP_LOOPING;
329 BONE_CTRL_DEBUG(clip.motion, "starts looping during fade out");
330 } else {
331 // No looping or ending segment, so toggle the motion off.
332 motion.status = MOTION_INACTIVE;
333 motion.fadeTime = 0.0f;
334 BONE_CTRL_DEBUG(clip.motion, "is now inactive");
335 }
336 }
337 }
338 else if (motion.clipStatus == CLIP_LOOPING) {
339 // Looping segment is done.
340 if (!isFadingOut) {
341 // Restart the looping segment.
342 //BONE_CTRL_DEBUG(clip.motion, "is now restarting");
343 motion.handle = startBoneAnimation(motion, clip.loop);
344 } else if (clip.end) {
345 // Switch to ending segment if we are fading out, i.e. if the motion is no longer desired.
346 if (motion.nextHandle != -1) {
347 motion.handle = motion.nextHandle;
348 motion.nextHandle = -1;
349 BONE_CTRL_DEBUG(clip.motion, REGEN_STRING("early faded to ending, time: "
350 << motion.fadeTime / motionFadeDuration_));
351 } else {
352 motion.handle = startBoneAnimation(motion, clip.end);
353 BONE_CTRL_DEBUG(clip.motion, "is now ending");
354 }
355 motion.clipStatus = CLIP_ENDING;
356 } else {
357 // No ending segment, so toggle the motion off.
358 if (motion.fadeTime > 1e-5f) {
359 motion.handle = startBoneAnimation(motion, clip.loop);
360 BONE_CTRL_DEBUG(clip.motion, "is now looping during fade out");
361 } else {
362 motion.status = MOTION_INACTIVE;
363 motion.fadeTime = 0.0f;
364 BONE_CTRL_DEBUG(clip.motion, "is now inactive");
365 }
366 }
367 }
368 else if (motion.clipStatus == CLIP_ENDING) {
369 // Ending segment is done, so toggle the motion off.
370 // Note: even if fading out, we do not restart the ending segment,
371 // hoping that the ending segment brings the mesh into good rest pose.
372 motion.status = MOTION_INACTIVE;
373 motion.fadeTime = 0.0f;
374 BONE_CTRL_DEBUG(clip.motion, "is now inactive");
375 }
376}

389void BoneController::updateMotionClip(MotionData &motion, float dt_s) {
390 // Each clip is made up of one or more segments that are played in sequence.
391 if (motion.status != MOTION_INACTIVE && motion.status != MOTION_STARTING) {
392 auto &clip = animItem_->clips[motion.clips[motion.lastClip]];
393 updateMotionClip(motion, clip);
394 }
395 // Update the motion state machine and blend weights.
396 switch (motion.status) {
397 case MOTION_ACTIVE:
398 case MOTION_INACTIVE:
399 // nothing to do
400 break;
401 case MOTION_STARTING:
402 if (startMotionClip(motion, 0.0f)) {
403 BONE_CTRL_DEBUG(lastActiveMotion(motion), "started");
404 } else {
405 // Could not start motion, so set to inactive again.
406 motion.status = MOTION_INACTIVE;
407 motion.fadeTime = 0.0f;
408 BONE_CTRL_DEBUG(motion.action, "could not be started");
409 }
410 break;
411 case MOTION_FADING_IN:
412 motion.fadeTime += dt_s;
413 if (motion.fadeTime >= motionFadeDuration_) {
414 BONE_CTRL_DEBUG(lastActiveMotion(motion), "fully faded in");
415 motion.status = MOTION_ACTIVE;
416 setMotionWeight(motion, 1.0f);
417 } else {
418 setMotionWeight(motion, motion.fadeTime / motionFadeDuration_);
419 }
420 break;
421 case MOTION_FADING_OUT:
422 auto undesiredMotion = lastActiveMotion(motion);
423 if (!isInterruptibleMotion(undesiredMotion)) {
424 // Avoid interrupting some motions in the middle by fading them out
425 // e.g. attack/block animations should play to the end.
426 // -> check if the remaining time is less than the fade duration.
427 auto remaining = animItem_->boneTree->remainingTime(instanceIdx_, motion.handle) * 0.001f;
428 // Start fading out a bit earlier to avoid overshooting.
429 remaining -= 0.1f*motionFadeDuration_;
430 if (remaining > motionFadeDuration_) {
431 break;
432 }
433 }
434
435 motion.fadeTime -= dt_s;
436 if (motion.fadeTime <= 0.0f) {
437 stopMotionClip(motion);
438 if (motion.nextHandle != -1) {
439 // Start the next handle if any.
440 motion.handle = motion.nextHandle;
441 motion.nextHandle = -1;
442 motion.fadeTime = motionFadeDuration_;
443 motion.status = MOTION_FADING_OUT;
444 motion.clipStatus = CLIP_ENDING;
445 setMotionWeight(motion, 1.0f);
446 BONE_CTRL_DEBUG(undesiredMotion, "fading out ending segment");
447 } else {
448 BONE_CTRL_DEBUG(undesiredMotion, "stopped after fade out");
449 }
450 } else {
451 float w = motion.fadeTime / motionFadeDuration_;
452 setMotionWeight(motion, w);
453 if (motion.nextHandle != -1) {
454 animItem_->boneTree->setAnimationWeight(instanceIdx_, motion.nextHandle, 1.0f - w);
455 }
456 }
457 break;
458 }
459}

475void BoneController::updateBoneController(float dt_s, const MotionType *desiredMotions, uint32_t numDesiredMotions) {
476 // Note: It is best to use low weights for IDLE state to avoid blending it in too aggressively
477 // when an active motion is set to undesired (eg. due to key release event).
478 // Unset all desired flags.
479 for (uint32_t activeIdx = 0; activeIdx < numActiveMotions_; ++activeIdx) {
480 const auto motionType = activeMotions_[activeIdx];
481 auto &motion = motionToData_[static_cast<int>(motionType)];
482 motion.desired = false;
483 }
484 // Mark only the given motions as desired.
485 for (uint32_t i = 0; i < numDesiredMotions; ++i) {
486 auto desiredMotion = desiredMotions[i];
487 updateDesiredMotion(desiredMotion, ActionType::LAST_ACTION, false);
488 }
489 // Update the state of all active motions.
490 updateBoneController(dt_s);

493void BoneController::updateBoneController(float dt_s) {
494 uint32_t newNumActiveMotions = 0;
495 for (uint32_t activeIdx = 0; activeIdx < numActiveMotions_; activeIdx++) {
496 auto &motion = motionToData_[static_cast<int>(activeMotions_[activeIdx])];
497 // Flip some motion state before update
498 if (!motion.desired && motion.status != MOTION_FADING_OUT) {
499 // Not desired anymore, so fade out.
500 BONE_CTRL_DEBUG(activeMotions_[activeIdx], "fading out");
501 motion.status = MOTION_FADING_OUT;
502 motion.fadeTime = motionFadeDuration_;
503 // Fade into the end segment if any.
504 auto &clip = animItem_->clips[motion.clips[motion.lastClip]];
505 if (motion.clipStatus != CLIP_ENDING && clip.end) {
506 motion.nextHandle = startBoneAnimation(motion, clip.end);
507 BONE_CTRL_DEBUG(activeMotions_[activeIdx], "starting end segment");
508 }
509 }
510 // Update the motion state, start/stop animations, set animation weights for fading etc.
511 updateMotionClip(motion, dt_s);
512 // Only add motions that are still active to the new active list.
513 // This will remove motions that have faded out completely.
514 if (motion.status != MOTION_INACTIVE) {
515 activeMotions_[newNumActiveMotions++] = activeMotions_[activeIdx];
516 }
517 }
518 numActiveMotions_ = newNumActiveMotions;
