48std::pair<bool, uint32_t> FreeList::reserve(uint32_t size) {
49 for (size_t i = 0; i < activeCount_; ++i) {
50 Node &node = nodes_[i];
51 if (node.size < size) continue;
52
53 uint32_t offset = node.offset;
54 node.offset += size;
55 node.size -= size;
56 if (node.size == 0) {
57 // The node is now empty, remove it.
58 // We do this here by shifting all nodes after it one position to the left, and
59 // reducing the active count.

61 nodes_[j - 1] = nodes_[j];
62 }
63 --activeCount_;
64 }

66 }

71void FreeList::release(uint32_t size, uint32_t offset) {
72 // Find insert position via linear search.
73 // Note: this is O(n) in the worst case. We could in fact use binary search here with log(n) complexity,
74 // as nodes are sorted by offset, but for small lists linear search will be ok.
75 size_t i = 0;
76 while (i < activeCount_ && nodes_[i].offset < offset) {
77 ++i;
78 }
79 // The node with index *i* is the first node >= offset.
80
81 // Try merge with left node, i.e. the first node < offset.
82 if (i > 0) {
83 uint32_t leftEnd = nodes_[i - 1].offset + nodes_[i - 1].size;
84
85 if (leftEnd > offset) {
86 // The offset is inside the previous node, which is not allowed.
87 REGEN_ERROR("Orphaning range at " << offset << " with size " << size
88 << " overlaps with existing range at " << nodes_[i - 1].offset
89 << " with size " << nodes_[i - 1].size);
90 size -= leftEnd - offset; // Adjust size
91 }
92
93 if (leftEnd == offset) {
94 // The offset is exactly at the end of the previous node, just how it should be!
95 nodes_[i - 1].size += size;
96
97 // Maybe also merge with right
98 if (i < activeCount_ && offset + size == nodes_[i].offset) {
99 nodes_[i - 1].size += nodes_[i].size;
100 // Remove right node by shifting all nodes after it one position to the left.
101 for (size_t j = i + 1; j < activeCount_; ++j) {
102 nodes_[j - 1] = nodes_[j];
103 }
104 --activeCount_;
105 }
106 return;
107 }
108 }
109
110 // Try merge with right only
111 if (i < activeCount_ && offset + size == nodes_[i].offset) {
112 nodes_[i].offset = offset;
113 nodes_[i].size += size;
114 return;
115 }
116
117 // No merge possible, insert new node at position i
118 if (activeCount_ >= nodes_.capacity()) {
119 // If we are at capacity, we need to extend the vector.
120 nodes_.reserve(nodes_.capacity() + NUM_RESERVED_NODES); // Reserve some more space
121 }
122 if (activeCount_ < nodes_.size()) {
123 // Reuse slot at the end
124 } else {
125 nodes_.emplace_back(); // Extend vector
126 }
127 // Shift all nodes after i one position to the right
128 for (size_t j = activeCount_; j > i; --j) {
129 nodes_[j] = nodes_[j - 1];
130 }
131 nodes_[i] = {offset, size};
132 ++activeCount_;
133}
