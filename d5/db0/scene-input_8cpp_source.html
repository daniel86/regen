57list<IndexRange> SceneInputNode::getIndexSequence(uint32_t numIndices) {
58 list<IndexRange> indices;
59 if (hasAttribute("index")) {
60 pushIndexToSequence(numIndices, indices,
61 IndexRange(getValue<int>("index", 0u)));
62 } else if (hasAttribute("instance")) {
63 pushIndexToSequence(numIndices, indices,
64 IndexRange(getValue<int>("instance", 0u)));
65 } else if (hasAttribute("from-index") || hasAttribute("to-index")) {
66 auto from = getValue<uint32_t>("from-index", 0u);
67 auto to = getValue<uint32_t>("to-index", numIndices - 1);
68 auto step = getValue<uint32_t>("index-step", 1u);
69 pushIndexToSequence(numIndices, indices,
70 IndexRange(from, to, step));
71 } else if (hasAttribute("from-instance") || hasAttribute("to-instance")) {
72 auto from = getValue<uint32_t>("from-instance", 0u);
73 auto to = getValue<uint32_t>("to-instance", numIndices - 1);
74 auto step = getValue<uint32_t>("instance-step", 1u);
75 pushIndexToSequence(numIndices, indices,
76 IndexRange(from, to, step));
77 } else if (hasAttribute("indices")) {
78 const auto indicesAtt = getValue<string>("indices", "0");
79 vector<string> indicesStr;
80 boost::split(indicesStr, indicesAtt, boost::is_any_of(","));
81 for (auto & it : indicesStr)
82 pushIndexToSequence(numIndices, indices, IndexRange(atoi(it.c_str())));
83 } else if (hasAttribute("instances")) {
84 const auto indicesAtt = getValue<string>("instances", "0");
85 vector<string> indicesStr;
86 boost::split(indicesStr, indicesAtt, boost::is_any_of(","));
87 for (auto & it : indicesStr)
88 pushIndexToSequence(numIndices, indices, IndexRange(atoi(it.c_str())));
89 } else if (hasAttribute("random-indices")) {
90 auto indexCount = getValue<uint32_t>("random-indices", numIndices);
91 while (indexCount > 0) {
92 --indexCount;
93 pushIndexToSequence(numIndices, indices, IndexRange(rand() % numIndices));
94 }
95 }
96
97 if (indices.empty()) {
98 // If no indices found, add all
99 indices.emplace_back(0, numIndices - 1, 1);
100 }
101 return indices;
102}
