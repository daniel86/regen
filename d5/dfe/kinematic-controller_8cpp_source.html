114 btCollisionRadius_, btCollisionHeight_);

116
117 // Create the ghost object for the character
118 auto ghostObject = ref_ptr<btPairCachingGhostObject>::alloc();
119 ghostObject->setCollisionShape(btShape_.get());
120 ghostObject->setCollisionFlags(btCollisionObject::CF_CHARACTER_OBJECT);
121 btGhostObject_ = ghostObject;
122
123 // Create the character controller
124 btController_ = ref_ptr<btKinematicCharacterController>::alloc(
125 ghostObject.get(), capsuleShape.get(), btStepHeight_);
126 setGravityForce(btGravityForce_);
127 setMaxSlope(btMaxSlope_);
128 // find platform the player is standing on
129 btRayAction_ = ref_ptr<PlatformRayAction>::alloc(
130 bt_->dynamicsWorld().get(),
131 btGhostObject_.get(),
132 btCollisionHeight_ * 0.5f + btCollisionRadius_,
133 [this](btRigidBody *btPlatform) {
134 btPlatform_ = btPlatform;
135 if (btPlatform) {
136 previousPlatformPos_ = btPlatform->getWorldTransform().getOrigin();
137 }
138 });
139
140 btQuaternion rotation;
141 rotation.setRotation(btVector3(0, 1, 0), meshHorizontalOrientation_);
142 btTransform initialTransform;
143 if (attachedToTransform_->hasModelMat()) {
144 auto m = attachedToTransform_->modelMat()->getVertex(0);
145 initialTransform.setFromOpenGLMatrix(
146 (const btScalar *) &m.r.x);
147 }
148 initialTransform.setRotation(rotation);
149 ghostObject->setWorldTransform(initialTransform);
150
151 // Add the character controller to the Bullet world
152 bt_->dynamicsWorld()->addCollisionObject(
153 btGhostObject_.get(),

168void KinematicPlayerController::cpuUpdate(double dt) {
169 updateStep(dt);
170 if (dt <= std::numeric_limits<double>::epsilon()) {
171 return;
172 }
173
174 if (!btController_.get()) {
175 if (!initializePhysics()) {
176 return;
177 }
178 }
179
180 // synchronize the ghost object's transform with the OpenGL transform
181 btTransform ghostTransform = btController_->getGhostObject()->getWorldTransform();
182 ghostTransform.getOpenGLMatrix((btScalar *) &matVal_);
183 // decrease the y position by the collision height, else
184 // the character would float above the ground
185 float heightAdjust = btCollisionHeight_ * 0.5f + btCollisionRadius_;
186 matVal_.x[13] -= heightAdjust;
187
188 btVector3 btVelocity = btController_->getLinearVelocity();
189 btVelocity.setX(0.0);
190 btVelocity.setZ(0.0);
191 bool isMoving = isMoving_;
192
193 // Apply the platform velocity to the character controller
194 if (btPlatform_) {
195 btVector3 currentPlatformPos = btPlatform_->getWorldTransform().getOrigin();
196 btVector3 platformVelocity = currentPlatformPos - previousPlatformPos_;
197 if (platformVelocity.length() > 0.0001) {
198 // not sure about the magic number 16.8, might
199 // be related to some hacks in the kinematics controller :/
200 btVelocity += 16.8 * platformVelocity / dt;
201 isMoving = true;
202 }
203 previousPlatformPos_ = currentPlatformPos;
204 }
205
206 if (!isMoving_) {
207 if (btIsMoving_ || isMoving) {
208 btVelocity.setY(btController_->getLinearVelocity().getY());
209 btController_->setLinearVelocity(btVelocity);
210 btIsMoving_ = false;
211 }
212 } else {
213 // Apply the step to the character controller
214 btVelocity += btVector3(step_.x, step_.y, step_.z)/dt;
215 btVelocity.setY(btController_->getLinearVelocity().getY());
216 btController_->setLinearVelocity(btVelocity);
217 btIsMoving_ = true;
218 }
219
220 // update the camera position and direction
221 updateCameraPose();
222 computeMatrices(camPos_, camDir_);
223 updateCamera(camPos_, camDir_, dt);
224 updateBoneController(dt);
