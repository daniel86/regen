10static float sampleNoise(
11 const NoiseGenerator &noiseGen,
12 double x, double y, double z,
13 bool isSeamless2D,
14 bool isSeamless3D) {
15 float val;
16 if (isSeamless2D) {
17 double a = noiseGen.GetValue(x, y, z);
18 double b = noiseGen.GetValue(x + 1.0, y, z);
19 double c = noiseGen.GetValue(x, y + 1.0, z);
20 double d = noiseGen.GetValue(x + 1.0, y + 1.0, z);
21 val = static_cast<float>(math::mix(
22 math::mix(a, b, 1.0 - x),
23 math::mix(c, d, 1.0 - x), 1.0 - y));
24 } else if (isSeamless3D) {
25 double a0 = noiseGen.GetValue(x, y, z);
26 double b0 = noiseGen.GetValue(x + 1.0, y, z);
27 double c0 = noiseGen.GetValue(x, y + 1.0, z);
28 double d0 = noiseGen.GetValue(x + 1.0, y + 1.0, z);
29 double val0 = math::mix(
30 math::mix(a0, b0, 1.0 - x),
31 math::mix(c0, d0, 1.0 - x), 1.0 - y);
32
33 double a1 = noiseGen.GetValue(x, y, z + 1.0);
34 double b1 = noiseGen.GetValue(x + 1.0, y, z + 1.0);
35 double c1 = noiseGen.GetValue(x, y + 1.0, z + 1.0);
36 double d1 = noiseGen.GetValue(x + 1.0, y + 1.0, z + 1.0);
37 double val1 = math::mix(
38 math::mix(a1, b1, 1.0 - x),
39 math::mix(c1, d1, 1.0 - x), 1.0 - y);
40
41 val = static_cast<float>(math::mix(val0, val1, 1.0 - z));
42 } else {
43 val = static_cast<float>(noiseGen.GetValue(x, y, z));
44 }
45 // map roughly to [0,1] and clamp to that range
46 return math::clamp((val + 1.0f) * 0.5f, 0.0f, 1.0f);
47}

160ref_ptr<NoiseGenerator> NoiseGenerator::preset_clouds(int randomSeed) {
161 // Base of the cloud texture.
162 // The billowy noise produces the basic shape of soft, fluffy clouds.
163 auto cloudBase = ref_ptr<module::Billow>::alloc();
164 cloudBase->SetSeed(randomSeed);
165 cloudBase->SetFrequency(2.0);

171
172 // Perturb the cloud texture for more realism.
173 auto finalClouds = ref_ptr<module::Turbulence>::alloc();

179 finalGen->addSource(baseGen);

183ref_ptr<NoiseGenerator> NoiseGenerator::preset_wood(int randomSeed) {
184 // Base wood texture. The base texture uses concentric cylinders aligned
185 // on the z axis, like a log.
186 auto baseWood = ref_ptr<module::Cylinders>::alloc();
187 baseWood->SetFrequency(16.0);
188 auto baseGen = ref_ptr<NoiseGenerator>::alloc("base", baseWood);
189
190 // Perlin noise to use for the wood grain.
191 auto woodGrainNoise = ref_ptr<module::Perlin>::alloc();
192 woodGrainNoise->SetSeed(randomSeed);
193 woodGrainNoise->SetFrequency(48.0);
194 woodGrainNoise->SetPersistence(0.5);
195 woodGrainNoise->SetLacunarity(2.20703125);
196 woodGrainNoise->SetOctaveCount(3);
197 woodGrainNoise->SetNoiseQuality(QUALITY_STD);
198 auto grainNoiseGen = ref_ptr<NoiseGenerator>::alloc("grain-noise", woodGrainNoise);
199
200 // Stretch the Perlin noise in the same direction as the center of the
201 // log. This produces a nice wood-grain texture.
202 auto scaledBaseWoodGrain = ref_ptr<module::ScalePoint>::alloc();
203 scaledBaseWoodGrain->SetYScale(0.25);
204 auto scaledGen = ref_ptr<NoiseGenerator>::alloc("scaled", scaledBaseWoodGrain);
205 scaledGen->addSource(grainNoiseGen);
206
207 // Scale the wood-grain values so that they may be added to the base wood
208 // texture.
209 auto woodGrain = ref_ptr<module::ScaleBias>::alloc();
210 woodGrain->SetScale(0.25);
211 woodGrain->SetBias(0.125);
212 auto grainGen = ref_ptr<NoiseGenerator>::alloc("grain", woodGrain);
213 grainGen->addSource(scaledGen);
214
215 // Add the wood grain texture to the base wood texture.
216 auto combinedGen = ref_ptr<NoiseGenerator>::alloc(
217 "combined", ref_ptr<module::Add>::alloc());
218 combinedGen->addSource(baseGen);
219 combinedGen->addSource(grainGen);
220
221 // Slightly perturb the wood texture for more realism.
222 auto perturbedWood = ref_ptr<module::Turbulence>::alloc();
223 perturbedWood->SetSeed(randomSeed + 1);
224 perturbedWood->SetFrequency(4.0);
225 perturbedWood->SetPower(1.0 / 256.0);
226 perturbedWood->SetRoughness(4);
227 auto perturbedGen = ref_ptr<NoiseGenerator>::alloc("perturbed", perturbedWood);
228 perturbedGen->addSource(combinedGen);
229
230 // Cut the wood texture a small distance from the center of the "log".
231 auto translatedWood = ref_ptr<module::TranslatePoint>::alloc();
232 translatedWood->SetZTranslation(1.48);
233 auto translatedGen = ref_ptr<NoiseGenerator>::alloc("translated", translatedWood);
234 translatedGen->addSource(perturbedGen);
235
236 // Cut the wood texture on an angle to produce a more interesting wood
237 // texture.
238 auto rotatedWood = ref_ptr<module::RotatePoint>::alloc();
239 rotatedWood->SetAngles(84.0, 0.0, 0.0);
240 auto rotatedGen = ref_ptr<NoiseGenerator>::alloc("rotated", rotatedWood);
241 rotatedGen->addSource(translatedGen);
242
243 // Finally, perturb the wood texture to produce the final texture.
244 auto finalWood = ref_ptr<module::Turbulence>::alloc();
245 finalWood->SetSeed(randomSeed + 2);
246 finalWood->SetFrequency(2.0);
247 finalWood->SetPower(1.0 / 64.0);
248 finalWood->SetRoughness(4);
249 auto finalGen = ref_ptr<NoiseGenerator>::alloc("final", finalWood);
250 finalGen->addSource(rotatedGen);
251
252 return finalGen;
253}

255ref_ptr<NoiseGenerator> NoiseGenerator::preset_granite(int randomSeed) {
256 // Primary granite texture. This generates the "roughness" of the texture
257 // when lit by a light source.
258 auto primaryGranite = ref_ptr<module::Billow>::alloc();
259 primaryGranite->SetSeed(randomSeed);
260 primaryGranite->SetFrequency(8.0);
261 primaryGranite->SetPersistence(0.625);
262 primaryGranite->SetLacunarity(2.18359375);
263 primaryGranite->SetOctaveCount(6);
264 primaryGranite->SetNoiseQuality(QUALITY_STD);
265 auto primaryGen = ref_ptr<NoiseGenerator>::alloc("primary", primaryGranite);
266
267 // Use Voronoi polygons to produce the small grains for the granite texture.
268 auto baseGrains = ref_ptr<module::Voronoi>::alloc();
269 baseGrains->SetSeed(randomSeed + 1);
270 baseGrains->SetFrequency(16.0);
271 baseGrains->EnableDistance(true);
272 auto baseGrainsGen = ref_ptr<NoiseGenerator>::alloc("base-grains", baseGrains);
273
274 // Scale the small grain values so that they may be added to the base
275 // granite texture. Voronoi polygons normally generate pits, so apply a
276 // negative scaling factor to produce bumps instead.
277 auto scaledGrains = ref_ptr<module::ScaleBias>::alloc();
278 scaledGrains->SetScale(-0.5);
279 scaledGrains->SetBias(0.0);
280 auto grainGen = ref_ptr<NoiseGenerator>::alloc("grains", scaledGrains);
281 grainGen->addSource(baseGrainsGen);
282
283 // Combine the primary granite texture with the small grain texture.
284 auto combinedGen = ref_ptr<NoiseGenerator>::alloc(
285 "combined", ref_ptr<module::Add>::alloc());
286 combinedGen->addSource(primaryGen);
287 combinedGen->addSource(grainGen);
288
289 // Finally, perturb the granite texture to add realism.
290 auto finalGranite = ref_ptr<module::Turbulence>::alloc();
291 finalGranite->SetSeed(randomSeed + 2);
292 finalGranite->SetFrequency(4.0);
293 finalGranite->SetPower(1.0 / 8.0);
294 finalGranite->SetRoughness(6);
295 auto finalGen = ref_ptr<NoiseGenerator>::alloc("final", finalGranite);
296 finalGen->addSource(combinedGen);
297 return finalGen;
298}

300ref_ptr<NoiseGenerator> loadGenerator(
301 LoadingContext &ctx,
302 scene::SceneInputNode &input,
303 const std::map<std::string, ref_ptr<NoiseGenerator>> &generators,
304 int randomSeed) {
305 const std::string generatorType = input.getValue("type");
306 ref_ptr<NoiseGenerator> generator;
307
308 if (generatorType == "perlin") {
309 auto perlin = ref_ptr<module::Perlin>::alloc();
310 perlin->SetSeed(randomSeed);
311 perlin->SetFrequency(input.getValue<double>("frequency", 4.0));
312 perlin->SetPersistence(input.getValue<double>("persistence", 0.5));
313 perlin->SetLacunarity(input.getValue<double>("lacunarity", 2.5));
314 perlin->SetOctaveCount(input.getValue<int>("octaves", 4));
315 generator = ref_ptr<NoiseGenerator>::alloc("perlin", perlin);
316 }
317 else if (generatorType == "billow") {
318 auto billow = ref_ptr<module::Billow>::alloc();
319 billow->SetSeed(randomSeed);
320 billow->SetFrequency(input.getValue<double>("frequency", 4.0));
321 billow->SetPersistence(input.getValue<double>("persistence", 0.5));
322 billow->SetLacunarity(input.getValue<double>("lacunarity", 2.5));
323 billow->SetOctaveCount(input.getValue<int>("octaves", 4));
324 //billow->SetNoiseQuality(QUALITY_STD);
325 generator = ref_ptr<NoiseGenerator>::alloc("billow", billow);
326 }
327 else if (generatorType == "turbulence") {
328 auto turbulence = ref_ptr<module::Turbulence>::alloc();
329 turbulence->SetSeed(randomSeed);
330 turbulence->SetFrequency(input.getValue<double>("frequency", 4.0));
331 turbulence->SetPower(input.getValue<double>("power", 1.0));
332 turbulence->SetRoughness(input.getValue<int>("roughness", 2));
333 generator = ref_ptr<NoiseGenerator>::alloc("turbulence", turbulence);
334 }
335 else if (generatorType == "voronoi") {
336 auto voronoi = ref_ptr<module::Voronoi>::alloc();
337 voronoi->SetSeed(randomSeed);
338 voronoi->SetFrequency(input.getValue<double>("frequency", 4.0));
339 voronoi->SetDisplacement(input.getValue<double>("displacement", 0.0));
340 voronoi->EnableDistance(input.getValue<bool>("distance", false));
341 generator = ref_ptr<NoiseGenerator>::alloc("voronoi", voronoi);
342 }
343 else if (generatorType == "cylinders") {
344 auto cylinders = ref_ptr<module::Cylinders>::alloc();
345 cylinders->SetFrequency(input.getValue<double>("frequency", 4.0));
346 generator = ref_ptr<NoiseGenerator>::alloc("cylinders", cylinders);
347 }
348 else if (generatorType == "scale-bias") {
349 auto scaleBias = ref_ptr<module::ScaleBias>::alloc();
350 scaleBias->SetScale(input.getValue<double>("scale", 1.0));
351 scaleBias->SetBias(input.getValue<double>("bias", 0.0));
352 generator = ref_ptr<NoiseGenerator>::alloc("scale-bias", scaleBias);
353 }
354 else if (generatorType == "rotate-point") {
355 auto rotatePoint = ref_ptr<module::RotatePoint>::alloc();
356 rotatePoint->SetAngles(
357 input.getValue<double>("angle-x", 0.0),
358 input.getValue<double>("angle-y", 0.0),
359 input.getValue<double>("angle-z", 0.0));
360 generator = ref_ptr<NoiseGenerator>::alloc("rotate-point", rotatePoint);
361 }
362 else if (generatorType == "translate-point") {
363 auto translatePoint = ref_ptr<module::TranslatePoint>::alloc();
364 translatePoint->SetXTranslation(input.getValue<double>("x-translation", 0.0));
365 translatePoint->SetYTranslation(input.getValue<double>("y-translation", 0.0));
366 translatePoint->SetZTranslation(input.getValue<double>("z-translation", 0.0));
367 generator = ref_ptr<NoiseGenerator>::alloc("translate-point", translatePoint);
368 }
369 else if (generatorType == "add") {
370 generator = ref_ptr<NoiseGenerator>::alloc("add", ref_ptr<module::Add>::alloc());
371 }
372 else if (generatorType == "multiply") {
373 generator = ref_ptr<NoiseGenerator>::alloc("multiply", ref_ptr<module::Multiply>::alloc());
374 }
375 else if (generatorType == "max") {
376 generator = ref_ptr<NoiseGenerator>::alloc("max", ref_ptr<module::Max>::alloc());
377 }
378 else if (generatorType == "min") {
379 generator = ref_ptr<NoiseGenerator>::alloc("min", ref_ptr<module::Min>::alloc());
380 }
381 else if (generatorType == "select") {
382 auto select = ref_ptr<module::Select>::alloc();
383 select->SetEdgeFalloff(input.getValue<double>("edge-falloff", 0.0));
384 select->SetBounds(
385 input.getValue<double>("lower-bound", -0.25),
386 input.getValue<double>("upper-bound", 0.25));
387 generator = ref_ptr<NoiseGenerator>::alloc("select", select);
388 }
389 else {
390 REGEN_WARN("Unknown noise generator type '" << generatorType << "'.");
391 return {};
392 }
393 for (auto &n: input.getChildren()) {
394 if (n->getCategory() != "source") {
395 REGEN_WARN("Unknown noise generator type '" << n->getCategory() <<
396 "' in node '" << n->getDescription() << "'.");
397 continue;
398 }
399 auto sourceId = n->getName();
400 auto needle = generators.find(sourceId);
401 if (needle == generators.end()) {
402 REGEN_WARN("Unable to find noise generator '" << sourceId <<
403 "' in node '" << n->getDescription() << "'.");
404 continue;
405 }
406 generator->addSource(needle->second);
407 }
408 return generator;
409}

411ref_ptr<NoiseGenerator> NoiseGenerator::load(LoadingContext &ctx, scene::SceneInputNode &input) {
412 auto randomSeed = input.getValue<int>("random-seed", math::randomInt());
413
414 if (input.hasAttribute("preset")) {
415 auto preset = input.getValue("preset");
416 if (preset == "perlin") {
417 return preset_perlin(randomSeed);
418 } else if (preset == "wood") {
419 return preset_wood(randomSeed);
420 } else if (preset == "granite") {
421 return preset_granite(randomSeed);
422 } else if (preset == "cloud" || preset == "clouds") {
423 return preset_clouds(randomSeed);
424 } else {
425 REGEN_WARN("Unknown noise generator preset '" << preset << "'.");
426 return {};
427 }
428 }
429 auto outputGenerator = input.getValue("output-generator");
430 if (outputGenerator.empty()) {
431 REGEN_WARN("Missing 'output-generator' attribute for noise generator '" << input.getDescription() << "'.");
432 return {};
433 }
434
435 std::map<std::string, ref_ptr<NoiseGenerator>> generators;
436 for (auto &n: input.getChildren()) {
437 if (n->getCategory() == "generator") {
438 auto noiseGen = loadGenerator(ctx, *n.get(), generators, randomSeed);
439 if (noiseGen.get() != nullptr) {
440 generators[n->getName()] = noiseGen;
441 }
442 } else {
443 REGEN_WARN("Unknown noise generator type '" << n->getCategory() << "'.");
444 }
445 }
446 if (generators.empty()) {
447 REGEN_WARN("No noise generators found for '" << input.getDescription() << "'.");
448 return {};
449 }
450
451 auto needle = generators.find(outputGenerator);
452 if (needle == generators.end()) {
453 REGEN_WARN("Unable to find noise generator '" << outputGenerator << "'.");
454 needle = generators.begin();
455 }
456 return needle->second;
457}
