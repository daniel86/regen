24osgHimmel::t_equf Sun::apparentPosition(const osgHimmel::t_julianDay &t) {
25 const osgHimmel::t_julianDay T(osgHimmel::jCenturiesSinceSE(t));
26 const float M = _rad(meanAnomaly(t));
27
28 osgHimmel::t_eclf ecl;
29 ecl.longitude = _deg(4.895048 + 628.331951 * T
30 + (0.033417 - 0.000084 * T) * sin(M) + 0.000351 * sin(2 * M));
31 ecl.latitude = 0;

33 osgHimmel::t_equf equ;
34 equ = ecl.toEquatorial(Earth::trueObliquity(t));

45float Sun::distance(const osgHimmel::t_julianDay &t) {
46 const osgHimmel::t_julianDay T(osgHimmel::jCenturiesSinceSE(t));
47 const float M = 6.24f + 628.302f * T;
48 const float R = 1.000140f - (0.016708f - 0.000042f * T) * cos(M) - 0.000141f * cos(2 * M); // in AU
49 return _kms(R);
