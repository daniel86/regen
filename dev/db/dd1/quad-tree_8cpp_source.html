176bool QuadTree::readd(uint32_t itemIdx, bool allowSubdivision) { // NOLINT(misc-no-recursion)
177 const auto &shape = itemBoundingShapes_[itemIdx];
178 const OrthogonalProjection &projection = shape->orthoProjection();
179
180 if (root_->contains(projection)) {
181 // the node fully contains the shape, try to insert it here
182 if (insert1(root_, itemIdx, allowSubdivision)) {
183 return true;
184 } else {
185 itemNodeIdx_[itemIdx] = root_->nodeIdx;
186 return false;
187 }
188 } else {
189 REGEN_WARN("Shape '" << shape->name() << "." << shape->instanceID() << "' is out of bounds!");
190 // the shape is out of bounds, reinsert at root
191 itemNodeIdx_[itemIdx] = root_->nodeIdx;
192 return false;
193 }
194}

196bool QuadTree::reinsert(uint32_t itemIdx, bool allowSubdivision) { // NOLINT(misc-no-recursion)
197 const auto &shape = itemBoundingShapes_[itemIdx];
198 const auto nodeIdx = itemNodeIdx_[itemIdx];
199 const auto idxInNode = itemIdxInNode_[itemIdx];
200 const OrthogonalProjection &projection = shape->orthoProjection();
201
202 Node *m = nodes_[nodeIdx];
203 removeFromNode(m, idxInNode);
204 itemNodeIdx_[itemIdx] = -1;
205
206 Node *n = m;
207 while (n->parentIdx != -1 && !n->contains(projection)) {
208 n = nodes_[n->parentIdx];
209 }
210 insert1(n, itemIdx, allowSubdivision);
211 if (n->nodeIdx != m->nodeIdx) collapse(m);
212 return true;
213}

223bool QuadTree::insert1(Node *node, uint32_t newItemIdx, bool allowSubdivision) { // NOLINT(misc-no-recursion)
224 if (node->isLeaf()) {
225 // the node does not have child nodes (yet).
226 // the shape can be added to the node in three cases:
227 // 1. the node has still not exceeded the maximum number of shapes per node
228 // 2. the node has reached the minimum size and cannot be subdivided further
229 // 3. the node was just created by subdividing a parent node
230 if (!allowSubdivision ||
231 node->items.size() < subdivisionThreshold_ ||
232 node->bounds.size() < minNodeSize_) {
233 itemIdxInNode_[newItemIdx] = node->items.size();
234 node->items.push_back(newItemIdx);
235 itemNodeIdx_[newItemIdx] = node->nodeIdx;
236 return true;
237 } else {
238 // split the node into four children
239 subdivide(node);
240 // Keep a copy of the existing shapes in shapesTmp for iteration below.
241 node->itemsTmp = std::move(node->items);
242 node->items.clear();
243 // reinsert the existing shapes into the new children nodes
244 for (uint32_t itemIdx: node->itemsTmp) {
245 if (insert(node, itemIdx, true)) {
246 continue;
247 }
248 // failed to insert, try root node instead
249 if (insert(root_, itemIdx, true)) {
250 continue;
251 }
252 // still failed, this should never happen
253 REGEN_WARN("Shape '" << itemBoundingShapes_[itemIdx]->name() << "." <<
254 itemBoundingShapes_[itemIdx]->instanceID() << "' is out of bounds!");
255 }
256 // also insert the new shape
257 return insert(node, newItemIdx, true);
258 }
259 } else {
260 // the node has child nodes, must insert into (at least) one of them.
261 // note: a shape can be inserted into multiple nodes, so we allways need to check all children
262 // (at least on the next level)
263 for (auto &childIdx: node->childrenIdx) {
264 auto *child = nodes_[childIdx];
265 auto &shape = itemBoundingShapes_[newItemIdx];
266 if (child->contains(shape->orthoProjection())) {
267 // the child node fully contains the shape, insert it
268 if (insert(child, newItemIdx, allowSubdivision)) return true;
269 }
270 }
271 itemNodeIdx_[newItemIdx] = node->nodeIdx;
272 itemIdxInNode_[newItemIdx] = node->items.size();
273 node->items.push_back(newItemIdx);
274 return true;
275 }

278void QuadTree::remove(const ref_ptr<BoundingShape> &shape) {
279 auto it = shapeToItem_.find(shape.get());
280 if (it == shapeToItem_.end()) {
281 REGEN_WARN("Shape not found in quad tree.");
282 return;
283 }
284 uint32_t itemIdx = it->second;
285 auto *node = nodes_[itemNodeIdx_[itemIdx]];
286 removeFromNode(node, itemIdxInNode_[itemIdx]);
287 collapse(node);
288 itemNodeIdx_[itemIdx] = -1;

291void QuadTree::removeFromNode(Node *node, uint32_t idxInNode) {
292 // Here we remove item at index `idxInNode` from node->shapes array.
293 // We do this by moving the last item of the array into the position of the removed item,
294 // and then popping the last item.
295 uint32_t lastIdx = node->items.size() - 1;
296 if (idxInNode != lastIdx) {
297 // move the last item into the position of the removed item
298 uint32_t movedItemIdx = node->items[lastIdx];
299 node->items[idxInNode] = movedItemIdx;
300 itemIdxInNode_[movedItemIdx] = idxInNode;
301 }
302 node->items.pop_back();
303}

305QuadTree::Node* QuadTree::collapse(Node *node) { // NOLINT(misc-no-recursion)
306 auto *parent = (node->parentIdx == -1) ? nullptr : nodes_[node->parentIdx];
307 if (!parent) return node;
308 auto c0 = nodes_[parent->childrenIdx[0]];
309 auto c1 = nodes_[parent->childrenIdx[1]];
310 auto c2 = nodes_[parent->childrenIdx[2]];
311 auto c3 = nodes_[parent->childrenIdx[3]];
312 // parent can only be collapsed if all its children are leaf nodes without shapes
313 if (!c0->isCollapsable() || !c1->isCollapsable() ||
314 !c2->isCollapsable() || !c3->isCollapsable()) return node;
315
316 // free all children nodes
317 freeNode(c0);
318 freeNode(c1);
319 freeNode(c2);
320 freeNode(c3);
321 for (int i = 0; i < 4; i++) {
322 parent->childrenIdx[i] = -1;
323 }
324 // four nodes are removed, parent turns into a leaf node
325 numNodes_ -= 4;
326 numLeaves_ -= 3;
327 // continue collapsing the parent node
328 return collapse(parent);
329}

331void QuadTree::subdivide(Node *node) {
332 // Subdivide the node into four children.
333 auto center = (node->bounds.min + node->bounds.max) * 0.5f;
334 Node *child;
335 // bottom-left
336 child = createNode(node->bounds.min, center);
337 child->parentIdx = node->nodeIdx;
338 node->childrenIdx[0] = child->nodeIdx;
339 // bottom-right
340 child = createNode(Vec2f(center.x, node->bounds.min.y), Vec2f(node->bounds.max.x, center.y));
341 child->parentIdx = node->nodeIdx;
342 node->childrenIdx[1] = child->nodeIdx;
343 // top-right
344 child = createNode(center, node->bounds.max);
345 child->parentIdx = node->nodeIdx;
346 node->childrenIdx[2] = child->nodeIdx;
347 // top-left
348 child = createNode(Vec2f(node->bounds.min.x, center.y), Vec2f(center.x, node->bounds.max.y));
349 child->parentIdx = node->nodeIdx;
350 node->childrenIdx[3] = child->nodeIdx;
351 // four new nodes created, parent turns into an internal node
352 numNodes_ += 4;
353 numLeaves_ += 3;
354}

389bool QuadTree::Node::intersects(const OrthogonalProjection &projection) const {
390 if (projection.type == OrthogonalProjection::Type::CIRCLE) {
391 const float radiusSqr = projection.points[1].x; // = radius * radius
392 const auto &p = projection.points[0];
393 // Calculate the squared distance from the circle's center to the AABB
394 float sqDist = 0.0f;
395 if (p.x < bounds.min.x) {
396 sqDist += (bounds.min.x - p.x) * (bounds.min.x - p.x);
397 } else if (p.x > bounds.max.x) {
398 sqDist += (p.x - bounds.max.x) * (p.x - bounds.max.x);
399 }
400 if (p.y < bounds.min.y) {
401 sqDist += (bounds.min.y - p.y) * (bounds.min.y - p.y);
402 } else if (p.y > bounds.max.y) {
403 sqDist += (p.y - bounds.max.y) * (p.y - bounds.max.y);
404 }
405 return sqDist < radiusSqr;
406 } else {
407 // Check for separation along the axes of the shape and the axis-aligned quad
408 const uint32_t numAxes = projection.axes.size();
409 bool isOutside = false;
410 for (uint32_t i = 0; i < numAxes && !isOutside; i++) {
411 auto &axis = projection.axes[i];
412 const float minXX = bounds.min.x * axis.dir.x;
413 const float maxXX = bounds.max.x * axis.dir.x;
414 const float minYY = bounds.min.y * axis.dir.y;
415 const float maxYY = bounds.max.y * axis.dir.y;
416
417 const float p0 = minXX + minYY;
418 const float p1 = maxXX + minYY;
419 const float p2 = minXX + maxYY;
420 const float p3 = maxXX + maxYY;
421
422 const float projMin = std::min(std::min(p0, p1), std::min(p2, p3));
423 const float projMax = std::max(std::max(p0, p1), std::max(p2, p3));
424
425 isOutside = (projMax < axis.min || axis.max < projMin);
426 }
427 return !isOutside;
428 }
429 return false;
430}

442template<uint32_t NumAxes> __attribute__((noinline))
443void QuadTree::Private::testNodesAndAxes(QuadTreeTraversal &td) {
444 using namespace regen::simd;
445 // Process numQueuedItems_ nodes from the queue, performing an intersection test with the shape's projection;
446 // and also writing nodeIdx to successor array if the test succeeds.
447 const auto &axes = td.projection->axes;
448 int32_t queueIdx = 0;
449
450#ifndef QUAD_TREE_DISABLE_SIMD
451 if constexpr (QUAD_TREE_DEFERRED_BATCH_STORE) {
452 td.batchCounter_ = 0;
453 }
454
455 for (; queueIdx + RegisterWidth <= static_cast<int32_t>(td.numQueuedItems_); queueIdx += RegisterWidth) {
456 // Load the bounds of the node at nodeIdx into the SIMD registers
457 td.batchBoundsMinX = BatchOf_float::loadAligned(td.queuedMinX_.data() + queueIdx);
458 td.batchBoundsMinY = BatchOf_float::loadAligned(td.queuedMinY_.data() + queueIdx);
459 td.batchBoundsMaxX = BatchOf_float::loadAligned(td.queuedMaxX_.data() + queueIdx);
460 td.batchBoundsMaxY = BatchOf_float::loadAligned(td.queuedMaxY_.data() + queueIdx);
461
462 uint8_t mask = RegisterMask;
463
464 for (uint32_t axisIdx = 0; axisIdx < NumAxes; ++axisIdx) {
465 // d_i = corners[i].dot(axis.dir);
466 const BatchOf_Vec2f axisDir =
467 BatchOf_Vec2f::fromScalar(td.projection->axes[axisIdx].dir);
468 BatchOf_float d0 =
469 (td.batchBoundsMinX * axisDir.x) +
470 (td.batchBoundsMinY * axisDir.y);
471 BatchOf_float d1 =
472 (td.batchBoundsMaxX * axisDir.x) +
473 (td.batchBoundsMinY * axisDir.y);
474 BatchOf_float d2 =
475 (td.batchBoundsMinX * axisDir.x) +
476 (td.batchBoundsMaxY * axisDir.y);
477 BatchOf_float d3 =
478 (td.batchBoundsMaxX * axisDir.x) +
479 (td.batchBoundsMaxY * axisDir.y);
480
481 // mm = min{d0, d1, d2, d3}
482 BatchOf_float min = BatchOf_float::min(
483 BatchOf_float::min(d0,d1),
484 BatchOf_float::min(d2,d3));
485 // mm = max{d0, d1, d2, d3}
486 BatchOf_float max = BatchOf_float::max(
487 BatchOf_float::max(d0,d1),
488 BatchOf_float::max(d2,d3));
489 // Compute `(max_n < axis.min) || (axis.max < min_n)`
490 BatchOf_float sep =
491 (BatchOf_float::fromScalar(axes[axisIdx].max) < min) ||
492 (max < BatchOf_float::fromScalar(axes[axisIdx].min));
493
494 // Convert mask to bits
495 int sepMask = movemask_ps(sep.c); // 1 = separated
496 mask &= ~sepMask; // Clear bits in intersectMask where sepMask is 1
497 if constexpr (QUAD_TREE_MASK_EARLY_EXIT) {
498 if (mask == 0) return;
499 }
500 }
501 if constexpr (QUAD_TREE_DEFERRED_BATCH_STORE) {
502 td.batchResults_[td.batchCounter_++] = mask;
503 } else {
504 // add nodeIdx to successorIdx_ for each bit set in intersectMask
505 while (mask) {
506 int bitIndex = simd::nextBitIndex<uint8_t>(mask);
507 td.successorIdx_[td.numSucceedingItems_++] = queueIdx + bitIndex;
508 }
509 }
510 }
511
512 if constexpr (QUAD_TREE_DEFERRED_BATCH_STORE) {
513 for (int32_t batchIdx = 0; batchIdx < td.batchCounter_; ++batchIdx) {
514 uint8_t mask = td.batchResults_[batchIdx];
515 int32_t startIdx = batchIdx * RegisterWidth;
516 while (mask) {
517 int bitIndex = simd::nextBitIndex<uint8_t>(mask);
518 td.successorIdx_[td.numSucceedingItems_++] = startIdx + bitIndex;
519 }
520 }
521 td.batchCounter_ = 0;
522 }
523#endif
524
525 // Process remaining nodes scalar way
526 for (; queueIdx < static_cast<int32_t>(td.numQueuedItems_); queueIdx++) {
527 const float minX = td.queuedMinX_[queueIdx];
528 const float minY = td.queuedMinY_[queueIdx];
529 const float maxX = td.queuedMaxX_[queueIdx];
530 const float maxY = td.queuedMaxY_[queueIdx];
531
532 bool hasIntersection = true;
533 for (uint32_t axisIdx = 0; axisIdx < NumAxes; ++axisIdx) {
534 auto &axis = axes[axisIdx];
535 const float minXX = minX * axis.dir.x;
536 const float maxXX = maxX * axis.dir.x;
537 const float minYY = minY * axis.dir.y;
538 const float maxYY = maxY * axis.dir.y;
539
540 const float p0 = minXX + minYY;
541 const float p1 = maxXX + minYY;
542 const float p2 = minXX + maxYY;
543 const float p3 = maxXX + maxYY;
544
545 const float projMin = std::min(std::min(p0, p1), std::min(p2, p3));
546 const float projMax = std::max(std::max(p0, p1), std::max(p2, p3));
547
548 if (projMax < axis.min || axis.max < projMin) {
549 hasIntersection = false; break;
550 }
551 }
552
553 // write to successor array
554 td.successorIdx_[td.numSucceedingItems_] = queueIdx;
555 // but only increment the count if the projection intersects with the node
556 td.numSucceedingItems_ += static_cast<int32_t>(hasIntersection);
557 }
558 if constexpr(QUAD_TREE_DEBUG_TESTS) {
559 td.num2DTests_ += td.numQueuedItems_;
560 }
561 td.numQueuedItems_ = 0;
562}

564__attribute__((noinline))
565void QuadTree::Private::testNodesAndSphere(QuadTreeTraversal &td) {
566 // Process numQueuedItems_ nodes from the queue, performing an intersection test with the shape's projection;
567 // and also writing nodeIdx to successor array if the test succeeds.
568 const auto &radiusSqr = td.projection->points[1].x; // = radius * radius
569 const auto &center = td.projection->points[0];
570 int32_t queueIdx = 0;
571
572#ifndef QUAD_TREE_DISABLE_SIMD
573 using namespace regen::simd;
574 if constexpr (QUAD_TREE_DEFERRED_BATCH_STORE) {
575 td.batchCounter_ = 0;
576 }
577
578 BatchOf_float r_centerX = BatchOf_float::fromScalar(center.x);
579 BatchOf_float r_centerY = BatchOf_float::fromScalar(center.y);
580 BatchOf_float r_radius = BatchOf_float::fromScalar(radiusSqr);
581
582 for (; queueIdx + RegisterWidth <= static_cast<int32_t>(td.numQueuedItems_); queueIdx += RegisterWidth) {
583 // Load the bounds of the node at nodeIdx into the SIMD registers
584 td.batchBoundsMinX = BatchOf_float::loadAligned(td.queuedMinX_.data() + queueIdx);
585 td.batchBoundsMinY = BatchOf_float::loadAligned(td.queuedMinY_.data() + queueIdx);
586 td.batchBoundsMaxX = BatchOf_float::loadAligned(td.queuedMaxX_.data() + queueIdx);
587 td.batchBoundsMaxY = BatchOf_float::loadAligned(td.queuedMaxY_.data() + queueIdx);
588
589 uint8_t mask = RegisterMask;
590 // Compute distance along X
591 BatchOf_float distL = (td.batchBoundsMinX - r_centerX);
592 BatchOf_float distR = (r_centerX - td.batchBoundsMaxX);
593 BatchOf_float distX =
594 ((r_centerX < td.batchBoundsMinX) && distL) ||
595 ((r_centerX > td.batchBoundsMaxX) && distR);
596
597 // Compute distance along Y
598 distL = (td.batchBoundsMinY - r_centerY);
599 distR = (r_centerY - td.batchBoundsMaxY);
600 BatchOf_float distY =
601 ((r_centerY < td.batchBoundsMinY) && distL) ||
602 ((r_centerY > td.batchBoundsMaxY) && distR);
603
604 // Compute total squared distance
605 BatchOf_float sqDist = (distX*distX + distY*distY);
606 // Finally, compare against radius², and push nodes that intersect
607 BatchOf_float sep = (sqDist < r_radius);
608
609 // Convert mask to bits
610 int sepMask = movemask_ps(sep.c); // 1 = separated
611 mask &= ~sepMask; // Clear bits in intersectMask where sepMask is 1
612 if constexpr (QUAD_TREE_DEFERRED_BATCH_STORE) {
613 td.batchResults_[td.batchCounter_++] = mask;
614 } else {
615 // add nodeIdx to successorIdx_ for each bit set in intersectMask
616 while (mask) {
617 int bitIndex = simd::nextBitIndex<uint8_t>(mask);
618 td.successorIdx_[td.numSucceedingItems_++] = queueIdx + bitIndex;
619 }
620 }
621 }
622
623 if constexpr (QUAD_TREE_DEFERRED_BATCH_STORE) {
624 for (int32_t batchIdx = 0; batchIdx < td.batchCounter_; ++batchIdx) {
625 uint8_t mask = td.batchResults_[batchIdx];
626 int32_t startIdx = batchIdx * RegisterWidth;
627 while (mask) {
628 int bitIndex = simd::nextBitIndex<uint8_t>(mask);
629 td.successorIdx_[td.numSucceedingItems_++] = startIdx + bitIndex;
630 }
631 }
632 td.batchCounter_ = 0;
633 }
634#endif
635
636 // Process remaining nodes scalar way
637 for (; queueIdx < static_cast<int32_t>(td.numQueuedItems_); queueIdx++) {
638 float minX = td.queuedMinX_[queueIdx];
639 float maxX = td.queuedMaxX_[queueIdx];
640 float minY = td.queuedMinY_[queueIdx];
641 float maxY = td.queuedMaxY_[queueIdx];
642
643 // Calculate the squared distance from the circle's center to the AABB
644 float sqDist = 0.0f;
645 if (center.x < minX) {
646 sqDist += (minX - center.x) * (minX - center.x);
647 } else if (center.x > maxX) {
648 sqDist += (center.x - maxX) * (center.x - maxX);
649 }
650 if (center.y < minY) {
651 sqDist += (minY - center.y) * (minY - center.y);
652 } else if (center.y > maxY) {
653 sqDist += (center.y - maxY) * (center.y - maxY);
654 }
655 bool hasIntersection = sqDist < radiusSqr;
656
657 // write to successor array
658 td.successorIdx_[td.numSucceedingItems_] = queueIdx;
659 // but only increment the count if the projection intersects with the node
660 td.numSucceedingItems_ += static_cast<int32_t>(hasIntersection);
661 }
662 td.numQueuedItems_ = 0;
663}

665template<uint32_t NumAxes, QuadTree::TestMode_3D TestMode3D>
666void QuadTree::Private::intersectionLoop(QuadTreeTraversal &td) {
667 // Process the queued nodes, performing an intersection test with the shape's projection;
668 // and also writing nodeIdx to successor array if the test succeeds.
669 while (td.numQueuedItems_ > 0) {
670 td.currIdx_ = td.nextIdx_;
671 td.nextIdx_ = 1 - td.nextIdx_;
672 testNodesAndAxes<NumAxes>(td);
673 processSuccessors<TestMode3D>(td);
674 }
675}

677template<QuadTree::TestMode_3D TestMode3D>
678void QuadTree::Private::intersectionLoop_sphere(QuadTreeTraversal &td) {
679 while (td.numQueuedItems_ > 0) {
680 td.currIdx_ = td.nextIdx_;
681 td.nextIdx_ = 1 - td.nextIdx_;
682 testNodesAndSphere(td);
683 processSuccessors<TestMode3D>(td);
684 }

687template <QuadTree::TestMode_3D TestMode3D>
688void QuadTree::Private::processSuccessors(QuadTreeTraversal &td) {
689 // this function implements a scalar loop over all nodes that passed the intersection test
690 // in the current iteration:
691 // (1) if inner node, then add children to the queue for the next iteration.
692 // (2) if leaf node, then process the shapes in the node.
693 const auto &treeNodes = td.tree->nodes_;
694 const auto &currArray = td.queuedNodes_[td.currIdx_];
695 const auto &nextArray = td.queuedNodes_[td.nextIdx_];
696
697 for (uint32_t i = 0; i < td.numSucceedingItems_; i++) {
698 auto successorIdx = td.successorIdx_[i];
699 auto successor = treeNodes[currArray[successorIdx]];
700
701 if (successor->items.size() > 0) {
702 processLeafNode<TestMode3D>(td, successor);
703 }
704 if (!successor->isLeaf()) {
705 // add child nodes to the queue for the next iteration
706 addNodeToQueue(td, nextArray, treeNodes[successor->childrenIdx[0]], td.numQueuedItems_++);
707 addNodeToQueue(td, nextArray, treeNodes[successor->childrenIdx[1]], td.numQueuedItems_++);
708 addNodeToQueue(td, nextArray, treeNodes[successor->childrenIdx[2]], td.numQueuedItems_++);
709 addNodeToQueue(td, nextArray, treeNodes[successor->childrenIdx[3]], td.numQueuedItems_++);
710 }
711 }
712 td.numSucceedingItems_ = 0;
713}

715template <QuadTree::TestMode_3D TestMode3D> __attribute__((noinline))
716void QuadTree::Private::processLeafNode(QuadTreeTraversal &td, Node *leaf) {
717 // 3D intersection test with the shapes in the node.
718 // However, we do not always perform the intersection test, depending
719 // on the test mode configured for the quad tree.
720 // Also we submit each shape to a batched intersection tester,
721 // which will perform the actual intersection tests later in a batch
722 // (latest when endFrame is called on the quad tree).
723
724 const uint32_t traversalMask = td.traversalMask;
725 auto &itemShapes = td.tree->itemBoundingShapes_;
726
727 if constexpr (TestMode3D == QUAD_TREE_3D_TEST_NONE) {
728 // no intersection test, just record the hit
729 for (uint32_t itemIdx: leaf->items) {
730 const BoundingShape &shape = *itemShapes[itemIdx].get();
731 if ((shape.traversalMask() & traversalMask) != 0) {
732 td.hits.push(itemIdx);
733 }
734 }
735 }
736 if constexpr (TestMode3D == QUAD_TREE_3D_TEST_ALL) {
737 // test all shapes, even if they are not close to the shape's projection origin
738 for (uint32_t itemIdx: leaf->items) {
739 const BoundingShape &shape = *itemShapes[itemIdx].get();
740 if ((shape.traversalMask() & traversalMask) == 0) continue;
741 if constexpr (QUAD_TREE_3D_BATCHING) {
742 td.batchTest3D->push(shape, itemIdx);
743 } else {
744 if (td.shape->hasIntersectionWith(shape)) {
745 td.hits.push(itemIdx);
746 }
747 }
748 if constexpr(QUAD_TREE_DEBUG_TESTS) {
749 td.num3DTests_ += 1;
750 }
751 }
752 }
753 if constexpr (TestMode3D == QUAD_TREE_3D_TEST_CLOSEST) {
754 // heuristic: only test shapes that are close to the shape's projection origin (e.g. camera position)
755 // This is a good approach because:
756 // (1) shapes that are close use higher level of detail -> more expensive to draw false positives
757 // (2) most false positives are close to camera position in case camera is above/below the ground level
758 float distSq = (td.basePoint - leaf->center).lengthSquared();
759 if (distSq > td.tree->closeDistanceSquared_) {
760 for (uint32_t itemIdx: leaf->items) {
761 const BoundingShape &shape = *itemShapes[itemIdx].get();
762 if ((shape.traversalMask() & traversalMask) != 0) {
763 td.hits.push(itemIdx);
764 }
765 }
766 } else {
767 for (uint32_t itemIdx: leaf->items) {
768 const BoundingShape &shape = *itemShapes[itemIdx].get();
769 if ((shape.traversalMask() & traversalMask) == 0) continue;
770 if constexpr (QUAD_TREE_3D_BATCHING) {
771 td.batchTest3D->push(shape, itemIdx);
772 } else {
773 if (td.shape->hasIntersectionWith(shape)) {
774 td.hits.push(itemIdx);
775 }
776 }
777 if constexpr(QUAD_TREE_DEBUG_TESTS) {
778 td.num3DTests_ += 1;
779 }
780 }
781 }
782 }
783}

785template <QuadTree::TestMode_3D TestMode3D>
786void QuadTree::Private::updateIntersections(QuadTreeTraversal &td, const OrthogonalProjection &projection) {
787 switch (projection.type) {
788 case OrthogonalProjection::Type::CIRCLE: [[unlikely]]
789 intersectionLoop_sphere<TestMode3D>(td);
790 break;
791 case OrthogonalProjection::Type::RECTANGLE:
792 intersectionLoop<4,TestMode3D>(td);
793 break;
794 case OrthogonalProjection::Type::CONVEX_HULL: [[likely]]
795 switch (projection.axes.size()) {
796 case 4: [[unlikely]] intersectionLoop<4,TestMode3D>(td); break;
797 case 5: [[unlikely]] intersectionLoop<5,TestMode3D>(td); break;
798 case 6: intersectionLoop<6,TestMode3D>(td); break;
799 case 7: [[likely]] intersectionLoop<7,TestMode3D>(td); break;
800 case 8: [[likely]] intersectionLoop<8,TestMode3D>(td); break;
801 case 9: [[likely]] intersectionLoop<9,TestMode3D>(td); break;
802 case 10: intersectionLoop<10,TestMode3D>(td); break;
803 default:
804 REGEN_ERROR("unsupported number of axes for intersection test: " << projection.axes.size());
805 break;
806 }
807 break;
808 }

811HitBuffer& QuadTree::foreachIntersection(const BoundingShape &shape, uint32_t mask) {
812 // use thread-local traversal data to avoid reallocations,
813 // and synchronization between threads.
814 thread_local QuadTreeTraversal td;
815 if (td.tree != this) {
816 // make sure the traversal data belongs to this quad tree
817 // (in case of multiple quad trees in the application).
818 // This also handles the very first initialization of td.
819 td.tree = this;
820 if constexpr (QUAD_TREE_3D_BATCHING) {
821 if (!td.batchTest3D) {
822 td.batchTest3D = ref_ptr<BatchedIntersectionTest>::alloc();
823 td.batchTest3D->setHitBuffer(&td.hits);
824 }
825 td.batchTest3D->setIndexedShapes(&itemBoundingShapes_);
826 }
827 }
828 td.hits.reset();
829
830 if (!root_) return td.hits;
831 if (root_->isLeaf() && root_->items.empty()) return td.hits;
832
833 auto &origin = shape.tfOrigin();
834 // project the shape onto the xz-plane for faster intersection tests
835 // with the quad tree nodes.
836 const OrthogonalProjection &projection = shape.orthoProjection();
837
838 if constexpr (QUAD_TREE_3D_BATCHING) {
839 td.batchTest3D->setBatchCapacity(batchSize3D_);
840 }
841 td.numQueuedItems_ = 0;
842 td.numSucceedingItems_ = 0;
843 td.shape = &shape;
844 td.projection = &projection;
845 td.basePoint = Vec2f(origin.x, origin.z);
846 td.traversalMask = mask;
847
848 if (nextBufferSize_ > td.queuedMinX_.size()) {
849 td.queuedNodes_[0].resize(nextBufferSize_);
850 td.queuedNodes_[1].resize(nextBufferSize_);
851 td.queuedMinX_.resize(nextBufferSize_);
852 td.queuedMinY_.resize(nextBufferSize_);
853 td.queuedMaxX_.resize(nextBufferSize_);
854 td.queuedMaxY_.resize(nextBufferSize_);
855 td.successorIdx_.resize(nextBufferSize_);
856#ifndef QUAD_TREE_DISABLE_SIMD
857 if constexpr (QUAD_TREE_DEFERRED_BATCH_STORE) {
858 td.batchResults_.resize(nextBufferSize_ / simd::RegisterWidth + 1);
859 }
860#endif
861 }
862 if constexpr(QUAD_TREE_DEBUG_TESTS) {
863 td.num2DTests_ = 0;
864 td.num3DTests_ = 0;
865 }
866
867 if (td.hits.data.capacity() < itemNodeIdx_.size()) {
868 td.hits.resize(itemNodeIdx_.size());
869 }
870
871 if constexpr (QUAD_TREE_3D_BATCHING) {
872 td.batchTest3D->beginFrame(shape);
873 }
874 priv_->addNodeToQueue(td, td.queuedNodes_[td.nextIdx_], root_, td.numQueuedItems_++);
875
876 switch (testMode3D_) {
877 case QUAD_TREE_3D_TEST_NONE: [[unlikely]]
878 priv_->updateIntersections<QUAD_TREE_3D_TEST_NONE>(td, projection);
879 break;
880 case QUAD_TREE_3D_TEST_ALL:
881 priv_->updateIntersections<QUAD_TREE_3D_TEST_ALL>(td, projection);
882 break;
883 case QUAD_TREE_3D_TEST_CLOSEST: [[likely]]
884 priv_->updateIntersections<QUAD_TREE_3D_TEST_CLOSEST>(td, projection);
885 break;
886 }
887 if constexpr (QUAD_TREE_3D_BATCHING) {
888 td.batchTest3D->endFrame();
889 }
890
891 if constexpr(QUAD_TREE_DEBUG_TESTS) {
892 static uint32_t numFrames = 0;
893 static std::vector<uint32_t> numTests_2d;
894 static std::vector<uint32_t> numTests_3d;
895 numFrames++;
896 numTests_2d.push_back(td.num2DTests_);
897 numTests_3d.push_back(td.num3DTests_);
898 if (numFrames > 1000) {
899 uint32_t avg2DTests = 0;
900 uint32_t avg3DTests = 0;
901 for (size_t i = 0; i < numFrames; i++) {
902 avg2DTests += numTests_2d[i];
903 avg3DTests += numTests_3d[i];
904 }
905 avg2DTests /= numFrames;
906 avg3DTests /= numFrames;
907 REGEN_INFO("QuadTree:"
908 << " #nodes: " << numNodes_
909 << " #leaves: " << numLeaves_
910 << " #avg-2d: " << avg2DTests
911 << " #avg-3d: " << avg3DTests);
912 numFrames = 0;
913 numTests_2d.clear();
914 numTests_3d.clear();
915 }
916 }
917
918 return td.hits;
919}

921void QuadTree::update(float dt) {
922 static constexpr float minFloat = std::numeric_limits<float>::lowest();
923 static constexpr float maxFloat = std::numeric_limits<float>::max();
924
925 if (itemNodeIdx_.empty() && newItems_.empty()) {
926 // nothing to do
927 return;
928 }
929 bool hasChanged;
930 if constexpr(QUAD_TREE_DEBUG_TIME) {
931 priv_->elapsedTime_.beginFrame();
932 priv_->elapsedTime_.push("starting");
933 }
934
935 changedItems_.clear();
936 reAddedItems_.clear();
937 newBounds_.min.x = maxFloat;
938 newBounds_.min.y = maxFloat;
939 newBounds_.max.x = minFloat;
940 newBounds_.max.y = minFloat;
941 if constexpr(QUAD_TREE_EVER_GROWING) {
942 if (root_ != nullptr) {
943 // never shrink the root node
944 newBounds_.extend(root_->bounds);
945 }
946 }
947
948 // go through all items and update their geometry and transform, and the new bounds
949 for (uint32_t itemIdx = 0; itemIdx < itemNodeIdx_.size(); itemIdx++) {
950 auto &shape = itemBoundingShapes_[itemIdx];
951 auto nodeIdx = itemNodeIdx_[itemIdx];
952 auto idxInNode = itemIdxInNode_[itemIdx];
953 OrthogonalProjection &projection = shape->orthoProjection();
954 // NOTE: items with traversalMask == 0 are currently disabled,
955 // so we can skip updating them, and remove them from the tree.
956 if (shape->traversalMask() == 0) {
957 if (nodeIdx != -1) {
958 auto &node = nodes_[nodeIdx];
959 removeFromNode(node, idxInNode);
960 collapse(node);
961 itemNodeIdx_[itemIdx] = -1;
962 }
963 continue;
964 }
965 hasChanged = shape->updateGeometry();
966 hasChanged = shape->updateTransform(hasChanged) || hasChanged;
967 if (hasChanged) {
968 projection.update(*shape.get());
969 if (nodeIdx != -1) {
970 if (const auto &node = nodes_[nodeIdx]; !node->isLeaf() || !node->contains(projection)) {
971 // item has moved outside its node, so we need to re-insert it
972 changedItems_.push_back(itemIdx);
973 }
974 } else {
975 reAddedItems_.push_back(itemIdx);
976 }
977 }
978 newBounds_.extend(projection.bounds);
979 }
980
981 // do the same for any additional items added to the tree
982 for (const auto &item: newItems_) {
983 OrthogonalProjection &projection = item->shape->orthoProjection();
984 hasChanged = item->shape->updateGeometry();
985 hasChanged = item->shape->updateTransform(hasChanged) || hasChanged;
986 if (hasChanged) {
987 projection.update(*item->shape.get());
988 }
989 newBounds_.extend(projection.bounds);
990 }
991 if constexpr(QUAD_TREE_SQUARED) {
992 // make the bounds square
993 newBounds_.min.x = std::min(newBounds_.min.x, newBounds_.min.y);
994 newBounds_.min.y = newBounds_.min.x;
995 newBounds_.max.x = std::max(newBounds_.max.x, newBounds_.max.y);
996 newBounds_.max.y = newBounds_.max.x;
997 }
998 if constexpr(QUAD_TREE_DEBUG_TIME) {
999 priv_->elapsedTime_.push("bounds-update");
1000 }
1001
1002 if (root_ == nullptr || newBounds_ != root_->bounds) {
1003 // if bounds have changed, re-initialize the tree
1004 // free the root node and start all over
1005 if (root_) freeNode(root_);
1006 root_ = createNode(newBounds_.min, newBounds_.max);
1007 numNodes_ = 1;
1008 numLeaves_ = 1;
1009
1010 for (uint32_t itemIdx = 0; itemIdx < itemNodeIdx_.size(); itemIdx++) {
1011 auto &shape = itemBoundingShapes_[itemIdx];
1012 if (shape->traversalMask() == 0) continue;
1013 itemNodeIdx_[itemIdx] = -1; // reset node index
1014 insert1(root_, itemIdx, true);
1015 }
1016 } else {
1017 // else remove/insert the changed items
1018 for (uint32_t itemIdx: changedItems_) {
1019 reinsert(itemIdx, true);
1020 }
1021 for (uint32_t itemIdx: reAddedItems_) {
1022 readd(itemIdx, true);
1023 }
1024 }
1025 if constexpr(QUAD_TREE_DEBUG_TIME) {
1026 priv_->elapsedTime_.push("re-insertions");
1027 }
1028
1029 // finally insert the new items
1030 for (auto item: newItems_) {
1031 uint32_t newItemIdx = static_cast<uint32_t>(itemNodeIdx_.size());
1032 itemIdxInNode_.push_back(0);
1033 itemNodeIdx_.push_back(-1);
1034 freeItem(item);
1035
1036 if (insert1(root_, newItemIdx, true)) {
1037 shapeToItem_[item->shape.get()] = newItemIdx;
1038 } else {
1039 REGEN_WARN("Failed to insert shape into quad tree. This should not happen!");
1040 }
1041 }
1042 newItems_.clear();
1043 if constexpr(QUAD_TREE_DEBUG_TIME) {
1044 priv_->elapsedTime_.push("new-insertions");
1045 }
1046
1047 // update buffer sizes for intersection tests.
1048 // at max we will have num-leaves nodes in the queue.
1049 // here we will resize the arrays to the next power of two to have sufficient space.
1050 nextBufferSize_ = (numLeaves_ * 2);
1051
1052 // make the visibility computations
1053 updateVisibility(traversalBit_);
1054 if constexpr(QUAD_TREE_DEBUG_TIME) {
1055 priv_->elapsedTime_.push("updated-visibility");
1056 priv_->elapsedTime_.endFrame();
1057 }
1058}
