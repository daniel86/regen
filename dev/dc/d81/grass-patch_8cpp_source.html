48void GrassPatch::generateLODLevel(uint32_t lodLevel) {
49 auto &grassLOD = meshLODs_[lodLevel];
50 // map client data for writing
51 auto grass_p = (Vec3f*) pos_->clientBuffer()->clientData(0);
52 auto grass_base = (Vec3f*) basePos_->clientBuffer()->clientData(0);
53 auto grass_i = (byte*)indices_->clientBuffer()->clientData(0);
54 auto indexType = indices_->baseType();
55 // offsets into data arrays
56 grass_p += grassLOD.vertexOffset();
57 grass_base += grassLOD.vertexOffset();
58 grass_i += grassLOD.indexOffset() * indices_->dataTypeBytes();
59 const uint32_t vBaseOffset = grassLOD.vertexOffset();
60 const auto &uvRects = silhouetteMesh_->silhouetteUVRects()[lodLevel];
61 const uint32_t numQuadsPerVertex = uvRects.size();
62
63 auto &maskLOD = maskMesh_->meshLODs()[lodLevel];
64 auto mask_p = maskMesh_->pos()->mapClientData<Vec3f>(BUFFER_GPU_READ);
65 const uint32_t maskOffset = maskLOD.vertexOffset();
66
67 Vec2f grassQuadSize = Vec2f::one();
68 if (lodLevel > 1) {
69 // HACK: increase quad size for higher LODs where we use less vertices.
70 // That will make the grass look more dense.
71 grassQuadSize.x *= static_cast<float>(lodLevel);
72 }
73
74 uint32_t vOffset = 0u;
75 uint32_t iOffset = 0u;
76 for (uint32_t objIdx = 0; objIdx < maskLOD.numVertices(); ++objIdx) {
77 // The base position is centered at the bottom of the quad.
78 const Vec3f &basePos = mask_p.r[maskOffset + objIdx];
79
80 for (uint32_t objQuadIdx = 0; objQuadIdx < numQuadsPerVertex; ++objQuadIdx) {
81 const auto &uvRect = uvRects[objQuadIdx];
82 // uvRect = (u0, v0, u1, v1) in normalized 0..1 space
83 float u0 = uvRect.x, v0 = uvRect.y, u1 = uvRect.z, v1 = uvRect.w;
84
85 // use (0..1) local coords around center:
86 Vec2f local0( (u0+u1)*0.5f - 0.5f, (v0+v1)*0.5f - 0.5f );
87 Vec2f localSize( u1 - u0, v1 - v0 );
88
89 // We'll create geometry in UV-space scaled to [-0.5..0.5] so that instancing can position it in world.
90 Vec3f p0(local0.x - localSize.x*0.5f, local0.y - localSize.y*0.5f + 0.5f, 0.0f);
91 Vec3f p1(local0.x + localSize.x*0.5f, local0.y - localSize.y*0.5f + 0.5f, 0.0f);
92 Vec3f p2(local0.x + localSize.x*0.5f, local0.y + localSize.y*0.5f + 0.5f, 0.0f);
93 Vec3f p3(local0.x - localSize.x*0.5f, local0.y + localSize.y*0.5f + 0.5f, 0.0f);
94 p0.xy() *= grassQuadSize;
95 p1.xy() *= grassQuadSize;
96 p2.xy() *= grassQuadSize;
97 p3.xy() *= grassQuadSize;
98
99 // Start with the 4 vertices of the quad.
100 grass_p[vOffset + 0] = basePos + p0;
101 grass_p[vOffset + 1] = basePos + p1;
102 grass_p[vOffset + 2] = basePos + p2;
103 grass_p[vOffset + 3] = basePos + p3;
104 for (uint32_t i=0; i<4; i++) {
105 minPosition_.setMin(grass_p[vOffset + i]);
106 maxPosition_.setMax(grass_p[vOffset + i]);
107 }
108 // Also set the base position for the grass quad.
109 grass_base[vOffset + 0] = basePos;
110 grass_base[vOffset + 1] = basePos;
111 grass_base[vOffset + 2] = basePos;
112 grass_base[vOffset + 3] = basePos;
113 // Finally, set the indices for the quad.
114 setIndexValue(grass_i, indexType, iOffset + 0, vBaseOffset + vOffset + 0);
115 setIndexValue(grass_i, indexType, iOffset + 1, vBaseOffset + vOffset + 1);
116 setIndexValue(grass_i, indexType, iOffset + 2, vBaseOffset + vOffset + 2);
117 setIndexValue(grass_i, indexType, iOffset + 3, vBaseOffset + vOffset + 2);
118 setIndexValue(grass_i, indexType, iOffset + 4, vBaseOffset + vOffset + 3);
119 setIndexValue(grass_i, indexType, iOffset + 5, vBaseOffset + vOffset + 0);
120 // Move to the next quad's vertices
121 vOffset += 4;
122 iOffset += 6;
123 }
124 }
125}

127void GrassPatch::updateAttributes() {
128 // Extrude each vertex either into (1) a quad or (2) a cross of 3 quads.
129 // Idea: only use cross for lod=0, else use a single quad.
130 maskMesh_->updateAttributes();
131
132 // count vertices and indices, and create LOD descriptions
133 uint32_t numVertices = 0u;
134 uint32_t numIndices = 0u;
135 for (uint32_t lodIdx = 0u; lodIdx < maskMesh_->meshLODs().size(); ++lodIdx) {
136 auto &maskLOD = maskMesh_->meshLODs()[lodIdx];
137 const auto &uvRects = silhouetteMesh_->silhouetteUVRects()[lodIdx];
138 const uint32_t numQuadsPerVertex = uvRects.size();
139 auto &x = meshLODs_.emplace_back();
140 x.d->numVertices = maskLOD.d->numVertices * numQuadsPerVertex * 4;
141 x.d->numIndices = maskLOD.d->numVertices * numQuadsPerVertex * 6;
142 x.d->vertexOffset = numVertices;
143 x.d->indexOffset = numIndices;
144 numVertices += x.d->numVertices;
145 numIndices += x.d->numIndices;
146 }
158 // allocate attributes
159 pos_->setVertexData(numVertices);
160 basePos_->setVertexData(numVertices);
161 indices_ = createIndexInput(numIndices, numVertices);
162 minPosition_ = Vec3f::zero();
163 maxPosition_ = Vec3f::zero();
164
165 for (auto lodIdx = 0u; lodIdx < maskMesh_->meshLODs().size(); ++lodIdx) {
166 generateLODLevel(lodIdx);
167 }
168
169 auto indexRef = setIndices(indices_, numVertices);
170 setInput(pos_);
171 setInput(basePos_);
172 updateVertexData();
173
174 for (auto &x: meshLODs_) {
175 // add the index buffer offset (in number of bytes)
176 x.d->indexOffset = indexRef->address() + x.d->indexOffset * indices_->dataTypeBytes();
177 }
178 activateLOD(0);
179}

250 maskTexture = scene->getResource<Texture2D>(materialTextureName);

255 input.getValue("silhouette-texture"));

261 auto m = ref_ptr<GrassPatch>::alloc(tf, maskCfg.quad.updateHint);
262 m->setMask(maskTexture, maskIndex, maskCfg);
263 m->setSilhouette(silhouetteTexture, silhouetteCfg);
