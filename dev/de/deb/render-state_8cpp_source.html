252 // init toggle states
253 GLenum enabledToggles[] = {
254 GL_CULL_FACE, GL_DEPTH_TEST,
255 GL_TEXTURE_CUBE_MAP_SEAMLESS
256 };
257 for (int i = 0; i < TOGGLE_STATE_LAST; ++i) {
258 GLenum e = toggleToID((Toggle) i);
259 // avoid initial state set for unsupported states
260 if (e == GL_NONE) continue;
261
262 bool enabled = false;
263 for (uint32_t j = 0; j < sizeof(enabledToggles) / sizeof(GLenum); ++j) {
264 if (enabledToggles[j] == e) {
265 enabled = true;
266 break;
267 }
268 }
269 toggles_.push(i, enabled);
270 }
271 toggles_.push(RenderState::BLEND, false);
272 // init value states
273 cullFace_.push(GL_BACK);
274 depthMask_.push(true);
275 depthFunc_.push(GL_LEQUAL);
276 depthRange_.push(DepthRange(0.0, 1.0));
277 blendEquation_.push(BlendEquation::create(GL_FUNC_ADD));
278 blendFunc_.push(BlendFunction(GL_ONE, GL_ONE, GL_ZERO, GL_ZERO));
279 polygonMode_.push(GL_FILL);
280 polygonOffset_.push(Vec2f::zero());
281 pointSize_.push(1.0);
282 lineWidth_.push(1.0);
283 colorMask_.push(ColorMask::create(true));
284 logicOp_.push(GL_COPY);
285 frontFace_.push(GL_CCW);
286 pointFadeThreshold_.push(1.0);
287 pointSpriteOrigin_.push(GL_UPPER_LEFT);
288 textureBuffer_.push(0);
289 // for pixel pack buffers we want that each time we have a push-pop sequence that
290 // "glBindBuffer(GL_PIXEL_[UN]PACK_BUFFER, 0)" is called.
291 pixelPackBuffer_.push(0);
292 pixelUnpackBuffer_.push(0);
293 copyWriteBuffer().push(0);
294 copyReadBuffer().push(0);
295 GL_ERROR_LOG();

367void RenderState::bindBufferBase(GLenum target, uint32_t index, uint32_t buffer) {
368 auto arrayIndex = bufferBaseIndex(target);
369 auto &bindings = bufferBaseBindings_[arrayIndex];
370 auto needle = bindings.find(index);
371 if (needle != bindings.end() && needle->second == buffer) {
372 // buffer is already bound
373 return;
374 }
375 bindings[index] = buffer;
376 Regen_BindBufferBase(target, index, buffer);
377}
