110void SkirtMesh::addSkirtLOD(
111 const Rectangle::Config &rectangleConfig,
112 const Tessellation &planeTess) {
113 // add skirt to tessellation, i.e. duplicate boundary vertices and pull them down.
114 // to this end, we first need to find the boundary edges, then we duplicate the vertices,
115 // and finally we add the skirt triangles.
116 auto makeEdge = [](uint32_t a, uint32_t b) {
117 return a < b ? std::make_pair(a, b) : std::make_pair(b, a);
118 };
119 Tessellation &skirtTess = skirtTess_.emplace_back();
120
121 // First find boundary edges, these are the ones that are only used by one triangle.
122 using Edge = std::pair<uint32_t, uint32_t>;
123 std::set<Edge> boundaryEdges; {
124 std::map<Edge, uint32_t> edgeCount;
125 for (const auto &face: planeTess.outputFaces) {
126 edgeCount[makeEdge(face.v1, face.v2)]++;
127 edgeCount[makeEdge(face.v2, face.v3)]++;
128 edgeCount[makeEdge(face.v3, face.v1)]++;
129 }
130 for (const auto &edge: edgeCount) {
131 if (edge.second == 1) {
132 boundaryEdges.insert(edge.first);
133 }
134 }
135 }
136
137 // create indices for skirt vertices
138 std::map<uint32_t, uint32_t> boundaryToSkirt;
139 uint32_t skirtVertexOffset = 0; //tessellation.vertices.size();
140 for (const auto &edge: boundaryEdges) {
141 if (boundaryToSkirt.find(edge.first) == boundaryToSkirt.end()) {
142 boundaryToSkirt[edge.first] = skirtVertexOffset++;
143 }
144 if (boundaryToSkirt.find(edge.second) == boundaryToSkirt.end()) {
145 boundaryToSkirt[edge.second] = skirtVertexOffset++;
146 }
147 }
148 // create vertex data for skirt vertices
149 Mat4f rotMat = Mat4f::rotationMatrix(
150 rectangleConfig.rotation.x,
151 rectangleConfig.rotation.y,
152 rectangleConfig.rotation.z);
153 skirtTess.vertices.resize(boundaryToSkirt.size()*2);
154 for (const auto [bIdx,sIdx] : boundaryToSkirt) {
155 auto boundaryPos = planeTess.vertices[bIdx];
156 skirtTess.vertices[sIdx*2] = boundaryPos;
157 skirtTess.vertices[sIdx*2+1] = boundaryPos +
158 rotMat.transformVector(Vec3f(0.0f, -skirtSize_, 0.0f));
159 }
160
161 // Finally, add skirt triangles: for each boundary edge, create two triangles.
162 // Iterate over the original faces to avoid winding issues (probably can be improved).
163 for (const auto &face : planeTess.outputFaces) {
164 const Edge edges[3] = {
165 Edge{face.v1, face.v2},
166 Edge{face.v2, face.v3},
167 Edge{face.v3, face.v1}
168 };
169 for (const auto &edge : edges) {
170 if (boundaryEdges.find(makeEdge(edge.first, edge.second)) != boundaryEdges.end()) {
171 auto skirtV1 = boundaryToSkirt[edge.first];
172 auto skirtV2 = boundaryToSkirt[edge.second];
173 skirtTess.outputFaces.emplace_back(skirtV1*2+1, skirtV2*2+1, skirtV1*2);
174 skirtTess.outputFaces.emplace_back(skirtV2*2+1, skirtV2*2, skirtV1*2);
175 }
176 }
177 }
178}

180void SkirtMesh::generateLODLevel(const Rectangle::Config &cfg,
181 const Tessellation &skirtTess,
182 const Mat4f &rotMat,
183 uint32_t vertexOffset,
184 uint32_t indexOffset) {
185 // map client data for writing
186 auto v_pos = (Vec3f*) pos_->clientBuffer()->clientData(0);
187 auto indices = (byte*)indices_->clientBuffer()->clientData(0);
188 auto indexType = indices_->baseType();
189
190 uint32_t nextIndex = indexOffset;
191 for (auto &tessFace: skirtTess.outputFaces) {
192 setIndexValue(indices, indexType, nextIndex++, vertexOffset + tessFace.v1);
193 setIndexValue(indices, indexType, nextIndex++, vertexOffset + tessFace.v2);
194 setIndexValue(indices, indexType, nextIndex++, vertexOffset + tessFace.v3);
195 }
196
197 Vec3f startPos;
198 if (cfg.centerAtOrigin) {
199 startPos = Vec3f(-cfg.posScale.x * 0.5f, 0.0f, -cfg.posScale.z * 0.5f);
200 } else {
201 startPos = Vec3f(0.0f, 0.0f, 0.0f);
202 }
203
204 for (uint32_t faceIndex = 0; faceIndex < skirtTess.outputFaces.size(); ++faceIndex) {
205 auto &face = skirtTess.outputFaces[faceIndex];
206
207 for (auto &tessIndex: {face.v1, face.v2, face.v3}) {
208 auto &vertex = skirtTess.vertices[tessIndex];
209 auto vertexIndex = vertexOffset + tessIndex;
210 Vec3f pos = rotMat.transformVector(
211 cfg.posScale * vertex + startPos) + cfg.translation;
212 v_pos[vertexIndex] = pos;
213 minPosition_.setMin(pos);
214 maxPosition_.setMax(pos);
215 }
216 }
