64 // Try to find the mesh
65 auto meshID = input.getValue("mesh-id");

67 auto compositeMesh = scene->getResource<CompositeMesh>(meshID);
68 if (compositeMesh.get() == nullptr || compositeMesh->meshes().empty()) {
69 REGEN_WARN("Unable to find Mesh for '" << input.getDescription() << "'.");
70 return {};
71 }
72 if (compositeMesh->meshes().size() <= meshIndex) {
73 REGEN_WARN("Mesh index " << meshIndex << " is out of range for '" << input.getDescription() << "'.");
74 meshIndex = 0u;
75 }
76 auto &mesh = compositeMesh->meshes()[meshIndex];
77 auto shaderKey = input.getValue("shader");
78 auto animation = ref_ptr<AttributeFeedbackAnimation>::alloc(mesh, shaderKey);
79 // create a state using children of the input node
80 auto &animationNode = animation->animationNode_;
81 for (auto &child: input.getChildren()) {
82 auto stateProcessor = scene->getStateProcessor(child->getCategory());
83 if (stateProcessor.get() != nullptr) {
84 stateProcessor->processInput(
85 scene, *child.get(),
86 animationNode, animationNode->state());
87 } else {
88 REGEN_WARN("No state processor registered for '" << child->getDescription() << "'.");
89 }
90 }
91 animation->initializeResources();
92 animation->startAnimation();
93 return animation;
