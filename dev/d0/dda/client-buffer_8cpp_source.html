136uint32_t ClientBuffer::swapData() {
137 // NOTE: This function should be very fast as potentially both animation and rendering threads
138 // are waiting for it to finish.
139 // flushing is only needed if the buffer is frame-locked.
140 if (!isFrameLocked_ || dataSize_ == 0u) return 0u;
141
142 const int32_t lastReadSlot = swapDataSlot_;
143 auto &dirtyLastFrame = dirtyLists_[lastReadSlot];
144 //assert(lastReadSlot == lastDataSlot_.load(std::memory_order_relaxed));
145
146 if (!dataSlots_[1]) {
147 // Single-buffered mode, no need to copy data.
148 // Just clear the dirty list for the last read slot.
149 dirtyLastFrame.clear();
150 return 0u;
151 } else if (dirtyLastFrame.empty()) {
152 const int32_t lastWriteSlot = 1 - lastReadSlot;
153 const auto &dirtyThisFrame = dirtyLists_[lastWriteSlot];
154
155 if(!dirtyThisFrame.empty()) {
156 // there was a write this frame -> need to do a swap
157 writeLockAll();
158 lastDataSlot_.store(lastWriteSlot, std::memory_order_release);
159 writeUnlockAll(0u, 0u);
160 swapDataSlot_ = lastWriteSlot;
161 }
162
163 return 0u;
164 } else {
165 // There were some dirty ranges in the last frame, we may need to copy the updated data
166 // over to have the full data in place for reading in the next frame.
167 const int32_t lastWriteSlot = 1 - lastReadSlot;
168 const auto &dirtyThisFrame = dirtyLists_[lastWriteSlot];
169
170 // pull the slot data pointers
171 const byte* __restrict lastReadData = dataSlots_[lastReadSlot];
172 byte* __restrict lastWriteData = dataSlots_[lastWriteSlot];
173
174 // We need to avoid race conditions of another thread getting a read lock
175 // while we do the switching, then the thread may attempt (while holding read)
176 // to acquire a write lock which will fail because the readers are blocking.
177 // Also we need to avoid any writer adding stuff to the dirty list while we read it.
178 writeLockAll();
179
180 // Delete all dirty ranges from the last read slot that have been written to this frame.
181 // It is certain that both dirty lists are coalesced, so calling subtract is safe.
182 dirtyLastFrame.subtract(dirtyThisFrame);
183 // Coalesce the remaining dirty ranges reducing the number of copy operations needed.
184 dirtyLastFrame.coalesce();
185
186 // Remaining are the ranges where data in the write slot is not up-to-date with the read slot,
187 // hence we copy it over.
188 const uint32_t numCopiesNeeded = dirtyLastFrame.count();
189 for (uint32_t rangeIdx = 0; rangeIdx < numCopiesNeeded; ++rangeIdx) {
190 const auto &range = dirtyLastFrame.ranges()[rangeIdx];
191 // Copy the data from the read slot to the write slot.
192 std::memcpy(
193 lastWriteData + range.offset,
194 lastReadData + range.offset,
195 range.size);
196 }
197
198 // For each write segment with stamp < read segment stamp: set the stamp to read segment stamp,
199 // as we have synced the data above.
200 const uint32_t readStamp = loadAtomicStamp(dataStamps_[lastReadSlot]);
201 const uint32_t writeStamp = loadAtomicStamp(dataStamps_[lastWriteSlot]);
202 if (writeStamp < readStamp) {
203 dataStamps_[lastWriteSlot].value.store(readStamp, std::memory_order_relaxed);
204 }
205 for (auto &segment: bufferSegments_) {
206 const auto s_writeStamp = loadAtomicStamp(segment->dataStamps_[lastWriteSlot]);
207 const auto s_readStamp = loadAtomicStamp(segment->dataStamps_[lastReadSlot]);
208 if (s_writeStamp < s_readStamp) {
209 segment->dataStamps_[lastWriteSlot].value.store(s_readStamp, std::memory_order_relaxed);
210 }
211 }
212
213 // Finally swap read and write idx, new read idx should have new data for reading next frame.
214 lastDataSlot_.store(lastWriteSlot, std::memory_order_release);
215
216 // Unlock the write locks on both slots.
217 writeUnlockAll(0u, 0u);
218
219 // Keep a cached copy of the last read slot for the next swap.
220 swapDataSlot_ = lastWriteSlot;
221
222 // clear dirty lists for the last read slot, such that it can be reused
223 // next frame for writing.
224 dirtyLastFrame.clear();
225
226 return numCopiesNeeded;
227 }

230void ClientBuffer::nextStamp(int32_t writeSlot) const {
231 // Increase the stamp for the given write slot to one higher than the current read slot.
232 const int32_t readSlot = (dataSlots_[1] ? (1 - writeSlot) : 0);
233 uint32_t stamp = loadAtomicStamp(dataStamps_[readSlot]);
234 dataStamps_[writeSlot].value.store(stamp + 1, std::memory_order_relaxed);
235
236 // Propagate to parent buffers.
237 auto *parent = parentBuffer_;
238 while (parent) {
239 stamp = loadAtomicStamp(parent->dataStamps_[readSlot]);
240 parent->dataStamps_[writeSlot].value.store(stamp + 1, std::memory_order_relaxed);
241 parent = parent->parentBuffer_;
242 }
243}

245void ClientBuffer::nextSegmentStamp(int32_t dataSlot, uint32_t writeBegin, uint32_t writeSize) const {
246 const uint32_t writeEnd = writeBegin + writeSize;
247
248 // Update the stamp for all segments that overlap with the updated range.
249 for (auto &segment: bufferSegments_) {
250 if (writeBegin < segment->dataOffset_ + segment->dataSize_ && writeEnd > segment->dataOffset_) {
251 // check if the segment overlaps with the updated range.
252 const int32_t readSlot = (segment->dataSlots_[1] ? (1 - dataSlot) : 0);
253 const uint32_t readStamp = loadAtomicStamp(segment->dataStamps_[readSlot]);
254 segment->dataStamps_[dataSlot].value.store(readStamp + 1, std::memory_order_relaxed);
255 } else if (segment->dataOffset_ >= writeEnd) {
256 // drop out if segment is located after the updated range
257 break;
258 }
259 }

262void ClientBuffer::nextStamp() const {
263 // Increase the stamp for both slots.
264 // This is only rarely used, e.g. when segments are changed or buffer resized.
265 uint32_t stamp = 1u + std::max(
266 loadAtomicStamp(dataStamps_[0]),
267 loadAtomicStamp(dataStamps_[1]));
268 dataStamps_[0].value.store(stamp, std::memory_order_relaxed);
269 dataStamps_[1].value.store(stamp, std::memory_order_relaxed);
270
271 // Propagate to parent buffers.
272 auto *parent = parentBuffer_;
273 while (parent != nullptr) {
274 stamp = 1u + std::max(
275 loadAtomicStamp(parent->dataStamps_[0]),
276 loadAtomicStamp(parent->dataStamps_[1]));
277 parent->dataStamps_[0].value.store(stamp, std::memory_order_relaxed);
278 parent->dataStamps_[1].value.store(stamp, std::memory_order_relaxed);
279 parent = parent->parentBuffer_;
280 }

283MappedClientData ClientBuffer::mapRange(int mapMode, uint32_t offset, uint32_t size) const {
284 const bool singleBufferMode = (clientBufferMode_ == SingleBuffer ||
285 (clientBufferMode_ == AdaptiveBuffer && !hasTwoSlots()));
286
287 if ((mapMode & BUFFER_GPU_WRITE) != 0) {
288 if (singleBufferMode) {
289 // Single-buffered mode, we can only write to the first slot.
290 return writeRange_SingleBuffer(offset, size);

292 // Double-buffered mode, we can write to either slot.
293 return writeRange_DoubleBuffer(offset, size);
294 }
295 } else if (singleBufferMode) {
296 // Single-buffered mode, we can only read from the first slot.
297 return readRange_SingleBuffer(offset, size);
298 } else { // clientBufferMode_ == DoubleBuffer
299 // Double-buffered mode, we can read from either slot.
300 return readRange_DoubleBuffer(offset, size);

310MappedClientData ClientBuffer::writeRange_DoubleBuffer(uint32_t offset, uint32_t size) const {
311 // we are in double-buffered mode, i.e. we have two slots.
312 // partial write can be expensive here!
313 // NOTE: no index mapping needed if there is only one vertex/array element
314 int w_index = writeLock_DoubleBuffer();
315 byte *data_w = dataSlots_[w_index];
316
317 if (dataSize_ == size) {
318 // FULL write
319 return {
320 dataSlots_[1 - w_index] + offset, -1,
321 data_w + offset, w_index};
322 } else { // PARTIAL WRITE
323 if (!isFrameLocked_) {
324 // we swap after each write operation, and a partial write is required.
325 // make sure to copy the data from the read slot to the write slot before we do the swap.
326 const int r_index = readLock();
327 std::memcpy(data_w, dataSlots_[r_index], dataSize_);
328 readUnlock(r_index);
329 }
330 return {
331 dataSlots_[1 - w_index] + offset, -1,
332 data_w + offset, w_index};
333 }
334}

336MappedClientData ClientBuffer::readRange_SingleBuffer(uint32_t offset, uint32_t size) const {
337 // we are still in single-buffered mode.
338 // first we try to get a read lock on the single slot.
339 if (readLock_SingleBuffer()) {
340 // got the read lock, return the data.
341 return {dataSlots_[0] + offset, 0};
342 } else if (isOwnerOfWriteLock(0)) {
343 // Read lock failed, which means there is a write operation in progress. There are two cases:
344 // (1) This thread holds the write lock on slot 0. If we wait here, then
345 // we would deadlock. But it is actually fine in this case to also
346 // read-lock the very same slot! then we can stay single-buffered.
347 dataOwner_->readerCounts_[0].value.fetch_add(1, std::memory_order_relaxed);
348 // Verify that we are still the last owner of the write lock on slot 0.
349 if (!isOwnerOfWriteLock(0)) {
350 dataOwner_->readerCounts_[0].value.fetch_sub(1, std::memory_order_relaxed);
351 return mapRange(BUFFER_GPU_READ, offset, size); // retry the read operation
352 }
353 // Got the read lock, return the data.
354 return {dataSlots_[0] + offset, 0};
355 } else {
356 // (2) Another thread holds the lock. Hence, it is not safe to copy data from
357 // slot 0 into slot 1 -> We need to wait until the other thread is done, then retry.
358 while (dataOwner_->writerFlags_[0].value.test(std::memory_order_acquire) != 0) {
359 // busy wait, we expect very short duration of wait here.
360 CPU_PAUSE();
361 }
362 // The concurrent write has finished, we can give it another try.
363 if (readLock_SingleBuffer()) {
364 // Attempt to switch to double-buffered mode.
365 // We do this here to avoid waiting like above in the future.
366 if (dataOwner_->writerFlags_[1].value.test_and_set(std::memory_order_acquire) == 0) {
367 setOwnerOfWriteLock(1);
368 if (dataSlots_[1] == nullptr) {
369 dataOwner_->createSecondSlot();
370 }
371 writeUnlock(1, 0, 0);
372 }
373 // Got the read lock, return the data.
374 return {dataSlots_[0] + offset, 0};
375 } else {
376 // Failed again, retry.
377 CPU_PAUSE();
378 return mapRange(BUFFER_GPU_READ, offset, size);
379 }
380 }
381}

383MappedClientData ClientBuffer::writeRange_SingleBuffer(uint32_t offset, uint32_t size) const {
384 if (writeLock_SingleBuffer()) {
385 return {dataSlots_[0] + offset, -1, dataSlots_[0] + offset, 0};
386 }
387 else if (writerFlags_[0].value.test(std::memory_order_acquire) != 0) {
388 // Another write operation is in progress on the first slot.
389 // Note: If the first slot is write-locked by this thread, then we would deadlock
390 // waiting here.
391 if (isOwnerOfWriteLock(0)) {
392 // For now, we return the data pointer without doing any locking, hoping that the original
393 // write lock will be lifted after the write operation.
394 return {
395 dataSlots_[0] + offset, -1,
396 dataSlots_[0] + offset, -1};
397 }
398 // If the first slot is write-locked by another thread,
399 // then we need to wait for it to be unlocked.
400 do {
401 // busy wait, we expect very short duration of wait here.
402 CPU_PAUSE();
403 } while (writerFlags_[0].value.test(std::memory_order_acquire) != 0);
404 // the concurrent write has finished, we can give it another try.
405 // note that in the meantime maybe we switched to double-buffered mode.
406 return mapRange(BUFFER_GPU_WRITE, offset, size);
407 } else {
408 // A read operation is in progress on the first slot.
409 // Try also to get a read lock on the first slot such that we can
410 // safely switch to double-buffered mode.
411 // Note: in case all reader are in this thread, we could skip switching to double-buffered mode,
412 // but currently the thread ids of readers are not tracked.
413 const int r_index = readLock();
414 if (r_index > 0) {
415 // seems someone else allocated the second slot already.
416 // release the read lock and do double-buffered write.
417 readUnlock(r_index);
418 return writeRange_DoubleBuffer(offset, size);
419 } else if (dataOwner_->writerFlags_[1].value.test_and_set(std::memory_order_acquire) == 0) {
420 // got a write lock on the second slot.
421 setOwnerOfWriteLock(1);
422 if (dataSlots_[1] == nullptr) {
423 // Still single-buffered, we can finally create the second slot.
424 // Note: In case of frame-locked mode, readers will continue using slot 0 for this frame.
425 // else switch to slot 1 after this write has finished.
426 dataOwner_->createSecondSlot();
427 return {dataSlots_[0] + offset, 0, dataSlots_[1] + offset, 1};
428 } else {
429 // someone else allocated the second slot already, retry.
430 writeUnlock(1, 0, 0);
431 readUnlock(r_index);
432 return writeRange_DoubleBuffer(offset, size);
433 }
434 } else {
435 // someone else holds the write lock on the second slot.
436 readUnlock(r_index);
437 if (dataSlots_[1] == nullptr) {
438 // still single-buffered. retry...
439 CPU_PAUSE();
440 return mapRange(BUFFER_GPU_WRITE, offset, size);
441 } else {
442 // there is a second slot, switch to double-buffered mode.
443 return writeRange_DoubleBuffer(offset, size);
444 }
445 }
446 }

507void ClientBuffer::updateBufferSize() {
508 if (!bufferSegments_.empty()) {
509 uint32_t offset = 0u;
510 for (auto &segment: bufferSegments_) {
511 // compute the offset for the segment, aligned to its base alignment.
512 offset = (offset + segment->baseAlignment_ - 1) & ~(segment->baseAlignment_ - 1);
513 // set the data size for the segment.
514 segment->lastOffset_ = segment->dataOffset_;
515 segment->dataOffset_ = dataOffset_ + offset;
516
517 segment->updateBufferSize();
518 offset += segment->dataSize_;
519 }
520 dataSize_ = offset + bufferSegments_.back()->dataSize_;
521 // Round total size up to next multiple of 16 (vec4 alignment for std140)
522 if (memoryLayout_ == BUFFER_MEMORY_STD140) {
523 static constexpr size_t std140Alignment = (16 - 1);
524 dataSize_ = (dataSize_ + std140Alignment) & ~std140Alignment;
525 }
526 }

529bool ClientBuffer::ownerResize() {
530 // keep a reference to the old data slots, for copying data over.
531 byte *oldData0 = dataSlots_[0];
532 byte *oldData1 = dataSlots_[1];
533 // compute the new data size, and update the offsets of the segments.
534 // note: that some segments may need padding to align to their base alignment.
535 // so the data size of a composed client buffer might be larger than the sum of the segment sizes.
536 updateBufferSize();
537 if (dataSize_==0u) {
538 // no data to allocate.
539 return false;
540 }
541
542 // allocate new data slots.
543 if constexpr(USE_CLIENT_BUFFER_POOL) {
544 auto oldDataRefs0 = dataRefs_[0];
545 auto oldDataRefs1 = dataRefs_[1];
546 auto memoryPool = getMemoryPool();
547 auto *allocator = getMemoryAllocator(dataSize_);
548 dataRefs_[0] = memoryPool->alloc(allocator, dataSize_);
549 dataSlots_[0] = dataRefs_[0].allocatorNode->allocatorRef;
550
551 if (clientBufferMode_ == SingleBuffer) {
552 dataSlots_[1] = nullptr;
553 } else if (clientBufferMode_ == DoubleBuffer) {
554 allocator = getMemoryAllocator(dataSize_);
555 dataRefs_[1] = memoryPool->alloc(allocator, dataSize_);
556 dataSlots_[1] = dataRefs_[1].allocatorNode->allocatorRef;
557 } else if (clientBufferMode_ == AdaptiveBuffer) {
558 if (dataSlots_[1]) {
559 allocator = getMemoryAllocator(dataSize_);
560 dataRefs_[1] = memoryPool->alloc(allocator, dataSize_);
561 dataSlots_[1] = dataRefs_[1].allocatorNode->allocatorRef;
562 }
563 }
564
565 if (dataSlots_[1]) {
566 resize_DoubleBuffer(
567 this,
568 oldData0,
569 oldData1,
570 dataSlots_[0],
571 dataSlots_[1]);
572 } else {
573 resize_SingleBuffer(
574 this,
575 oldData0,
576 dataSlots_[0]);
577 }
578
579 // delete the old data slots.
580 if (oldData0) memoryPool->free(oldDataRefs0);
581 if (oldData1) memoryPool->free(oldDataRefs1);
582 } else { // not using memory pool
583 dataSlots_[0] = new byte[dataSize_];
584 if (clientBufferMode_ == SingleBuffer) {
585 dataSlots_[1] = nullptr;
586 } else if (clientBufferMode_ == DoubleBuffer) {
587 dataSlots_[1] = new byte[dataSize_];
588 } else if (clientBufferMode_ == AdaptiveBuffer) {
589 if (dataSlots_[1]) {
590 dataSlots_[1] = new byte[dataSize_];
591 }
592 }
593
594 if (dataSlots_[1]) {
595 resize_DoubleBuffer(
596 this,
597 oldData0,
598 oldData1,
599 dataSlots_[0],
600 dataSlots_[1]);
601 } else {
602 resize_SingleBuffer(
603 this,
604 oldData0,
605 dataSlots_[0]);
606 }
607
608 delete[] oldData0;
609 delete[] oldData1;
610 }
611
612 return true;
613}

615void ClientBuffer::resize_SingleBuffer(ClientBuffer *owner, const byte *oldDataPtr, byte *newDataPtr) {
616 const byte *localOldDataPtr = nullptr;
617 if (owner != this && dataOwner_ == this) {
618 // write-lock the slots, avoiding any concurrent reads/writes.
619 writeLockAll();
620 localOldDataPtr = dataSlots_[0];
621 oldDataPtr = localOldDataPtr;
622 markWrittenTo(0, 0, dataSize_);
623 }
624
625 if (dataSize_ == allocatedSize_) {
626 // no resize, just copy over the data from old to new slot.
627 if (oldDataPtr) {
628 std::memcpy(newDataPtr, oldDataPtr, dataSize_);
629 }
630 setDataPointer(owner, newDataPtr, 0);
631 markWrittenTo(0, 0, dataSize_);
632 } else {
633 if (bufferSegments_.empty()) {
634 dataSlots_[0] = newDataPtr;
635 } else {
636 for (auto &segment: bufferSegments_) {
637 if (oldDataPtr) {
638 segment->resize_SingleBuffer(
639 owner, oldDataPtr + segment->lastOffset_,
640 newDataPtr + segment->dataOffset_);
641 } else if (segment->hasClientData()) {
642 segment->resize_SingleBuffer(
643 owner, segment->dataSlots_[0],
644 newDataPtr + segment->dataOffset_);
645 } else {
646 segment->resize_SingleBuffer(
647 owner, nullptr,
648 newDataPtr + segment->dataOffset_);
649 }
650 }
651 }
652 allocatedSize_ = dataSize_;
653 dataOwner_ = owner;
654 }
655
656 if (localOldDataPtr) {
657 // we had a local copy of the data, let's clean up the local locks,
658 // and delete the local data pointer.
659 readerCounts_[0].value.store(0, std::memory_order_release);
660 readerCounts_[1].value.store(0, std::memory_order_release);
661 writerFlags_[0].value.clear(std::memory_order_release);
662 writerFlags_[1].value.clear(std::memory_order_release);
663 if constexpr(USE_CLIENT_BUFFER_POOL) {
664 getMemoryPool()->free(dataRefs_[0]);
665 } else {

671void ClientBuffer::resize_DoubleBuffer(
672 ClientBuffer *owner,
673 const byte *oldDataPtr0,
674 const byte *oldDataPtr1,
675 byte *newDataPtr0,
676 byte *newDataPtr1) {
677 const byte *localOldDataPtr0 = nullptr;
678 const byte *localOldDataPtr1 = nullptr;
679 if (owner != this && dataOwner_ == this && dataOwner_ != owner) {
680 // write-lock the slots, avoiding any concurrent reads/writes.
681 writeLockAll();
682 localOldDataPtr0 = dataSlots_[0];
683 localOldDataPtr1 = dataSlots_[1];
684 // Make sure we only use the last data slot, i.e. newest data.
685 oldDataPtr0 = dataSlots_[lastDataSlot_.load(std::memory_order_acquire)];
686 oldDataPtr1 = oldDataPtr0;
687 }
688 if (!oldDataPtr1) {
689 oldDataPtr1 = oldDataPtr0;
690 }
691
692 if (dataSize_ == allocatedSize_) {
693 // no resize, just copy over the data from old to new slot.
694 if (oldDataPtr0) {
695 std::memcpy(newDataPtr0, oldDataPtr0, dataSize_);
696 std::memcpy(newDataPtr1, oldDataPtr1, dataSize_);
697 }
698 setDataPointer(owner, newDataPtr0, 0);
699 setDataPointer(owner, newDataPtr1, 1);
700 markWrittenTo(currentWriteSlot(), 0, dataSize_);
701 } else {
702 if (bufferSegments_.empty()) {
703 dataSlots_[0] = newDataPtr0;
704 dataSlots_[1] = newDataPtr1;
705 markWrittenTo(currentWriteSlot(), 0, dataSize_);
706 } else {
707 for (auto &segment: bufferSegments_) {
708 if (oldDataPtr0) {
709 segment->resize_DoubleBuffer(
710 owner,
711 oldDataPtr0 + segment->lastOffset_,
712 (oldDataPtr1 ? oldDataPtr1 : oldDataPtr0) + segment->lastOffset_,
713 newDataPtr0 + segment->dataOffset_,
714 newDataPtr1 + segment->dataOffset_);
715 } else if (segment->hasClientData()) {
716 segment->resize_DoubleBuffer(
717 owner,
718 segment->dataSlots_[0],
719 segment->hasTwoSlots() ? segment->dataSlots_[1] : segment->dataSlots_[0],
720 newDataPtr0 + segment->dataOffset_,
721 newDataPtr1 + segment->dataOffset_);
722 } else {
723 segment->resize_DoubleBuffer(
724 owner,
725 nullptr,
726 nullptr,
727 newDataPtr0 + segment->dataOffset_,
728 newDataPtr1 + segment->dataOffset_);
729 }
730 }
731 }
732 allocatedSize_ = dataSize_;
733 dataOwner_ = owner;
734 }
735
736 if (localOldDataPtr0) {
737 // we had a local copy of the data, let's clean up the local locks,
738 // and delete the local data pointer.
739 readerCounts_[0].value.store(0, std::memory_order_release);
740 readerCounts_[1].value.store(0, std::memory_order_release);
741 writerFlags_[0].value.clear(std::memory_order_release);
742 writerFlags_[1].value.clear(std::memory_order_release);
743 if constexpr(USE_CLIENT_BUFFER_POOL) {
744 getMemoryPool()->free(dataRefs_[0]);
745 if (localOldDataPtr1 != nullptr && localOldDataPtr1 != localOldDataPtr0) {
746 getMemoryPool()->free(dataRefs_[1]);
747 }
748 } else {
749 delete[] localOldDataPtr0;
750 if (localOldDataPtr1 != nullptr && localOldDataPtr1 != localOldDataPtr0) {
751 delete[] localOldDataPtr1;
752 }
753 }
754 }

769void ClientBuffer::writeLockAll() const {
770 auto *owner0 = dataOwner_;
771
772 // Indicate intent to acquire all write locks.
773 // This is used to prevent new readers from acquiring read locks meanwhile.
774 // However, this has one exception: if the thread that wants to acquire read
775 // lock is also the owner of the write lock on the slot, then it is allowed
776 // to acquire the read lock even with writeAllPending=true to avoid deadlocks.
777 owner0->writeAllPending_.store(true, std::memory_order_release);
778
779 while (true) {
780 auto *currentOwner = dataOwner_;
781
782 // If owner changed, move intent declaration to new owner
783 if (currentOwner != owner0) {
784 owner0->writeAllPending_.store(false, std::memory_order_release);
785 owner0 = currentOwner;
786 owner0->writeAllPending_.store(true, std::memory_order_release);
787 }
788
789 const int readSlot = currentOwner->lastDataSlot_.load(std::memory_order_acquire);
790 const int writeSlot = 1 - readSlot;
791 // pull atomics
792 auto &writeFlag = currentOwner->writerFlags_[writeSlot].value;
793 auto &readFlag = currentOwner->writerFlags_[readSlot].value;
794 auto &readerCount = currentOwner->readerCounts_[readSlot].value;
795
796 // First try to acquire write lock on current write slot.
797 // This will prevent any *new* attempts to write to this slot.
798 if (writeFlag.test_and_set(std::memory_order_acquire)) {
799 // Failed, meaning there is another active writer on this slot.
800 CPU_PAUSE();
801 continue; // try again
802 }
803
804 // Check if there are any active readers on the read slot.
805 if (readerCount.load(std::memory_order_acquire) != 0) {
806 // With active readers we must lift the lock again as it could be that the thread
807 // holding the read lock will also attempt to acquire the write lock.
808 writeFlag.clear(std::memory_order_release);
809 CPU_PAUSE();
810 continue; // try again
811 }
812
813 // Also acquire write lock on the read slot.
814 // This will prevent any *new* attempts to read or write.
815 if (readFlag.test_and_set(std::memory_order_acquire)) {
816 // failed to acquire write lock on read slot, release write lock on write slot.
817 writeFlag.clear(std::memory_order_release);
818 CPU_PAUSE();
819 continue; // try again
820 }
821
822 // To be safe, make sure that no readers sneaked in meanwhile on the read slot.
823 if (readerCount.load(std::memory_order_acquire) != 0) {
824 // Readers sneaked in, release both write locks as it is not safe to read during swapping.
825 writeFlag.clear(std::memory_order_release);
826 readFlag.clear(std::memory_order_release);
827 CPU_PAUSE();
828 continue; // try again
829 }
830
831 break; // got both write locks
832 }
833
834 // Writer now owns all slots
835 owner0->writeAllPending_.store(false, std::memory_order_release);
836}

843int ClientBuffer::readLock() const {
844 auto thisThreadId = std::this_thread::get_id();
845 while (true) {
846 // Note: ownership may change while waiting for the lock.
847 auto *currentOwner = dataOwner_;
848
849 // Get the current slot index for reading.
850 // note that the slot index can be flipped through data swapping, so we need to keep loading
851 // it within this loop in case we cannot obtain the lock on first try, e.g.
852 // because there are active writers on the slot which in turn will flip the slot index once done.
853 // Even worse, the flip can happen *during* this call, so we need to verify again after acquiring
854 // the read lock that we in fact acquired the lock on the correct slot.
855 const int dataSlot = currentOwner->lastDataSlot_.load(std::memory_order_acquire);
856 // pull atomics
857 auto &readerCount = currentOwner->readerCounts_[dataSlot].value;
858
859 // Check if a writer is pending, if so give them priority.
860 // However, if this thread is the owner of the write lock on this slot, then
861 // we must ignore the intent as otherwise we definitely would deadlock in this situation!
862 if (currentOwner->writeAllPending_.load(std::memory_order_acquire) &&
863 currentOwner->writerThreads_[dataSlot] != thisThreadId) {
864 // there is a pending writer, we need to wait for them to finish.
865 waitOnAtomic<bool,false>(currentOwner->writeAllPending_);
866 continue; // try again
867 }
868
869 // First step: increment the reader count for this slot.
870 readerCount.fetch_add(1, std::memory_order_relaxed);
871
872 // However, maybe there is an active writer on this slot already, we need to check that.
873 if (dataOwner_->writerFlags_[dataSlot].value.test(std::memory_order_acquire) != 0 &&
874 dataOwner_->writerThreads_[dataSlot] != thisThreadId) {
875 // Seems there is an active writer on this slot, we need to wait for them to finish.
876 // first decrement the reader count, so that we do not block writer in the meanwhile.
877 readerCount.fetch_sub(1, std::memory_order_relaxed);
878 // then wait until there are no active writers on `dataSlot`.
879 waitOnFlag<false>(dataOwner_->writerFlags_[dataSlot].value);
880 continue;
881 }
882
883 if (dataOwner_ != currentOwner ||
884 currentOwner->lastDataSlot_.load(std::memory_order_acquire) != dataSlot) {
885 // data owner has changed, or the read/write slot swapped meanwhile.
886 // better to retry in this case.
887 readerCount.fetch_sub(1, std::memory_order_relaxed);
888 CPU_PAUSE();
889 continue;
890 }
891
892 // no writer has locked the slot, other ones are prevented from doing so,
893 // hence we can safely read from this slot.
894 return dataSlot;
895 }
896}

898int ClientBuffer::writeLock_DoubleBuffer() const {
899 auto thisThreadId = std::this_thread::get_id();
900 while (true) {
901 // Note: ownership may change while waiting for the lock.
902 auto *currentOwner = dataOwner_;
903
904 // get the current slot index for writing.
905 // note that every writer will flip the slot index, so we need to keep loading
906 // it within this loop in case we cannot obtain the lock on first try.
907 const int currentReadSlot = currentOwner->lastDataSlot_.load(std::memory_order_acquire);
908 const int currentWriteSlot = 1 - currentReadSlot;
909 // pull atomics
910 auto &readerCount = currentOwner->readerCounts_[currentWriteSlot].value;
911 auto &writeFlag = currentOwner->writerFlags_[currentWriteSlot].value;
912
913 // check if there are any active readers on the write slot.
914 if (readerCount.load(std::memory_order_acquire) != 0) {
915 // seems there are some remaining readers on the write slot, we need to wait for them to finish.
916 waitOnAtomic<uint32_t,0u>(readerCount);
917 continue; // try again
918 }
919
920 if (writeFlag.test_and_set(std::memory_order_acquire) &&
921 currentOwner->writerThreads_[currentWriteSlot] != thisThreadId) {
922 // seems someone else is writing to this slot, we need to wait for them to finish.
923 waitOnFlag<false>(writeFlag);
924 continue; // try again
925 }
926
927 if (readerCount.load(std::memory_order_acquire) != 0 ||
928 dataOwner_ != currentOwner ||
929 currentOwner->lastDataSlot_.load(std::memory_order_acquire) != currentReadSlot) {
930 // a reader sneaked in while we were waiting for the write lock,
931 // data owner has changed, or read/write slot swapped meanwhile.
932 writeFlag.clear(std::memory_order_relaxed);
933 CPU_PAUSE();
934 continue;
935 }
936
937 // we got the exclusive write lock for this slot, so we can safely write to it.
938 setOwnerOfWriteLock(currentWriteSlot);
939 return currentWriteSlot;
940 }

943bool ClientBuffer::readLock_SingleBuffer() const {
944 // We are here in single buffer mode, and only quickly try to get a lock in the one
945 // slot (with index 0), or else return false.
946 // and the only thing preventing us from doing so would be a writer that is currently writing to the slot
947 // which would be indicated by the writerFlags_[0] being set.
948 auto *owner = dataOwner_;
949
950 if (owner->writeAllPending_.load(std::memory_order_acquire) &&
951 owner->writerThreads_[0] != std::this_thread::get_id()) return false;
952
953 owner->readerCounts_[0].value.fetch_add(1, std::memory_order_relaxed);
954 if (owner->writerFlags_[0].value.test(std::memory_order_acquire) != 0) {
955 owner->readerCounts_[0].value.fetch_sub(1, std::memory_order_relaxed);
956 return false; // Busy writing
957 } else {
958 return true;
959 }

971 return false; // Busy reading
972 }
973 setOwnerOfWriteLock(0);

981void ClientBuffer::writeUnlock(int32_t dataSlot, uint32_t writeOffset, uint32_t writeSize) const {
982 if (writeSize > 0u) {
983 // increment the data stamp, and remember the last slot that was written to.
984 // consecutive reads will be done from this slot, next write will be done to the other slot.
985 // If the write operation did not change the data, the stamp is not incremented,
986 // and the last slot is not updated.
987 nextStamp(dataSlot);
988 if (!bufferSegments_.empty()) {
989 // Advance the stamp for all overlapping segments.
990 nextSegmentStamp(dataSlot, writeOffset, writeSize);
991 }
992
993 if (isFrameLocked_) {
994 // If frame-locked, the swap to the other slot is done centrally, not on write unlock.
995 // But we still need to remember which data range was written to this frame.
996 // This is done to avoid unnecessary copies.
997 markWrittenTo(dataSlot, writeOffset, writeSize);
998 } else {
999 // swap to the other slot.
1000 dataOwner_->lastDataSlot_.store(dataSlot, std::memory_order_release);
1001 }
1002 }
1003 // clear the exclusive write lock for this slot, allowing any waiting writer to proceed.
1004 // NOTE: reader will only proceed once all writing is done.
1005 dataOwner_->writerFlags_[dataSlot].value.clear(std::memory_order_relaxed);
1006}
