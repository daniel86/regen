324void ProcTree::computeTan(TreeMesh &treeMesh, const ProcMesh &procMesh, int vertexOffset, Vec4f *tanData) {
325 for (int i = 0; i < procMesh.mFaceCount; i++) {
326 auto &v0 = *((Vec3f *) &procMesh.mVert[procMesh.mFace[i].x]);
327 auto &v1 = *((Vec3f *) &procMesh.mVert[procMesh.mFace[i].y]);
328 auto &v2 = *((Vec3f *) &procMesh.mVert[procMesh.mFace[i].z]);
329 auto &uv0 = *((Vec2f *) &procMesh.mUV[procMesh.mFace[i].x]);
330 auto &uv1 = *((Vec2f *) &procMesh.mUV[procMesh.mFace[i].y]);
331 auto &uv2 = *((Vec2f *) &procMesh.mUV[procMesh.mFace[i].z]);
332 auto e1 = v1 - v0;
333 auto e2 = v2 - v0;
334 auto deltaUV1 = uv1 - uv0;
335 auto deltaUV2 = uv2 - uv0;
336 float f = 1.0f / (deltaUV1.x * deltaUV2.y - deltaUV2.x * deltaUV1.y);
337 Vec3f tangent3 = (e1 * deltaUV2.y - e2 * deltaUV1.y) * f;
338 Vec4f tangent(tangent3.x, tangent3.y, tangent3.z, 1.0f);
339 tanData[vertexOffset + procMesh.mFace[i].x] = tangent;
340 tanData[vertexOffset + procMesh.mFace[i].y] = tangent;
341 tanData[vertexOffset + procMesh.mFace[i].z] = tangent;
342 }

345static int addQuadPoint(Vec3f *quadPos, const Vec3f &pos, const Vec2f &uv) {
346 int quadIndex;
347 if (uv.y < 0.5 && uv.x < 0.5) {
348 quadIndex = 0;
349 } else if (uv.y < 0.5 && uv.x > 0.5) {
350 quadIndex = 2;
351 } else if (uv.x < 0.5) {
352 quadIndex = 1;
353 } else {
354 quadIndex = 3;
355 }
356 quadPos[quadIndex] = pos;
357 return quadIndex;
358}

360void ProcTree::computeBasePos(TreeMesh &treeMesh, const ProcMesh &procMesh, int vertexOffset, Vec3f *basePosData) {
361 for (int i = 0; i < procMesh.mFaceCount; i++) {
362 auto &v0 = *((Vec3f *) &procMesh.mVert[procMesh.mFace[i].x]);
363 auto &v1 = *((Vec3f *) &procMesh.mVert[procMesh.mFace[i].y]);
364 auto &v2 = *((Vec3f *) &procMesh.mVert[procMesh.mFace[i].z]);
365 auto &uv0 = *((Vec2f *) &procMesh.mUV[procMesh.mFace[i].x]);
366 auto &uv1 = *((Vec2f *) &procMesh.mUV[procMesh.mFace[i].y]);
367 auto &uv2 = *((Vec2f *) &procMesh.mUV[procMesh.mFace[i].z]);
368
369 Vec3f quadPos[4] = {Vec3f::zero(), Vec3f::zero(),
370 Vec3f::zero(), Vec3f::zero()};
371 int vIndex0 = addQuadPoint(quadPos, v0, uv0);
372 int vIndex1 = addQuadPoint(quadPos, v1, uv1);
373 int vIndex2 = addQuadPoint(quadPos, v2, uv2);
374 int missingIndex = 6 - vIndex0 - vIndex1 - vIndex2;
375 if (missingIndex == 0) {
376 quadPos[0] = quadPos[1] + quadPos[2] - quadPos[3];
377 } else if (missingIndex == 1) {
378 quadPos[1] = quadPos[0] + quadPos[3] - quadPos[2];
379 } else if (missingIndex == 2) {
380 quadPos[2] = quadPos[3] + quadPos[0] - quadPos[1];
381 } else {
382 quadPos[3] = quadPos[2] + quadPos[1] - quadPos[0];
383 }
384 // first base point
385 if (vIndex0 == 0 || vIndex0 == 2) {
386 basePosData[vertexOffset + procMesh.mFace[i].x] = v0;
387 } else if (vIndex0 == 1) { // quadPos[0] is base point
388 basePosData[vertexOffset + procMesh.mFace[i].x] = quadPos[0];
389 } else { // quadPos[2] is base point
390 basePosData[vertexOffset + procMesh.mFace[i].x] = quadPos[2];
391 }
392 // second base point
393 if (vIndex1 == 0 || vIndex1 == 2) {
394 basePosData[vertexOffset + procMesh.mFace[i].y] = v1;
395 } else if (vIndex1 == 1) { // quadPos[0] is base point
396 basePosData[vertexOffset + procMesh.mFace[i].y] = quadPos[0];
397 } else { // quadPos[2] is base point
398 basePosData[vertexOffset + procMesh.mFace[i].y] = quadPos[2];
399 }
400 // third base point
401 if (vIndex2 == 0 || vIndex2 == 2) {
402 basePosData[vertexOffset + procMesh.mFace[i].z] = v2;
403 } else if (vIndex2 == 1) { // quadPos[0] is base point
404 basePosData[vertexOffset + procMesh.mFace[i].z] = quadPos[0];
405 } else { // quadPos[2] is base point
406 basePosData[vertexOffset + procMesh.mFace[i].z] = quadPos[2];
407 }
408 }

411void ProcTree::updateAttributes_(
412 TreeMesh &treeMesh,
413 const ProcMesh &lod0,
414 const std::vector<Mesh::MeshLOD> &lodLevels,
415 uint32_t maxVIndex) {
416 auto indexRef = treeMesh.mesh->setIndices(treeMesh.indices, maxVIndex);
417 treeMesh.mesh->setInput(treeMesh.pos);
418 if (treeMesh.basePos.get()) {
419 treeMesh.mesh->setInput(treeMesh.basePos);
420 }
421 treeMesh.mesh->setInput(treeMesh.nor);
422 treeMesh.mesh->setInput(treeMesh.tan);
423 treeMesh.mesh->setInput(treeMesh.texco);
424 treeMesh.mesh->updateVertexData();
425
426 Vec3f min = *((Vec3f *) &lod0.mVert[0]);
427 Vec3f max = *((Vec3f *) &lod0.mVert[0]);
428 // find bounding box of LOD=0
429 for (int i = 1; i < lod0.mVertCount; i++) {
430 auto &v = *((Vec3f *) &lod0.mVert[i]);
431 min.x = std::min(min.x, v.x);
432 min.y = std::min(min.y, v.y);
433 min.z = std::min(min.z, v.z);
434 max.x = std::max(max.x, v.x);
435 max.y = std::max(max.y, v.y);
436 max.z = std::max(max.z, v.z);
437 }
438 treeMesh.mesh->set_bounds(min, max);
439
440 if (!lodLevels.empty()) {
441 for (auto &x: lodLevels) {
442 // add the index buffer offset (in number of bytes)
443 x.d->indexOffset = indexRef->address() + x.d->indexOffset * treeMesh.indices->dataTypeBytes();
444 }
445 treeMesh.mesh->setMeshLODs(lodLevels);
446 treeMesh.mesh->activateLOD(0);
447 }

450void ProcTree::updateTrunkAttributes(TreeMesh &treeMesh, const std::vector<ProcMesh> &procLODs) const {
451 std::vector<Mesh::MeshLOD> lodLevels;
452 auto &lod0 = procLODs[0];
453 int numVertices = 0, numIndices = 0;
454 for (auto &lod: procLODs) {
455 numVertices += lod.mVertCount;
456 numIndices += lod.mFaceCount * 3;
457 }
458
459 // allocate memory then copy each LOD into the vertex data array
460 treeMesh.indices = createIndexInput(numIndices, numVertices);
461 treeMesh.pos->setVertexData(numVertices);
462 treeMesh.nor->setVertexData(numVertices);
463 treeMesh.texco->setVertexData(numVertices);
464 treeMesh.tan->setVertexData(numVertices);
465 // map client data for writing
466 auto indices = treeMesh.indices->mapClientDataRaw(BUFFER_GPU_WRITE);
467 auto v_pos = treeMesh.pos->mapClientData<float>(BUFFER_GPU_WRITE);
468 auto v_nor = treeMesh.nor->mapClientData<float>(BUFFER_GPU_WRITE);
469 auto v_tan = treeMesh.tan->mapClientData<float>(BUFFER_GPU_WRITE);
470 auto v_texco = treeMesh.texco->mapClientData<float>(BUFFER_GPU_WRITE);
471 auto *ptr_indices = indices.w;
472 auto *ptr_pos = v_pos.w.data();
473 auto *ptr_nor = v_nor.w.data();
474 auto *ptr_texco = v_texco.w.data();
475 auto indexType = treeMesh.indices->baseType();
476 // copy data from Proctree to Mesh
477 // also create LOD descriptions on the way.
478 unsigned int vertexOffset = 0u, indexOffset = 0u;
479 for (auto &lod: procLODs) {
480 // create LOD description
481 auto &lodLevel = lodLevels.emplace_back();
482 lodLevel.d->numVertices = lod.mVertCount;
483 lodLevel.d->numIndices = lod.mFaceCount * 3;
484 lodLevel.d->vertexOffset = vertexOffset;
485 lodLevel.d->indexOffset = indexOffset;
486 // copy data
487 for (int i = 0; i < lod.mFaceCount; i++) {
488 setIndexValue(ptr_indices, indexType, i * 3 + 0, lod.mFace[i].x + vertexOffset);
489 setIndexValue(ptr_indices, indexType, i * 3 + 1, lod.mFace[i].y + vertexOffset);
490 setIndexValue(ptr_indices, indexType, i * 3 + 2, lod.mFace[i].z + vertexOffset);
491 }
492 memcpy(ptr_pos, &lod.mVert[0].x, lod.mVertCount * 3 * sizeof(float));
493 memcpy(ptr_nor, &lod.mNormal[0].x, lod.mVertCount * 3 * sizeof(float));
494 memcpy(ptr_texco, &lod.mUV[0].u, lod.mVertCount * 2 * sizeof(float));
495 // compute tangents
496 computeTan(treeMesh, lod, vertexOffset, (Vec4f *) v_tan.w.data());
497 // increase offsets
498 vertexOffset += lodLevel.d->numVertices;
499 indexOffset += lodLevel.d->numIndices;
500 ptr_indices += lodLevel.d->numIndices * treeMesh.indices->dataTypeBytes();
501 ptr_pos += lod.mVertCount * 3;
502 ptr_nor += lod.mVertCount * 3;
503 ptr_texco += lod.mVertCount * 2;
504 }
505
506 indices.unmap();
507 v_pos.unmap();
508 v_nor.unmap();
509 v_tan.unmap();
510 v_texco.unmap();
511
512 updateAttributes_(treeMesh, lod0, lodLevels, numVertices);
513}

549void ProcTree::updateTwigAttributes() {
550 const ProcMesh lod0 = twigProcMesh(handle);
551 std::vector<Mesh::MeshLOD> lodLevels;
552
553 if (useSilhouetteMesh_ && twigSilhouette_.get()) {
554 const uint32_t nq_in = lod0.mFaceCount / 2;
555 uint32_t numVertices = 0u; // Total number of vertices over all LODs
556 uint32_t numIndices = 0u; // Total number of indices over all LODs
557 for (uint32_t lodIdx=0; lodIdx < 2; lodIdx++) {
558 auto &uvRects = twigSilhouette_->silhouetteUVRects()[lodIdx];
559 numVertices += nq_in * uvRects.size() * 4; // 4 vertices per quad
560 numIndices += nq_in * uvRects.size() * 6; // 6 indices per quad
561 }
562
563 // Allocate client memory
564 twig.indices = createIndexInput(numIndices, numVertices);
565 twig.pos->setVertexData(numVertices);
566 twig.texco->setVertexData(numVertices);
567 if (twig.basePos.get()) {
568 twig.basePos->setVertexData(numVertices);
569 }
570
571 // map client data for writing
572 auto twig_p = (Vec3f*) twig.pos->clientBuffer()->clientData(0);
573 auto twig_uv = (Vec2f*) twig.texco->clientBuffer()->clientData(0);
574 auto twig_bp = (twig.basePos.get() ?
575 (Vec3f*) twig.basePos->clientBuffer()->clientData(0) : nullptr);
576 auto twig_i = (byte*)twig.indices->clientBuffer()->clientData(0);
577 const auto indexType = twig.indices->baseType();
578
579 // Running offsets for vertices and indices
580 uint32_t vOffset = 0u;
581 uint32_t iOffset = 0u;
582
583 for (uint32_t lodIdx=0; lodIdx < 2; lodIdx++) {
584 auto &uvRects = twigSilhouette_->silhouetteUVRects()[lodIdx];
585
586 for (uint32_t quadIdx_in=0u; quadIdx_in < nq_in; quadIdx_in++) {
587 // For each input quad, we create uvRects.size() output quads.
588 // For this we need to compute tangents along u/v directions.
589
590 auto &face_in = lod0.mFace[quadIdx_in * 2];
591 const Vec3f *topLeft, *topRight, *botRight;
592 if (quadIdx_in % 2 == 0) {
593 // input ordering: (0,0), (1,0), (1,1)
594 topLeft = (Vec3f*)&lod0.mVert[face_in.x];
595 topRight = (Vec3f*)&lod0.mVert[face_in.y];
596 botRight = (Vec3f*)&lod0.mVert[face_in.z];
597 } else {
598 // input ordering: (1,1), (1,0), (0,0)
599 botRight = (Vec3f*)&lod0.mVert[face_in.x];
600 topRight = (Vec3f*)&lod0.mVert[face_in.y];
601 topLeft = (Vec3f*)&lod0.mVert[face_in.z];
602 }
603 Vec3f origin = *topLeft;
604
605 // base position for the quad
606 Vec3f basePos = ((*topLeft) + (*topRight)) * 0.5f + ((*botRight) - (*topRight));
607 // compute direction vectors along face plane
608 // length of the vectors should be width/height of the quad.
609 Vec3f u_axis = (*topRight) - (*topLeft);
610 Vec3f v_axis = (*botRight) - (*topRight);
611 if (quadIdx_in % 2 != 0) {
612 // for odd numbered quads, flip the u axis to maintain winding order
613 u_axis = -u_axis;
614 origin = *topRight;
615 }
616
617 for (uint32_t uvRectIdx = 0u; uvRectIdx < uvRects.size(); uvRectIdx++) {
618 // For each output quad, we need to compute the position and UVs.
619 auto &uvRect = uvRects[uvRectIdx];
620 // uvRect = (u0, v0, u1, v1) in normalized 0..1 space
621 float u0 = uvRect.x, v0 = uvRect.y, u1 = uvRect.z, v1 = uvRect.w;
622 // corners in UV space (BL, BR, TR, TL)
623 float us[4] = { u0, u1, u1, u0 };
624 float vs[4] = { v0, v0, v1, v1 };
625
626 for (int c = 0; c < 4; ++c) {
627 // world position = origin + u_axis * uq + v_axis * vq
628 twig_p[vOffset + c] = origin + (u_axis * us[c]) + (v_axis * vs[c]);
629 // UVs unchanged (still 0..1)
630 twig_uv[vOffset + c] = Vec2f(us[c], 1.0 - vs[c]);
631 // basePos per-vertex
632 if (twig_bp) twig_bp[vOffset + c] = basePos;
633 }
634
635 // Finally, set the indices for the quad.
636 setIndexValue(twig_i, indexType, iOffset + 0, vOffset + 0);
637 setIndexValue(twig_i, indexType, iOffset + 1, vOffset + 1);
638 setIndexValue(twig_i, indexType, iOffset + 2, vOffset + 2);
639 setIndexValue(twig_i, indexType, iOffset + 3, vOffset + 2);
640 setIndexValue(twig_i, indexType, iOffset + 4, vOffset + 3);
641 setIndexValue(twig_i, indexType, iOffset + 5, vOffset + 0);
642 // increment offsets
643 vOffset += 4u;
644 iOffset += 6u;
645 }
646 }
647 }
648
649 twig.nor->setUniformData(Vec3f::up());
650 twig.tan->setUniformData(Vec4f::create(Vec3f::right(), 1.0f));
651
652 // create 2 LOD levels, one for each silhouette map
653 lodLevels.resize(2);
654 const auto &rects0 = twigSilhouette_->silhouetteUVRects()[0];
655 const auto &rects1 = twigSilhouette_->silhouetteUVRects()[1];
656 // LOD 0
657 lodLevels[0].d->numVertices = nq_in * rects0.size() * 4;
658 lodLevels[0].d->numIndices = nq_in * rects0.size() * 6;
659 lodLevels[0].d->vertexOffset = 0;
660 lodLevels[0].d->indexOffset = 0;
661 // LOD 1
662 lodLevels[1].d->numVertices = nq_in * rects1.size() * 4;
663 lodLevels[1].d->numIndices = nq_in * rects1.size() * 6;
664 lodLevels[1].d->vertexOffset = lodLevels[0].d->numVertices;
665 lodLevels[1].d->indexOffset = lodLevels[0].d->numIndices;
666
667 updateAttributes_(twig, lod0, lodLevels, numVertices);
668 } else {
669 uint32_t nv = lod0.mVertCount;
670 uint32_t ni = lod0.mFaceCount * 3;
671 // use proc tree data directly
672#define PROC_DATA_PTR_(arg) reinterpret_cast<const unsigned char *>(&(arg))
673 auto indices = (int32_t*)PROC_DATA_PTR_(lod0.mFace[0].x);
674 twig.indices = createIndexInput(ni, nv);
675 auto m_indices = twig.indices->clientData();
676 for (uint32_t idx=0; idx < ni; idx++) {
677 setIndexValue(m_indices, twig.indices->baseType(), idx, indices[idx]);
678 }
679 twig.pos->setVertexData(nv, PROC_DATA_PTR_(lod0.mVert[0].x));
680 twig.texco->setVertexData(nv, PROC_DATA_PTR_(lod0.mUV[0].u));
681#undef PROC_DATA_PTR_
682
683 twig.nor->setUniformData(Vec3f::up());
684 twig.tan->setUniformData(Vec4f::create(Vec3f::right(), 1.0f));
685
686 if (twig.basePos.get()) {
687 twig.basePos->setVertexData(nv);
688 auto v_basePos = twig.basePos->mapClientData<float>(BUFFER_GPU_WRITE);
689 computeBasePos(twig, lod0, 0, (Vec3f *) v_basePos.w.data());
690 v_basePos.unmap();
691 }
692
693 updateAttributes_(twig, lod0, lodLevels, nv);
694 }
695}
