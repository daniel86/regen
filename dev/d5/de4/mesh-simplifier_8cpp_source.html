12 case GL_TRIANGLES:
13 break;
14 case GL_TRIANGLE_FAN:
15 case GL_TRIANGLE_STRIP:
16 case GL_QUADS:
17 case GL_QUAD_STRIP:
18 case GL_LINES:
19 case GL_LINE_LOOP:
20 case GL_LINE_STRIP:
21 default:
22 REGEN_ERROR("Mesh simplifier only supports triangle meshes.");
23 hasValidAttributes_ = false;
24 break;
25 }
26
27 // lookup mesh attributes
28 for (auto &in: mesh_->inputs()) {

30 if (!addInputAttribute(in)) {
31 hasValidAttributes_ = false;
32 }

36 quadrics_.resize(inputPos_->numVertices());
37 neighbors_.resize(inputPos_->numVertices());

41 hasValidAttributes_ = false;
42 }

44 (inputBoneIndices_.get() != nullptr ? 1 : 0);

58 "#verts=" << inputPos_->numVertices() << " " <<
59 "#attributes=" << (inputAttributes_.size() + 1) << " ");

93bool MeshSimplifier::addInputAttribute(const NamedShaderInput &namedAttribute) {
94 if (namedAttribute.name_.find(ATTRIBUTE_NAME_POS) == 0) {
95 if (!ensureBaseType(namedAttribute.in_, GL_FLOAT)) return false;
96 if (!ensureValueSize(namedAttribute.in_, 3)) return false;
97 if (!ensureNoArray(namedAttribute.in_)) return false;
98 inputPos_ = ref_ptr<ShaderInput3f>::dynamicCast(namedAttribute.in_);
99 return inputPos_.get() != nullptr;
100 }
101 else if (namedAttribute.name_.find(ATTRIBUTE_NAME_NOR) == 0) {
102 if (!ensureBaseType(namedAttribute.in_, GL_FLOAT)) return false;
103 if (!ensureValueSize(namedAttribute.in_, 3)) return false;
104 if (!ensureNoArray(namedAttribute.in_)) return false;
105 inputAttributes_.emplace_back(AttributeSemantic::NORMAL, namedAttribute.in_);
106 inputNor_ = ref_ptr<ShaderInput3f>::dynamicCast(namedAttribute.in_);
107 norIndex_ = inputAttributes_.size() - 1;
108 }
109 else if (namedAttribute.name_.find("texco") == 0 ||
110 namedAttribute.name_.find("uv") == 0) {
111 if (!ensureBaseType(namedAttribute.in_, GL_FLOAT)) return false;
112 if (!ensureNoArray(namedAttribute.in_)) return false;
113 inputAttributes_.emplace_back(AttributeSemantic::TEXCOORD, namedAttribute.in_);
114 }
115 else if (namedAttribute.name_.find("col") == 0) {
116 if (!ensureBaseType(namedAttribute.in_, GL_FLOAT)) return false;
117 if (!ensureNoArray(namedAttribute.in_)) return false;
118 inputAttributes_.emplace_back(AttributeSemantic::COLOR, namedAttribute.in_);
119 }
120 else if (namedAttribute.name_.find(ATTRIBUTE_NAME_TAN) == 0) {
121 if (!ensureBaseType(namedAttribute.in_, GL_FLOAT)) return false;
122 if (!ensureValueSize(namedAttribute.in_, 4)) return false;
123 if (!ensureNoArray(namedAttribute.in_)) return false;
124 inputAttributes_.emplace_back(AttributeSemantic::TANGENT, namedAttribute.in_);
125 }
126 else if (namedAttribute.name_.find("bitan") == 0) {
127 if (!ensureBaseType(namedAttribute.in_, GL_FLOAT)) return false;
128 if (!ensureValueSize(namedAttribute.in_, 4)) return false;
129 if (!ensureNoArray(namedAttribute.in_)) return false;
130 inputAttributes_.emplace_back(AttributeSemantic::BITANGENT, namedAttribute.in_);
131 }
132 else if (namedAttribute.name_.find("boneWeights") == 0) {
133 if (!ensureBaseType(namedAttribute.in_, GL_FLOAT)) return false;
134 inputAttributes_.emplace_back(AttributeSemantic::BONE_WEIGHTS, namedAttribute.in_);
135 inputBoneWeights_ = ref_ptr<ShaderInput1f>::dynamicCast(namedAttribute.in_);
136 }
137 else if (namedAttribute.name_.find("boneIndices") == 0) {
138 if (!ensureBaseType(namedAttribute.in_, GL_UNSIGNED_INT)) return false;
139 inputAttributes_.emplace_back(AttributeSemantic::BONE_INDICES, namedAttribute.in_);
140 inputBoneIndices_ = ref_ptr<ShaderInput1ui>::dynamicCast(namedAttribute.in_);
141 }
142 else {
143 REGEN_DEBUG("Unknown attribute semantics for attribute '" << namedAttribute.name_ << "'." <<
144 " Treating it as generic float attribute.");
145 if (!ensureBaseType(namedAttribute.in_, GL_FLOAT)) return false;
146 inputAttributes_.emplace_back(AttributeSemantic::UNKNOWN, namedAttribute.in_);
147 }
148 return true;

164void MeshSimplifier::computeQuadrics(const std::vector<Triangle> &faces, const Vec3f *posData, uint32_t numVertices) {
165 // this is done for each LOD level, so first clear the quadrics and neighbors
166 quadrics_.clear();
167 quadrics_.resize(numVertices);
168 neighbors_.clear();
169 neighbors_.resize(numVertices);
170 // count edges for boundary detection
171 std::map<std::pair<uint32_t, uint32_t>, int> inputEdges;
172 isBoundary_.clear();
173 isBoundary_.resize(numVertices, false);
174
175 for (auto &face: faces) {
176 neighbors_[face.v0].insert(face.v1);
177 neighbors_[face.v0].insert(face.v2);
178 neighbors_[face.v1].insert(face.v0);
179 neighbors_[face.v1].insert(face.v2);
180 neighbors_[face.v2].insert(face.v0);
181 neighbors_[face.v2].insert(face.v1);
182 // store the edges
183 inputEdges[makeEdge(face.v0, face.v1)]++;
184 inputEdges[makeEdge(face.v1, face.v2)]++;
185 inputEdges[makeEdge(face.v2, face.v0)]++;
186
187 // update vertex quadrics
188 auto &p0 = posData[face.v0];
189 auto &p1 = posData[face.v1];
190 auto &p2 = posData[face.v2];
191 auto faceNormal = (p1 - p0).cross(p2 - p0);
192 faceNormal.normalize();
193 quadric faceQuadric(
194 faceNormal.x,
195 faceNormal.y,
196 faceNormal.z,
197 -faceNormal.dot(p0));
198 quadrics_[face.v0] += faceQuadric;
199 quadrics_[face.v1] += faceQuadric;
200 quadrics_[face.v2] += faceQuadric;
201 }
202
203 // mark boundary vertices, we will avoid collapsing them
204 for (auto &edge: inputEdges) {
205 if (edge.second == 1) {
206 isBoundary_[edge.first.first] = true;
207 isBoundary_[edge.first.second] = true;
208 }
209 }
210}

212bool MeshSimplifier::solve(const quadric &Q, Vec3f &outPos) {
213 // Build the system from the quadric: A * v = -b
214 Mat3f inv;
215 if (!Q.toMatrix().inverse(inv)) {
216 // Degenerate quadric
217 return false;
218 }
219 float b[3] = {-Q.a[3], -Q.a[6], -Q.a[8]};
220 outPos.x = inv.x[0] * b[0] + inv.x[1] * b[1] + inv.x[2] * b[2];
221 outPos.y = inv.x[3] * b[0] + inv.x[4] * b[1] + inv.x[5] * b[2];
222 outPos.z = inv.x[6] * b[0] + inv.x[7] * b[1] + inv.x[8] * b[2];
223 return true;
224}

226void MeshSimplifier::pushEdge(uint32_t idx0, uint32_t idx1, const quadric &Q, const LODLevel &lod) {
227 auto posData = lod.pos.data();
228
229 // Compute cost based on quadric error
230 Vec3f collapsePos;
231 if (!solve(Q, collapsePos)) {
232 // pick the midpoint of the edge in case of degenerate quadric
233 collapsePos = (posData[idx0] + posData[idx1]) * 0.5f;
234 }
235 auto cost = Q.evaluate(collapsePos);
236
237 // TODO: Add additional constraints to cost function.
238 // - Experiment with penalty for boundary condition.
239 // Currently boundary edges are skipped entirely. But
240 // some meshes use them for small details in which case we actually want to collapse them.
241 // - discourage collapsing across visible UV seams
242 // - float uvDistance = length(v1.uv - v2.uv);
243 // - if (uvDistance > threshold) return;
244#ifdef SIMPLIFIER_USE_AREA_COST
245 auto area = averageAreaOfAffectedTriangles(idx0, idx1);
246 auto areaCost = 1.0f / (area + 1e-6f);
247 cost += areaCost * areaPenalty_;
248#endif
249#ifdef SIMPLIFIER_USE_VALENCE_COST
250 // Add a Valence Preservation Heuristic
251 auto diff = std::abs(
252 static_cast<int>(neighbors_[idx0].size()) -
253 static_cast<int>(neighbors_[idx1].size()));
254 cost += (1.0f / (1.0f + static_cast<float>(diff))) * valencePenalty_;
255#endif
256 // Check Attribute Continuity Constraints
257 if (inputNor_.get()) {
258 // add a penalty for normal difference based on angle diff
259 auto faceNor = (Vec3f *) lod.attributes[norIndex_]->clientData();
260 auto angle = acosf(std::clamp(
261 faceNor[idx0].dot(faceNor[idx1]), -1.0f, 1.0f));
262 if (normalMaxAngle_ > 0.0 && angle > normalMaxAngle_) {
263 // skip collapse if angle is too large
264 return;
265 }
266 auto norCost = (angle / M_PIf);
267 // add squared penalty to cost
268 cost += norCost * norCost * normalPenalty_;
269 }
270
271 edgeCollapses_.push({idx0, idx1, collapsePos, cost});
272}

274void MeshSimplifier::buildEdgeQueue(const std::vector<Triangle> &faces, const LODLevel &data) {
275 // Build the edge queue from the faces.
276 // It orders the edges by their cost, which is the sum of the quadrics
277 // of the two vertices of the edge.
278 std::set<std::pair<uint32_t, uint32_t>> usedEdges;
279 std::array<std::pair<uint32_t, uint32_t>, 3> edges;
280
281 // remove any remaining edges from the previous LOD level
282 edgeCollapses_ = {};
283
284 for (auto &face: faces) {
285 edges[0] = makeEdge(face.v0, face.v1);
286 edges[1] = makeEdge(face.v1, face.v2);
287 edges[2] = makeEdge(face.v2, face.v0);
288
289 for (auto &edge: edges) {
290 if (usedEdges.find(edge) != usedEdges.end()) continue;
291 usedEdges.insert(edge);
292 pushEdge(edge.first, edge.second,
293 quadrics_[edge.first] + quadrics_[edge.second],
294 data);
295 }
296 }
297}

376static void interpolateBones(
377 LODAttribute &indexAttr,
378 LODAttribute *weightAttr,
379 uint32_t i1, uint32_t i2, float w1, float w2) {
380 auto numWeights = static_cast<uint32_t>(indexAttr.attribute->numArrayElements());
381 auto offset1 = i1 * numWeights;
382 auto offset2 = i2 * numWeights;
383 auto &indexData = ((LODAttributeT<uint32_t> *) (&indexAttr))->data;
384 // temporary map to accumulate weights
385 static std::unordered_map<uint32_t, float> boneWeightMap;
386 boneWeightMap.clear();
387 // Accumulate bone weights
388 auto *indices1 = indexData.data() + offset1;
389 auto *indices2 = indexData.data() + offset2;
390 float *weights1 = nullptr, *weights2 = nullptr;
391 if (weightAttr != nullptr) {
392 auto &weightData = ((LODAttributeT<float> *) weightAttr)->data;
393 weights1 = weightData.data() + offset1;
394 weights2 = weightData.data() + offset2;
395 for (uint32_t i = 0; i < numWeights; ++i) { boneWeightMap[indices1[i]] += weights1[i] * w1; }
396 for (uint32_t i = 0; i < numWeights; ++i) { boneWeightMap[indices2[i]] += weights2[i] * w2; }
397 } else {
398 for (uint32_t i = 0; i < numWeights; ++i) { boneWeightMap[indices1[i]] += w1; }
399 for (uint32_t i = 0; i < numWeights; ++i) { boneWeightMap[indices2[i]] += w2; }
400 }
401 // Flatten to vector
402 static std::vector<BoneInfluence> merged;
403 merged.clear();
404 merged.reserve(boneWeightMap.size());
405 for (const auto &[idx, w]: boneWeightMap) {
406 merged.push_back({idx, w});
407 }
408 // Sort descending by weight
409 std::sort(merged.begin(), merged.end(), [](const BoneInfluence &a, const BoneInfluence &b) {
410 return a.weight > b.weight;
411 });
412 // Keep top N and normalize
413 merged.resize(std::min<uint32_t>(numWeights, merged.size()));
414 normalizeWeights(merged);
415 // Write output
416 if (weightAttr != nullptr) {
417 auto &weightData = ((LODAttributeT<float> *) weightAttr)->data;
418 for (uint32_t i = 0; i < numWeights; ++i) {
419 if (i >= merged.size()) {
420 weightData.emplace_back(0.0f);
421 indexData.emplace_back(0);
422 } else {
423 weightData.emplace_back(merged[i].weight);
424 indexData.emplace_back(merged[i].index);
425 }
426 }
427 } else {
428 for (uint32_t i = 0; i < numWeights; ++i) {
429 if (i >= merged.size()) {
430 indexData.emplace_back(0);
431 } else {
432 indexData.emplace_back(merged[i].index);
433 }
434 }
435 }
436}

472static void interpolate(
473 LODLevel &level,
474 LODAttribute &attr,
475 uint32_t i1, uint32_t i2,
476 float w1, float w2) {
477 switch (attr.semantic) {
478 case AttributeSemantic::NORMAL:
479 interpolateNormal(attr, i1, i2, w1, w2);
480 break;
481 case AttributeSemantic::BITANGENT:
482 case AttributeSemantic::TANGENT:
483 interpolateTangent(attr, i1, i2, w1, w2);

485 case AttributeSemantic::TEXCOORD:

492 // NOTE: indices are handled together with weights

495 // assert(attr.attribute->baseType() == GL_FLOAT);
496 interpolateLinear(attr, i1, i2, w1, w2);
497 break;

501uint32_t MeshSimplifier::collapseEdge(uint32_t i1, uint32_t i2, const Vec3f &opt, LODLevel &level) {
502 // Get positions
503 auto &p1 = level.pos[i1];
504 auto &p2 = level.pos[i2];
505 // Compute distances to optimal position
506 auto d1 = (opt - p1).length();
507 auto d2 = (opt - p2).length();
508 // Threshold for reusing existing vertex
509 const float reuseThreshold = 1e-6f;
510 uint32_t newIdx;
511
512 if (d1 < reuseThreshold || isBoundaryVertex(i1)) {
513 // Collapse to v1
514 newIdx = i1;
515 } else if (d2 < reuseThreshold || isBoundaryVertex(i2)) {
516 // Collapse to v2
517 newIdx = i2;
518 } else {
519 // Compute weights for interpolation
520 auto invSum = 1.0f / (d1 + d2 + 1e-6f);
521 auto w1 = d2 * invSum;
522 auto w2 = d1 * invSum;
523
524 // Create new vertex with interpolated attributes
525 newIdx = static_cast<uint32_t>(level.pos.size());
526 level.pos.push_back(opt);
527 for (auto &attr: level.attributes) {
528 interpolate(level, *attr, i1, i2, w1, w2);
529 }
530 }
531
532 return newIdx;
533}

535void MeshSimplifier::updateEdgeCosts(uint32_t vNew,
536 const LODLevel &levelData,
537 std::vector<uint32_t> &mapping) {
538 const quadric &Qv = quadrics_[vNew];
539 for (uint32_t neighbor: neighbors_[vNew]) {
540 // Skip if vertex is no longer active
541 if (mapping[neighbor] != neighbor) continue;
542 // Else create new edge, as we do not want to modify items in the priority queue
543 pushEdge(
544 std::min(vNew, neighbor),
545 std::max(vNew, neighbor),
546 Qv + quadrics_[neighbor],
547 levelData);
548 }
549}

551uint32_t MeshSimplifier::generateLodLevel(size_t targetFaceCount, LODLevel &lodData) {
552 const auto &inputTriangles = lodLevels_.back(); // Last LOD level
553 auto triangles = inputTriangles; // Working copy
554 // LOD data has initially the same size as the last LOD level output.
555 // and also holds a copy of the vertex data.
556 // In the loop below, we might create additional vertices which
557 // are pushed to the back of the lodData.
558 // The output of generateLodLevel then may has some unused vertices and faces,
559 // which we will remove in compactLodLevel.
560 // Map from old vertex index to its current representative
561 std::vector<uint32_t> vertexMapping(lodData.pos.size());
562 std::iota(vertexMapping.begin(), vertexMapping.end(), 0); // initially identity mapping
563 uint32_t collapseCount = 0;
564 uint32_t numActiveFaces = triangles.size();
565 uint32_t r1, r2, newIdx;
566
567 while (numActiveFaces > targetFaceCount && !edgeCollapses_.empty()) {
568 { // collapse next edge
569 auto &collapse = edgeCollapses_.top();
570 r1 = resolve(collapse.v1, vertexMapping);
571 r2 = resolve(collapse.v2, vertexMapping);
572 bool preventCollapse = (r1 == r2);
573 if (useStrictBoundary_) {
574 if (isBoundaryVertex(r1) || isBoundaryVertex(r2)) {
575 preventCollapse = true;
576 }
577 } else {
578 if (isBoundaryVertex(r1) && isBoundaryVertex(r2)) {
579 preventCollapse = true;
580 }
581 }
582
583 if (preventCollapse ||
584 // Already collapsed, and another edge was added to the queue
585 r1 != collapse.v1 || r2 != collapse.v2) {
586 edgeCollapses_.pop();
587 continue;
588 }
589 collapseCount += 1;
590
591 // Collapse v2 into v1 or use a new vertex
592 newIdx = collapseEdge(r1, r2, collapse.optimalPos, lodData);
593 edgeCollapses_.pop();
594 }
595 // clean up the neighborhood
596 if (r1 != newIdx) {
597 for (auto n: neighbors_[r1]) {
598 neighbors_[resolve(n, vertexMapping)].erase(r1);
599 }
600 neighbors_[r1].clear();
601 }
602 if (r2 != newIdx) {
603 for (auto n: neighbors_[r2]) {
604 neighbors_[resolve(n, vertexMapping)].erase(r2);
605 }
606 neighbors_[r2].clear();
607 }
608 // Update mapping: collapsed vertices now point to new vertex
609 vertexMapping[r1] = newIdx;
610 vertexMapping[r2] = newIdx;
611 if (newIdx >= vertexMapping.size()) {
612 // New vertex created, update mapping and add to neighbors (empty set of neighbors)
613 vertexMapping.push_back(newIdx);
614 neighbors_.emplace_back();
615 quadrics_.emplace_back();
616 }
617 // Compute quadric of new vertex
618 quadrics_[newIdx] = quadrics_[r1] + quadrics_[r2];
619
620 // Update triangles
621 for (auto &tri: triangles) {
622 if (!tri.active) continue;
623 tri.v0 = resolve(tri.v0, vertexMapping);
624 tri.v1 = resolve(tri.v1, vertexMapping);
625 tri.v2 = resolve(tri.v2, vertexMapping);
626 if (tri.isDegenerate()) {
627 tri.active = false;
628 numActiveFaces -= 1;
629 continue;
630 }
631 // Update neighbors, we only insert here and skip the inactive neighbors later in edge collapses
632 if (tri.v0 == newIdx) {
633 neighbors_[tri.v0].insert(tri.v1);
634 neighbors_[tri.v0].insert(tri.v2);
635 neighbors_[tri.v1].insert(tri.v0);
636 neighbors_[tri.v2].insert(tri.v0);
637 } else if (tri.v1 == newIdx) {
638 neighbors_[tri.v1].insert(tri.v0);
639 neighbors_[tri.v1].insert(tri.v2);
640 neighbors_[tri.v0].insert(tri.v1);
641 neighbors_[tri.v2].insert(tri.v1);
642 } else if (tri.v2 == newIdx) {
643 neighbors_[tri.v2].insert(tri.v0);
644 neighbors_[tri.v2].insert(tri.v1);
645 neighbors_[tri.v0].insert(tri.v2);
646 neighbors_[tri.v1].insert(tri.v2);
647 }
648 }
649
650 // rebuild affected edge collapses around `newIndex`
651 updateEdgeCosts(newIdx, lodData, vertexMapping);
652 }
653
654 // Store the result
655 auto &finalLOD = lodLevels_.emplace_back();
656 for (const auto &tri: triangles) {
657 if (tri.active) finalLOD.push_back(tri);
658 }
659
660 return collapseCount;
661}

663static void compactLodLevel(const LODLevel &data, LODLevel &compactData, std::vector<Triangle> &faces) {
664 // after collapse some vertices may be unused.
665 // we remove them here, and remap the indices accordingly.
666 std::unordered_set<uint32_t> usedIndices;
667 for (const Triangle &tri: faces) {
668 if (!tri.active) continue;
669 usedIndices.insert(tri.v0);
670 usedIndices.insert(tri.v1);
671 usedIndices.insert(tri.v2);
672 }
673
674 std::unordered_map<uint32_t, uint32_t> indexRemap;
675 // allocate compacted position buffer
676 compactData.pos.resize(usedIndices.size());
677 for (auto &a: data.attributes) {
678 auto sibling = a->makeSibling(usedIndices.size());
679 compactData.attributes.push_back(sibling);
680 }
681 // copy the data
682 uint32_t newIdx = 0u;
683 for (uint32_t oldIdx: usedIndices) {
684 compactData.pos[newIdx] = data.pos[oldIdx];
685 for (uint32_t attributeIndex = 0; attributeIndex < data.attributes.size(); ++attributeIndex) {
686 auto &attr_in = data.attributes[attributeIndex];
687 auto &attr_out = compactData.attributes[attributeIndex];
688 attr_out->setVertex(newIdx, attr_in, oldIdx);
689 }
690 indexRemap[oldIdx] = newIdx++;
691 }
692
693 // Rewrite Indices for new vertex buffer data
694 for (Triangle &tri: faces) {
695 if (!tri.active) continue;
696 tri.v0 = indexRemap[tri.v0];
697 tri.v1 = indexRemap[tri.v1];
698 tri.v2 = indexRemap[tri.v2];
699 }
700}

759void MeshSimplifier::simplifyMesh() {
760 if (!hasValidAttributes_) {
761 REGEN_ERROR("Mesh simplifier has invalid attributes, simplification will not be performed.");
762 return;
763 }
764
765 { // construct first LOD level.
766 auto numOriginalFaces = mesh_->numIndices() / 3;
767 auto &lodLevel0 = lodLevels_.emplace_back();
768 lodLevel0.reserve(numOriginalFaces);
769 if (inputIndices_->baseType() == GL_UNSIGNED_BYTE) {
770 auto m_inIdx = (uint8_t *) inputIndices_->clientData();
771 for (int i = 0; i < numOriginalFaces; ++i) {
772 auto j = i * 3;
773 lodLevel0.emplace_back(
774 (uint32_t)m_inIdx[j],
775 (uint32_t)m_inIdx[j + 1],
776 (uint32_t)m_inIdx[j + 2]);
777 }
778 } else if (inputIndices_->baseType() == GL_UNSIGNED_SHORT) {
779 auto m_inIdx = (uint16_t *) inputIndices_->clientData();
780 for (int i = 0; i < numOriginalFaces; ++i) {
781 auto j = i * 3;
782 lodLevel0.emplace_back(
783 (uint32_t)m_inIdx[j],
784 (uint32_t)m_inIdx[j + 1],
785 (uint32_t)m_inIdx[j + 2]);
786 }
787 } else {
788 auto m_inIdx = (uint32_t *) inputIndices_->clientData();
789 for (int i = 0; i < numOriginalFaces; ++i) {
790 auto j = i * 3;
791 lodLevel0.emplace_back(
792 (uint32_t)m_inIdx[j],
793 (uint32_t)m_inIdx[j + 1],
794 (uint32_t)m_inIdx[j + 2]);
795 }
796 }
797 }
798
799 // compute number of faces for each LOD level
800 std::vector<unsigned int> lodFaces = {0, 0, 0, 0, 0};
801 lodFaces[0] = lodLevels_.front().size();
802 if (useOriginalData()) {
803 lodFaces[1] = getNumFaces(thresholds_[1]);
804 lodFaces[2] = getNumFaces(thresholds_[2]);
805 lodFaces[3] = getNumFaces(thresholds_[3]);
806 lodFaces[4] = 0.0f;
807 } else {
808 lodFaces[1] = getNumFaces(thresholds_[0]);
809 lodFaces[2] = getNumFaces(thresholds_[1]);
810 lodFaces[3] = getNumFaces(thresholds_[2]);
811 lodFaces[4] = getNumFaces(thresholds_[3]);
812 }
813 uint32_t numLodLevels = 1;
814 for (int i=0; i < 4; ++i) {
815 if (i!=0 && thresholds_[i] >= thresholds_[i-1]) {
816 break;
817 }
818 if (thresholds_[i] > 0.0001f && thresholds_[i] < 0.9999f) {
819 numLodLevels++;
820 }
821 }
822
823 // First level offset is num vertices of the original mesh
824 vertexOffset_ = useOriginalData() ? inputPos_->numVertices() : 0u;
825
826 // Iteratively collapse the cheapest edge, updating affected geometry and edge costs.
827 // - Stop once your desired number of triangles is reached (or vertices),
828 // and record the current mesh as LOD level.
829 uint32_t collapseCount = 0;
830 for (size_t lodLevel = 1; lodLevel < numLodLevels; ++lodLevel) {
831 // temporary level data
832 LODLevel levelData;
833 { // generate LOD level
834 levelData.offset = vertexOffset_;
835 if (lodLevel == 1) {
836 levelData.pos.resize(inputPos_->numVertices());
837 std::memcpy(
838 (byte *) levelData.pos.data(),
839 inputPos_->clientData(),
840 inputPos_->inputSize());
841 for (auto &it: inputAttributes_) {
842 auto &attr = it.second;
843 initializeLevelData(
844 levelData,
845 attr,
846 it.first,
847 attr->numVertices() * attr->numArrayElements(),
848 attr->clientData());
849 }
850 computeQuadrics(lodLevels_.back(), (Vec3f *) inputPos_->clientData(), levelData.pos.size());
851 } else {
852 // copy the previous LOD level
853 auto &lastLevelData = lodData_[lodLevel - 2];
854 levelData.pos.resize(lastLevelData.pos.size());
855 std::memcpy(
856 (byte *) levelData.pos.data(),
857 (byte *) lastLevelData.pos.data(),
858 levelData.pos.size() * inputPos_->vertexSize());
859 for (unsigned int attributeIndex = 0u; attributeIndex < inputAttributes_.size(); ++attributeIndex) {
860 auto lodAttr = lastLevelData.attributes[attributeIndex];
861 initializeLevelData(
862 levelData,
863 lodAttr->attribute,
864 lodAttr->semantic,
865 lodAttr->numElements(),
866 (byte *) lodAttr->clientData());
867 }
868 computeQuadrics(lodLevels_.back(), lastLevelData.pos.data(), levelData.pos.size());
869 }
870 buildEdgeQueue(lodLevels_.back(), levelData);
871 collapseCount = generateLodLevel(lodFaces[lodLevel], levelData);
872 }
873 if (collapseCount == 0) {
874 REGEN_DEBUG("Mesh simplifier: No more collapses possible, stopping.");
875 // remove all remaining LOD levels
876 lodLevels_.resize(lodLevels_.size() - 1);
877 break;
878 }
879 { // compact the LOD level
880 auto &compactData = lodData_.emplace_back();
881 compactData.offset = vertexOffset_;
882 compactLodLevel(levelData, compactData, lodLevels_[lodLevel]);
883 REGEN_DEBUG("LOD level " << lodLevels_.size() - 1 << ":"
884 << " #faces=" << lodLevels_[lodLevel].size()
885 << " #verts=" << compactData.pos.size()
886 << " #collapses=" << collapseCount
887 << " offset=" << compactData.offset
888 << " target=" << lodFaces[lodLevel]);
889 // increment vertex offset for next LOD level
890 vertexOffset_ += compactData.pos.size();
891 }
892 }
893
894 applyAttributes();
895}

897uint32_t MeshSimplifier::createOutputAttributes() {
898 bool useOriginal = useOriginalData();
899 uint32_t numVertices = inputPos_->numVertices();
900 uint32_t numIndices = 0;
901 for (auto &faces: lodLevels_) {
902 numIndices += faces.size() * 3;
903 }
904 for (auto &lod: lodData_) {
905 numVertices += lod.pos.size();
906 }
907 if (!useOriginal) {
908 numVertices -= inputPos_->numVertices();
909 numIndices -= lodLevels_.front().size() * 3;
910 }
911
912 // create a new index buffer
913 outputIndices_ = createIndexInput(numIndices, numVertices);
914 REGEN_INFO("Mesh simplified:"
915 << " #lods=" << (lodLevels_.size() - (useOriginal ? 0 : 1))
916 << " #faces=" << numIndices / 3
917 << " #verts=" << numVertices
918 << " #indices=" << numIndices
919 << " out-index-type=0x" << std::hex << outputIndices_->baseType() << std::dec
920 << " in-index-type=0x" << std::hex << inputIndices_->baseType() << std::dec);
921
922 auto v_indices = (byte*) outputIndices_->clientData();
923 auto indexType = outputIndices_->baseType();
924 uint32_t v_idx = 0;
925 for (uint32_t lod_i = (useOriginal ? 0 : 1); lod_i < lodLevels_.size(); ++lod_i) {
926 auto &faces = lodLevels_[lod_i];
927 auto vertexOffset = (lod_i == 0 ? 0u : lodData_[lod_i - 1].offset);
928 for (auto &face: faces) {
929 // faces count from 0 to numVertices, so we need to add the vertexOffset
930 setIndexValue(v_indices, indexType, v_idx++, vertexOffset + face.v0);
931 setIndexValue(v_indices, indexType, v_idx++, vertexOffset + face.v1);
932 setIndexValue(v_indices, indexType, v_idx++, vertexOffset + face.v2);
933 }
934 }
935
936 // allocate new attributes
937 outputPos_ = ref_ptr<ShaderInput3f>::alloc("pos");
938 outputPos_->setVertexData(numVertices);
939 for (auto &typedAttribute: inputAttributes_) {
940 auto att = ShaderInput::create(typedAttribute.second);
941 att->setVertexData(numVertices);
942 outputAttributes_.emplace_back(typedAttribute.first, att);
943 }
944 // and copy data into the new attributes using std::memcpy into clientData() of the new attributes
945 // first copy the original mesh data to the beginning of the new attributes
946 if (useOriginal) {
947 std::memcpy(outputPos_->clientData(), inputPos_->clientData(), inputPos_->inputSize());
948 for (uint32_t att_i = 0; att_i < inputAttributes_.size(); ++att_i) {
949 auto &in = inputAttributes_[att_i].second;
950 auto &out = outputAttributes_[att_i].second;
951 std::memcpy(out->clientData(), in->clientData(), in->inputSize());
952 }
953 }
954 // then copy the LOD data into the new attributes
955 for (auto &lod: lodData_) {
956 std::memcpy(
957 outputPos_->clientData() + lod.offset * outputPos_->vertexSize(),
958 (byte *) lod.pos.data(),
959 lod.pos.size() * outputPos_->vertexSize());
960 for (unsigned int attributeIndex = 0; attributeIndex < lod.attributes.size(); ++attributeIndex) {
961 auto &in = lod.attributes[attributeIndex];
962 auto &out = outputAttributes_[attributeIndex].second;
963 std::memcpy(
964 out->clientData() + lod.offset * out->vertexSize(),
965 in->clientData(),
966 in->inputSize());
967 }
968 }
969
970 return numVertices;
971}

973void MeshSimplifier::applyAttributes() {
974 // create the output attributes from the generated LOD levels
975 auto numVertices = createOutputAttributes();
976
977 // remove the input attributes from the mesh
978 mesh_->removeInput(inputPos_);
979 for (auto &pair: inputAttributes_) {

982
983 // add the output attributes to the mesh
984 mesh_->set_numVertices(numVertices);

986 mesh_->setInput(outputPos_);
987 for (auto &pair: outputAttributes_) {
988 mesh_->setInput(pair.second);
989 }
990 mesh_->updateVertexData();
991
992 // create the mesh LODs
993 std::vector<Mesh::MeshLOD> meshLODs(useOriginalData() ? lodLevels_.size() : lodLevels_.size() - 1);

995 for (size_t i = (useOriginalData() ? 0 : 1); i < lodLevels_.size(); ++i) {
996 auto &faces = lodLevels_[i];

999 meshLODs[lodIndex].d->vertexOffset = vertexOffset;

1003 meshLODs[lodIndex].d->numVertices = lodData_[i - 1].pos.size();
1004 }
1005 indexOffset += meshLODs[lodIndex].d->numIndices;
1006 vertexOffset += meshLODs[lodIndex].d->numVertices;
1007 lodIndex += 1;
1008 }

1010 mesh_->activateLOD(0);
