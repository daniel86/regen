8static uint32_t computeMaxInstances(const BlanketTrail::TrailConfig &cfg) {
9 uint32_t maxInstances = cfg.numTrails;
10 if (cfg.blanketLifetime > 0.0f && cfg.blanketFrequency > 0.0f) {
11 // compute number of instances based on walk speed and lifetime
12 maxInstances += static_cast<uint32_t>(cfg.numTrails *
13 std::ceil(2.0 * cfg.blanketLifetime / cfg.blanketFrequency));
14 }
15 return maxInstances;
16}

51void BlanketTrail::setModelTransform(const ref_ptr<ModelTransformation> &tf) {
52 tf_ = tf;
53
54 // initialize TF to num instances
55 if (tf_.get()) {
56 tf_->set_numInstances(numBlankets_);
57 auto modelMat = tf_->modelMat();
58 modelMat->setInstanceData(numBlankets_, 1, nullptr);
59
60 Mat4f *mat0 = (Mat4f*) modelMat->clientData(0);
61 Mat4f *mat1 = (Mat4f*) modelMat->clientData(1);
62 for (uint32_t j = 0; j < numBlankets_; j += 1) {
63 mat0[j] = Mat4f::identity();
64 if(mat1) mat1[j] = Mat4f::identity();
65 }
66
67 // distribute initially randomly
68 for (uint32_t i = 0; i < numBlankets_; ++i) {
69 float randX = (math::random<float>() - 0.5f) * 2.0f * 1000.0f;
70 float randZ = (math::random<float>() - 0.5f) * 2.0f * 1000.0f;
71 mat0[i].translate(Vec3f(randX, 2000.0f, randZ));
72 if(mat1) mat1[i].translate(Vec3f(randX, 2000.0f, randZ));
73 }
74 }
75
76 joinStates(tf);
77}

79void BlanketTrail::insertBlanket(const Vec3f &pos, const Vec3f &dir, uint32_t maskIndex) {
80 uint32_t instanceIdx = reviveBlanket();
81
82 // update TF
83 Quaternion q(0.0, 0.0, 0.0, 1.0);
84 q.setEuler(dir.x, dir.y, dir.z);
85 tmpMat_ = q.calculateMatrix();
86 tmpMat_.translate(Vec3f(pos.x, insertHeight_, pos.z));
87 tf_->setModelMat(instanceIdx, tmpMat_);
88
89 if(indexedShapes_.get()) {
90 // Also update bounding shape position in spatial index.
91 auto &iShape = indexedShape(instanceIdx);
92 iShape->setBaseOffset(Vec3f(0.0f, pos.y - insertHeight_, 0.0f));
93 iShape->nextLocalStamp();
94 }
95
96 // update mask index
97 if (!maskIndex_.empty() && maskIndex_[instanceIdx] != maskIndex) {
98 maskIndex_[instanceIdx] = maskIndex;
99 u_maskIndex_->setVertex(instanceIdx, maskIndex);

124 updateFlags.compute = input.getValue<BufferComputeMode>("compute-mode", BUFFER_NO_COMPUTE);
125 BlanketTrail::TrailConfig meshCfg;
126 meshCfg.levelOfDetails = lodLevels;
127 meshCfg.texcoScale = input.getValue<Vec2f>("texco-scaling", Vec2f::one());
128 meshCfg.blanketSize = input.getValue<Vec2f>("blanket-size", Vec2f(1.0f, 1.0f));
129 meshCfg.isInitiallyDead = input.getValue<bool>("initially-dead", false);
130 meshCfg.blanketLifetime = input.getValue<float>("blanket-lifetime", 0.0f);
131 meshCfg.updateHint = updateFlags;

133 meshCfg.accessMode = input.getValue<ClientAccessMode>("access-mode", BUFFER_CPU_WRITE);
134 meshCfg.numTrails = input.getValue<uint32_t>("blanket-trails", 1u);
135 meshCfg.blanketFrequency = input.getValue<float>("blanket-frequency", 0.5f);
136 auto blanket = ref_ptr<BlanketTrail>::alloc(groundMesh, meshCfg, scene->systemTime());
137
138 // early add mesh vector to scene for children handling below
139 auto out_ = ref_ptr<CompositeMesh>::alloc();
140 out_->addMesh(blanket);
141 ctx.scene()->putResource<CompositeMesh>(input.getName(), out_);
142

144 if (tfChild.get()) {

147 scene->putResource("ModelTransformation", tfChild->getValue("id"), tf);
148 input.removeChild(tfChild);

152 tf->setModelMat(0, Mat4f::identity());
153 tf->tfBuffer()->updateBuffer();

160 // avoid re-processing
161 input.removeChild(maskNode);
162

165 auto fbo = ctx.scene()->getResource<FBO>(fboName);
166 if (fbo.get()) {

168 blanketMask = fbo->colorTextures()[attachmentIdx];

172 }
173 }

175 blanket->setBlanketMask(blanketMask);
176 }
177 }
178 groundMesh->setupGroundBlanket(*blanket.get());
179
180 // Load shapes, this is needed to initially set the traversal mask for all shapes.
181 std::vector<ref_ptr<scene::SceneInputNode>> handledChildren;
182 for (auto &n: input.getChildren("shape")) {
183 handledChildren.push_back(n);
184 // Initialize shapes. For this we need to make sure that the shape children are loaded.
185 auto processor = ctx.scene()->getStateProcessor(n->getCategory());
186 if (processor.get() == nullptr) {
187 REGEN_WARN("No processor registered for '" << n->getDescription() << "'.");
188 } else {
189 ref_ptr<State> dummyState = ref_ptr<State>::alloc();
190 processor->processInput(ctx.scene(), *n.get(), ctx.parent(), dummyState);
191 }
192 }

194 // make sure mesh loading does not attempt to load materials again (this will cause a warning)
195 input.removeChild(n);
196 }
