11float traitStrength(const std::vector<float> &positive, const std::vector<float> &negative) {
12 if (positive.empty() && negative.empty()) return 0.5f;
13 float pos = 0.f, neg = 0.f;
14 for (float t : positive) pos += t;
15 for (float t : negative) neg += t;
16 // Combine into [0,1] range. 0.5 = neutral, >0.5 favors positive traits.
17 pos = positive.empty() ? 0.0f : pos / positive.size();
18 neg = negative.empty() ? 0.0f : neg / negative.size();
19 return math::clamp(0.5f + 0.5f * (pos - neg), 0.0f, 1.0f);
20}

64BehaviorStatus SelectTargetPlace::tick(Blackboard& kb, float /*dt_s*/) {
65 auto &targetPlace = kb.targetPlace();
66 if (targetPlace.get() && (kb.lingerTime() > 0.0f || kb.activityTime() > 0.0f)) {
67 return BehaviorStatus::SUCCESS;
68 }
69 if (desiredPlaceType == PlaceType::LAST) {
70 // Compute randomized weights for the places.
71 float homeWeight = (0.5f + 0.25f * math::random<float>() +
72 0.25f * traitStrength({kb.laziness()}, {kb.sociability(), kb.spirituality()}));
73 float marketWeight = (0.5f + 0.25f * math::random<float>() +
74 0.25f * traitStrength({kb.sociability()}, {kb.laziness()}));
75 float shrineWeight = (0.5f + 0.25f * math::random<float>() +
76 0.25f * traitStrength({kb.spirituality()}, {kb.laziness()}));
77 if (!kb.homePlace()) homeWeight = 0.0f;
78 auto &gatheringPlaces = kb.getPlacesByType(PlaceType::GATHERING);
79 auto &spiritualPlaces = kb.getPlacesByType(PlaceType::SPIRITUAL);
80 if (gatheringPlaces.empty()) marketWeight = 0.0f;
81 if (spiritualPlaces.empty()) shrineWeight = 0.0f;
82
83 float r = math::random<float>() * (homeWeight + marketWeight + shrineWeight);
84 marketWeight += homeWeight;
85 shrineWeight += marketWeight;
86
87 if (r < homeWeight) {
88 kb.setTargetPlace(kb.homePlace());
89 } else if (r < marketWeight) {
90 kb.setTargetPlace(gatheringPlaces[math::randomInt() % gatheringPlaces.size()]);
91 } else {
92 kb.setTargetPlace(spiritualPlaces[math::randomInt() % spiritualPlaces.size()]);
93 }
94 } else {
95 auto &places = kb.getPlacesByType(desiredPlaceType);
96 if (places.empty()) {
97 return BehaviorStatus::FAILURE;
98 }
99 kb.setTargetPlace(places[math::randomInt() % places.size()]);
100 }
101 setLingerTime(kb, kb.targetPlace()->placeType());
102 if constexpr(NPC_ACTIONS_DEBUG) {
103 REGEN_INFO("["<<kb.instanceId()<<"] Selected target place " <<
104 (kb.targetPlace().get() ? kb.targetPlace()->name() : "null"));
105 }
106 return BehaviorStatus::SUCCESS;

186void SelectPlaceActivity::updateActionPossibilities(Blackboard& kb) {
187 static const std::vector<WeightedAction> PossibleActions = {
188 { ActionType::OBSERVING, 0.25f, 0.25f, 0.5f,
189 {Trait::ALERTNESS}, {Trait::SOCIALABILITY} },
190 { ActionType::CONVERSING, 0.25f, 0.25f, 0.5f,
191 {Trait::SOCIALABILITY}, {Trait::LAZINESS} },
192 { ActionType::PRAYING, 0.25f, 0.25f, 0.5f,
193 {Trait::SPIRITUALITY}, {Trait::LAZINESS} },
194 { ActionType::SLEEPING, 0.1f, 0.1f, 0.5f,
195 {Trait::LAZINESS}, {Trait::ALERTNESS, Trait::SOCIALABILITY} },
196 { ActionType::ATTACKING, 0.15f, 0.15f, 0.4f,
197 {Trait::BRAVERY}, {Trait::LAZINESS, Trait::SOCIALABILITY} },
198 };
199 static const std::vector<WeightedNavigation> PossibleNavigations = {
200 { PathwayType::PATROL, ActionType::PATROLLING, 0.1f, 0.1f, 0.4f,
201 {Trait::ALERTNESS, Trait::BRAVERY}, {Trait::LAZINESS, Trait::SOCIALABILITY} },
202 { PathwayType::STROLL, ActionType::STROLLING, 0.15f, 0.15f, 0.4f,
203 {Trait::LAZINESS}, {Trait::SOCIALABILITY, Trait::ALERTNESS, Trait::BRAVERY} },
204 };
205
206 actionPossibilities_.clear();
207 actionPossibilities_.emplace_back(ActionType::IDLE, (
208 0.05f +
209 0.05f * math::random<float>() +
210 0.1f * traitStrength({kb.laziness()}, {kb.alertness(), kb.sociability()})));
211
212 // Add possible navigation actions based on pathways.
213 for (const auto &wn : PossibleNavigations) {
214 if (canWalkAt(kb, wn.pathway, lastPlace_)) {
215 std::vector<float> posTraits, negTraits;
216 for (const auto &t : wn.positiveTraits) {
217 posTraits.push_back(kb.traitStrength(t));
218 }
219 for (const auto &t : wn.negativeTraits) {
220 negTraits.push_back(kb.traitStrength(t));
221 }
222 actionPossibilities_.emplace_back(wn.action, (
223 wn.baseWeight +
224 wn.randomWeight * math::random<float>() +
225 wn.traitWeight * traitStrength(posTraits, negTraits)));
226 }
227 }
228
229 // Add other possible actions based on affordances.
230 for (const auto &wa : PossibleActions) {
231 if (canUseAt(kb, wa.action, lastPlace_)) {
232 std::vector<float> posTraits, negTraits;
233 for (const auto &t : wa.positiveTraits) {
234 posTraits.push_back(kb.traitStrength(t));
235 }
236 for (const auto &t : wa.negativeTraits) {
237 negTraits.push_back(kb.traitStrength(t));
238 }
239 actionPossibilities_.emplace_back(wa.action, (
240 wa.baseWeight +
241 wa.randomWeight * math::random<float>() +
242 wa.traitWeight * traitStrength(posTraits, negTraits)));
243 }
244 }
245
246 // Sum and normalize weights.
247 float totalWeight = 0.0f;
248 for (auto &p : actionPossibilities_) {
249 totalWeight += p.second;
250 }
251 if (!actionPossibilities_.empty()) {
252 for (auto &p : actionPossibilities_) {
253 p.second /= totalWeight;
254 }
255 }
256}

308BehaviorStatus SelectPlaceActivity::tick(Blackboard& kb, float /*dt_s*/) {
309 auto &place = kb.currentPlace();
310 if (kb.lingerTime() <= 0.0f) {
311 // No place, or time to leave
312 kb.setActivityTime(0.0f);
313 return BehaviorStatus::SUCCESS;
314 }
315 if (!place.get()) {
316 // No place, or time to leave
317 if constexpr(NPC_ACTIONS_DEBUG) {
318 REGEN_WARN("["<<kb.instanceId()<<"] No current place for activity selection.");
319 }
320 return BehaviorStatus::FAILURE;
321 }
322 if (kb.activityTime() > 0.0f) {
323 // Continue current activity, skip selection for now and move one.
324 return BehaviorStatus::SUCCESS;
325 }
326 // Update weights for action possibilities at this place.
327 if (place.get() != lastPlace_.get()) {
328 lastPlace_ = place;
329 updateActionPossibilities(kb);
330 }
331 // Get a random number [0,1] and pick an activity based on the weights.
332 ActionType nextActivity = ActionType::IDLE;
333 float r = math::random<float>();
334 float scan = 0.0f;
335 for (uint32_t i = 0; i < actionPossibilities_.size(); i++) {
336 scan += actionPossibilities_[i].second;
337 if (r <= scan) {
338 nextActivity = actionPossibilities_[i].first;
339 break;
340 }
341 }
342 kb.setDesiredAction(nextActivity);
343 setActionTime(kb, nextActivity);
344 // If we use an object affordance of another type, unset the interaction target, else
345 // we may continue using the same object.
346 auto &currentInteraction = kb.interactionTarget();
347 if (currentInteraction.affordance.get() && currentInteraction.affordance->type != nextActivity) {
348 kb.unsetInteractionTarget();
349 }
350 if constexpr(NPC_ACTIONS_DEBUG) {
351 REGEN_INFO("["<<kb.instanceId()<<"] Selected activity " << nextActivity <<
352 " at place '" << place->name() << "'.");
353 }
354 return BehaviorStatus::SUCCESS;
355}

392BehaviorStatus SelectPlacePatient::tick(Blackboard& kb, float /*dt_s*/) {
393 auto &place = kb.currentPlace();
394 if (kb.lingerTime() <= 0.0f) {
395 // No time left at this place
396 kb.setActivityTime(-1.0f);
397 return BehaviorStatus::SUCCESS;
398 }
399 if (!place.get()) {
400 // No place
401 if constexpr(NPC_ACTIONS_DEBUG) {
402 REGEN_WARN("["<<kb.instanceId()<<"] No current place for activity selection.");
403 }
404 return BehaviorStatus::FAILURE;
405 }
406 const ActionType desiredAction = kb.desiredAction();
407 const bool hasCurrentPatient = kb.hasInteractionTarget();
408 const bool isContinuingSame = desiredAction == lastDesiredAction_ && place.get() == lastPlace_.get();
409
410 if (hasCurrentPatient && isContinuingSame) {
411 // No change, stick with last selection.
412 return BehaviorStatus::SUCCESS;
413 }
414 lastDesiredAction_ = desiredAction;
415 lastPlace_ = place;
416 if (hasCurrentPatient) {
417 kb.unsetInteractionTarget();
418 }
419
420 ref_ptr<WorldObject> selected;
421 ref_ptr<Affordance> aff;
422
423 // First try to continue using the same object if possible.
424 if (isContinuingSame && lastSelected_.get()) {
425 auto candidateAff = lastSelected_->getAffordance(desiredAction);
426 if (candidateAff->hasFreeSlot()) {
427 selected = lastSelected_;
428 aff = candidateAff;
429 }
430 }
431
432 if (!selected) {
433 // Retrieve affordances from place.
434 auto &objects = place->getAffordanceObjects(desiredAction);
435 if (!objects.empty()) {
436 // For now pick a random affordance.
437 uint32_t startIdx = math::randomInt() % objects.size();
438 for (uint32_t i = 0; i < objects.size(); i++) {
439 auto candidate = objects[(i + startIdx) % objects.size()];
440 auto candidateAff = candidate->getAffordance(desiredAction);
441 if (candidateAff->hasFreeSlot()) {
442 selected = candidate;
443 aff = candidateAff;
444 break;
445 }
446 }
447 }
448 }
449
450 if (!selected) {
451 // No available slots on any object.
452 lastSelected_ = {};
453 return BehaviorStatus::FAILURE;
454 }
455 int slotIdx = reserveAffordanceSlot(kb, aff);
456 if (slotIdx == -1) {
457 lastSelected_ = {};
458 return BehaviorStatus::FAILURE;
459 } else {
460 if constexpr(NPC_ACTIONS_DEBUG) {
461 REGEN_INFO("["<<kb.instanceId()<<"] Setting patient to '" <<
462 selected->name() << "' for action " << desiredAction);
463 }
464 lastSelected_ = selected;
465 return selectPatient(kb, selected, aff, slotIdx);
466 }

621 if (kb.navigationTarget().object.get() == kb.currentGroup().get()) {
622 if (kb.distanceToTarget() < kb.currentGroup()->radius()) {
623 // Reached group.
624 if constexpr(NPC_ACTIONS_DEBUG) {
625 REGEN_INFO("["<<kb.instanceId()<<"] Reached group.");
626 }
627 kb.unsetNavigationTarget();
628 return BehaviorStatus::SUCCESS;
629 } else {
630 // Already moving to group.
631 kb.setCurrentAction(ActionType::NAVIGATING);
632 return BehaviorStatus::RUNNING;
633 }
634 }
635 kb.setCurrentAction(ActionType::NAVIGATING);
636 kb.setNavigationTarget(kb.currentGroup(), {}, -1);
637 if (!setInitialDistance(kb, kb.navigationTarget())) {
638 REGEN_WARN("["<<kb.instanceId()<<"] Unable to set initial distance, no object set.");
639 }
640 if constexpr(NPC_ACTIONS_DEBUG) {
641 REGEN_INFO("["<<kb.instanceId()<<"] Moving to group at position " <<
642 kb.currentGroup()->position2D() << ".");
643 }
644 return BehaviorStatus::RUNNING;
645}

653BehaviorStatus MoveToPatient::tick(Blackboard& kb, float /*dt_s*/) {
654 if (!kb.interactionTarget().object) {
655 // no patient set.
656 REGEN_WARN("["<<kb.instanceId()<<"] No patient set.");
657 kb.unsetInteractionTarget();
658 kb.unsetNavigationTarget();
659 return BehaviorStatus::FAILURE;
660 }
661 if (kb.activityTime() <= 0.0f) {
662 // No remaining time to reach patient.
663 kb.unsetInteractionTarget();
664 kb.unsetNavigationTarget();
665 return BehaviorStatus::SUCCESS;
666 }
667 auto &it = kb.interactionTarget();
668 if (kb.distanceToPatient() <= getReachRadius(it)) {
669 // Reached patient/affordance
670 if constexpr(NPC_ACTIONS_DEBUG) {
671 REGEN_INFO("["<<kb.instanceId()<<"] Reached patient '" << it.object->name() << "'.");
672 }
673 kb.unsetNavigationTarget();
674 return BehaviorStatus::SUCCESS;
675 }
676 auto currentNavTarget = kb.navigationTarget().object.get();
677 if (currentNavTarget == kb.interactionTarget().object.get() &&
678 kb.navigationTarget().affordance.get() == kb.interactionTarget().affordance.get() &&
679 kb.navigationTarget().affordanceSlot == kb.interactionTarget().affordanceSlot) {
680 // Already moving to patient.
681 kb.setCurrentAction(ActionType::NAVIGATING);
682 return BehaviorStatus::RUNNING;
683 } else if (currentNavTarget != nullptr) {
684 // There can only be one navigation target at a time.
685 kb.unsetNavigationTarget();
686 }
687 if constexpr(NPC_ACTIONS_DEBUG) {
688 REGEN_INFO("["<<kb.instanceId()<<"] Moving to patient '" << it.object->name() << "'.");
689 }
690 kb.setCurrentAction(ActionType::NAVIGATING);
691 kb.setNavigationTarget(kb.interactionTarget());
692 return BehaviorStatus::RUNNING;
693}

721BehaviorStatus PerformAffordedAction::tick(Blackboard& kb, float /*dt_s*/) {
722 auto &patient = kb.interactionTarget();
723 if (!patient.object || !patient.affordance || patient.affordanceSlot == -1) {
724 // Invalid state when entering this action
725 REGEN_WARN("["<<kb.instanceId()<<"] No valid patient or affordance in PerformAffordedAction.");
726 return BehaviorStatus::FAILURE;
727 }
728 if (kb.activityTime() <= 0.0f) {
729 // Time ran out, stop action.
730 kb.unsetInteractionTarget();
731 return BehaviorStatus::SUCCESS;
732 }
733 if (kb.distanceToPatient() > getReachRadius(patient)) {
734 // Not in reach of the affordance -> failure.
735 // Usually you want to have PerformAffordedAction in a selection followed by MoveToPatient
736 // which will become active in this case.
737 return BehaviorStatus::FAILURE;
738 }
739 // Make afforded action current, this will make the controller play the action animation.
740 kb.setCurrentAction(patient.affordance->type);
741 return BehaviorStatus::RUNNING;

744BehaviorStatus FormLocationGroup::tick(Blackboard &kb, float /*dt_s*/) {
745 auto &location = kb.currentLocation();
746 if (!location) {
747 // Invalid state when entering this action
748 REGEN_WARN("["<<kb.instanceId()<<"] No current location.");
749 return BehaviorStatus::FAILURE;
750 }
751 if (location->numVisitors() < 2) {
752 // Not enough characters at location to form a group.
753 return BehaviorStatus::FAILURE;
754 }
755 if (kb.isPartOfGroup()) {
756 // Already in a group, nothing to do.
757 if constexpr(NPC_ACTIONS_DEBUG) {
758 REGEN_INFO("["<<kb.instanceId()<<"] Already part of a group.");
759 }
760 return BehaviorStatus::SUCCESS;
761 }
762 Vec2f currentPos2D(kb.currentPosition().x, kb.currentPosition().z);
763
764 // Find a friendly character at the location to form/join a group with.
765 // Consider characters that are closest first, for this we need to sort by distance.
766 std::vector<std::pair<uint32_t, float>> candidates(location->numVisitors());
767 uint32_t candidateIdx = 0;
768 for (int32_t characterIdx = 0; characterIdx < location->numVisitors(); characterIdx++) {
769 WorldObject *character = location->visitor(characterIdx);
770 if (!character) continue;
771 float dist = (character->position2D() - currentPos2D).lengthSquared();
772 candidates[candidateIdx++] = std::make_pair(characterIdx, dist);
773 }
774 std::sort(candidates.begin(), candidates.end(),
775 [](const std::pair<uint32_t,float> &a, const std::pair<uint32_t,float> &b) {
776 return a.second < b.second;
777 });
778
779 WorldObject *friendlyCharacter = nullptr;
780 for (uint32_t i = 0; i < candidateIdx; i++) {
781 WorldObject *character = location->visitor(candidates[i].first);
782 if (character == kb.characterObject()) continue;
783 if (character->isPartOfGroup()) {
784 if (character->currentGroup()->isFull()) {
785 // Try next character
786 continue;
787 }
788 }
789 friendlyCharacter = character;
790 break;
791 }
792 if (!friendlyCharacter) {
793 // No friendly character found to form/join a group with.
794 return BehaviorStatus::FAILURE;
795 }
796 if (friendlyCharacter->isPartOfGroup()) {
797 // Join existing group.
798 kb.joinGroup(friendlyCharacter->currentGroup());
799 if constexpr(NPC_ACTIONS_DEBUG) {
800 REGEN_INFO("["<<kb.instanceId()<<"] Joining existing group with '" << friendlyCharacter->name() << "'.");
801 }
802 } else {
803 // Form new group.
804 kb.formGroup(kb.desiredAction(), friendlyCharacter);
805 if constexpr(NPC_ACTIONS_DEBUG) {
806 REGEN_INFO("["<<kb.instanceId()<<"] Forming new group with '" << friendlyCharacter->name() << "'.");
807 }
808 }
809 return BehaviorStatus::SUCCESS;
810}
