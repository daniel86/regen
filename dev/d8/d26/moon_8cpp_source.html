113t_eclf Moon::position(const t_julianDay &t) {
114 const float sM = _rad(Sun::meanAnomaly(t));
115 const float mL = _rad(meanLongitude(t));
116 const float mM = _rad(meanAnomaly(t));
117 const float mD = _rad(meanElongation(t));
118 const float mF = _rad(meanLatitude(t));
119
120 // ("A Physically-Based Night Sky Model" - 2001 - Wann Jensen et al.)
121
122 float Sl = mL;
123
124 Sl += 0.1098f * sinf(+1 * mM);
125 Sl += 0.0222f * sinf(2 * mD - 1 * mM);
126 Sl += 0.0115f * sinf(2 * mD);
127 Sl += 0.0037f * sinf(+2 * mM);
128 Sl -= 0.0032f * sinf(+1 * sM);
129 Sl -= 0.0020f * sinf(+2 * mF);
130 Sl += 0.0010f * sinf(2 * mD - 2 * mM);
131 Sl += 0.0010f * sinf(2 * mD - 1 * sM - 1 * mM);
132 Sl += 0.0009f * sinf(2 * mD + 1 * mM);
133 Sl += 0.0008f * sinf(2 * mD - 1 * sM);
134 Sl -= 0.0007f * sinf(+1 * sM - 1 * mM);
135 Sl -= 0.0006f * sinf(1 * mD);
136 Sl -= 0.0005f * sinf(+1 * sM + 1 * mM);
137
138 float Sb = 0.0;
139
140 Sb += 0.0895f * sinf(+1 * mF);
141 Sb += 0.0049f * sinf(+1 * mM + 1 * mF);
142 Sb += 0.0048f * sinf(+1 * mM - 1 * mF);
143 Sb += 0.0030f * sinf(2 * mD - 1 * mF);
144 Sb += 0.0010f * sinf(2 * mD - 1 * mM + 1 * mF);
145 Sb += 0.0008f * sinf(2 * mD - 1 * mM - 1 * mF);
146 Sb += 0.0006f * sinf(2 * mD + 1 * mF);
147
148 t_eclf ecl;
149
150 ecl.longitude = _deg(Sl);
151 ecl.latitude = _deg(Sb);
152
153 return ecl;
154}

168float Moon::distance(const t_julianDay &t) {
169 // NOTE: This gives the distance from the center of the moon to the
170 // center of the earth.
171 const float sM = _rad(Sun::meanAnomaly(t));
172 const float mM = _rad(meanAnomaly(t));
173 const float mD = _rad(meanElongation(t));
174
175 float Sr = 0.016593;
176 Sr += 0.000904f * cosf(1 * mM);
177 Sr += 0.000166f * cosf(2 * mD - 1 * mM);
178 Sr += 0.000137f * cosf(2 * mD);
179 Sr += 0.000049f * cosf(2 * mM);
180 Sr += 0.000015f * cosf(2 * mD + 1 * mM);
181 Sr += 0.000009f * cosf(2 * mD - 1 * sM);
182 return Earth::meanRadius() / Sr;
183}

185void Moon::opticalLibrations(const t_julianDay &t, float &l, float &b) {
186 // (AA.51.1)
187 const float Dr = _rad(Earth::longitudeNutation(t));
188 const float F = _rad(meanLatitude(t));
189 const float O = _rad(meanOrbitLongitude(t));
190
191 const t_eclf ecl = position(t);
192 const float lo = _rad(ecl.longitude);
193 const float la = _rad(ecl.latitude);
194
195 static const float I = _rad(1.54242f);
196
197 const float cos_la = cosf(la);
198 const float sin_la = sinf(la);
199 const float cos_I = cosf(I);
200 const float sin_I = sinf(I);
201
202 const float W = _rev(lo - Dr - O);
203 const float sin_W = sinf(W);
204
205 const float A = _rev(atan2(sin_W * cos_la * cos_I - sin_la * sin_I, cos(W) * cos_la));
206
207 l = _deg(A - F);
208 b = _deg(asinf(-sin_W * cos_la * sin_I - sin_la * cos_I));
209}

211float Moon::parallacticAngle(const t_aTime &aTime, float latitude, float longitude) {
212 // (AA.13.1)
213 const t_julianDay t(jd(aTime));
214
215 const float la = _rad(latitude);
216 const float lo = _rad(longitude);
217
218 const t_equf pos = apparentPosition(t);
219 const float ra = _rad(pos.right_ascension);
220 const float de = _rad(pos.declination);
221
222 const float s = _rad(siderealTime(aTime));
223
224 // (AA.p88) - local hour angle
225
226 const float H = s + lo - ra;
227
228 const float cos_la = cos(la);
229 const float P = atan2f(sin(H) * cos_la, sin(la) * cos(de) - sin(de) * cos_la * cos(H));
230
231 return _deg(P);
232}

234float Moon::positionAngleOfAxis(const t_julianDay &t) {
235 // (AA.p344)
236 const t_equf pos = apparentPosition(t);
237
238 const float a = _rad(pos.right_ascension);
239 const float e = _rad(Earth::meanObliquity(t));
240
241 const float Dr = _rad(Earth::longitudeNutation(t));
242 const float O = _rad(meanOrbitLongitude(t));
243
244 const float V = O + Dr;
245
246 static const float I = _rad(1.54242);
247 const float sin_I = sinf(I);
248
249 const float X = sin_I * sinf(V);
250 const float Y = sin_I * cosf(V) * cosf(e) - cosf(I) * sinf(e);
251
252 // optical libration in latitude
253
254 const t_eclf ecl = position(t);
255
256 const float lo = _rad(ecl.longitude);
257 const float la = _rad(ecl.latitude);
258
259 const float W = _rev(lo - Dr - O);
260 const float b = asinf(-sinf(W) * cosf(la) * sin_I - sinf(la) * cosf(I));
261
262 // final angle
263
264 const float w = _rev(atan2(X, Y));
265 const float P = asinf(sqrt(X * X + Y * Y) * cosf(a - w) / cosf(b));
266
267 return _deg(P);
268}
