18static Vec3f getPerpendicular1(const Vec3f &v) {
19 if (std::abs(v.x) < std::abs(v.y) && std::abs(v.x) < std::abs(v.z)) {
20 return v.cross(Vec3f::right());
21 } else if (std::abs(v.y) < std::abs(v.z)) {
22 return v.cross(Vec3f::up());
23 } else {
24 return v.cross(Vec3f::front());
25 }
26}

28static Vec3f getPerpendicular(const Vec3f &v) {
29 // Compute a vector perpendicular to v
30 auto w = getPerpendicular1(v);
31 w.normalize();
32 // Generate a random angle
33 auto phi = static_cast<float>(math::random<float>() * 2.0f * M_PI);
34 // Rotate the perpendicular vector around v by phi
35 return w * std::cos(phi) + v.cross(w) * std::sin(phi);
36}

38static Vec3f branch(
39 const Vec3f &start,
40 const Vec3f &dir,
41 const Vec3f &segmentEnd,
42 float offsetAmount,
43 float branchLength) {
44 Vec3f v = segmentEnd;
45 v += getPerpendicular(dir) * (offsetAmount * (math::random<float>() * 2.0f - 1.0f));
46 v -= start;
47 v.normalize();
48 // pull the end point of the branch closer to origin by factor lengthScale
49 return start + v * dir.length() * branchLength;
50}

89void LightningStrike::updateSegmentData(const Vec3f &source, const Vec3f &target) {
90 // update CPU segment data.
91 // we use two std::vectors in the process which are swapped for each subdivision.
92 // internally the vector should keep the memory allocated, so there should be no
93 // reallocation of memory after the first iteration.
94 // the memory in the vector is further aligned with the GL buffer such that we can
95 // directly copy the data to the buffer.
96 const uint32_t currentDrawIdx = drawIndex_.load(std::memory_order_acquire);
97 // Obtain the two update indices, which are not the current draw index
98 // as we should not modify the segment being drawn.
99 // NOTE: this logic will break if the number of segment buffers is changed from 3 to something else.
100 const uint32_t updateIdx[2] = {
101 (currentDrawIdx == 0u) ? 1u : 0u,
102 (currentDrawIdx == 2u) ? 1u : 2u
103 };
104 SegmentData *updateSegments[2] = {
105 &segments_[updateIdx[0]],
106 &segments_[updateIdx[1]]
107 };
108
109 updateSegments[1]->clear();
110 updateSegments[0]->clear();
111 updateSegments[0]->push_back(source, target, strikeIdx_);
112 uint32_t lastIndex = 0, nextIndex = 1;
113 float offsetAmount = jitterOffset_;
114 float heightRange = source.y - target.y;
115
116 int numMainBranchVertices = static_cast<int>(pow(2, maxSubDivisions_));
117 int numSubBranchVertices = numMainBranchVertices;
118 int numRemainingVertices = numMainBranchVertices * static_cast<int>(maxBranches_ - 1);
119
120 for (unsigned int i = 0u; i < maxSubDivisions_; ++i) {
121 numSubBranchVertices /= 2;
122
123 auto &segmentData = updateSegments[lastIndex];
124 for (uint32_t segmentIdx = 0u; segmentIdx < segmentData->pos_.size() / 2; ++segmentIdx) {
125 const uint32_t startIdx = segmentIdx * 2;
126 const uint32_t endIdx = startIdx + 1;
127
128 const Vec3f &startPos = segmentData->pos_[startIdx];
129 const Vec3f &endPos = segmentData->pos_[endIdx];
130
131 const float startBrightness = segmentData->brightness_[startIdx];
132 const float endBrightness = segmentData->brightness_[endIdx];
133
134 Vec3f midPoint = (startPos + endPos) * 0.5f;
135 Vec3f direction = endPos - startPos;
136 direction.normalize();
137 midPoint += getPerpendicular(direction) * (offsetAmount * (math::random<float>() * 2.0f - 1.0f));
138 updateSegments[nextIndex]->push_back(
139 startPos, midPoint,
140 strikeIdx_,
141 startBrightness);
142 updateSegments[nextIndex]->push_back(
143 midPoint, endPos,
144 strikeIdx_,
145 endBrightness);
146
147 direction = midPoint - startPos;
148 // with some probability create a new branch.
149 if (numRemainingVertices > numSubBranchVertices &&
150 math::random<float>() < branchProbability_ *
151 // decrease the probability for sub-branches
152 std::min(1.0f, startBrightness + 0.25f) *
153 // decrease the probability further away from the target
154 std::min(1.0f, (1.0f - (midPoint.y - target.y) / heightRange) + 0.25f)) {
155 updateSegments[nextIndex]->push_back(
156 midPoint,
157 branch(midPoint, direction, endPos,
158 offsetAmount * branchOffset_,
159 branchLength_),
160 strikeIdx_,
161 // decrease brightness for sub-branches
162 startBrightness * branchDarkening_);
163 numRemainingVertices -= numSubBranchVertices;
164 }
165 }
166
167 updateSegments[lastIndex]->clear();
168 nextIndex = lastIndex;
169 lastIndex = 1 - lastIndex;
170 offsetAmount *= 0.5f;
171 }
172
173 const uint32_t nextDrawIdx = updateIdx[lastIndex];
174 drawIndex_.store(nextDrawIdx, std::memory_order_release);
175}

177bool LightningStrike::updateStrike(double dt_s) {
178 u_time_ += dt_s;
179
180 if (active_) {
181 static const float maxAlpha_ = 1.0f;
182
183 if (u_time_ >= u_maxLifetime_) {
184 u_time_ = 0.0;
185 active_ = false;
186 u_nextStrike_ = frequencyConfig_.x + frequencyConfig_.y * (math::random<float>() * 2.0f - 1.0f);
187 alpha_->setVertex(strikeIdx_, 0.0f);
188 } else {
189 float fade = (1.0f - static_cast<float>(u_time_ / u_maxLifetime_)) * maxAlpha_;
190 fade = std::min(1.0f, std::max(0.0f, fade));
191 alpha_->setVertex(strikeIdx_, fade);
192 }
193 return true;
194 }
195 else if (u_time_ >= u_nextStrike_) {
196 active_ = true;
197 u_time_ = 0.0;
198 u_maxLifetime_ = lifetimeConfig_.x + lifetimeConfig_.y * (math::random<float>() * 2.0f - 1.0f);
199 if (source_.randomizeInstance) {
200 source_.instance = static_cast<uint32_t>(math::randomInt() % source_.numInstances);
201 }
202 if (target_.randomizeInstance) {
203 target_.instance = static_cast<uint32_t>(math::randomInt() % target_.numInstances);
204 }
205 alpha_->setVertex(strikeIdx_, 0.0f);
206 updateSegmentData();
207 return true;
208 } else {
209 return false;
210 }

300static std::vector<StrikeDescr> getStrikePoints(LoadingContext &ctx, const ref_ptr<scene::SceneInputNode> &input) {
301 std::vector<StrikeDescr> points;
302 if (input->hasAttribute("ring-tf")) {
303 // Create a ring of strikes ranging over all instances
304 LightningStrike::StrikePoint ringPoint = getStrikePoint_tf(ctx, input, "ring-tf");
305 const uint32_t ringStrength = input->getValue<unsigned int>("ring-strength", 1u);
306 for (uint32_t ringIdx= 0u; ringIdx < ringStrength; ++ringIdx) {
307 for (uint32_t instance=0u; instance < ringPoint.numInstances; ++instance) {
308 auto source = ringPoint;
309 auto target = ringPoint;
310 source.instance = instance;
311 target.instance = (instance + 1) % ringPoint.numInstances;
312 points.emplace_back(source, target);
313 }
314 }
315 } else {
316 auto source = getStrikeSource(ctx, input);
317 auto target = getStrikeTarget(ctx, input);
318 if (input->hasAttribute("source-instance")) {
319 selectTargetInstance(source, target, input);
320 points.emplace_back(source, target);
321 } else {
322 uint32_t maxInstances = source.numInstances;
323 if (input->hasAttribute("source-max-instances")) {
324 maxInstances = std::min(maxInstances,
325 input->getValue<unsigned int>("source-max-instances", maxInstances));
326 }
327 for (uint32_t instance=0u; instance < maxInstances; ++instance) {
328 source.instance = instance;
329 selectTargetInstance(source, target, input);
330 points.emplace_back(source, target);
331 }
332 }
333 }
334 return points;

396void LightningBolt::createResources() {
397 // allocate vertex data for max number of segments and branches,
398 // particular bolts will have less segments and branches.
399 unsigned int numSegments = 0;
400 unsigned int numVertices = 0;
401 for (auto &strike : strikes_) {
402 numSegments = (static_cast<int>(pow(2, strike->maxSubDivisions_)) * strike->maxBranches_);
403 numVertices += numSegments * 2;
404 }
405 REGEN_INFO("Allocating LightningBolt with " <<
406 numVertices << " vertices for " << strikes_.size() << " strikes.");
407
408 {
409 pos_->setVertexData(numVertices);
410 brightness_->setVertexData(numVertices);
411 strikeIdx_->setVertexData(numVertices);
412
413 setInput(pos_);
414 setInput(brightness_);
415 setInput(strikeIdx_);
416 updateVertexData();
417
418 // create single LOD level
419 auto &lod = meshLODs_.emplace_back();
420 lod.d->numVertices = numVertices;
421 lod.d->vertexOffset = 0;
422 }
423
424 {
425 std::vector<float> alphas(strikes_.size(), 0.0f);
426 std::vector<float> widths(strikes_.size());
427 for (size_t i = 0; i < strikes_.size(); ++i) {
428 widths[i] = strikes_[i]->widthFactor_;
429 }
430
431 strikeAlpha_->set_numArrayElements(strikes_.size());
432 strikeAlpha_->set_forceArray(true);

434
435 strikeWidth_->set_numArrayElements(strikes_.size());
436 strikeWidth_->set_forceArray(true);
437 strikeWidth_->setUniformUntyped((byte*)widths.data());

440 }

455void LightningBolt::gpuUpdate(RenderState *rs, double dt) {
456 if (!isActive_) return;
457
458 const uint32_t maxVertices = pos_->numVertices();
459 uint32_t numVertices = 0u;
460
461 auto m_pos = pos_->mapClientDataRaw(BUFFER_GPU_WRITE);
462 auto m_brightness = brightness_->mapClientDataRaw(BUFFER_GPU_WRITE);
463 auto m_strikeIdx = strikeIdx_->mapClientDataRaw(BUFFER_GPU_WRITE);
464
465 auto* posData = (Vec3f*)m_pos.w;
466 auto* brightnessData = (float*)m_brightness.w;
467 auto* strikeIdxData = (uint32_t*)m_strikeIdx.w;
468
469 for (auto &strike : strikes_) {
470 // Load the index of last updated segment.
471 // NOTE: This is safe because draw and update thread are frame-synced, i.e. update will swap
472 // the draw idx only after writing, and won't write again until draw is definitely done.
473 // Draw will either use update data of last frame, or if it is slower it may also pick
474 // up the data just written, which is also fine.
475 const uint32_t drawIdx = strike->drawIndex_.load(std::memory_order_acquire);
476 LightningStrike::SegmentData &segment = strike->segments_[drawIdx];
477 const uint32_t strikeVertices = segment.pos_.size();
478 if (strikeVertices == 0 || numVertices + strikeVertices > maxVertices) { continue; }
479
480 // Copy position data
481 std::memcpy(
482 posData + numVertices,
483 segment.pos_.data(),
484 strikeVertices * pos_->vertexSize());
485
486 // Copy brightness data
487 std::memcpy(
488 brightnessData + numVertices,
489 segment.brightness_.data(),
490 strikeVertices * brightness_->vertexSize());
491
492 // Copy strikeIdx data
493 std::memcpy(
494 strikeIdxData + numVertices,
495 segment.strikeIdx_.data(),
496 strikeVertices * strikeIdx_->vertexSize());
497
498 numVertices += strikeVertices;
499 }
500
501 m_pos.unmap();
502 m_brightness.unmap();
503 m_strikeIdx.unmap();
504
505 // update the number of vertices
506 set_numVertices(numVertices);
507 if (!meshLODs_.empty()) {
508 meshLODs_.front().d->numVertices = numVertices;
509 }
