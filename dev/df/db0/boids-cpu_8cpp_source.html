201void BoidsCPU::initBoidSimulation() {
202 const auto maxNumNeighbors = maxNumNeighbors_->getVertex(0).r;
203
204 setAnimationName("boids");
205 // use a dedicated thread for the boids simulation which is not synchronized with the graphics thread,
206 // i.e. it can be slower or faster than the graphics thread.
207 setSynchronized(false);
208 priv_->baseOrientation_ = baseOrientation_->getVertex(0).r;
209 priv_->yawAdjust_.setAxisAngle(Vec3f(0, 1, 0), priv_->baseOrientation_);
210 priv_->boidsScale_ = boidsScale_->getVertex(0).r;
211
212 // Create slices for multithreading
213 if constexpr (BOID_USE_MULTITHREADING) {
214 JobPool& pool = threading::getJobPool();
215 const uint32_t numWorkerThreads = pool.numThreads();
216 // Compute the number of instances in each slice.
217 const uint32_t numSlices = std::min(numBoids_, numWorkerThreads + 1);
218 priv_->boidSlices_.resize(numSlices);
219 priv_->sliceSize_ = (numBoids_ + numSlices - 1) / numSlices;
220 // Make sure each slice starts aligned with simd RegisterWidth
221 priv_->sliceSize_ = (priv_->sliceSize_ + simd::RegisterWidth - 1) & ~(simd::RegisterWidth - 1);
222
223 auto &firstSlice = priv_->boidSlices_[0];
224 firstSlice.sim = this;
225 firstSlice.priv = priv_;
226 firstSlice.frame.startIdx = 0;
227 firstSlice.frame.endIdx = std::min(priv_->sliceSize_, numBoids_);
228
229 for (uint32_t i = 1; i < numSlices; ++i) {
230 auto &slice = priv_->boidSlices_[i];
231 slice.sim = this;
232 slice.priv = priv_;
233 slice.frame.startIdx = i * priv_->sliceSize_;
234 slice.frame.endIdx = std::min(slice.frame.startIdx + priv_->sliceSize_, numBoids_);
235 }
236
237 // resize slice buffers
238 for (auto &slice : priv_->boidSlices_) {
239 slice.frame.resize(maxNumNeighbors + 1);
240 }
241 } else {
242 // Only allocate one slice for single-threaded execution.
243 auto &firstSlice = priv_->boidSlices_.emplace_back();
244 firstSlice.sim = this;
245 firstSlice.priv = priv_;
246 firstSlice.frame.startIdx = 0;
247 firstSlice.frame.endIdx = numBoids_;
248 firstSlice.frame.resize(maxNumNeighbors + 1);
249 priv_->sliceSize_ = numBoids_;
250 }
251
252 const auto numCells = priv_->gridSize_.x * priv_->gridSize_.y * priv_->gridSize_.z;
253
254 // Initialize grid memory.
255 for (auto &cell : priv_->grid_) {
256 cell.elements.resize(maxNumNeighbors + 1); // +1 to avoid branching
257 }
258 priv_->cellCounts_.resize(numCells);
259
260 // Initialize per-boid memory.
261 for (uint32_t i = 0; i < numBoids_; ++i) {
262 // note: an additional element is added to the end of the vector which is used
263 // to avoid branching.
264 priv_->boidNeighbors_[i].resize(maxNumNeighbors + 1);
265 priv_->boidNumNeighbors_[i] = 0;
266 }
267
268 animationState()->setInput(coherenceWeight_);
269 animationState()->setInput(alignmentWeight_);
270 animationState()->setInput(separationWeight_);
271 animationState()->setInput(avoidanceWeight_);
272 animationState()->setInput(avoidanceDistance_);
273 animationState()->setInput(visualRange_);
274 animationState()->setInput(lookAheadDistance_);
275 animationState()->setInput(repulsionFactor_);
276 animationState()->setInput(maxNumNeighbors_);
277 animationState()->setInput(maxBoidSpeed_);
278 animationState()->setInput(maxAngularSpeed_);
279 REGEN_INFO("CPU Boids simulation with " << numBoids_ << " boids");
280}

315void BoidsCPU::cpuUpdate(double dt) {
316 #ifdef REGEN_BOID_DEBUG_TIME
317 thread_local ElapsedTimeDebugger elapsedTime("BoidsCPU Update",
318 1000, ElapsedTimeDebugger::CPU_ONLY);
319 elapsedTime.beginFrame();
320 elapsedTime.push("starting");
321 #endif
322 const auto dt_f = static_cast<float>(dt) * 0.001f;
323 priv_->simBounds_.min = simulationBoundsMin_->getVertex(0).r;
324 priv_->simBounds_.max = simulationBoundsMax_->getVertex(0).r;
325 priv_->avoidanceDistance_ = avoidanceDistance_->getVertex(0).r;
326 priv_->avoidanceDistanceSq_ = priv_->avoidanceDistance_ * priv_->avoidanceDistance_;
327 priv_->avoidanceDistanceHalf_ = priv_->avoidanceDistance_ * 0.5f;
328 priv_->repulsionTimesSeparation_ = repulsionFactor_->getVertex(0).r * separationWeight_->getVertex(0).r;
329 priv_->visualRange_ = visualRange_->getVertex(0).r;
330 priv_->visualRangeSq_ = priv_->visualRange_ * priv_->visualRange_;
331 priv_->attractionRange_ = attractionRange_->getVertex(0).r;
332 priv_->separationWeight_ = separationWeight_->getVertex(0).r;
333 priv_->alignmentWeight_ = alignmentWeight_->getVertex(0).r;
334 priv_->coherenceWeight_ = coherenceWeight_->getVertex(0).r;
335 priv_->lookAheadDistance_ = lookAheadDistance_->getVertex(0).r;
336 priv_->maxBoidSpeed_ = maxBoidSpeed_->getVertex(0).r;
337 priv_->maxAngularSpeed_ = maxAngularSpeed_->getVertex(0).r;
338 priv_->cos_maxAngularSpeed_ = cosf(priv_->maxAngularSpeed_);
339 priv_->maxNumNeighbors_ = maxNumNeighbors_->getVertex(0).r;
340 priv_->cellSize_ = cellSize_->getVertex(0).r;
341 // update dt in all slices
342 for (auto &slice : priv_->boidSlices_) {
343 slice.frame.dt = dt_f;
344 }
345
346 Bounds<Vec3f> bounds = Bounds<Vec3f>::create(Vec3f::posMax(), Vec3f::negMax());
347 if constexpr (BOID_USE_MULTITHREADING) {
348 priv_->performSlicedJob<Private::updateForceJob>();
349 priv_->performSlicedJob<Private::advanceBoidJob>();
350 // Accumulate the new bounds from all slices
351 for (const auto &slice : priv_->boidSlices_) {
352 bounds.extend(slice.frame.localBounds);
353 }
354 }
355 else {
356 // Single-threaded update
357 const auto numBoids = numBoids_;
358 auto &slice = priv_->boidSlices_.front();
359 priv_->updateForces(slice);
360
361 for (uint32_t boidIdx = 0; boidIdx < numBoids; ++boidIdx) {
362 advanceBoid(boidIdx, dt_f);
363 // update the grid bounds
364 bounds.min.x = std::min(bounds.min.x, priv_->boidPositionsX_[boidIdx]);
365 bounds.min.y = std::min(bounds.min.y, priv_->boidPositionsY_[boidIdx]);
366 bounds.min.z = std::min(bounds.min.z, priv_->boidPositionsZ_[boidIdx]);
367 bounds.max.x = std::max(bounds.max.x, priv_->boidPositionsX_[boidIdx]);
368 bounds.max.y = std::max(bounds.max.y, priv_->boidPositionsY_[boidIdx]);
369 bounds.max.z = std::max(bounds.max.z, priv_->boidPositionsZ_[boidIdx]);
370 }
371 }
372 boidBounds_ = bounds;
373 #ifdef REGEN_BOID_DEBUG_TIME
374 elapsedTime.push("simulation");
375 #endif
376
377 // update boids model transformation using the boids data
378 updateTransforms();
379 #ifdef REGEN_BOID_DEBUG_TIME
380 elapsedTime.push("transforms");
381 #endif
382
383 // resize the grid, and clear all cells
384 clearGrid();
385 if (priv_->grid_.empty()) { return; }
386
387 // Update boidGridIndex_: per boid grid cell index
388 updateCellIndex();
389 // add boids to the grid and compute their neighborhood relations.
390 insertIntoCells();
391 #ifdef REGEN_BOID_DEBUG_TIME
392 elapsedTime.push("binning");
393 #endif
394
395 // Update neighbors of each boid (this is expensive)
396 if constexpr (BOID_USE_MULTITHREADING) {
397 priv_->performSlicedJob<Private::updateNeighborsJob>();
398 } else {
399 priv_->updateNeighbours(priv_->boidSlices_.front());
400 }
401 #ifdef REGEN_BOID_DEBUG_TIME
402 elapsedTime.push("neighbours");
403 elapsedTime.endFrame();
404 #endif
405}

501template <void(*Fn)(void*)>
502void BoidsCPU::Private::performSlicedJob() {
503 JobPool& pool = threading::getJobPool();
504 const uint32_t numSlices = boidSlices_.size();
505 auto &jobFrame = jobFrame_;
506
507 jobFrame.numPushed = 0;
508
509 for (uint32_t i = 1; i < numSlices; ++i) {
510 auto &slice = boidSlices_[i];
511 pool.addJobPreFrame(jobFrame, Job{ .fn = Fn, .arg = &slice });
512 }
513
514 // Execute jobs
515 pool.beginFrame(jobFrame, 1u); // one local job
516 FramedJob localJob = { { Fn, &boidSlices_[0] }, &jobFrame };
517 do {
518 JobPool::performJob(localJob);
519 } while (pool.stealJob(localJob));
520 JobPool::endFrame(jobFrame);
521}

523void BoidsCPU::updateCellIndex() {
524 // NOTE: This could easily be done in parallel if needed.
525 // But it seems superfast already, but in case it becomes a bottleneck,
526 // we can parallelize it using multi threading.
527 const auto numBoids = numBoids_;
528
529 const auto* __restrict boidPosX = static_cast<float*>(__builtin_assume_aligned(priv_->boidPositionsX_.data(), 32));
530 const auto* __restrict boidPosY = static_cast<float*>(__builtin_assume_aligned(priv_->boidPositionsY_.data(), 32));
531 const auto* __restrict boidPosZ = static_cast<float*>(__builtin_assume_aligned(priv_->boidPositionsZ_.data(), 32));
532 auto* __restrict boidGridIndex = static_cast<uint32_t*>(__builtin_assume_aligned(priv_->boidGridIndex_.data(), 32));
533
534 uint32_t boidIdx = 0u;
535
536 if constexpr (BOID_USE_SIMD) {
537 const BatchOf_Vec3f gridMin = BatchOf_Vec3f::fromScalar(gridBounds_.min);
538 const BatchOf_Vec3i gridSize = BatchOf_Vec3i::fromScalar(priv_->gridSize_);
539 const BatchOf_int32 allOne = BatchOf_int32::fromScalar(1);
540 const BatchOf_float allZero = BatchOf_float::allZeros();
541 const BatchOf_float simd_cellSize = BatchOf_float::fromScalar(priv_->cellSize_);
542
543 BatchOf_Vec3f boidBatch; // NOLINT(cppcoreguidelines-pro-type-member-init)
544 // we compute the grid index in batches
545 for (; boidIdx + simd::RegisterWidth <= numBoids; boidIdx += simd::RegisterWidth) {
546 // load the boid positions into a SIMD register
547 boidBatch.setAligned(
548 boidPosX + boidIdx,
549 boidPosY + boidIdx,
550 boidPosZ + boidIdx);
551 // x = (x - gridBounds_.min) / cellSize
552 boidBatch = (boidBatch - gridMin) / simd_cellSize;
553 // clamp to 0+
554 boidBatch.x = BatchOf_float::max(boidBatch.x, allZero);
555 boidBatch.y = BatchOf_float::max(boidBatch.y, allZero);
556 boidBatch.z = BatchOf_float::max(boidBatch.z, allZero);
557
558 // floor to integer grid indices using cvttps_epi32 + clamp to max grid bounds
559 BatchOf_Vec3i gridIndices = boidBatch.floor().min(gridSize - allOne);
560 // i_f = ix + iy_f + iz_f;
561 // - iy_f = iy * gridSize.x
562 // - iz_f = iz * gridSize.x * gridSize.y;
563 const BatchOf_int32 i_f = (gridIndices.x +
564 (gridIndices.y * gridSize.x) +
565 (gridIndices.z * gridSize.x * gridSize.y));
566 // store results in local array
567 i_f.storeAligned(boidGridIndex + boidIdx);
568 }
569 }
570
571 // Fallback to scalar loop for grid index computation
572 for (; boidIdx < numBoids; ++boidIdx) {
573 const Vec3f boidPos = Vec3f::max(
574 (getBoidPosition(boidIdx) - gridBounds_.min) / priv_->cellSize_,
575 // ensure the boid position is within the grid bounds
576 Vec3f::zero());
577 const Vec3i gridIndex{
578 static_cast<int>(std::trunc(boidPos.x)),
579 static_cast<int>(std::trunc(boidPos.y)),
580 static_cast<int>(std::trunc(boidPos.z))};
581 // ensure the boid position is within the grid bounds

584 boidGridIndex[boidIdx] = getGridIndex(idx3D, priv_->gridSize_);

625void BoidsCPU::Private::updateNeighbours(BoidSimulationFrame &frame, uint32_t boidIdx, uint32_t cellIdx) {
626 const auto* __restrict globalX = static_cast<float*>(__builtin_assume_aligned(boidPositionsX_.data(), 32));
627 const auto* __restrict globalY = static_cast<float*>(__builtin_assume_aligned(boidPositionsY_.data(), 32));
628 const auto* __restrict globalZ = static_cast<float*>(__builtin_assume_aligned(boidPositionsZ_.data(), 32));
629 const auto* __restrict cellCounts = static_cast<const uint32_t*>(__builtin_assume_aligned(cellCounts_.data(), 32));
630 auto* __restrict queueX = static_cast<float*>(__builtin_assume_aligned(frame.queuePosX_.data(), 32));
631 auto* __restrict queueY = static_cast<float*>(__builtin_assume_aligned(frame.queuePosY_.data(), 32));
632 auto* __restrict queueZ = static_cast<float*>(__builtin_assume_aligned(frame.queuePosZ_.data(), 32));
633 auto* __restrict boidQueue = static_cast<uint32_t*>(__builtin_assume_aligned(frame.boidQueue_.data(), 32));
634 auto* __restrict neighbors = boidNeighbors_[boidIdx].data();
635
636 const uint32_t neighborCount = cellCounts[cellIdx];
637 const auto* __restrict neighborIndices = grid_[cellIdx].elements.data();
638
639 const float boidPosX = globalX[boidIdx];
640 const float boidPosY = globalY[boidIdx];
641 const float boidPosZ = globalZ[boidIdx];
642 const uint32_t maxNumNeighbors = maxNumNeighbors_;
643
644 uint32_t numHits = 0;
645 uint32_t queueIdx = 0;
646
647 // Load the neighbor positions into local SOA arrays
648 for (uint32_t i = 0; i < neighborCount; ++i) {
649 const uint32_t neighborIdx = neighborIndices[i];
650 queueX[i] = globalX[neighborIdx];
651 queueY[i] = globalY[neighborIdx];
652 queueZ[i] = globalZ[neighborIdx];
653 }
654
655 if constexpr (BOID_USE_SIMD) {
656 const BatchOf_Vec3f b_boidPos{
657 BatchOf_float::fromScalar(boidPosX),
658 BatchOf_float::fromScalar(boidPosY),
659 BatchOf_float::fromScalar(boidPosZ) };
660 const BatchOf_float b_visualRangeSq = BatchOf_float::fromScalar(visualRangeSq_);
661
662 for (; queueIdx + simd::RegisterWidth <= neighborCount; queueIdx += simd::RegisterWidth) {
663 // load the positions of the neighbors into a SIMD register
664 const BatchOf_Vec3f candidatePos = BatchOf_Vec3f::loadAligned(
665 queueX + queueIdx,
666 queueY + queueIdx,
667 queueZ + queueIdx);
668
669 // finally compute the distance to the boid position for a batch of neighbors
670 const BatchOf_float lengthSq = (candidatePos - b_boidPos).lengthSquared();
671 const BatchOf_float mask = (lengthSq < b_visualRangeSq);
672
673 // Convert lanes (1/0) to bitmask value
674 uint8_t maskBits = mask.toBitmask8();
675 while (maskBits) {
676 int bitIndex = simd::nextBitIndex<uint8_t>(maskBits);
677 boidQueue[numHits++] = queueIdx + bitIndex;
678 }
679 }
680 }
681
682 // Fallback to scalar loop for remaining elements
683 for (; queueIdx < neighborCount; queueIdx++) {
684 const Vec3f dx {
685 queueX[queueIdx] - boidPosX,
686 queueY[queueIdx] - boidPosY,
687 queueZ[queueIdx] - boidPosZ };
688 if (dx.lengthSquared() <= visualRangeSq_) {
689 boidQueue[numHits++] = queueIdx;
690 }
691 }
692
693 // Process the found neighbors
694 uint32_t numNeighbors = 0u;
695 for (uint32_t hitIdx = 0;
696 hitIdx < numHits &&
697 numNeighbors < maxNumNeighbors; ++hitIdx) {
698 const uint32_t neighborIdx = neighborIndices[boidQueue[hitIdx]];
699 if (neighborIdx == boidIdx) [[unlikely]] continue; // skip self
700 neighbors[numNeighbors++] = neighborIdx;
701 }
702 boidNumNeighbors_[boidIdx] = numNeighbors;
703}

714void BoidsCPU::Private::updateForces(const BoidSimulationFrame &frame, uint32_t boidIdx) {
715 const uint32_t numNeighbors = boidNumNeighbors_[boidIdx];
716 const float alpha = 1.0f / static_cast<float>(std::max(numNeighbors,1u));
717 const float boidPosX = boidPositionsX_[boidIdx];
718 const float boidPosY = boidPositionsY_[boidIdx];
719 const float boidPosZ = boidPositionsZ_[boidIdx];
720
721 auto* __restrict queuePosX = static_cast<float*>(__builtin_assume_aligned(frame.queuePosX_.data(), 32));
722 auto* __restrict queuePosY = static_cast<float*>(__builtin_assume_aligned(frame.queuePosY_.data(), 32));
723 auto* __restrict queuePosZ = static_cast<float*>(__builtin_assume_aligned(frame.queuePosZ_.data(), 32));
724 auto* __restrict queueVelX = static_cast<float*>(__builtin_assume_aligned(frame.queueVelX_.data(), 32));
725 auto* __restrict queueVelY = static_cast<float*>(__builtin_assume_aligned(frame.queueVelY_.data(), 32));
726 auto* __restrict queueVelZ = static_cast<float*>(__builtin_assume_aligned(frame.queueVelZ_.data(), 32));
727 const auto* __restrict globalPosX = static_cast<const float*>(__builtin_assume_aligned(boidPositionsX_.data(), 32));
728 const auto* __restrict globalPosY = static_cast<const float*>(__builtin_assume_aligned(boidPositionsY_.data(), 32));
729 const auto* __restrict globalPosZ = static_cast<const float*>(__builtin_assume_aligned(boidPositionsZ_.data(), 32));
730 const auto* __restrict globalVelX = static_cast<const float*>(__builtin_assume_aligned(boidVelocityX_.data(), 32));
731 const auto* __restrict globalVelY = static_cast<const float*>(__builtin_assume_aligned(boidVelocityY_.data(), 32));
732 const auto* __restrict globalVelZ = static_cast<const float*>(__builtin_assume_aligned(boidVelocityZ_.data(), 32));
733 const auto* __restrict neighborIndices = boidNeighbors_[boidIdx].data();
734
735 // Load the neighbor positions into local SOA arrays
736 for (uint32_t i = 0; i < numNeighbors; ++i) {
737 const uint32_t neighborIdx = neighborIndices[i];
738 queuePosX[i] = globalPosX[neighborIdx];
739 queuePosY[i] = globalPosY[neighborIdx];
740 queuePosZ[i] = globalPosZ[neighborIdx];
741 queueVelX[i] = globalVelX[neighborIdx];
742 queueVelY[i] = globalVelY[neighborIdx];
743 queueVelZ[i] = globalVelZ[neighborIdx];
744 }
745
746 Vec3f boidSum;
747 const float aaw = alpha * alignmentWeight_;
748 const float acw = alpha * coherenceWeight_;
749 const float sw = separationWeight_;
750 uint32_t queueIdx = 0;
751
752 if constexpr (BOID_USE_SIMD) {
753 const BatchOf_Vec3f fixedPos {
754 BatchOf_float::fromScalar(boidPosX),
755 BatchOf_float::fromScalar(boidPosY),
756 BatchOf_float::fromScalar(boidPosZ) };
757 const BatchOf_float avoidance = BatchOf_float::fromScalar(avoidanceDistanceSq_);
758 const BatchOf_float v_aaw = BatchOf_float::fromScalar(aaw);
759 const BatchOf_float v_acw = BatchOf_float::fromScalar(acw);
760 const BatchOf_float v_sw = BatchOf_float::fromScalar(sw);
761
762 BatchOf_Vec3f boidSumVec = BatchOf_Vec3f::fromScalar(Vec3f::zero());
763
764 for (; queueIdx + simd::RegisterWidth <= numNeighbors; queueIdx += simd::RegisterWidth) {
765 // ---- Phase 1: velocity term ----
766 {
767 const BatchOf_Vec3f vel = BatchOf_Vec3f::loadAligned(
768 queueVelX + queueIdx,
769 queueVelY + queueIdx,
770 queueVelZ + queueIdx);
771 boidSumVec += vel * v_aaw;
772 }
773
774 // ---- Phase 2: cohesion + separation ----
775 {
776 const BatchOf_Vec3f pos = BatchOf_Vec3f::loadAligned(
777 queuePosX + queueIdx,
778 queuePosY + queueIdx,
779 queuePosZ + queueIdx);
780 // Accumulate position for the cohesion term.
781 boidSumVec += pos * v_acw;
782
783 // Compute vector from neighbor (x) to boid (fixedPos) for the separation term.
784 const BatchOf_Vec3f sep = fixedPos - pos;
785 // Compute distance squared which is used for weighting the separation force
786 // (closer neighbors contribute more).
787 BatchOf_float distSq = sep.lengthSquared();
788 // Use masking to only consider neighbors within the avoidance distance.
789 //distSq = (distSq < avoidance).maskToFloat() / (distSq + 0.0001f);
790 distSq = (distSq < avoidance).maskToFloat() * (distSq + 0.0001f).reciprocal();
791 // Finally, accumulate the separation force.
792 boidSumVec += sep * distSq * v_sw;
793 }
794 }
795 // Horizontal sum of SIMD registers to get final results
796 boidSum = boidSumVec.hsum();
797 } else {
798 boidSum = Vec3f::zero();
799 }
800
801 // Fallback to scalar loop for remaining elements
802 Vec3f scalarPosSum = Vec3f::zero();
803 Vec3f scalarVelSum = Vec3f::zero();
804 Vec3f scalarSepSum = Vec3f::zero();
805 for (; queueIdx < numNeighbors; queueIdx++) {
806 scalarPosSum.x += queuePosX[queueIdx];
807 scalarPosSum.y += queuePosY[queueIdx];
808 scalarPosSum.z += queuePosZ[queueIdx];
809
810 const Vec3f boidDirection {
811 boidPosX - queuePosX[queueIdx],
812 boidPosY - queuePosY[queueIdx],
813 boidPosZ - queuePosZ[queueIdx] };
814 const float dSq = boidDirection.lengthSquared();
815 const auto mask = static_cast<float>(dSq < avoidanceDistanceSq_);
816 scalarSepSum += boidDirection * mask / (dSq + 0.0001f); // avoid division by zero
817
818 scalarVelSum.x += queueVelX[queueIdx];
819 scalarVelSum.y += queueVelY[queueIdx];
820 scalarVelSum.z += queueVelZ[queueIdx];
821 }
822
823 const auto boidPos = Vec3f{ boidPosX, boidPosY, boidPosZ };
824 const auto boidVel = Vec3f{
825 boidVelocityX_[boidIdx],
826 boidVelocityY_[boidIdx],
827 boidVelocityZ_[boidIdx] };
828 boidForce_[boidIdx] = boidSum +
829 scalarPosSum * acw +
830 scalarVelSum * aaw +
831 scalarSepSum * sw -
832 boidVel*alignmentWeight_ -
833 boidPos*coherenceWeight_;
834}

836void BoidsCPU::Private::advanceBoid(BoidSliceData &slice) {
837 const auto startIdx = slice.frame.startIdx;
838 const auto endIdx = slice.frame.endIdx;
839 const auto dt = slice.frame.dt;
840 auto &sim = *slice.sim;
841
842 // Each frame, we compute new bounds. In a threaded setting
843 // each thread must compute local bounds, these are later consolidated
844 // into the global bounds.
845 Bounds<Vec3f> &localBounds = slice.frame.localBounds;
846 localBounds.min = Vec3f::posMax();
847 localBounds.max = Vec3f::negMax();
848
849 for (uint32_t boidIdx = startIdx; boidIdx < endIdx; ++boidIdx) {
850 sim.advanceBoid(boidIdx, dt);
851 // update the local bounds
852 localBounds.min.x = std::min(localBounds.min.x, boidPositionsX_[boidIdx]);
853 localBounds.min.y = std::min(localBounds.min.y, boidPositionsY_[boidIdx]);
854 localBounds.min.z = std::min(localBounds.min.z, boidPositionsZ_[boidIdx]);
855 localBounds.max.x = std::max(localBounds.max.x, boidPositionsX_[boidIdx]);
856 localBounds.max.y = std::max(localBounds.max.y, boidPositionsY_[boidIdx]);
857 localBounds.max.z = std::max(localBounds.max.z, boidPositionsZ_[boidIdx]);
858 }
859}

861void BoidsCPU::advanceBoid(uint32_t boidIdx, float dt) {
862 const uint32_t numNeighbors = priv_->boidNumNeighbors_[boidIdx];
863 const Vec3f boidPos = getBoidPosition(boidIdx);
864 Vec3f boidVel = getBoidVelocity(boidIdx);
865 Vec3f boidForce = priv_->boidForce_[boidIdx];
866
867 // a boid is lost if it is outside the bounds
868 const bool isBoidLost = (numNeighbors == 0 || !priv_->simBounds_.contains(boidPos));
869 const bool isInCollision = avoidCollisions(boidPos, boidVel, boidForce, dt);
870 const bool isInDanger = !dangers_.empty() && avoidDanger(boidPos, boidForce);
871
872 if (!isInDanger) {
873 // note: ignore attractors if in danger
874 attract(boidPos, boidForce);
875 }
876 if (isBoidLost || isInCollision) {
877 // drift towards home if lost or in collision
878 homesickness(boidPos, boidForce);
879 }
880
881 const auto dx = boidVel * dt;
882 priv_->boidPositionsX_[boidIdx] += dx.x;
883 priv_->boidPositionsY_[boidIdx] += dx.y;
884 priv_->boidPositionsZ_[boidIdx] += dx.z;
885 auto lastVelNorm = boidVel;
886 lastVelNorm.normalize();
887
888 float boidSpeed = 0.0f;
889 boidVel = limitVelocity(
890 lastVelNorm,
891 boidVel + boidForce * dt,
892 boidSpeed);
893 setBoidVelocity(boidIdx, boidVel);
894 if (boidSpeed > 0.001f) {
895 Quaternion boidRotation;
896 boidRotation.setLookRotation(boidVel / boidSpeed);
897 setBoidOrientation(boidIdx, boidRotation);
898 }
899}

901Vec3f BoidsCPU::limitVelocity(const Vec3f &lastVelNorm, const Vec3f &nextVel, float &boidSpeed) const {
902 boidSpeed = nextVel.length();
903 const Vec3f nextVelNorm = nextVel / boidSpeed;
904
905 // limit translation speed
906 boidSpeed = std::min(boidSpeed, priv_->maxBoidSpeed_);
907 Vec3f limitedVel = nextVelNorm * boidSpeed;
908
909 // limit angular speed
910 if (nextVelNorm.dot(lastVelNorm) < priv_->cos_maxAngularSpeed_) {
911 auto axis = nextVelNorm.cross(lastVelNorm);
912 axis.normalize();
913 Quaternion boidRotation;
914 boidRotation.setAxisAngle(axis, priv_->maxAngularSpeed_);
915 const auto newDirection = boidRotation.rotate(lastVelNorm);
916 limitedVel = newDirection * boidSpeed;
917 }
918
919 return limitedVel;
920}

922void BoidsCPU::homesickness(const Vec3f &boidPos, Vec3f &boidForce) const {
923 static constexpr float MaxFloat = std::numeric_limits<float>::max();
924 if (homePoints_.empty()) { return; }
925
926 // a boid seems to have lost track, and wants to go home!
927 // first find closest home point...
928 const Vec3f *closestHomePoint = &Vec3f::zero();
929 float minDistance = MaxFloat;
930 for (auto &home: homePoints_) {
931 auto distance = (boidPos - home).lengthSquared();
932 if (distance < minDistance) {
933 minDistance = distance;
934 closestHomePoint = &home;
935 }
936 }
937 const Vec3f boidDirection = (*closestHomePoint - boidPos) / sqrt(minDistance);
938 boidForce += boidDirection * priv_->repulsionTimesSeparation_ * 0.1;
939}

941bool BoidsCPU::avoidDanger(const Vec3f &boidPos, Vec3f &boidForce) const {
942 bool isInDanger = false;
943 Vec3f dir;
944
945 for (auto &danger: dangers_) {
946 dir = Vec3f::zero();
947 if (danger.pos.get()) {
948 dir = danger.pos.get()->getVertex(0).r;
949 }
950 if (danger.tf.get()) {
951 dir += danger.tf.get()->getVertex(0).r.position();
952 }
953 dir -= boidPos;
954 auto distance = dir.lengthSquared();
955 if (distance < priv_->visualRangeSq_) {
956 if (distance < 0.001f) {
957 dir = Vec3f::random();
958 dir.normalize();
959 } else {
960 dir /= sqrt(distance);
961 }
962 boidForce += dir * priv_->repulsionTimesSeparation_;
963 isInDanger = true;
964 }
965 }
966 return isInDanger;
967}

969void BoidsCPU::attract(const Vec3f &boidPos, Vec3f &boidForce) const {
970 auto maxDistance = priv_->attractionRange_;
971 Vec3f dir;
972 for (auto &attractor: attractors_) {
973 dir = Vec3f::zero();
974 if (attractor.pos.get()) {
975 dir = attractor.pos.get()->getVertex(0).r;
976 }
977 if (attractor.tf.get()) {
978 dir += attractor.tf.get()->getVertex(0).r.position();
979 }
980 dir -= boidPos;
981 auto distance = dir.length();
982 if (distance < maxDistance && distance > 0.001f) {
983 dir /= distance;
984 boidForce += dir * priv_->repulsionTimesSeparation_;
985 }
986 }

989bool BoidsCPU::avoidCollisions(
990 const Vec3f &boidPos,
991 const Vec3f &boidVel,
992 Vec3f &boidForce,
993 float dt) const {
994 auto nextVelocity = boidVel + boidForce * dt;
995 nextVelocity.normalize();
996 auto lookAhead = boidPos + nextVelocity * priv_->lookAheadDistance_;
997 bool isCollisionFree = true;
998
1002 if (lookAhead.x < priv_->simBounds_.min.x + priv_->avoidanceDistance_) {
1003 boidForce.x += priv_->repulsionTimesSeparation_;
1004 } else if (lookAhead.x > priv_->simBounds_.max.x - priv_->avoidanceDistance_) {
1005 boidForce.x -= priv_->repulsionTimesSeparation_;
1006 }
1007 if (lookAhead.y < priv_->simBounds_.min.y + priv_->avoidanceDistance_) {
1008 boidForce.y += priv_->repulsionTimesSeparation_;
1009 } else if (lookAhead.y > priv_->simBounds_.max.y - priv_->avoidanceDistance_) {
1010 boidForce.y -= priv_->repulsionTimesSeparation_;
1011 }
1012 if (lookAhead.z < priv_->simBounds_.min.z + priv_->avoidanceDistance_) {
1013 boidForce.z += priv_->repulsionTimesSeparation_;
1014 } else if (lookAhead.z > priv_->simBounds_.max.z - priv_->avoidanceDistance_) {
1015 boidForce.z -= priv_->repulsionTimesSeparation_;
1016 }
1017
1021 if (collisionMap_.get()) {
1022 auto uv0 = computeUV(boidPos, collisionMapCenter_, collisionMapSize_);
1023 uv0.x = math::clamp(uv0.x, 0.0f, 1.0f);
1024 uv0.y = math::clamp(uv0.y, 0.0f, 1.0f);
1025 auto uv1 = computeUV(lookAhead, collisionMapCenter_, collisionMapSize_);
1026 uv1.x = math::clamp(uv1.x, 0.0f, 1.0f);
1027 uv1.y = math::clamp(uv1.y, 0.0f, 1.0f);
1028
1029 if (collisionMapType_ == COLLISION_VECTOR_FIELD) {
1030 auto sample0 = collisionMap_->sampleLinear<Vec3f,3>(uv0, collisionMap_->textureData());
1031 auto sample1 = collisionMap_->sampleLinear<Vec3f,3>(uv1, collisionMap_->textureData());
1032 Vec2f collisionFlow = sample0.xy();
1033 float collisionStrength = sample0.z;
1034 if (sample1.z > collisionStrength) {
1035 collisionStrength = sample1.z;
1036 collisionFlow = sample1.xy();
1037 }
1038
1039 if (collisionStrength > 0.01f) {
1040 // Compute avoidance force proportional to strength
1041 float strength = priv_->repulsionTimesSeparation_ * collisionStrength * 3.0f;
1042 boidForce.x += collisionFlow.x * strength;
1043 boidForce.z += collisionFlow.y * strength;
1044 // dampen forward velocity slightly near obstacles
1045 boidForce -= nextVelocity * (strength * 0.2f);
1046 }
1047 } else {
1048 // Sample at current and lookahead positions
1049 float collisionValue = 0.0f;
1050 static constexpr int NumSamples = 3; // sample along velocity
1051 for (int i = 0; i < NumSamples; ++i) {
1052 float t = (float)i / (NumSamples - 1);
1053 Vec3f pos = boidPos + nextVelocity * t;
1054 auto uvSample = computeUV(pos, collisionMapCenter_, collisionMapSize_);
1055 uvSample.x = 1.0f - std::clamp(uvSample.x, 0.0f, 1.0f);
1056 uvSample.y = std::clamp(uvSample.y, 0.0f, 1.0f);
1057 collisionValue = std::max(collisionValue,
1058 collisionMap_->sampleLinear<float,1>(uvSample, collisionMap_->textureData()));
1059 }
1060
1061 if (collisionValue > 0.05f) {
1062 const float eps = 1.0f / float(collisionMap_->width());
1063 uv0.x = 1.0f - uv0.x; // HACK: flip x for sampling
1064 float c = collisionMap_->sampleLinear<float,1>(uv0, collisionMap_->textureData()); // central sample
1065 float dx = collisionMap_->sampleLinear<float,1>(uv0 + Vec2f(eps,0), collisionMap_->textureData())
1066 - collisionMap_->sampleLinear<float,1>(uv0 - Vec2f(eps,0), collisionMap_->textureData());
1067 float dy = collisionMap_->sampleLinear<float,1>(uv0 + Vec2f(0,eps), collisionMap_->textureData())
1068 - collisionMap_->sampleLinear<float,1>(uv0 - Vec2f(0,eps), collisionMap_->textureData());
1069
1070 Vec3f normal(dx, 0.0f, dy);
1071 float normalLength = normal.lengthSquared();
1072 if (normalLength > 1e-8f) normal /= sqrt(normalLength);
1073 else {
1074 Vec3f up(0,1,0);
1075 normal = nextVelocity.cross(up);
1076 normalLength = normal.lengthSquared();
1077 if (normalLength > 1e-8f) normal /= sqrt(normalLength);
1078 }
1079
1080 // use both the max along lookahead and the center sample to compute final strength
1081 float strength = priv_->repulsionTimesSeparation_ * std::clamp(0.75f * collisionValue + 0.25f * c, 0.0f, 1.0f);
1082
1083 boidForce -= normal * strength;
1084 boidForce -= nextVelocity * (strength * 0.3f);
1085 }
1086 }
1087 }
1088
1092 if (heightMap_.get()) {
1093 // boid position in height map space [0, 1]
1094 auto boidCoord = computeUV(boidPos, mapCenter_, mapSize_);
1095 auto currentY = heightMap_->sampleLinear<float,1>(boidCoord, heightMap_->textureData());
1096 currentY = currentY*heightMapFactor_ + mapCenter_.y + priv_->avoidanceDistanceHalf_;
1097
1098 // sample the height map at the projected boid position
1099 boidCoord = computeUV(lookAhead, mapCenter_, mapSize_);
1100 auto nextY = heightMap_->sampleLinear<float,1>(boidCoord, heightMap_->textureData());
1101 nextY = nextY*heightMapFactor_ + mapCenter_.y + priv_->avoidanceDistanceHalf_;
1102
1103 if (boidPos.y < currentY) {
1104 // The boid is below the minimum height of the height map -> push boid up.
1105 if (currentY < priv_->simBounds_.max.y - priv_->avoidanceDistanceHalf_) {
1106 // only push up in case the height map y is below the maximum y of boids
1107 boidForce.y += priv_->repulsionTimesSeparation_;
1108 }
1109 isCollisionFree = false;
1110 }
1111 if (lookAhead.y < nextY) {
1112 // Assuming the boid follows the lookAhead direction, it will eventually reach a point where
