52 void sort(std::vector<IndexType> &indices, const std::vector<KeyType> &keys) {
53 // The actual number of keys to sort for this sort.
54 // This can be less than the reserved maximum number of keys.
55 // NOTE: The size of keys and indices must not match, only requirement
56 // is that values in indices are valid indices into keys.
57 const uint32_t numKeys = indices.size();

76 // Unroll radix passes at compile time
77 template<int I, int N> constexpr void unrolledRadix(uint32_t numKeys,
78 IndexType *&src, IndexType *&dst, const KeyType *keys) {
79 if constexpr (I < N) {
80 // Perform a single radix pass.
81 // We do attempt to run some parts with SIMD if possible, the histogram
82 // counting can benefit from it easily.
83 // However, for huge gain we would need to load the histogram into
84 // SIMD registers as well, which is only practical with small radix width.
85 constexpr int SHIFT = I * BITS_PER_PASS;
86 // Limit bit width for last pass.
87 constexpr int BITS_THIS_PASS = (I+1 == N &&
88 (KEY_BITS_EFFECTIVE % BITS_PER_PASS != 0)) ?
89 (KEY_BITS_EFFECTIVE % BITS_PER_PASS) : BITS_PER_PASS;
90 constexpr int MASK = (1 << BITS_THIS_PASS) - 1;
91 std::ranges::fill(histogram_, 0);
92
93 // 1. Histogram: Count how many keys fall into each bucket
94 radixHistogram<SHIFT,MASK>(numKeys, src, keys);
95 // 2. Prefix sum: Compute the starting index for each bucket
96 size_t sum = 0;
97 for (int i = 0; i < BUCKETS; ++i) {
98 const uint32_t tmp = histogram_[i];
99 histogram_[i] = sum;
100 sum += tmp;
101 }
102 // 3. Scatter: Place each key in its corresponding bucket
103 for (uint32_t i = 0u; i < numKeys; ++i) {
104 const auto digit = (keys[src[i]] >> SHIFT) & MASK;
105 dst[histogram_[digit]++] = src[i];
106 }
107 // Swap buffers
108 std::swap(src, dst);
109 // Recurse to the next pass
110 unrolledRadix<I + 1, N>(numKeys, src, dst, keys);
111 }
112 }

114 template <int SHIFT, int MASK>
115 void radixHistogram(const uint32_t numKeys, IndexType *src, const KeyType *keys) {
116 // Histogram: Count how many keys fall into each bucket
117 // note: Depending on key type we need different API for SIMD unfortunately, as only
118 // 256-bit registers are available with AVX, so we can only load
119 // e.g. 8 uint32_t keys or 4 uint64_t keys in one go.
120 uint32_t keyIdx = 0;
121#ifdef REGEN_RADIX_SORT_CPU_USE_SIMD
122 {
123 // Load constant: (BUCKETS-1) across SIMD lane of 256-bit register
124 simd::Register_i mask;
125 if constexpr (KEY_TYPE_BITS == 16) {
126 mask = simd::set1_epi16(MASK);
127 } else if constexpr (KEY_TYPE_BITS == 32) {
128 mask = simd::set1_epi32(MASK);
129 } else if constexpr (KEY_TYPE_BITS == 64) {
130 mask = simd::set1_epi64u(MASK);
131 } else {
132 static_assert("Unsupported key size for SIMD radix sort");
133 return;
134 }
135
136 while (keyIdx+KEYS_PER_SIMD_PASS <= numKeys) {
137 simd::Register_i r0;
138
139 if constexpr (KEY_TYPE_BITS == 16) {
140 // Gather 8 keys manually, and promote to 32-bit
141 for (int k = 0; k < 8; ++k) tmpKeys32[k] = static_cast<int32_t>(keys[src[keyIdx+k]]);
142 r0 = _mm256_load_si256(reinterpret_cast<const __m256i*>(tmpKeys32));
143 r0 = _mm256_and_si256(_mm256_srli_epi32(r0, SHIFT), mask);
144 keyIdx += 8; // processed 8 keys, not 16!
145 }
146 else if constexpr (KEY_TYPE_BITS == 32) {
147 simd::Register_i idx = _mm256_loadu_si256(reinterpret_cast<const __m256i*>(&src[keyIdx]));
148 // Gather 8 scattered keys, and apply shift and mask to get bucket ids
149 r0 = _mm256_i32gather_epi32(reinterpret_cast<const int*>(keys), idx, 4);
150 r0 = _mm256_and_si256(_mm256_srli_epi32(r0, SHIFT), mask);
151 keyIdx += KEYS_PER_SIMD_PASS;
152 }
153 else if constexpr (KEY_TYPE_BITS == 64) {
154 // note: values have 32 bits, use __m128i to load only 4
155 __m128i idx32 = _mm_loadu_si128(reinterpret_cast<const __m128i*>(&src[keyIdx]));
156 // Gather 4 scattered keys, and apply shift and mask to get bucket ids
157 r0 = _mm256_i32gather_epi64(reinterpret_cast<const long long*>(keys), idx32, 8);
158 r0 = _mm256_and_si256(_mm256_srli_epi64(r0, SHIFT), mask);
159 keyIdx += KEYS_PER_SIMD_PASS;
160 }
161 else {
162 static_assert("Unsupported key size for SIMD radix sort");
163 break;
164 }
165 // Store results into tmpBins_ and increment histogram
166 _mm256_storeu_si256(reinterpret_cast<__m256i*>(tmpBins_), r0);
167 for (auto x : tmpBins_) ++histogram_[x];
168 }
169 }
170#endif
171 for (; keyIdx < numKeys; ++keyIdx) {
172 ++histogram_[(keys[src[keyIdx]] >> SHIFT) & MASK];
173 }
174 }
175 };
