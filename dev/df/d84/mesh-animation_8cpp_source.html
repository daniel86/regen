188void MeshAnimation::loadFrame(uint32_t frameIndex, bool isPongFrame) {
189 KeyFrame &frame = frames_[frameIndex];
190
191 auto &dataBuffer = (isPongFrame ? pongBuffer_ : pingBuffer_);
192 auto m_staged = dataBuffer->mapClientDataRaw(BUFFER_GPU_WRITE);
193 byte* d_staged = m_staged.w;
194
195 uint32_t offset = 0u;
196 for (auto & attribute : frame.attributes) {
197 const ref_ptr<ShaderInput> &input = attribute.input;
198 const uint32_t numBytes = input->alignedInputSize();
199 const auto m_frame = input->mapClientDataRaw(BUFFER_GPU_READ);
200 // align to base alignment
201 offset = (offset + input->baseAlignment() - 1) & ~(input->baseAlignment() - 1);
202 std::memcpy(
203 d_staged + offset,
204 m_frame.r, numBytes);
205 offset += numBytes;
206 }
207
208 if (isPongFrame) {
209 pongFrame_ = frameIndex;
210 frame.buffer = pongBuffer_;
211 } else {
212 pingFrame_ = frameIndex;
213 frame.buffer = pingBuffer_;
214 }

217void MeshAnimation::gpuUpdate(RenderState *rs, double dt) {
218 if (dt <= 0.00001) return;
219
220 elapsedTime_ += dt;
221
222 // map into anim's duration
223 const double duration = tickRange_.y - tickRange_.x;
224 const double timeInTicks = elapsedTime_ * ticksPerSecond_ / 1000.0;
225 if (timeInTicks > duration) {
226 REGEN_DEBUG("Mesh animation stopped at tick " <<
227 timeInTicks << " (duration: " << duration << ", frame:" << nextFrame_ << ")");
228 elapsedTime_ = 0.0;
229 lastTime_ = 0.0;
230 tickRange_.x = 0.0;
231 tickRange_.y = 0.0;
232 stopAnimation();
233 return;
234 }
235
236 // Look for present frame number.
237 int lastFrame = lastFramePosition_;
238 int frame = (timeInTicks >= lastTime_ ? lastFrame : startFramePosition_);
239 findFrameAfterTick(timeInTicks, frame, frames_);
240 lastFramePosition_ = frame;
241
242 // keep two frames in animation buffer
243 lastFrame = frame - 1;
244
245 KeyFrame &frame0 = frames_[lastFrame];
246 if (lastFrame != pingFrame_ && lastFrame != pongFrame_) {
247 loadFrame(lastFrame, frame == pingFrame_);
248 }
249 if (lastFrame != lastFrame_) {
250 lastFrame_ = lastFrame;
251 }
252
253 KeyFrame &frame1 = frames_[frame];
254 if (frame != pingFrame_ && frame != pongFrame_) {
255 loadFrame(frame, lastFrame == pingFrame_);
256 }
257 if (frame != nextFrame_) {
258 nextFrame_ = frame;
259 REGEN_DEBUG("Next frame: " << nextFrame_ << " (time: " << timeInTicks << ")");
260 }
261
262 frameTimeUniform_->setVertex(0,
263 (timeInTicks - frame1.startTick) / frame1.timeInTicks);
264
265 // bind frame data buffers
266 frame0.buffer->bind(lastFrameBindingPoint_);
267 frame1.buffer->bind(nextFrameBindingPoint_);
268
269 // Compute next vertex data
270 interpolationState_->enable(rs);
271 interpolationState_->disable(rs);
272
273 lastTime_ = tickRange_.x + timeInTicks;
274}

278void MeshAnimation::addFrame(
279 const std::list<ref_ptr<ShaderInput> > &attributes,
280 double timeInTicks) {
281 MeshAnimation::KeyFrame frame;
282
283 frame.timeInTicks = timeInTicks;
284 frame.startTick = 0.0;
285 for (auto it = frames_.rbegin(); it != frames_.rend(); ++it) {
286 MeshAnimation::KeyFrame &parentFrame = *it;
287 frame.startTick += parentFrame.timeInTicks;
288 }
289 frame.endTick = frame.startTick + frame.timeInTicks;
290
291 REGEN_DEBUG("Adding frame at tick " << frame.startTick << " (duration: " << frame.timeInTicks << ")");
292
293 // add attributes
294 for (const auto & input : mesh_->inputs()) {
295 const ref_ptr<ShaderInput> &in0 = input.in_;
296 if (!in0->isVertexAttribute()) continue;
297 if (!animAttributes_.contains(in0->name())) continue;
298 ref_ptr<ShaderInput> att;
299
300 // find specified attribute
301 for (const auto & in1 : attributes) {
302 if (in0->name() == in1->name()) {
303 att = in1;
304 break;
305 }
306 }
307 if (att.get() == nullptr) {
308 // find attribute from previous frames
309 att = findLastAttribute(in0->name());
310 }
311 if (att.get() != nullptr) {
312 frame.attributes.emplace_back(att, -1);
313 } else {
314 REGEN_WARN("No attribute found for '" << in0->name() << "'");
315 }
316 }
317
318 frames_.push_back(frame);
319}

344void MeshAnimation::addSphereAttributes(
345 float horizontalRadius,
346 float verticalRadius,
347 double timeInTicks,
348 const Vec3f &offset) {
349 if (!mesh_->hasInput(ATTRIBUTE_NAME_POS)) {
350 REGEN_WARN("mesh has no input named '" << ATTRIBUTE_NAME_POS << "'");
351 return;
352 }
353 if (!mesh_->hasInput(ATTRIBUTE_NAME_NOR)) {
354 REGEN_WARN("mesh has no input named '" << ATTRIBUTE_NAME_NOR << "'");
355 return;
356 }
357 //float radiusScale = horizontalRadius / verticalRadius;
358 //Vec3f scale(radiusScale, 1.0, radiusScale);
359
360 ref_ptr<ShaderInput3f> posAtt = ref_ptr<ShaderInput3f>::dynamicCast(mesh_->positions());
361 ref_ptr<ShaderInput3f> norAtt = ref_ptr<ShaderInput3f>::dynamicCast(mesh_->normals());
362 // allocate memory for the animation attributes
363 ref_ptr<ShaderInput3f> spherePos = ref_ptr<ShaderInput3f>::alloc(ATTRIBUTE_NAME_POS);
364 ref_ptr<ShaderInput3f> sphereNor = ref_ptr<ShaderInput3f>::alloc(ATTRIBUTE_NAME_NOR);
365 spherePos->setMemoryLayout(BUFFER_MEMORY_STD430);
366 sphereNor->setMemoryLayout(BUFFER_MEMORY_STD430);
367 spherePos->setVertexData(posAtt->numVertices(), nullptr);
368 sphereNor->setVertexData(norAtt->numVertices(), nullptr);
369
370 // find the centroid of the mesh
371 Vec3f minPos = posAtt->getVertex(0).r;
372 Vec3f maxPos = posAtt->getVertex(0).r;
373 for (uint32_t i = 1; i < posAtt->numVertices(); ++i) {
374 auto v = posAtt->getVertex(i);
375 if(minPos.x > v.r.x) minPos.x = v.r.x;
376 if(minPos.y > v.r.y) minPos.y = v.r.y;
377 if(minPos.z > v.r.z) minPos.z = v.r.z;
378 if(maxPos.x < v.r.x) maxPos.x = v.r.x;
379 if(maxPos.y < v.r.y) maxPos.y = v.r.y;
380 if(maxPos.z < v.r.z) maxPos.z = v.r.z;
381 }
382 Vec3f centroid = (minPos + maxPos) * 0.5f + offset;
383
384 // set sphere vertex data
385 // Note: this is a very simple sphere mapping, it is not perfect.
386 // There might be artifacts caused by faces that are flipped.
387 // Also make sure to use polygon offset to avoid shadow map fighting.
388 // NOTE: It should be possible to do this with fewer artifacts by taking the faces into account.
389 for (uint32_t i = 0; i < spherePos->numVertices(); ++i) {
390 Vec3f v = posAtt->getVertex(i).r;
391 Vec3f direction = v - centroid;
392 Vec3f n;
393 double l = direction.length();
394 if (l == 0) {
395 continue;
396 }
397
398 // take normalized direction vector as normal
399 n = direction;
400 n.normalize();
401 // and scaled normal as sphere position
402 // 1e-1 to avoid fighting
403 v = centroid + n * verticalRadius * (1.0f + l * 1e-6);
404
405 spherePos->setVertex(i, v);
406 sphereNor->setVertex(i, n);
407 }
408
409 std::list<ref_ptr<ShaderInput> > attributes;
410 attributes.emplace_back(spherePos);
411 attributes.emplace_back(sphereNor);
412 addFrame(attributes, timeInTicks);
413}
