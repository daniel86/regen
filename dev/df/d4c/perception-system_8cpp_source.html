76void PerceptionSystem::handleIntersection(const BoundingShape &other) {
77 if (indexedShape_.get() == &other) return; // skip self
78 const Vec3f &thisCenter = indexedShape_->tfOrigin();
79 const Vec3f &otherCenter = other.tfOrigin();
80 auto &collisionData = collisionEvt_.data;
81
82 // Compute perception data.
83 collisionData.other = &other;
84 collisionData.delta = thisCenter - otherCenter;
85 collisionData.distance = collisionData.delta.length();
86 collisionData.dir = collisionData.delta / (collisionData.distance + 1e-6f);
87 collisionData.isBehind = (Vec3f::front().dot(collisionData.delta) < 0.0f);
88 if (collisionData.distance < 0.01f) return;
89
90 // Notify all monitors.
91 for (auto &monitor : collisionMonitors_) {
92 monitor->emitCollisionEvent(collisionEvt_);
93 }
