12 shape_(shape) {
13 numLODs_ = 1;
14 if (shape->mesh().get()) {
15 numLODs_ = std::max(numLODs_, shape->mesh()->numLODs());
16 }
17 for (const auto &part : shape->parts()) {
18 numLODs_ = std::max(numLODs_, part->numLODs());
19 }
20 // remember LOD thresholds
21 const ref_ptr<Mesh> &mesh = shape->baseMesh();
22 const Vec3f &lodThresholds = mesh->lodThresholds();
23 lodThresholds_ = lodThresholds;
24 // move the thresholds around based on LOD shift
25 for (int32_t i = 0; i < 3; ++i) {
26 int shift = lodShift[i];
27 if (shift > 0) {
28 for (int32_t j = 0; j < shift && (i+j) < 3; ++j) {
29 lodThresholds_[i+j] = (i==0 ? 0.0f : lodThresholds[i-1]);
30 }
31 }
32 else if (shift < 0) {
33 for (int32_t j = -shift; j>0 && (i-j) > 0; --j) {
34 lodThresholds_[i-j] = (i==2 ? std::numeric_limits<float>::max() : lodThresholds[i+1]);
35 }
36 }
37 }
38 // set LOD thresholds for out-of-rand LOD levels to max
39 for (uint32_t unusedIdx = numLODs_; unusedIdx < 4; ++unusedIdx) {
40 lodThresholds_[unusedIdx-1] = std::numeric_limits<float>::max();
41 }
42}
