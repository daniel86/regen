52void UserController::updateBoneController(double dt) {
53 if (!boneController_) return;
54
55 float dt_s = dt / 1000.0f;
56 // Update the list of active motions from the motion state array.
57 numActiveMotions_ = 0;
58 for (uint32_t motionIdx = 0; motionIdx < motionTypes_.size(); ++motionIdx) {
59 auto &motionFlag = motionState_[motionIdx];
60 if (motionFlag.load()) {
61 activeMotions_[numActiveMotions_++] = motionTypes_[motionIdx];
62 }
63 }
64 if (numActiveMotions_ == 0) {
65 // Ensure at least idle motion is active.
66 activeMotions_[numActiveMotions_++] = MotionType::IDLE;
67 }
68 // Update the bone controller with the desired motions.
69 boneController_->updateBoneController(dt_s,
70 activeMotions_.data(), numActiveMotions_);
71}
