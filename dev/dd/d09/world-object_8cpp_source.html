161static void loadChildren(LoadingContext &ctx, scene::SceneInputNode &n, const ref_ptr<WorldObject> &wo) {
162 auto world = ctx.scene()->getResources()->getWorldModel();
163 // load child objects
164 for (const auto &childNode: n.getChildren("object")) {
165 auto childObj = WorldObject::load(ctx, *childNode.get());
166 switch (wo->objectType()) {
167 case ObjectType::LOCATION:
168 childObj->setCurrentLocation(ref_ptr<Location>::dynamicCast(wo));
169 if (wo->placeOfObject().get()) {
170 ref_ptr<Place> place = ref_ptr<Place>::dynamicCast(wo->placeOfObject());
171 if (place.get()) {
172 place->addWorldObject(childObj);
173 childObj->setPlaceOfObject(place);
174 }
175 }
176 break;
177 case ObjectType::PLACE:
178 childObj->setPlaceOfObject(wo);
179 break;
180 case ObjectType::COLLECTION: {
181 auto group = ref_ptr<ObjectGroup>::dynamicCast(wo);
182 if (group.get()) {
183 childObj->joinGroup(group);
184 }
185 break;
186 }
187 default:
188 break;
189 }
190 world->addWorldObject(childObj);
191 }
192}

194ref_ptr<WorldObject> WorldObject::load(LoadingContext &ctx, scene::SceneInputNode &n) {
195 ObjectType type = ObjectType::THING;
196 if (n.hasAttribute("type")) {
197 type = n.getValue<ObjectType>("type", ObjectType::THING);
198 }
199 ref_ptr<WorldObject> wo;
200 if (type == ObjectType::CHARACTER) {
201 wo = ref_ptr<PersonObject>::alloc(n.getName());
202 } else if (type == ObjectType::ANIMAL) {
203 wo = ref_ptr<AnimalObject>::alloc(n.getName());
204 } else if (type == ObjectType::PLAYER) {
205 wo = ref_ptr<PlayerObject>::alloc(n.getName());
206 } else if (type == ObjectType::LOCATION) {
207 wo = ref_ptr<Location>::alloc(n.getName());
208 } else {
209 wo = ref_ptr<WorldObject>::alloc(n.getName());
210 wo->setObjectType(type);
211 }
212 if (n.hasAttribute("static")) {
213 wo->setStatic(n.getValue<bool>("static", true));
214 }
215 wo->setDanger(n.getValue<float>("danger", 0.0f));
216 wo->setRadius(n.getValue<float>("radius", 1.0f));
217
218 // Load objects that are part of this world object.
219 loadChildren(ctx, n, wo);
220
221 // load affordances
222 for (const auto &a: n.getChildren("affordance")) {
223 ref_ptr<Affordance> affordance = ref_ptr<Affordance>::alloc(wo);
224 affordance->type = a->getValue<ActionType>("type", ActionType::IDLE);
225 affordance->minDistance = a->getValue<float>("min-distance", 0.0f);
226 if (a->hasAttribute("max-participants")) {
227 affordance->slotCount = a->getValue<int>("max-participants", 1);
228 } else {
229 affordance->slotCount = a->getValue<int>("num-slots", 1);
230 }
231 affordance->spacing = a->getValue<float>("spacing", 2.0f);
232 affordance->layout = a->getValue<SlotLayout>("slot-layout", SlotLayout::CIRCULAR);
233 affordance->radius = a->getValue<float>("radius", 1.0f);
234 affordance->baseOffset = a->getValue<Vec3f>("base-offset", Vec3f(0.0f, 0.0f, 0.0f));
235 affordance->name = a->getName();
236 wo->addAffordance(affordance);
237 }
238
239 // load factions
240 if (n.hasAttribute("factions")) {
241 const auto indicesAtt = n.getValue<std::string>("factions", "0");
242 std::vector<std::string> indicesStr;
243 boost::split(indicesStr, indicesAtt, boost::is_any_of(","));
244 for (auto & it : indicesStr) {
245 int i = atoi(it.c_str());
246 if (i >= 0 && i < 32) {
247 wo->addFaction(1u << i);
248 } else {
249 REGEN_WARN("Ignoring " << n.getDescription() << ", invalid faction index '" << it << "'.");
250 }
251 }
252 } else if (n.hasAttribute("faction")) {
253 int i = n.getValue<int>("faction", -1);
254 if (i >= 0 && i < 32) {
255 wo->addFaction(1u << i);
256 } else {
257 REGEN_WARN("Ignoring " << n.getDescription() << ", invalid faction index '" << i << "'.");
258 }
259 }
260
261 ctx.scene()->getResources()->getWorldModel()->addWorldObject(wo);
262 return wo;
263}

265ref_ptr<WorldObjectVec> WorldObjectVec::load(LoadingContext &ctx, scene::SceneInputNode &n) {
266 ref_ptr<WorldObjectVec> objVec = ref_ptr<WorldObjectVec>::alloc();
267
268 if (n.hasAttribute("mesh")) {
269 uint32_t meshIdx = n.getValue<uint32_t>("mesh-index", 0u);
270 auto compositeMesh = ctx.scene()->getResources()->getMesh(ctx.scene(), n.getValue("mesh"));
271 if (!compositeMesh || compositeMesh->meshes().empty()) {
272 REGEN_WARN("Cannot find mesh in `" << n.getDescription() << "`.");
273 } else if (meshIdx >= compositeMesh->meshes().size()) {
274 REGEN_WARN("Invalid mesh index in `" << n.getDescription() << "`.");
275 } else {
276 auto mesh = compositeMesh->meshes()[meshIdx];
277 auto shape = mesh->indexedShape(0);
278 uint32_t numInstances = shape->transform()->numInstances();
279 // load range of instances
280 std::list<scene::IndexRange> instanceRange = n.getIndexSequence(numInstances);
281 for (const auto &r: instanceRange) {
282 for (uint32_t instance = r.from; instance <= r.to; instance += r.step) {
283 auto wo = WorldObject::load(ctx, n);
284 wo->setShape(mesh->indexedShape(instance), instance);
285
286 // Add to semantic places.
287 auto worldModel = ctx.scene()->getResources()->getWorldModel();
288 for (const auto &place: worldModel->places) {
289 if ((wo->position2D() - place->position2D()).lengthSquared() <= place->radius() * place->radius()) {
290 place->addWorldObject(wo);
291 wo->setPlaceOfObject(place);
292 break;
293 }
294 }
295
296 objVec->push_back(wo);
297 }
298 }
299 }
300 }
301 if (objVec->empty()) {
302 return {};
303 }
304 ctx.scene()->putResource<WorldObjectVec>(n.getName(), objVec);
305 return objVec;
306}
