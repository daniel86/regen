77static std::vector<s_BrightStar> loadStarsData(const char *fileName) {
78 // Retrieve file size.
79 FILE *f;
80#ifdef __GNUC__
81 f = std::fopen(fileName, "r");
82#else // __GNUC__
83 fopen_s(&f, fileName, "r");
84#endif // __GNUC__
85 if (!f) { return {}; }
86
87 std::fseek(f, 0, SEEK_END);
88 const auto fileSize = std::ftell(f);
89 std::fclose(f);
90
91 auto numStars = fileSize / sizeof(s_BrightStar);
92
93 std::vector<s_BrightStar> stars(numStars);
94 std::ifstream in_stream(fileName, std::ios::binary);
95 in_stream.read(reinterpret_cast<char *>(stars.data()), fileSize);
96 return stars;
97}

99void BrightStars::set_brightStarsFile(const std::string &brightStars) {
100 auto starsData = loadStarsData(brightStars.c_str());
101 numStars_ = static_cast<unsigned int>(starsData.size());
102 if (numStars_ == 0) {
103 REGEN_WARN("Unable to load bright stars catalog at " << brightStars << ".");
104 return;
105 }
106 REGEN_INFO("Loaded " << numStars_ << " bright stars from " << brightStars << ".");
107
108 pos_->setVertexData(numStars_);
109 col_->setVertexData(numStars_);
110
111 auto stars = starsData.data();
112 for (unsigned int i = 0; i < numStars_; ++i) {
113 osgHimmel::t_equf equ;
114 equ.right_ascension = _rightascd(stars[i].RA, 0, 0);
115 equ.declination = stars[i].DE;
116
117 pos_->setVertex(i, Vec4f::create(equ.toEuclidean(), static_cast<float>(i)));
118 col_->setVertex(i, Vec4f(
119 stars[i].sRGB_R,
120 stars[i].sRGB_G,
121 stars[i].sRGB_B,
122 stars[i].Vmag + 0.4f // the 0.4 accounts for magnitude decrease due to the earth's atmosphere
123 ));
124 }
125
126 meshState_->setInput(pos_);
127 meshState_->setInput(col_);
128 meshState_->updateVertexData();
129}
