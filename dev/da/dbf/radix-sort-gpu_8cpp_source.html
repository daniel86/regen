228void RadixSort_GPU::sortContiguous(RenderState *rs) {
229 // now we can make the radix passes starting with values_[0] as input
230 // and writing to values_[1]. Then we swap the buffers each pass.
231 // In the end, we will have the sorted instanceIDs in values_[0].
232 uint32_t readOffset = 0u;
233 uint32_t writeOffset = numKeys_;
234 for (uint32_t bitOffset = 0u; bitOffset < 32u; bitOffset += radixBits_) {
235 // Run histogram pass. As a result we will have global counts for each bucket
236 // and work group in the globalHistogramBuffer_.
237 radixHistogramPass_->enable(rs);
238 glUniform1ui(histogramBitOffsetIndex_, bitOffset);
239 glUniform1ui(histogramReadIndex_, readOffset);
240 radixHistogramPass_->disable(rs);
241#ifdef RADIX_DEBUG_HISTOGRAM
242 printHistogram(rs);
243#endif
244
245 // Run offsets pass. As a result we will have the offsets for each work group
246 // in the globalHistogramBuffer_.
247 prefixScan_->enable(rs);
248 prefixScan_->disable(rs);
249#ifdef RADIX_DEBUG_HISTOGRAM
250 printHistogram(rs);
251#endif
252
253 // Finally, run the scatter pass. As a result we will have the sorted instanceIDs
254 // in the values_[writeIndex] buffer.
255 radixScatterPass_->enable(rs);
256 glUniform1ui(scatterBitOffsetIndex_, bitOffset);
257 glUniform1ui(scatterReadIndex_, readOffset);
258 glUniform1ui(scatterWriteIndex_, writeOffset);
259 radixScatterPass_->disable(rs);
260
261 // swap read and write buffers
262 std::swap(readOffset, writeOffset);
263 }

266void RadixSort_GPU::sort(RenderState *rs) {
267 // now we can make the radix passes starting with values_[0] as input
268 // and writing to values_[1]. Then we swap the buffers each pass.
269 // In the end, we will have the sorted instanceIDs in values_[0].
270 uint32_t readIndex = 0u;
271 uint32_t writeIndex = 1u;
272 for (uint32_t bitOffset = 0u; bitOffset < 32u; bitOffset += radixBits_) {
273 // Run histogram pass. As a result we will have global counts for each bucket

275 radixHistogramPass_->enable(rs);

278 radixHistogramPass_->disable(rs);

326void RadixSort_GPU::printInstanceMap(RenderState *rs) {
327 // debug sorted output
328 std::vector<double> distances(numKeys_);
329 auto sortKeys = (uint32_t *) glMapNamedBufferRange(
330 keyBuffer_->drawBufferRef()->bufferID(),
331 keyBuffer_->drawBufferRef()->address(),
332 keyBuffer_->drawBufferRef()->allocatedSize(),
333 GL_MAP_READ_BIT);
334 if (sortKeys) {
335 for (uint32_t i = 0; i < numKeys_; ++i) {
336 distances[i] = conversion::uintToFloat(sortKeys[i]);
337 }
338 glUnmapNamedBuffer(keyBuffer_->drawBufferRef()->bufferID());
339 }
340
341 auto idRef = valueBuffer_[outputIdx_]->drawBufferRef();
342 //auto idRef = valueBuffer_->drawBufferRef();
343 auto instanceIDs = (uint32_t *) glMapNamedBufferRange(
344 idRef->bufferID(),
345 idRef->address(),
346 idRef->allocatedSize(),
347 GL_MAP_READ_BIT);
348 if (instanceIDs) {
349 double lastDistance = 0.0;
350 bool validSortedIDs_ = true;
351 for (uint32_t i = 0; i < numKeys_; ++i) {
352 auto mappedID = instanceIDs[i];
353 if (mappedID >= numKeys_) {
354 REGEN_ERROR("mappedID " << mappedID << " >= numInstances_ " << numKeys_);
355 validSortedIDs_ = false;
356 break;
357 }
358 auto distance = distances[mappedID];
359 if (distance < lastDistance) {
360 validSortedIDs_ = false;
361 }
362 lastDistance = distance;
363 }
364 if (validSortedIDs_) {
365 REGEN_INFO(" sortedIDs are valid");
366 } else {
367 REGEN_INFO(" sortedIDs are INVALID");
368 }
369#ifdef RADIX_DEBUG_RESULT
370 {
371 std::stringstream sss;
372 sss << " ID data (" << numInstances_ << "): ";
373 for (uint32_t i = 0; i < numInstances_; ++i) {
374 if (instanceIDs[i] >= numInstances_) {
375 break;
376 }
377 sss << instanceIDs[i] << " (" << distances[instanceIDs[i]] << ") ";
378 }
379 REGEN_INFO(" " << sss.str());
380 }
381#endif
382 glUnmapNamedBuffer(idRef->bufferID());
383 }
