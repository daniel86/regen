157void Ground::updatePatchSize() {
158 // compute number of patches in x/z direction based on map size
159 // and number of patches per row (this is used for direction with less extent)
160 if (mapSize_.x > mapSize_.z) {
161 patchSize_.y = mapSize_.z / static_cast<float>(numPatchesPerRow_);
162 // set number of patches
163 numPatches_.y = numPatchesPerRow_;
164 numPatches_.x = static_cast<uint32_t>(std::ceil(mapSize_.x / patchSize_.y));
165 // adjust patch size in x direction to evenly cover the ground
166 patchSize_.x = patchSize_.y + fmod(mapSize_.x, patchSize_.y) / static_cast<float>(numPatches_.x);
167 }
168 else {
169 patchSize_.x = mapSize_.x / static_cast<float>(numPatchesPerRow_);
170 // set number of patches
171 numPatches_.x = numPatchesPerRow_;
172 numPatches_.y = static_cast<uint32_t>(std::ceil(mapSize_.z / patchSize_.x));
173 // adjust patch size in z direction to evenly cover the ground
174 patchSize_.y = patchSize_.x + fmod(mapSize_.z, patchSize_.x) / static_cast<float>(numPatches_.y);
175 }
176 rectangleConfig_.posScale = Vec3f(patchSize_.x, 1.0f, patchSize_.y);
177 REGEN_INFO("Ground num patches: " << numPatches_ << " with patch size: " << patchSize_);
178}

180void Ground::updateGroundPatches() {
181 const auto numPatches = numPatches_.x * numPatches_.y;
182 const float offsetX = mapCenter_.x - (mapSize_.x / 2.0f);
183 const float offsetZ = mapCenter_.z - (mapSize_.z / 2.0f);
184 const auto patchHalfSize = patchSize_ / 2.0f;
185 uint32_t tfIndex = 0;
186
187 tf_->set_numInstances(numPatches);
188 tf_->modelOffset()->setInstanceData(numPatches, 1, nullptr);
189 auto tfData = tf_->modelOffset()->mapClientData<Vec4f>(BUFFER_GPU_WRITE);
190 for (uint32_t xIdx=0; xIdx<numPatches_.x; ++xIdx) {
191 for (uint32_t zIdx=0; zIdx<numPatches_.y; ++zIdx) {
192 const auto xPos = offsetX + (static_cast<float>(xIdx) * patchSize_.x) + patchHalfSize.x;
193 const auto zPos = offsetZ + (static_cast<float>(zIdx) * patchSize_.y) + patchHalfSize.y;
194 auto &patchTF = tfData.w[tfIndex++];
195 patchTF.x = xPos;
196 patchTF.y = mapCenter_.y - mapSize_.y * 0.5f;
197 patchTF.z = zPos;
198 }
199 }

202void Ground::updateAttributes() {
203 SkirtQuad::updateAttributes();
204 updateGroundPatches();
205 // update bounding box
206 Vec3f minPos = rectangleConfig_.translation;
207 Vec3f maxPos = rectangleConfig_.translation;
208 minPos.x -= patchSize_.x * 0.5f;
209 minPos.z -= patchSize_.y * 0.5f;
210 maxPos.x += patchSize_.x * 0.5f;
211 maxPos.z += patchSize_.y * 0.5f;
212 // extend bounds in y direction by half the map size.
213 // This is exessive, and usually far too much, but e.g. if we have a patch
214 // with a steep cliff, we want to make sure the entire patch is always
215 // contained in the bounds.
216 // Note: we could compute tight bounds on per-patch basis, but
217 // that would require quite some changes. For CPU path, we have a shape
218 // per patch (per instance), so we can give them tight bounds at least
219 // based on heightmap min/max.
220 maxPos.y += 0.5f*mapSize_.y;
221 minPos.y -= 0.5f*mapSize_.y;
222
223 set_bounds(minPos, maxPos);
224 if(skirtMesh_.get()) {
225 skirtMesh_->set_bounds(minPos, maxPos);
226 }
227}

581void Ground::createWeightPass() {
582 auto fbo = ref_ptr<FBO>::alloc(weightMapSize_, weightMapSize_);
583 std::vector<GLenum> attachments;
584
585 // we create weight maps that store weights for 4 materials each.
586 auto numWeightMaps = static_cast<uint32_t>(std::ceil(
587 static_cast<float>(materialConfigs_.size()) / 4.0f));
588 auto numBiomeMaps = static_cast<uint32_t>(std::ceil(
589 static_cast<float>(biomeConfigs_.size()) / 4.0f));
590 attachments.resize(numWeightMaps);
591 weightMaps_.resize(numWeightMaps);
592 groundShaderDefines_->shaderDefine(
593 "NUM_WEIGHT_MAPS", REGEN_STRING(numWeightMaps));
594 groundShaderDefines_->shaderDefine(
595 "NUM_BIOME_MAPS", REGEN_STRING(numBiomeMaps));
596
597 for (uint32_t i=0; i<numWeightMaps; ++i) {
598 auto tex = fbo->addTexture(1,
599 GL_TEXTURE_2D,
600 GL_RGBA,
601 GL_RGBA8,
602 GL_UNSIGNED_BYTE);
603 weightMaps_[i] = ref_ptr<Texture2D>::dynamicCast(tex);
604 attachments[i] = GL_COLOR_ATTACHMENT0 + i;
605 if (useWeightMapMips_) {
606 tex->set_filter(TextureFilter(GL_LINEAR_MIPMAP_LINEAR, GL_LINEAR));
607 }
608 joinSkirtStates(ref_ptr<TextureState>::alloc(
609 weightMaps_[i], "groundMaterialWeights" + REGEN_STRING(i)));
610 }
611
612 for (uint32_t i=0; i<numBiomeMaps; ++i) {
613 auto tex = fbo->addTexture(1,
614 GL_TEXTURE_2D,
615 GL_RGBA,
616 GL_RGBA8,
617 GL_UNSIGNED_BYTE);
618 biomeMaps_.push_back(ref_ptr<Texture2D>::dynamicCast(tex));
619 attachments.push_back(GL_COLOR_ATTACHMENT0 + numWeightMaps + i);
620 }
621
622 weightFBO_ = ref_ptr<FBOState>::alloc(fbo);
623 weightFBO_->setDrawBuffers(attachments);
624 weightUpdateState_->joinStates(weightFBO_);
625
626 // bind ground parameter for size information
627 weightUpdateState_->setInput(u_mapCenter_);
628 weightUpdateState_->setInput(u_skirtSize_);
629 weightUpdateState_->setInput(u_mapSize_);
630 // bind material masks
631 if (materialMaskState_.get()) {
632 weightUpdateState_->joinStates(materialMaskState_);
633 }
634 { // bind height map and normal map
635 weightUpdateState_->joinStates(ref_ptr<TextureState>::alloc(heightMap_, "heightMap"));
636 weightUpdateState_->joinStates(ref_ptr<TextureState>::alloc(normalMap_, "normalMap"));
637 }
638 { // disable depth test/write
639 auto depth = ref_ptr<DepthState>::alloc();
640 depth->set_useDepthTest(false);
641 depth->set_useDepthWrite(false);
642 weightUpdateState_->joinStates(depth);
643 }
644 // run weight pass
645 weightUpdatePass_ = ref_ptr<FullscreenPass>::alloc("regen.objects.terrain.ground.weights");
646 weightUpdateState_->joinStates(weightUpdatePass_);
647
648 StateConfigurer shaderConfigurer;
649 shaderConfigurer.addState(weightUpdateState_.get());
650 weightUpdatePass_->createShader(shaderConfigurer.cfg());
651}

664ref_ptr<Ground> Ground::load(LoadingContext &ctx, scene::SceneInputNode &input) {
665 auto scene = ctx.scene();
666 auto ground = ref_ptr<Ground>::alloc();
667
668 if (input.hasAttribute("weight-map-size")) {
669 ground->setWeightMapSize(input.getValue<uint32_t>("weight-map-size", 2048));
670 }
671 if (input.hasAttribute("weight-mips")) {
672 ground->setUseWeightMapMips(input.getValue<bool>("weight-mips", false));
673 }
674 if (input.hasAttribute("skirt-size")) {
675 ground->setSkirtSize(input.getValue<float>("skirt-size", 0.05f));
676 }
677 // early add mesh vector to scene for children handling below
678 auto out_ = ref_ptr<CompositeMesh>::alloc();
679 out_->addMesh(ground);
680 ctx.scene()->putResource<CompositeMesh>(input.getName(), out_);
681
682 std::vector<ref_ptr<scene::SceneInputNode>> handledChildren;
683 for (auto &n: input.getChildren()) {
684 if (n->getCategory() == "materials") {
685 MaterialConfig materialCfg;
686 for (auto &materialSpec: n->getChildren()) {
687 auto matName = materialSpec->getValue<std::string>("name", "");
688 auto variant = materialSpec->getValue<std::string>("variant", "0");
689 MaterialDescription materialDescr(matName, variant);
690 materialCfg.type = materialSpec->getValue<std::string>("type", "dirt");
691 if (materialDescr.hasMap(TextureState::MAP_TO_DIFFUSE)) {
692 materialCfg.colorFile = materialDescr.getMaps(TextureState::MAP_TO_DIFFUSE).front();
693 } else if (materialDescr.hasMap(TextureState::MAP_TO_COLOR)) {
694 materialCfg.colorFile = materialDescr.getMaps(TextureState::MAP_TO_COLOR).front();
695 } else {
696 REGEN_WARN("No color map found for " << input.getDescription() << ".");
697 continue;
698 }
699 if (materialDescr.hasMap(TextureState::MAP_TO_NORMAL)) {
700 materialCfg.normalFile = materialDescr.getMaps(TextureState::MAP_TO_NORMAL).front();
701 } else {
702 materialCfg.normalFile = "";
703 }
704 materialCfg.maskFile = materialSpec->getValue<std::string>("mask-file", "");
705 materialCfg.isFallback = materialSpec->getValue<bool>("is-fallback", false);
706 materialCfg.uvScale = materialSpec->getValue<float>("uv-scale", 0.5f);
707 materialCfg.slope = SmoothRange(materialSpec->getValue<Vec3f>(
708 "slope", Vec3f(0.0f,180.0f, 10.0f)));
709 materialCfg.height = SmoothRange(materialSpec->getValue<Vec3f>(
710 "height", Vec3f(0.0f,1.0f,0.1f)));
711
712 ground->setMaterial(materialCfg);
713 }
714 handledChildren.push_back(n);
715 }
716 else if (n->getCategory() == "biomes") {
717 for (auto &m: n->getChildren()) {
718 auto biomeType = m->getValue<std::string>("type", "");
719 BiomeDescription biome(biomeType);
720 biome.height = SmoothRange(m->getValue<Vec3f>(
721 "height", Vec3f(0.0f,1.0f,0.1f)));
722 biome.slope = SmoothRange(m->getValue<Vec3f>(
723 "slope", Vec3f(0.0f,180.0f, 10.0f)));
724 biome.temperature = SmoothRange(m->getValue<Vec3f>(
725 "temperature", Vec3f(0.0f,1.0f, 0.1f)));
726 biome.humidity = SmoothRange(m->getValue<Vec3f>(
727 "humidity", Vec3f(0.0f,1.0f, 0.1f)));
728 biome.rockiness = SmoothRange(m->getValue<Vec3f>(
729 "rockiness", Vec3f(0.0f,1.0f, 0.1f)));
730 biome.concavity = SmoothRange(m->getValue<Vec3f>(
731 "concavity", Vec3f(0.0f,1.0f, 0.1f)));
732 ground->setBiome(biome);
733 }
734 handledChildren.push_back(n);
735 }
736 else if (n->getCategory() == "transform") {
737 // note: skip transform, we create one below
738 handledChildren.push_back(n);
739 }
740 else if (n->getCategory() == "masks") {
741 handledChildren.push_back(n);
742 // Add screen-space masks that modulate the blanket weights
743 // (e.g. to remove snow where there are footprints)
744 for (auto &maskNode : n->getChildren("mask")) {
745 ref_ptr<Texture> maskTexture;
746 if (maskNode->hasAttribute("fbo")) {
747 auto fboName = maskNode->getValue<std::string>("fbo", "");
748 auto fbo = ctx.scene()->getResource<FBO>(fboName);
749 if (fbo.get()) {
750 uint32_t attachmentIdx = maskNode->getValue<uint32_t>("attachment", 0);
751 maskTexture = fbo->colorTextures()[attachmentIdx];
752 }
753 } else if (maskNode->hasAttribute("texture")) {
754 auto texName = maskNode->getValue<std::string>("texture", "");
755 maskTexture = ctx.scene()->getResource<Texture2D>(texName);
756 if (!maskTexture) {
757 REGEN_WARN("No valid texture found for blanket mask in " << maskNode->getDescription() << ".");
758 }
759 }
760 if (!maskTexture.get()) {
761 REGEN_WARN("No valid FBO/attachment found for blanket mask in " << maskNode->getDescription() << ".");
762 continue;
763 }
764 MaskFallback maskFallback;
765 maskFallback.channelIdx = maskNode->getValue<uint32_t>("fallback-channel", 0);
766 maskFallback.intensity = maskNode->getValue<float>("fallback-intensity", 1.0f);
767
768 std::vector<MaskChannel> maskChannels;
769 for (auto &n: maskNode->getChildren("channel")) {
770 auto &ch = maskChannels.emplace_back();
771 ch.channelIdx = n->getValue<uint32_t>("index", 0);
772 ch.blendMode = n->getValue<BlendMode>("blend-mode", BLEND_MODE_MULTIPLY);
773 ch.blendFactor = n->getValue<float>("blend-factor", 1.0f);
774 ch.invert = n->getValue<bool>("invert", false);
775 }
776
777 if (maskChannels.empty()) {
778 REGEN_WARN("No valid mask channels found for blanket mask in " << maskNode->getDescription() << ".");
779 } else {
780 ground->addMask(maskTexture, maskChannels, maskFallback);
781 }
782 }
783 }
784 }
785 for (auto &n: handledChildren) {
786 // make sure mesh loading does not attempt to load materials again (this will cause a warning)
787 input.removeChild(n);
788 }
789
790 auto lodVec = input.getValue<Vec4ui>(
791 "lod-levels", Vec4ui(5,3,2, 1));
792 auto patchDensity = input.getValue<uint32_t>("patch-density", 9);
793 ground->setLODConfig(patchDensity, {lodVec.x, lodVec.y, lodVec.z, lodVec.w});
794 ground->setMapGeometry(
795 input.getValue<Vec3f>("map-center", Vec3f::zero()),
796 input.getValue<Vec3f>("map-size", Vec3f::one()));
797
798 auto heightMap = scene->getResource<Texture2D>(input.getValue("height-map"));
799 auto normalMap = scene->getResource<Texture2D>(input.getValue("normal-map"));
800 if (!heightMap.get()) {
801 REGEN_WARN("No height map found for " << input.getDescription() << ".");
802 return {};
803 }
804 if (!normalMap.get()) {
805 REGEN_WARN("No normal map found for " << input.getDescription() << ".");
806 return {};
807 }
808 ground->setMapTextures(heightMap, normalMap);
809
810 auto tfName = input.getValue("tf");
811 auto modelTransform = ref_ptr<ModelTransformation>::alloc(
812 ModelTransformation::TF_OFFSET, BufferUpdateFlags::NEVER);
813 scene->putResource<ModelTransformation>(tfName, modelTransform);
814 ground->setModelTransform(modelTransform);
815 ground->updateAttributes();
816 modelTransform->tfBuffer()->updateBuffer();
817
818 ground->createResources();
819 ground->updateWeightMaps();
820
821 for (uint32_t mapIdx = 0; mapIdx < ground->weightMaps_.size(); ++mapIdx) {
822 scene->putResource<Texture>(
823 REGEN_STRING(input.getName() << "-weight-map-" << mapIdx),
824 ground->weightMaps_[mapIdx]);
825 }
826 for (uint32_t mapIdx = 0; mapIdx < ground->biomeMaps_.size(); ++mapIdx) {
827 scene->putResource<Texture>(
828 REGEN_STRING(input.getName() << "-biome-map-" << mapIdx),
829 ground->biomeMaps_[mapIdx]);
830 }
831
832 return ground;
833}
