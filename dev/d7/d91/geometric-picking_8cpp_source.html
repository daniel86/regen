78void GeomPicking::updateMouse() {
79 auto &inverseProjectionMatrix = camera_->projectionInverse(0);
80 auto mouse = mouseTexco_->getVertex(0);
81 // find view space mouse ray intersecting the frustum
82 Vec2f mouseNDC = mouse.r * 2.0 - Vec2f::one();
83 // in NDC space the ray starts at (mx,my,0) and ends at (mx,my,1)
84 Vec4f mouseRayNear = inverseProjectionMatrix ^ Vec4f::create(mouseNDC, 0.0, 1.0);
85 Vec4f mouseRayFar = inverseProjectionMatrix ^ Vec4f::create(mouseNDC, 1.0, 1.0);
86 mouseRayNear.xyz() /= mouseRayNear.w;
87 mouseRayFar.xyz() /= mouseRayFar.w;
88 mousePosVS_->setVertex(0, mouseRayNear.xyz());
89 mouseDirVS_->setVertex(0, mouseRayNear.xyz() - mouseRayFar.xyz());
90}

92void GeomPicking::traverse(RenderState *rs) {
93 updateMouse();
94 state_->enable(rs);
95
96 int feedbackCount = 0;
97 uint32_t feedbackQuery = 0;
98 glGenQueries(1, &feedbackQuery);
99 for (auto &pickableNode: childs()) {
100 auto pickableMesh = pickableNode->findStateWithType<Mesh>();
101 if (pickableMesh == nullptr) {
102 REGEN_WARN("No mesh found in pickable node.");
103 continue;
104 }
105 glBeginQuery(GL_PRIMITIVES_GENERATED, feedbackQuery);
106 // update buffer range of pickable
107 bufferRange_->offset_ = feedbackCount * sizeof(PickData);
108 bufferRange_->size_ = bufferSize_ - bufferRange_->offset_;
109 bufferRange_->offset_ += feedbackRange_.offset_;
110 pickableMesh->setFeedbackRange(bufferRange_.get());
111 // render pickable
112 pickableNode->traverse(rs);
113
114 glEndQuery(GL_PRIMITIVES_GENERATED);
115 feedbackCount += static_cast<int>(getGLQueryResult(feedbackQuery));
116 if (feedbackCount >= static_cast<int>(maxPickedObjects_)) {
117 break;
118 }
119 }
120 glDeleteQueries(1, &feedbackQuery);
121
122 if (feedbackCount > 0) {
123 // NOTE: the staging buffer pickMapping_ is currently not wrapped by a BO that adds itself to the
124 // StagingSystem, so we need to manually update it. reason: currently VBOs are not
125 // supported in the StagingSystem.
126 // Later, we might want to add the picking buffer to staging system to gain advantages of
127 // centralized update and synchronization.
128 if(!pickMapping_->readBuffer(vboRef_, feedbackRange_, 0u)) {
129 REGEN_WARN("Failed to read feedback buffer for picking.");
130 hasPickedObject_ = false;
131 // avoid further processing
132 state_->disable(rs);
133 return;
134 }
135 if (pickMapping_->hasReadData()) {
136 auto &pickData = pickMapping_->stagingReadValue();
137 pickedObject_.depth = pickData.depth;
138 pickedObject_.instanceID = pickData.instanceID;
139 pickedObject_.objectID = pickData.objectID;
140 hasPickedObject_ = true;
141 } else {
142 hasPickedObject_ = false;
143 }
144 } else {
145 hasPickedObject_ = false;
146 }
147
148 state_->disable(rs);
149 GL_ERROR_LOG();
150}
