413ref_ptr<Texture> TextureState::getTexture(
414 scene::SceneLoader *scene,
415 scene::SceneInputNode &input,
416 const std::string &idKey,
417 const std::string &bufferKey,
418 const std::string &attachmentKey) {
419 ref_ptr<Texture> tex;
420 // Find the texture resource
421 if (input.hasAttribute(idKey)) {
422 tex = scene->getResource<Texture>(input.getValue(idKey));
423 } else if (input.hasAttribute(bufferKey)) {
424 ref_ptr<FBO> fbo = scene->getResource<FBO>(input.getValue(bufferKey));
425 if (fbo.get() == nullptr) {
426 REGEN_WARN("Unable to find FBO '" << input.getValue(bufferKey) <<
427 "' for " << input.getDescription() << ".");
428 return tex;
429 }
430 const auto val = input.getValue<std::string>(attachmentKey, "0");
431 if (val == "depth") {
432 tex = fbo->depthTexture();
433 if (!tex) {
434 tex = fbo->depthStencilTexture();
435 }
436 } else if (val == "stencil") {
437 tex = fbo->stencilTexture();
438 if (!tex) {
439 tex = fbo->depthStencilTexture();
440 }
441 } else if (val == "depth-stencil" || val == "stencil-depth") {
442 tex = fbo->depthStencilTexture();
443 } else {
444 std::vector<ref_ptr<Texture> > &textures = fbo->colorTextures();
445
446 unsigned int attachment;
447 std::stringstream ass(val);
448 ass >> attachment;
449
450 if (attachment < textures.size()) {
451 tex = textures[attachment];
452 } else {
453 REGEN_WARN("Invalid attachment '" << val <<
454 "' for " << input.getDescription() << ".");
455 }
456 }
457 }
458 if (!tex.get()) {
459 REGEN_WARN("No texture found for " << input.getDescription() << ".");
460 }
461 return tex;
462}

464ref_ptr<TextureState> TextureState::load(LoadingContext &ctx, scene::SceneInputNode &input) {
465 auto scene = ctx.scene();
466
467 ref_ptr<Texture> tex = getTexture(scene, input);
468 if (tex.get() == nullptr) {
469 REGEN_WARN("Skipping unidentified texture node for " << input.getDescription() << ".");
470 return {};
471 }
472 if (input.hasAttribute("mip-level") && input.getValue<int>("mip-level", 0) > 0) {
473 auto mipLevel = input.getValue<int>("mip-level", 1);
474 auto mipTex = dynamic_cast<TextureMips2D *>(tex.get());
475 if (mipTex) {
476 uint32_t maxLevel = mipTex->numCustomMips();
477 if (static_cast<uint32_t>(mipLevel) < maxLevel) {
478 tex = mipTex->mipRefs()[mipLevel - 1];
479 } else {
480 REGEN_WARN("Mip level " << mipLevel << " is out of range for texture '" << tex->name() << "'.");
481 }
482 } else {
483 REGEN_WARN("Texture '" << tex->name() << "' is not a mip texture.");
484 }
485 }
486
487 // Set-Up the texture state
488 ref_ptr<TextureState> texState = ref_ptr<TextureState>::alloc(
489 tex, input.getValue("name"));
490
491 texState->set_ignoreAlpha(
492 input.getValue<bool>("ignore-alpha", false));
493 texState->set_mapTo(input.getValue<TextureState::MapTo>(
494 "map-to", TextureState::MAP_TO_CUSTOM));
495
496 // Describes how a texture will be mixed with existing pixels.
497 auto customBlend = ShaderFunction::load(input, "blend-function");
498 if (customBlend.get()) {
499 texState->set_blendMode(customBlend);
500 } else {
501 texState->set_blendMode(
502 input.getValue<BlendMode>("blend-mode", BLEND_MODE_SRC));
503 }
504 texState->set_blendFactor(
505 input.getValue<float>("blend-factor", 1.0f));
506
507 // Defines how a texture should be mapped on geometry.
508 auto customMapping = ShaderFunction::load(input, "mapping-function");
509 if (customMapping.get()) {
510 texState->set_mapping(customMapping);
511 } else {
512 texState->set_mapping(input.getValue<TextureState::Mapping>(
513 "mapping", TextureState::MAPPING_TEXCO));
514 }
515
516 // texel transfer wraps sampled texels before returning them.
517 if (input.hasAttribute("texel-transfer")) {
518 texState->set_texelTransfer(
519 input.getValue<TextureState::TexelTransfer>("texel-transfer",
520 TextureState::TEXEL_TRANSFER_IDENTITY));
521 } else {
522 auto customTexelTransfer = ShaderFunction::load(input, "texel-transfer");
523 if (customTexelTransfer.get()) {
524 texState->set_texelTransfer(customTexelTransfer);
525 }
526 }
527
528 // texel transfer wraps computed texture coordinates before returning them.
529 if (input.hasAttribute("texco-transfer")) {
530 texState->set_texcoTransfer(input.getValue<TextureState::TransferTexco>(
531 "texco-transfer", TextureState::TRANSFER_TEXCO_RELIEF));
532 } else {
533 auto customTexcoTransfer = ShaderFunction::load(input, "texco-transfer");
534 if (customTexcoTransfer.get()) {
535 texState->set_texcoTransfer(customTexcoTransfer);
536 }
537 }
538
539 if (input.hasAttribute("texco-scale")) {
540 auto scale = input.getValue<float>("texco-scale", 1.0f);
541 texState->set_texcoScale(scale);
542 }
543
544 if (input.hasAttribute("texco-flipping")) {
545 auto flipModeName = input.getValue("texco-flipping");
546 if (flipModeName == "x") {
547 texState->set_texcoFlipping(TextureState::TEXCO_FLIP_X);
548 } else if (flipModeName == "y") {
549 texState->set_texcoFlipping(TextureState::TEXCO_FLIP_Y);
550 } else {
551 texState->set_texcoFlipping(TextureState::TEXCO_FLIP_NONE);
552 }
553 }
554 if (input.hasAttribute("sampler-type")) {
555 texState->set_samplerType(input.getValue("sampler-type"));
556 }
557
558 return texState;
559}
