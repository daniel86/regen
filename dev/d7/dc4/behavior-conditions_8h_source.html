29 // If condition is true (or false if negated), tick the child node.
30 // Otherwise, return FAILURE.
31 BehaviorStatus tick(Blackboard& bb, float dt_s) override {
32 if (cond->evaluate(bb)) {
33 auto childStatus = child->tick(bb, dt_s);
34 // In case of "exclusive" conditions, we want to prevent that neighbor branches
35 // are activated in case the subtree of the condition fails which would
36 // happen in case of forwarding the FAILURE status.
37 if (childStatus == BehaviorStatus::FAILURE && cond->isExclusive()) {
38 return BehaviorStatus::SUCCESS;
39 } else {
40 return childStatus;
41 }
42 }
43 return cond->failureStatus();
44 }
