32Vec3f Affordance::computeSlotPosition(int idx) const {

77 const auto col = static_cast<float>(idx % cols);
78

81 forward.y = 0.0;
82 const float l = forward.length();

84 forward = Vec3f(0.0f, 0.0f, 1.0f);
85 } else {
86 forward /= l;
87 }

89 right.normalize();

93 const float gridHeight = std::ceil(static_cast<float>(slotCount) /

95
