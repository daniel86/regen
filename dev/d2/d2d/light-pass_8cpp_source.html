12LightPass::LightPass(Light::Type type, std::string_view shaderKey)
13 : State(), lightType_(type), shaderKey_(shaderKey) {
14 switch (lightType_) {
15 case Light::DIRECTIONAL:
16 mesh_ = Rectangle::getUnitQuad();
17 break;
18 case Light::SPOT:
19 mesh_ = ConeClosed::getBaseCone();
20 joinStates(ref_ptr<CullFaceState>::alloc(GL_FRONT));
21 // enable depth clamping to avoid faces to be clipped away when out of depth range.
22 joinStates(ref_ptr<ToggleState>::alloc(RenderState::DEPTH_CLAMP, true));
23 break;
24 case Light::POINT:
25 mesh_ = Box::getUnitCube();
26 joinStates(ref_ptr<CullFaceState>::alloc(GL_FRONT));
27 // enable depth clamping to avoid faces to be clipped away when out of depth range.
28 joinStates(ref_ptr<ToggleState>::alloc(RenderState::DEPTH_CLAMP, true));
29 break;
30 }
31 shadowFiltering_ = SHADOW_FILTERING_NONE;
32 numShadowLayer_ = 1;
33
34 shader_ = ref_ptr<ShaderState>::alloc();
35 joinStates(shader_);
36 setShadowFiltering(SHADOW_FILTERING_NONE);
37}

110void LightPass::createShader(const StateConfig &cfg) {
111 StateConfigurer _cfg(cfg);
112 _cfg.addState(this);
113 _cfg.addState(mesh_.get());
114 if (!lights_.empty()) {
115 // add first light to shader to set up shader defines and also instance count
116 auto &firstLight = lights_.front();
117 for (auto &in: firstLight.light->inputs()) {
118 if (in.in_->isBufferBlock()) {
119 auto *block = dynamic_cast<BufferBlock *>(in.in_.get());
120 _cfg.addInput(in.in_->name(), in.in_);
121 for (auto &blockUniform: block->stagedInputs()) {
122 numInstances_ = std::max(numInstances_, blockUniform.in_->numInstances());
123 }
124 } else {
125 _cfg.addInput(in.in_->name(), in.in_);
126 numInstances_ = std::max(numInstances_, in.in_->numInstances());
127 }
128 }
129 if (firstLight.camera.get()) {
130 auto &shadowBuffer = firstLight.camera->shadowBuffer();
131 _cfg.addInput(shadowBuffer->name(), shadowBuffer);
132 _cfg.addBufferContainer("LightCamera",
133 firstLight.camera->lightCamera()->cameraBuffer(), "_Light");
134 }
135 }
136 if (numInstances_ > 1) {
137 _cfg.define("HAS_INSTANCES", "TRUE");
138 _cfg.cfg().numInstances_ = std::max(_cfg.cfg().numInstances_, numInstances_);
139 }
140 _cfg.define("NUM_SHADOW_LAYER", REGEN_STRING(numShadowLayer_));
141
142 // replicate each mesh LOD numLayer times for indirect multi-layer rendering
143 const uint32_t numRenderLayer = _cfg.cfg().numRenderLayer();
144 if (numRenderLayer > 1) {
145 REGEN_INFO("Using indirect multi-layer light-pass with " << numRenderLayer << " layers.");
146 mesh_->createIndirectDrawBuffer(numRenderLayer);
147 }
148
149 bool isShaderValid = shader_->createShader(_cfg.cfg(), shaderKey_);
150 mesh_->updateVAO(_cfg.cfg(), shader_->shader());
151 if (!isShaderValid) {
152 REGEN_WARN("LightPass shader creation failed for key '" << shaderKey_ << "'.");
153 set_isHidden(true);
154 return;
155 }
156
157 for (auto &light: lights_) { addLightInput(light); }
158 shadowMapLoc_ = shader_->shader()->uniformLocation("shadowTexture");
159 shadowColorLoc_ = shader_->shader()->uniformLocation("shadowColorTexture");
160}

222 }

229ref_ptr<LightPass> LightPass::load(LoadingContext &ctx, scene::SceneInputNode &input) {
230 auto scene = ctx.scene();
231
232 if (!input.hasAttribute("shader")) {
233 REGEN_WARN("Missing shader attribute for " << input.getDescription() << ".");
234 return {};

237 input.getValue<Light::Type>("type", Light::SPOT),

261 }
262 }
263 if (n->hasAttribute("shadow-buffer") || n->hasAttribute("shadow-texture")) {
264 shadowMap = TextureState::getTexture(scene, *n.get(),
265 "shadow-texture", "shadow-buffer",
266 "shadow-attachment");
267 if (shadowMap.get() == nullptr) {
268 REGEN_WARN("Unable to find ShadowMap for '" << n->getDescription() << "'.");
269 }
270 }
271 if ((n->hasAttribute("shadow-buffer") && n->hasAttribute("shadow-color-attachment")) ||
272 n->hasAttribute("shadow-color-texture")) {
273 shadowColorMap = TextureState::getTexture(scene, *n.get(),
274 "shadow-color-texture", "shadow-buffer",
275 "shadow-color-attachment");
276 }

278 useShadows = (shadowMap.get() != nullptr);
279 toggle = false;
280 } else if ((shadowMap.get() != nullptr) != useShadows) {
281 REGEN_WARN("Ignoring " << input.getDescription() << ".");
282 continue;
283 }
284
285 // Each light pass can have a set of ShaderInput's
286 for (auto &m: n->getChildren()) {
287 if (m->getCategory() == "input") {
288 inputs.push_back(scene::ShaderInputProcessor::createShaderInput(scene, *m.get(), x));
289 } else {
290 REGEN_WARN("Unhandled node " << m->getDescription() << ".");
291 }
292 }
293
294 x->addLight(light, shadowCamera, shadowMap, shadowColorMap, inputs);
295 }
296
297 StateConfigurer shaderConfigurer;
298 shaderConfigurer.addNode(ctx.parent().get());
299 shaderConfigurer.addState(x.get());
300 x->createShader(shaderConfigurer.cfg());
