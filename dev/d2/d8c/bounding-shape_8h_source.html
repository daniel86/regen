411 ~BatchedBoundingShape() override {
412 const uint32_t numCopies = copyCounter_.refCount();
413 if (numCopies == 1) {

415 }

422 static const BatchType &globalBatchData_t() { return globalBatchData_; }
423
424 protected:
425 // A global contiguous buffer holding batched shape data for all instances in SOA layout.
426 inline static BatchType globalBatchData_ = BatchType();
427 // A pool of free indices into the global batch data buffer, only indices < nextGlobalIndex_ are added,
428 // initially this pool is empty.
429 inline static std::vector<uint32_t> freeGlobalIndices_ = std::vector<uint32_t>();
430 // The number of free global indices available in the pool (size() could be larger)
431 inline static uint32_t numFreeGlobalIndices_ = 0u;
432 // The index into the global batch data buffer for the next new shape instance.
433 inline static uint32_t nextGlobalIndex_ = 0u;
434
435 private:
436 // we use this only for counting how many copies are made, such that we can
437 // safely release the global index on destruction of the last copy.
438 ref_ptr<bool> copyCounter_ = ref_ptr<bool>::alloc(true);
439
440 static uint32_t reserveGlobalIndex() {
441 if (numFreeGlobalIndices_ > 0) {
442 // reuse an index from the free pool
443 return freeGlobalIndices_[--numFreeGlobalIndices_];
444 } else {
445 // assign a new index
446 uint32_t globalIndex = nextGlobalIndex_++;
447 // ensure global batch data has enough capacity
448 if (globalBatchData_.capacity < nextGlobalIndex_) {
449 static constexpr bool preserveData = true;
450 globalBatchData_.resize(nextGlobalIndex_ * 2, preserveData);
451 }
452 return globalIndex;
453 }
454 }
455
456 static void releaseGlobalIndex(uint32_t globalIndex) {
457 // add the index to the free pool
458 if (numFreeGlobalIndices_ < freeGlobalIndices_.size()) {
459 freeGlobalIndices_[numFreeGlobalIndices_++] = globalIndex;
460 } else {
461 freeGlobalIndices_.push_back(globalIndex);
462 numFreeGlobalIndices_++;
463 }
464 }
465 };
