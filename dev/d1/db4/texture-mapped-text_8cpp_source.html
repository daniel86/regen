73 const uint32_t numRequiredVertices = numCharacters_ * 6;

77 norData_.resize(numRequiredVertices + 36);

91
92 // actual width for this line

94 // remember space for splitting string at words
95 int lastSpaceIndex = 0;
96 float lastSpaceWidth = 0.0;
97
98 // get line width and split the line
99 // where it exceeds the width limit
100 for (uint32_t i = 0; i < it->size(); ++i) {
101 const wchar_t &ch = (*it)[i];
102 const float buf = lineWidth + font_->faceData(ch).advanceX * height_;
103 if (maxLineWidth > 0.0 && buf > maxLineWidth && lastSpaceIndex != 0) {
104 // maximal line length reached
105 // split string at remembered space
106
107 // insert the rest after current iterator
108 // it cannot be placed in one line with
109 // the words before lastSpaceIndex
110 auto nextIt = it;
111 nextIt++;
112 value_.insert(nextIt, it->substr(lastSpaceIndex + 1,
113 it->size() - lastSpaceIndex - 1));
114 // set the string width validated width
115 *it = it->substr(0, lastSpaceIndex);
116
117 lineWidth = lastSpaceWidth;
118 break;
119 } else if (ch == ' ' || ch == '\t') {
120 // remember spaces
121 lastSpaceIndex = i;
122 lastSpaceWidth = lineWidth;
123 }
124 lineWidth = buf;
125 }
126 if (lineWidth > actualMaxLineWidth) {
127 actualMaxLineWidth = lineWidth;
128 }
129
130 // calculate line advance!
131 switch (alignment) {
132 case ALIGNMENT_CENTER:
133 translation.x = 0.5f * (maxLineWidth - lineWidth);
134 break;
135 case ALIGNMENT_RIGHT:
136 translation.x = maxLineWidth - lineWidth;
137 break;
138 case ALIGNMENT_LEFT:
139 translation.x = 0.0f;
140 break;
141 }
142
143 // create the geometry with appropriate
144 // translation and size for each glyph
145 for (uint32_t i = 0; i < it->size(); ++i) {
146 const wchar_t &ch = (*it)[i];
147 const Font::FaceData &data = font_->faceData(ch);
148
149 glyphTranslation = Vec3f(

151 (data.top - data.height) * height_,
152 0.001 * (i + 1));
153 makeGlyphGeometry(data, translation + glyphTranslation, (float) ch,
154 v_pos,
155 v_nor,
156 v_texco,
157 &vertexCounter);
158
159 // move cursor to next glyph
160 translation.x += data.advanceX * height_;
161 }
162
163 translation.x = 0.0;
164 }
165
166 // apply offset to each vertex
167 if (centerAtOrigin_) {
168 float centerOffset = actualMaxLineWidth * 0.5f;
169 for (uint32_t i = 0; i < vertexCounter; ++i) {
170 v_pos[i].x -= centerOffset;
171 }
172 }
173
174 // set center and extends for bounding box
175 minPosition_ = v_pos[0];
176 maxPosition_ = v_pos[0];
177 for (uint32_t i = 1; i < vertexCounter; ++i) {
178 minPosition_.setMin(v_pos[i]);
179 maxPosition_.setMax(v_pos[i]);
180 }
181
182 if (numCharactersChanged) {
183 if (bufferRef_.get()) {
184 BufferObject::orphanBufferRange(bufferRef_.get());
185 }
186 posAttribute_->setVertexData(numRequiredVertices, (const byte*)v_pos);
187 texcoAttribute_->setVertexData(numRequiredVertices, (const byte*)v_texco);
188 norAttribute_->setVertexData(numRequiredVertices, (const byte*)v_nor);
189 set_numVertices(numRequiredVertices);
190 lastNumCharacters_ = numCharacters_;
191 bufferRef_ = updateVertexData();
192 updateVAO();
193 }
194
195 auto m_pos = posAttribute_->mapClientDataRaw(BUFFER_GPU_WRITE);
196 auto m_nor = norAttribute_->mapClientDataRaw(BUFFER_GPU_WRITE);
197 auto m_texco = texcoAttribute_->mapClientDataRaw(BUFFER_GPU_WRITE);
198 std::memcpy(m_pos.w, v_pos, numRequiredVertices * sizeof(Vec3f));
199 std::memcpy(m_nor.w, v_nor, numRequiredVertices * sizeof(Vec3f));
200 std::memcpy(m_texco.w, v_texco, numRequiredVertices * sizeof(Vec3f));
201 m_pos.unmap();
202 m_nor.unmap();
203 m_texco.unmap();
204}
