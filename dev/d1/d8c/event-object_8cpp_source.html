129void EventObject::dispatchEvents() {
130 // Swap queues
131 int dispatchIdx = staticData_.eventPushIndex_.value.load(std::memory_order_acquire);
132 int nextPushIdx = (dispatchIdx == 0) ? 1 : 0;
133 staticData_.eventPushIndex_.value.store(nextPushIdx, std::memory_order_release);
134
135 // Wait for any concurrent pushers on dispatchIdx to finish
136 while (staticData_.eventPusherCount_[dispatchIdx].value.load(std::memory_order_acquire) > 0) {
137 CPU_PAUSE();
138 }
139
140 // Process events
141 auto &queue = staticData_.eventQueue_[dispatchIdx];
142 for (const auto &ev : queue) {
143 if (ev.emitter) {
144 ev.emitter->emitEvent(ev.eventID, ev.data);
145 }
146 }
147 queue.clear();
148}

150int EventObject::pushLock() {
151 // load the current push index
152 int insertIdx = staticData_.eventPushIndex_.value.load(std::memory_order_acquire);
153 // increment pusher count
154 auto &count = staticData_.eventPusherCount_[insertIdx];
155 count.value.fetch_add(1, std::memory_order_acquire);
156 // make sure insertIdx is still valid, as it could be that dispatch has swapped the queues
157 // in the meantime. If not, we are protected by the counter and can proceed.
158 if (insertIdx != staticData_.eventPushIndex_.value.load(std::memory_order_acquire)) {
159 count.value.fetch_sub(1, std::memory_order_release);
160 return -1;
161 }
162 return insertIdx;
163}
