113template <BoundingShapeType BatchType> void copyBatchData(BatchedIntersectionCase &buffer) {
114 static constexpr int NUM = ShapeTraits<BatchType>::NumSoAArrays;
115 static constexpr int PREFETCH_WIDTH = 16;
116 static constexpr int PREFETCH_READ = 0;
117
118 auto& local = buffer.batchData->soaData_;
119 auto& global = buffer.globalBatchData->soaData_;
120
121 // Preload base pointers once
122 float* __restrict ld[NUM];
123 float* __restrict gl[NUM];
124 for (int i=0; i<NUM; ++i) {
125 ld[i] = static_cast<float*>(__builtin_assume_aligned(local[i].data(), 32));
126 gl[i] = static_cast<float*>(__builtin_assume_aligned(global[i].data(), 32));
127 }
128
129 // Preload global indices
130 uint32_t* __restrict gl_i = static_cast<uint32_t*>(
131 __builtin_assume_aligned(buffer.globalIndices.data(), 32));
132
133 const uint32_t numItems = buffer.numQueued;
134
135 for (uint32_t localIdx = 0; localIdx < numItems; ++localIdx) {
136 const uint32_t globalIdx = gl_i[localIdx];
137
138 // Prefetch next element (if in bounds)
139 if (localIdx + PREFETCH_WIDTH < numItems) {
140 // index prefetch: fetch global index PREFETCH_WIDTH ahead
141 __builtin_prefetch(gl_i + localIdx + PREFETCH_WIDTH, PREFETCH_READ, 1);
142 // read data prefetches: fetch all shape data arrays PREFETCH_WIDTH ahead
143 const uint32_t g2 = gl_i[localIdx + PREFETCH_WIDTH];
144 for (int i=0; i<NUM; ++i) {
145 __builtin_prefetch(gl[i] + g2, PREFETCH_READ, 1);
146 }
147 }
148
149 // Copy all shape data arrays
150 for (int i = 0; i < NUM; ++i) {
151 ld[i][localIdx] = gl[i][globalIdx];
152 }
153 }
154}

175void BatchedIntersectionTest::push(const BoundingShape &shape, uint32_t shapeIdx) {
176 // Increment per-shape-type counter and push into the correct case buffer
177 switch (shape.shapeType()) {
178 case BoundingShapeType::SPHERE:
179 pushBatchData<BoundingShapeType::SPHERE>(
180 *activeCases_[SPHERE_IDX], shape.globalIndex(), shapeIdx);
181 break;
182 case BoundingShapeType::AABB:
183 pushBatchData<BoundingShapeType::AABB>(
184 *activeCases_[AABB_IDX], shape.globalIndex(), shapeIdx);
185 break;
186 case BoundingShapeType::OBB:
187 pushBatchData<BoundingShapeType::OBB>(
188 *activeCases_[OBB_IDX], shape.globalIndex(), shapeIdx);
189 break;
190 case BoundingShapeType::FRUSTUM: [[unlikely]]
191 pushBatchData<BoundingShapeType::FRUSTUM>(
192 *activeCases_[FRUSTUM_IDX], shape.globalIndex(), shapeIdx);
193 break;
194 case BoundingShapeType::LAST: [[unlikely]]
195 break;
196 }
