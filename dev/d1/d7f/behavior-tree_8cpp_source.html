107static BehaviorTree::Node* addConditionDecorator(
108 LoadingContext &ctx,
109 scene::SceneInputNode &xmlNode,
110 BehaviorTree::Node *node) {
111 std::vector<std::unique_ptr<BehaviorTree::Condition>> conditions;
112
113 auto conditionChildren = xmlNode.getChildren("condition");
114 for (auto &xmlCond : conditionChildren) {
115 auto cond = makeCondition(xmlCond.get());
116 if (cond) {
117 conditions.push_back(std::move(cond));
118 } else {
119 REGEN_WARN("Ignoring child condition in " << xmlNode.getDescription() << ".");
120 }
121 }
122 if (conditions.size() > 1) {
123 BehaviorConditionSequence *seq = new BehaviorConditionSequence();
124 for (auto &c : conditions) {
125 seq->addCondition(std::move(c));
126 }
127 std::unique_ptr<BehaviorTree::Condition> seqPtr(seq);
128 std::unique_ptr<BehaviorTree::Node> nodePtr(node);
129 return new BehaviorConditionNode(std::move(seqPtr), std::move(nodePtr));
130 } else if (conditions.size() == 1) {
131 std::unique_ptr<BehaviorTree::Node> nodePtr(node);
132 return new BehaviorConditionNode(std::move(conditions[0]), std::move(nodePtr));
133 } else {
134 return node;
135 }
