43unsigned int BuddyAllocator::createPartition(BuddyNode *n, unsigned int size) {
44 if (size == n->size) {
45 // the buddy node fits perfectly with the requested size.
46 // Just mark the node as full and return the address.
47 n->state = FULL;
48 n->maxSpace = 0;
49 return n->address;
50 }
51#ifndef EXACT_BUDDY_ALLOCATION
52 else if(size*2 > n->size) {
53 // half of the node size is not enough for the requested memory.
54#endif
55 // Create a node that fits exactly with the requested size
56 // and the rest of the node space remains free for allocation.
57 // No intern fragmentation occurs.
58 unsigned int size0 = size;
59 unsigned int size1 = n->size - size;
60 n->state = PARTIAL;
61 n->maxSpace = size1;
62 n->leftChild = createNode(n->address, size0, n);
63 n->leftChild->state = FULL;
64 n->leftChild->maxSpace = 0;
65 n->rightChild = createNode(n->address + size0, size1, n);
66 return n->address;
67#ifndef EXACT_BUDDY_ALLOCATION
68 }
69 else {
70 // half of the node size is enough for the requested memory.
71 // Split the node with half size and continue for left child.
72 unsigned int size0 = n->size/2;
73 unsigned int size1 = n->size-size0;
74 n->state = PARTIAL;
75 n->maxSpace = size1;
76 n->leftChild = new BuddyNode(n->address, size0, n);
77 n->rightChild = new BuddyNode(n->address+size0, size1, n);
78 return createPartition(n->leftChild, size);
79 }
80#endif
81}

83bool BuddyAllocator::alloc(unsigned int size, unsigned int *x) {
84 if (buddyTree_->maxSpace < size) return false;
85
86 BuddyNode *n = buddyTree_;
87 while (true) {
88 switch (n->state) {
89 case FREE:
90 // free node with enough space. Create partitions inside the
91 // node to avoid internal fragmentation.
92 *x = createPartition(n, size);
93 computeMaxSpace(n);
94 return true;
95 case PARTIAL: {
96 // walk down the tree until we reach a FREE node.
97 int d1 = n->leftChild->maxSpace - size;
98 int d2 = n->rightChild->maxSpace - size;
99 if (d2 < 0 || (d1 >= 0 && d1 < d2)) { n = n->leftChild; }
100 else { n = n->rightChild; }
101 break;
102 }
103 default:
104 REGEN_ERROR("Invalid state " << n->state << ".");
105 REGEN_ASSERT(0);//should not be reached
106 return false;
107 }
108 }
109 return false;
110}

130void BuddyAllocator::free(unsigned int address) {
131 // find the node that allocates given address
132 BuddyNode *t = buddyTree_;
133 while (t->rightChild) {
134 t = (address >= t->rightChild->address) ? t->rightChild : t->leftChild;
135 }
136
137 // mark as free space
138 t->state = FREE;
139 t->maxSpace = t->size;
140 if (t->leftChild) {
141 deleteNode(t->leftChild);
142 t->leftChild = nullptr;
143 }
144 if (t->rightChild) {
145 deleteNode(t->rightChild);
146 t->rightChild = nullptr;
147 }
148 // join parents with both childs becoming FREE with this call
149 while (t->parent) {
150 if (t->parent->leftChild->state == FREE &&
151 t->parent->rightChild->state == FREE) {
152 t = t->parent;
153 deleteNode(t->leftChild);
154 deleteNode(t->rightChild);
155 t->leftChild = nullptr;
156 t->rightChild = nullptr;
157 // mark as free space
158 t->state = FREE;
159 t->maxSpace = t->size;
160 } else {
161 computeMaxSpace(t->parent);
162 break;
163 }
