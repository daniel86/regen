13static Vec3f computeBasePosition(const ref_ptr<Mesh> &mesh, const std::vector<ref_ptr<Mesh>> &parts) {

118void BoundingSphere::batchTest_Spheres(BatchedIntersectionCase &td) {
119 const auto numQueued = static_cast<int32_t>(td.numQueued);
120 auto *testShape = static_cast<const BoundingSphere *>(td.testShape);
121
122 REGEN_SPHERE_BATCH_DATA(t, testShape->globalBatchData_t());
123 const uint32_t t_idx = testShape->globalIndex();
124
125 auto *batchData = static_cast<const BatchOfSpheres *>(td.batchData);
126 const float *d_spherePosX = batchData->posX().data();
127 const float *d_spherePosY = batchData->posY().data();
128 const float *d_spherePosZ = batchData->posZ().data();
129 const float *d_sphereRadius = batchData->radius().data();
130
131 int32_t queuedIdx = 0;
132 for (; queuedIdx + simd::RegisterWidth <= numQueued; queuedIdx += simd::RegisterWidth) {
133 // Load the sphere data into SIMD registers
134 BatchOf_float spherePosX = BatchOf_float::loadAligned(d_spherePosX + queuedIdx);
135 BatchOf_float spherePosY = BatchOf_float::loadAligned(d_spherePosY + queuedIdx);
136 BatchOf_float spherePosZ = BatchOf_float::loadAligned(d_spherePosZ + queuedIdx);
137 BatchOf_float sphereRadius = BatchOf_float::loadAligned(d_sphereRadius + queuedIdx);
138 // r_sum = r1 + r2
139 sphereRadius += t_radius[t_idx];
140 // r_sum = r_sum * r_sum
141 sphereRadius *= sphereRadius;
142 // delta = p0 - p1
143 spherePosX -= t_posX[t_idx];
144 spherePosY -= t_posY[t_idx];
145 spherePosZ -= t_posZ[t_idx];
146 // tmp = delta(dot)delta = lengthSquared(delta)
147 BatchOf_float tmp = (
148 (spherePosX * spherePosX) +
149 (spherePosY * spherePosY) +
150 (spherePosZ * spherePosZ));
151 // tmp = (delta(dot)delta) < r_sum*r_sum
152 tmp = (tmp < sphereRadius);
153 // Convert lanes (1/0) to bitmask value
154 uint8_t mask = tmp.toBitmask8();
155 while (mask) {
156 int bitIndex = simd::nextBitIndex<uint8_t>(mask);
157 td.hits->push(td.queuedIndices[queuedIdx + bitIndex]);
158 }
159 }
160
161 // Scalar fallback for remaining items
162 for (; queuedIdx < numQueued; queuedIdx++) {
163 auto itemIdx = td.queuedIndices[queuedIdx];
164 const Vec3f delta(
165 d_spherePosX[queuedIdx] - t_posX[t_idx],
166 d_spherePosY[queuedIdx] - t_posY[t_idx],
167 d_spherePosZ[queuedIdx] - t_posZ[t_idx]);
168 // r_sum = r1 + r2
169 const float r_sum = testShape->radius() + d_sphereRadius[queuedIdx];
170 // test if squared distance between centers <= squared sum of radii
171 if (delta.lengthSquared() <= (r_sum * r_sum)) {
172 td.hits->push(itemIdx);
173 }
174 }
175}

187void BoundingSphere::batchTest_AABBs(BatchedIntersectionCase &td) {
188 const auto numQueued = static_cast<int32_t>(td.numQueued);
189 auto *testShape = static_cast<const BoundingSphere *>(td.testShape);
190
191 REGEN_SPHERE_BATCH_DATA(t, testShape->globalBatchData_t());
192 const uint32_t t_idx = testShape->globalIndex();
193
194 auto *batchData = static_cast<const BatchOfAABBs *>(td.batchData);
195 const float *d_aabbMinX = batchData->minX().data();
196 const float *d_aabbMinY = batchData->minY().data();
197 const float *d_aabbMinZ = batchData->minZ().data();
198 const float *d_aabbMaxX = batchData->maxX().data();
199 const float *d_aabbMaxY = batchData->maxY().data();
200 const float *d_aabbMaxZ = batchData->maxZ().data();
201
202 // Load sphere data into SIMD registers, we need 4 registers.
203 const BatchOf_float pX = BatchOf_float::fromScalar(t_posX[t_idx]);
204 const BatchOf_float pY = BatchOf_float::fromScalar(t_posY[t_idx]);
205 const BatchOf_float pZ = BatchOf_float::fromScalar(t_posZ[t_idx]);
206 const BatchOf_float r = BatchOf_float::fromScalar(t_radius[t_idx]);
207
208 int32_t queuedIdx = 0;
209 for (; queuedIdx + simd::RegisterWidth <= numQueued; queuedIdx += simd::RegisterWidth) {
210 // Load the AABB data into SIMD registers, 6 registers.
211 const BatchOf_float aabbMinX = BatchOf_float::loadAligned(d_aabbMinX + queuedIdx);
212 const BatchOf_float aabbMinY = BatchOf_float::loadAligned(d_aabbMinY + queuedIdx);
213 const BatchOf_float aabbMinZ = BatchOf_float::loadAligned(d_aabbMinZ + queuedIdx);
214 const BatchOf_float aabbMaxX = BatchOf_float::loadAligned(d_aabbMaxX + queuedIdx);
215 const BatchOf_float aabbMaxY = BatchOf_float::loadAligned(d_aabbMaxY + queuedIdx);
216 const BatchOf_float aabbMaxZ = BatchOf_float::loadAligned(d_aabbMaxZ + queuedIdx);
217 // Intersection test:
218 // ((spherePos + sphereRadius) > aabbMin) &&
219 // ((spherePos - sphereRadius) < aabbMax)
220 const BatchOf_float isInside = (
221 // x axis
222 ((pX + r) > aabbMinX) && ((pX - r) < aabbMaxX) &&
223 // y axis
224 ((pY + r) > aabbMinY) && ((pY - r) < aabbMaxY) &&
225 // z axis
226 ((pZ + r) > aabbMinZ) && ((pZ - r) < aabbMaxZ));
227 // Convert lanes (1/0) to bitmask value
228 uint8_t mask = isInside.toBitmask8();
229 while (mask) {
230 int bitIndex = simd::nextBitIndex<uint8_t>(mask);
231 td.hits->push(td.queuedIndices[queuedIdx + bitIndex]);
232 }
233 }
234
235 // Scalar fallback for remaining items
236 for (; queuedIdx < numQueued; queuedIdx++) {
237 auto itemIdx = td.queuedIndices[queuedIdx];
238 bool isOutside =
239 t_posX[t_idx] + t_radius[t_idx] < d_aabbMinX[queuedIdx] ||
240 t_posX[t_idx] - t_radius[t_idx] > d_aabbMaxX[queuedIdx] ||
241 t_posY[t_idx] + t_radius[t_idx] < d_aabbMinY[queuedIdx] ||
242 t_posY[t_idx] - t_radius[t_idx] > d_aabbMaxY[queuedIdx] ||
243 t_posZ[t_idx] + t_radius[t_idx] < d_aabbMinZ[queuedIdx] ||
244 t_posZ[t_idx] - t_radius[t_idx] > d_aabbMaxZ[queuedIdx];
245 if (!isOutside) {
246 td.hits->push(itemIdx);
247 }
248 }

261void BoundingSphere::batchTest_OBBs(BatchedIntersectionCase &td) {
262 auto *shapes = td.indexedShapes->data();
263 const auto numQueued = static_cast<int32_t>(td.numQueued);
264 auto *testShape = static_cast<const BoundingSphere *>(td.testShape);
265
266 REGEN_SPHERE_BATCH_DATA(t, testShape->globalBatchData_t());
267 const uint32_t t_idx = testShape->globalIndex();
268 const float sphereRadiusSq = t_radius[t_idx] * t_radius[t_idx];
269
270 auto *batchData = static_cast<const BatchOfOBBs *>(td.batchData);
271 const float *d_obbCenterX = batchData->centerX().data();
272 const float *d_obbCenterY = batchData->centerY().data();
273 const float *d_obbCenterZ = batchData->centerZ().data();
274 const float *d_obbHalfSizeX = batchData->halfSizeX().data();
275 const float *d_obbHalfSizeY = batchData->halfSizeY().data();
276 const float *d_obbHalfSizeZ = batchData->halfSizeZ().data();
277 auto d_axes = batchData->axes();
278
279 int32_t queuedIdx = 0;
280 for (; queuedIdx + simd::RegisterWidth <= numQueued; queuedIdx += simd::RegisterWidth) {
281 // Load some of the ABB data into SIMD registers, too many registers are needed so
282 // we leave out the axes and rather load them in a loop below.
283 BatchOf_float closestX = BatchOf_float::loadAligned(d_obbCenterX + queuedIdx);
284 BatchOf_float closestY = BatchOf_float::loadAligned(d_obbCenterY + queuedIdx);
285 BatchOf_float closestZ = BatchOf_float::loadAligned(d_obbCenterZ + queuedIdx);
286 const BatchOf_float obbHalfSize[3] = {
287 BatchOf_float::loadAligned(d_obbHalfSizeX + queuedIdx),
288 BatchOf_float::loadAligned(d_obbHalfSizeY + queuedIdx),
289 BatchOf_float::loadAligned(d_obbHalfSizeZ + queuedIdx)};
290
291 // Find closest point on OBB to sphere center, start with:
292 // closest = obbCenter (we will just accumulate in td.batch_obbCenterX)
293 // delta = p1 - obbCenter
294 BatchOf_float deltaX = BatchOf_float::fromScalar(t_posX[t_idx]) - closestX;
295 BatchOf_float deltaY = BatchOf_float::fromScalar(t_posY[t_idx]) - closestY;
296 BatchOf_float deltaZ = BatchOf_float::fromScalar(t_posZ[t_idx]) - closestZ;
297
298 // We need to project delta onto each OBB axis, clamp to halfSize, and accumulate
299 for (int i = 0; i < 3; ++i) {
300 // axis = obbAxes[i]
301 const auto &axisBatch = d_axes[i];
302 const BatchOf_float axisX = BatchOf_float::loadAligned(axisBatch.x.data() + queuedIdx);
303 const BatchOf_float axisY = BatchOf_float::loadAligned(axisBatch.y.data() + queuedIdx);
304 const BatchOf_float axisZ = BatchOf_float::loadAligned(axisBatch.z.data() + queuedIdx);
305 // dist = delta.dot(axis)
306 BatchOf_float dist = (deltaX * axisX) + (deltaY * axisY) + (deltaZ * axisZ);
307 // dist = clamp(dist, -halfSize, halfSize)
308 const BatchOf_float halfSizeOnAxis(obbHalfSize[i]);
309 dist.c = simd::min_ps(simd::max_ps(dist.c, -halfSizeOnAxis.c), halfSizeOnAxis.c);
310 // closest += axis * clamped, use fused-multiply-add
311 closestX.c = simd::mul_add_ps(dist.c, axisX.c, closestX.c);
312 closestY.c = simd::mul_add_ps(dist.c, axisY.c, closestY.c);
313 closestZ.c = simd::mul_add_ps(dist.c, axisZ.c, closestZ.c);
314 }
315 // delta = closest - p1
316 deltaX = closestX - BatchOf_float::fromScalar(t_posX[t_idx]);
317 deltaY = closestY - BatchOf_float::fromScalar(t_posY[t_idx]);
318 deltaZ = closestZ - BatchOf_float::fromScalar(t_posZ[t_idx]);
319 BatchOf_float tmp = (deltaX * deltaX + deltaY * deltaY + deltaZ * deltaZ);
320 // tmp = (delta(dot)delta) < r_sum*r_sum
321 tmp = (tmp < BatchOf_float::fromScalar(sphereRadiusSq));
322
323 // Convert lanes (1/0) to bitmask value
324 uint8_t mask = tmp.toBitmask8();
325 while (mask) {
326 int bitIndex = simd::nextBitIndex<uint8_t>(mask);
327 td.hits->push(td.queuedIndices[queuedIdx + bitIndex]);
328 }
329 }
330
331 // Scalar fallback for remaining items
332 for (; queuedIdx < numQueued; queuedIdx++) {
333 auto itemIdx = td.queuedIndices[queuedIdx];
334 const OBB &box = *static_cast<OBB *>(shapes[itemIdx].get());
335 if (testShape->hasIntersectionWithOBB(box)) {
336 td.hits->push(itemIdx);
337 }
338 }
339}
