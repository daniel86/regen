126 const float *d_spherePosY = batchData->posY().data();
127 const float *d_spherePosZ = batchData->posZ().data();

130 int32_t queuedIdx = 0;
131 for (; queuedIdx + simd::RegisterWidth <= numQueued; queuedIdx += simd::RegisterWidth) {
132 // Load the sphere data into SIMD registers
133 BatchOf_float spherePosX = BatchOf_float::loadAligned(d_spherePosX + queuedIdx);
134 BatchOf_float spherePosY = BatchOf_float::loadAligned(d_spherePosY + queuedIdx);

137 // r_sum = r1 + r2
138 sphereRadius += t_radius[t_idx];
139 // r_sum = r_sum * r_sum
140 sphereRadius *= sphereRadius;
141 // delta = p0 - p1
142 spherePosX -= t_posX[t_idx];
143 spherePosY -= t_posY[t_idx];
144 spherePosZ -= t_posZ[t_idx];
145 // tmp = delta(dot)delta = lengthSquared(delta)
146 BatchOf_float tmp = (
147 (spherePosX * spherePosX) +
148 (spherePosY * spherePosY) +
149 (spherePosZ * spherePosZ));
150 // tmp = (delta(dot)delta) < r_sum*r_sum
151 tmp = (tmp < sphereRadius);
152 // Convert lanes (1/0) to bitmask value
153 uint8_t mask = tmp.toBitmask8();
154 while (mask) {
155 int bitIndex = simd::nextBitIndex<uint8_t>(mask);
156 td.hits->push(td.queuedIndices[queuedIdx + bitIndex]);
157 }
158 }
159
160 // Scalar fallback for remaining items
161 for (; queuedIdx < numQueued; queuedIdx++) {
162 auto itemIdx = td.queuedIndices[queuedIdx];
163 const Vec3f delta(
164 d_spherePosX[queuedIdx] - t_posX[t_idx],
165 d_spherePosY[queuedIdx] - t_posY[t_idx],
166 d_spherePosZ[queuedIdx] - t_posZ[t_idx]);

169 // test if squared distance between centers <= squared sum of radii

260void BoundingSphere::batchTest_OBBs(BatchedIntersectionCase &td) {
261 auto *shapes = td.indexedShapes->data();
262 const auto numQueued = static_cast<int32_t>(td.numQueued);
263 auto *testShape = static_cast<const BoundingSphere *>(td.testShape);
264
265 REGEN_SPHERE_BATCH_DATA(t, testShape->globalBatchData_t());
266 const uint32_t t_idx = testShape->globalIndex();
267 const float sphereRadiusSq = t_radius[t_idx] * t_radius[t_idx];
268
269 auto *batchData = static_cast<const BatchOfOBBs *>(td.batchData);
270 const float *d_obbCenterX = batchData->centerX().data();
271 const float *d_obbCenterY = batchData->centerY().data();
272 const float *d_obbCenterZ = batchData->centerZ().data();
273 const float *d_obbHalfSizeX = batchData->halfSizeX().data();
274 const float *d_obbHalfSizeY = batchData->halfSizeY().data();
275 const float *d_obbHalfSizeZ = batchData->halfSizeZ().data();
276 auto d_axes = batchData->axes();
277
278 int32_t queuedIdx = 0;
279 for (; queuedIdx + simd::RegisterWidth <= numQueued; queuedIdx += simd::RegisterWidth) {
280 // Load some of the ABB data into SIMD registers, too many registers are needed so
281 // we leave out the axes and rather load them in a loop below.
282 BatchOf_float closestX = BatchOf_float::loadAligned(d_obbCenterX + queuedIdx);
283 BatchOf_float closestY = BatchOf_float::loadAligned(d_obbCenterY + queuedIdx);
284 BatchOf_float closestZ = BatchOf_float::loadAligned(d_obbCenterZ + queuedIdx);
285 const BatchOf_float obbHalfSize[3] = {
286 BatchOf_float::loadAligned(d_obbHalfSizeX + queuedIdx),
287 BatchOf_float::loadAligned(d_obbHalfSizeY + queuedIdx),
288 BatchOf_float::loadAligned(d_obbHalfSizeZ + queuedIdx)};
289
290 // Find closest point on OBB to sphere center, start with:
291 // closest = obbCenter (we will just accumulate in td.batch_obbCenterX)
292 // delta = p1 - obbCenter
293 BatchOf_float deltaX = BatchOf_float::fromScalar(t_posX[t_idx]) - closestX;
294 BatchOf_float deltaY = BatchOf_float::fromScalar(t_posY[t_idx]) - closestY;
295 BatchOf_float deltaZ = BatchOf_float::fromScalar(t_posZ[t_idx]) - closestZ;
296
297 // We need to project delta onto each OBB axis, clamp to halfSize, and accumulate
298 for (int i = 0; i < 3; ++i) {
299 // axis = obbAxes[i]
300 const auto &axisBatch = d_axes[i];
301 const BatchOf_float axisX = BatchOf_float::loadAligned(axisBatch.x.data() + queuedIdx);
302 const BatchOf_float axisY = BatchOf_float::loadAligned(axisBatch.y.data() + queuedIdx);
303 const BatchOf_float axisZ = BatchOf_float::loadAligned(axisBatch.z.data() + queuedIdx);
304 // dist = delta.dot(axis)
305 BatchOf_float dist = (deltaX * axisX) + (deltaY * axisY) + (deltaZ * axisZ);
306 // dist = clamp(dist, -halfSize, halfSize)
307 const BatchOf_float halfSizeOnAxis(obbHalfSize[i]);
308 dist.c = simd::min_ps(simd::max_ps(dist.c, -halfSizeOnAxis.c), halfSizeOnAxis.c);
309 // closest += axis * clamped, use fused-multiply-add
310 closestX.c = simd::mul_add_ps(dist.c, axisX.c, closestX.c);
311 closestY.c = simd::mul_add_ps(dist.c, axisY.c, closestY.c);
312 closestZ.c = simd::mul_add_ps(dist.c, axisZ.c, closestZ.c);
313 }
314 // delta = closest - p1
315 deltaX = closestX - BatchOf_float::fromScalar(t_posX[t_idx]);
316 deltaY = closestY - BatchOf_float::fromScalar(t_posY[t_idx]);
317 deltaZ = closestZ - BatchOf_float::fromScalar(t_posZ[t_idx]);
318 BatchOf_float tmp = (deltaX * deltaX + deltaY * deltaY + deltaZ * deltaZ);
319 // tmp = (delta(dot)delta) < r_sum*r_sum
320 tmp = (tmp < BatchOf_float::fromScalar(sphereRadiusSq));
321
322 // Convert lanes (1/0) to bitmask value
323 uint8_t mask = tmp.toBitmask8();
324 while (mask) {
325 int bitIndex = simd::nextBitIndex<uint8_t>(mask);
326 td.hits->push(td.queuedIndices[queuedIdx + bitIndex]);
327 }
328 }
329
330 // Scalar fallback for remaining items
331 for (; queuedIdx < numQueued; queuedIdx++) {
332 auto itemIdx = td.queuedIndices[queuedIdx];
333 const OBB &box = *static_cast<OBB *>(shapes[itemIdx].get());
334 if (testShape->hasIntersectionWithOBB(box)) {
335 td.hits->push(itemIdx);
336 }
337 }
338}
