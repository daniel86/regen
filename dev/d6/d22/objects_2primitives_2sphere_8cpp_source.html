53static Vec3f computeSphereTangent(const Vec3f &v) {
54 Vec3f vAbs = Vec3f(abs(v.x), abs(v.y), abs(v.z));
55 Vec3f v_;
56 if (1.0f - v.z < std::numeric_limits<float>::epsilon()) {
57 // there is a singularity at the back pole
58 v_ = Vec3f(1.0, 0.0, 0.0);
59 } else if (vAbs.x < vAbs.y && vAbs.x < vAbs.z) {
60 v_ = Vec3f(0.0, -v.z, v.y);
61 } else if (vAbs.y < vAbs.x && vAbs.y < vAbs.z) {
62 v_ = Vec3f(-v.z, 0, v.x);
63 } else {
64 v_ = Vec3f(-v.y, v.x, 0);
65 }
66 v_.normalize();
67 return v.cross(v_);
68}

70void pushVertex(
71 uint32_t vertexIndex,
72 double u,
73 double v,
74 const Sphere::Config &cfg,
75 Vec3f *pos,
76 Vec3f *nor,
77 Vec4f *tan,
78 Vec2f *texco) {
79 double r = std::sin(M_PI * v);
80 pos[vertexIndex] = Vec3f(
81 static_cast<float>(r * std::cos(2.0 * M_PI * u)),
82 static_cast<float>(r * std::sin(2.0 * M_PI * u)),
83 static_cast<float>(std::cos(M_PI * v))
84 );
85 if (cfg.isNormalRequired) {
86 nor[vertexIndex] = pos[vertexIndex];
87 }
88 if (cfg.isTangentRequired) {
89 Vec3f t = computeSphereTangent(pos[vertexIndex]);
90 tan[vertexIndex] = Vec4f(t.x, t.y, t.z, 1.0);
91 }
92 if (texco) {
93 texco[vertexIndex] = Vec2f(
94 static_cast<float>(u) * cfg.texcoScale.x,
95 static_cast<float>(v) * cfg.texcoScale.y);
96 }
97 pos[vertexIndex] *= cfg.posScale * 0.5;
98}

100void Sphere::generateLODLevel(const Config &cfg,
101 uint32_t lodLevel,
102 uint32_t vertexOffset,
103 uint32_t indexOffset) {
104 // map client data for writing
105 auto v_pos = (Vec3f*) pos_->clientBuffer()->clientData(0);
106 auto v_nor = (cfg.isNormalRequired ?
107 (Vec3f*) nor_->clientBuffer()->clientData(0) : nullptr);
108 auto v_tan = (cfg.isTangentRequired ?
109 (Vec4f*) tan_->clientBuffer()->clientData(0) : nullptr);
110 auto v_texco = (texco_.get() ?
111 (Vec2f*) texco_->clientBuffer()->clientData(0) : nullptr);
112 // map untyped index data (we switch type depending on the number of vertices)
113 auto indices = (byte*)indices_->clientBuffer()->clientData(0);
114 auto indexType = indices_->baseType();
115
116 double stepSizeInv = 1.0 / (double) lodLevel;
117 uint32_t vertexIndex = vertexOffset, faceIndex = indexOffset / 6;
118
119 for (uint32_t i = 0; i < lodLevel; i++) {
120 for (uint32_t j = 0; j < lodLevel; j++) {
121 double u0 = (double) i * stepSizeInv;
122 double u1 = (double) (i + 1) * stepSizeInv;
123 double v0 = (double) j * stepSizeInv;
124 double v1 = (double) (j + 1) * stepSizeInv;
125
126 if (cfg.isHalfSphere && u0 < 0.5) continue;
127
128 // create two triangles for each quad
129 uint32_t index = (faceIndex++) * 6;
130 setIndexValue(indices, indexType, index + 0, vertexIndex + 0);
131 setIndexValue(indices, indexType, index + 1, vertexIndex + 1);
132 setIndexValue(indices, indexType, index + 2, vertexIndex + 2);
133 setIndexValue(indices, indexType, index + 3, vertexIndex + 2);
134 setIndexValue(indices, indexType, index + 4, vertexIndex + 1);
135 setIndexValue(indices, indexType, index + 5, vertexIndex + 3);
136
137 // they are made of 4 vertices
138 pushVertex(vertexIndex++, u1, v1, cfg, v_pos, v_nor, v_tan, v_texco);
139 pushVertex(vertexIndex++, u1, v0, cfg, v_pos, v_nor, v_tan, v_texco);
140 pushVertex(vertexIndex++, u0, v1, cfg, v_pos, v_nor, v_tan, v_texco);
141 pushVertex(vertexIndex++, u0, v0, cfg, v_pos, v_nor, v_tan, v_texco);
142 }
143 }

146void Sphere::updateAttributes(const Config &cfg) {
147 std::vector<uint32_t> LODs;
148 uint32_t vertexOffset = 0;
149 uint32_t indexOffset = 0;
150 for (auto &lod: cfg.levelOfDetails) {
151 uint32_t lodLevel = 4 + lod * lod;
152 LODs.push_back(lodLevel);
153 uint32_t numFaces;
154 if (cfg.isHalfSphere) {
155 numFaces = lodLevel * lodLevel;
156 } else {
157 numFaces = 2 * lodLevel * lodLevel;
158 }
159 auto &x = meshLODs_.emplace_back();
160 x.d->numVertices = numFaces * 2;
161 x.d->numIndices = numFaces * 3;
162 x.d->vertexOffset = vertexOffset;
163 x.d->indexOffset = indexOffset;
164 vertexOffset += x.d->numVertices;
165 indexOffset += x.d->numIndices;
166 }
167 uint32_t numVertices = vertexOffset;
168 uint32_t numIndices = indexOffset;
169
170 // allocate attributes
171 pos_->setVertexData(numVertices);
172 if (cfg.isNormalRequired) {
173 nor_->setVertexData(numVertices);
174 }
175 if (cfg.isTangentRequired) {
176 tan_->setVertexData(numVertices);
177 }
178 TexcoMode texcoMode = cfg.texcoMode;
179 if (cfg.isTangentRequired && cfg.texcoMode == TEXCO_MODE_NONE) {
180 texcoMode = TEXCO_MODE_UV;
181 }
182 if (texcoMode == TEXCO_MODE_UV) {
183 texco_ = ref_ptr<ShaderInput2f>::alloc("texco0");
184 texco_->setVertexData(numVertices);
185 }
186 indices_ = createIndexInput(numIndices, numVertices);
187
188 for (auto i = 0u; i < LODs.size(); ++i) {
189 generateLODLevel(cfg,
190 LODs[i],
191 meshLODs_[i].d->vertexOffset,
192 meshLODs_[i].d->indexOffset);
193 }
194
195 auto indexRef = setIndices(indices_, numVertices);
196 setInput(pos_);
197 if (cfg.isNormalRequired) {
198 setInput(nor_);
199 }
200 if (cfg.isTangentRequired) {
201 setInput(tan_);
202 }
203 if (texcoMode != TEXCO_MODE_NONE) {
204 setInput(texco_);

209 // add the index buffer offset (in number of bytes)

211 }
212 activateLOD(0);
213 minPosition_ = -cfg.posScale * 0.5f;
214 maxPosition_ = cfg.posScale * 0.5f;
