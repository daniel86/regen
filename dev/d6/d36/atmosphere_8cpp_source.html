220ref_ptr<SSBO> Atmosphere::createIndirectDrawBuffer() {
221 constexpr uint32_t numLayer = 5;
222 constexpr uint32_t numLODs = 1;
223 auto &lodData = updateMesh_->meshLODs()[0];
224
225 // One draw command per cube map face (except bottom face).
226 std::array<DrawCommand,5> drawCommands;
227 DrawCommand &drawParams = drawCommands[0];
228 drawParams.mode = 1u; // 1=elements, 2=arrays
229 drawParams.setCount(lodData.d->numIndices);
230 drawParams.setFirstElement(lodData.d->indexOffset / updateMesh_->indices()->dataTypeBytes());
231 drawParams.data[3] = 0; // base vertex
232 drawParams.setInstanceCount(1);
233 drawParams.setBaseInstance(0);
234 // Copy over the data for the remaining layers.
235 // Actually, we currently use the same mesh for each face, so the draw commands
236 // are identical, however the vertex shader selects render layer based on gl_DrawID,
237 // and maps the vertex positions accordingly.
238 for (uint32_t layerIdx = 1; layerIdx < numLayer; ++layerIdx) {
239 const uint32_t lodLayerIdx = layerIdx;;
240 std::memcpy(&drawCommands[lodLayerIdx], &drawCommands[0], sizeof(DrawCommand));
241 }
242
243 indirectDrawBuffer_ = ref_ptr<DrawIndirectBuffer>::alloc(
244 "IndirectDrawBuffer", BufferUpdateFlags::NEVER);
245 auto input = ref_ptr<ShaderInputStruct<DrawCommand>>::alloc(
246 "DrawCommand", "drawParams", numLODs * numLayer);
247 input->setInstanceData(1, 1, (byte*)drawCommands.data());
248 indirectDrawBuffer_->addStagedInput(input);
249 indirectDrawBuffer_->update();
250 //indirectDrawBuffer_->setBufferData((byte*)initialData.data());
251 return indirectDrawBuffer_;
252}
