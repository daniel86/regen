81void Torus::generateLODLevel(const Config &cfg,
82 uint32_t lodLevel,
83 uint32_t vertexOffset,
84 uint32_t indexOffset) {
85 // map client data for writing
86 auto v_pos = (Vec3f*) pos_->clientBuffer()->clientData(0);
87 auto v_nor = (cfg.isNormalRequired ?
88 (Vec3f*) nor_->clientBuffer()->clientData(0) : nullptr);
89 auto v_tan = (cfg.isTangentRequired ?
90 (Vec4f*) tan_->clientBuffer()->clientData(0) : nullptr);
91 auto v_texco = (texco_.get() ?
92 (float*) texco_->clientBuffer()->clientData(0) : nullptr);
93 auto indices = (byte*)indices_->clientBuffer()->clientData(0);
94 auto indexType = indices_->baseType();
95
96 uint32_t vertexIndex = vertexOffset;
97 const float ringStep = 2.0f * M_PI / lodLevel;
98 const float tubeStep = 2.0f * M_PI / lodLevel;
99
100 for (uint32_t i = 0; i <= lodLevel; ++i) {
101 float theta = i * ringStep;
102 float cosTheta = cos(theta);
103 float sinTheta = sin(theta);
104
105 for (uint32_t j = 0; j <= lodLevel; ++j) {
106 float phi = j * tubeStep;
107 float cosPhi = cos(phi);
108 float sinPhi = sin(phi);
109
110 Vec3f pos(
111 (cfg.ringRadius + cfg.tubeRadius * cosPhi) * cosTheta,
112 cfg.tubeRadius * sinPhi,
113 (cfg.ringRadius + cfg.tubeRadius * cosPhi) * sinTheta
114 );
115
116 pos = cfg.posScale * pos;
117 v_pos[vertexIndex] = pos;
118
119 if (cfg.isNormalRequired) {
120 v_nor[vertexIndex] = Vec3f(
121 cosPhi * cosTheta,
122 sinPhi,
123 cosPhi * sinTheta);
124 }
125
126 if (cfg.texcoMode == TEXCO_MODE_UV) {
127 Vec2f texco((float) i / lodLevel, (float) j / lodLevel);
128 ((Vec2f*)v_texco)[vertexIndex] = texco * cfg.texcoScale;
129 } else if (cfg.texcoMode == TEXCO_MODE_CUBE_MAP) {
130 Vec3f texco = pos;
131 texco.normalize();
132 ((Vec3f*)v_texco)[vertexIndex] = texco;
133 }
134
135 if (cfg.isTangentRequired) {
136 v_tan[vertexIndex] = Vec4f(
137 -sinTheta,
138 0.0f,
139 cosTheta, 1.0f);
140 }
141
142 ++vertexIndex;
143 }
144 }
145
146 uint32_t iOffset = indexOffset;
147 for (uint32_t i = 0; i < lodLevel; ++i) {
148 for (uint32_t j = 0; j < lodLevel; ++j) {
149 uint32_t first = (i * (lodLevel + 1)) + j;
150 uint32_t second = vertexOffset + first + lodLevel + 1;
151 first += vertexOffset;
152
153 setIndexValue(indices, indexType, iOffset++, first);
154 setIndexValue(indices, indexType, iOffset++, first + 1);
155 setIndexValue(indices, indexType, iOffset++, second);
156
157 setIndexValue(indices, indexType, iOffset++, second);
158 setIndexValue(indices, indexType, iOffset++, first + 1);
159 setIndexValue(indices, indexType, iOffset++, second + 1);
160 }
161 }

164void Torus::updateAttributes(const Config &cfg) {
165 // Generate multiple LOD levels of the torus
166 std::vector<uint32_t> LODs;
167 uint32_t vertexOffset = 0;
168 uint32_t indexOffset = 0;
169 for (auto &lod: cfg.levelOfDetails) {
170 uint32_t lodLevel = 4u * pow(2u, lod);
171 LODs.push_back(lodLevel);
172 auto &x = meshLODs_.emplace_back();
173 x.d->numVertices = (lodLevel + 1) * (lodLevel + 1);
174 x.d->numIndices = lodLevel * lodLevel * 6;
175 x.d->vertexOffset = vertexOffset;
176 x.d->indexOffset = indexOffset;
177 vertexOffset += x.d->numVertices;
178 indexOffset += x.d->numIndices;
179 }
180 uint32_t numVertices = vertexOffset;
181 uint32_t numIndices = indexOffset;
182
183 pos_->setVertexData(numVertices);
184 if (cfg.isNormalRequired) {
185 nor_->setVertexData(numVertices);
186 }
187 if (cfg.isTangentRequired) {
188 tan_->setVertexData(numVertices);
189 }
190 if (cfg.texcoMode == TEXCO_MODE_UV) {
191 texco_ = ref_ptr<ShaderInput2f>::alloc("texco0");
192 texco_->setVertexData(numVertices);
193 } else if (cfg.texcoMode == TEXCO_MODE_CUBE_MAP) {
194 texco_ = ref_ptr<ShaderInput3f>::alloc("texco0");
195 texco_->setVertexData(numVertices);
196 }
197 indices_ = createIndexInput(numIndices, numVertices);
198
199 for (auto i = 0u; i < LODs.size(); ++i) {
200 generateLODLevel(cfg,
201 LODs[i],
202 meshLODs_[i].d->vertexOffset,
203 meshLODs_[i].d->indexOffset);
204 }
205
206 // Set up the vertex attributes
207 auto indexRef = setIndices(indices_, numVertices);
208 setInput(pos_);
209 if (cfg.isNormalRequired) {
210 setInput(nor_);
211 }
212 if (cfg.texcoMode != TEXCO_MODE_NONE) {
213 setInput(texco_);
214 }
215 if (cfg.isTangentRequired) {
216 setInput(tan_);
217 }
218 updateVertexData();
219
220 for (auto &x: meshLODs_) {
221 // add the index buffer offset (in number of bytes)
222 x.d->indexOffset = indexRef->address() + x.d->indexOffset * indices_->dataTypeBytes();
223 }
224 activateLOD(0);
225
226 minPosition_ = Vec3f::create(-cfg.tubeRadius);
227 minPosition_.x -= cfg.ringRadius;
228 minPosition_.z -= cfg.ringRadius;
229 minPosition_ *= cfg.posScale;
230
231 maxPosition_ = Vec3f::create(cfg.tubeRadius);
232 maxPosition_.x += cfg.ringRadius;
233 maxPosition_.z += cfg.ringRadius;
234 maxPosition_ *= cfg.posScale;
235}
