48void BufferObject::createMemoryPools() {
49 auto *pools = bufferPools();
50 for (int i = 0; i < (int) BufferTarget::TARGET_LAST * (int) BUFFER_STORAGE_MODE_LAST; ++i) {
51 if (pools[i] == nullptr) {
52 pools[i] = new BufferPool();
53 pools[i]->set_index(i);
54 }
55 }
56 // some buffer semantics need special attention as they require
57 // alignment to be set, i.e. when using shared buffers consecutive
58 // allocations need to be aligned to the size of the buffer.
59 for (int i = 0; i < BUFFER_STORAGE_MODE_LAST; ++i) {
60 int poolIndex;
61
62 poolIndex = (int) ARRAY_BUFFER * (int) BUFFER_STORAGE_MODE_LAST + i;
63 pools[poolIndex]->set_alignment(PACKED_BASE_ALIGNMENT);
64
65 // Element array buffers need to be aligned to 4 bytes such that binding offsets
66 // are multiple of 4 bytes (which is needed for u_32 data)
67 poolIndex = (int) ELEMENT_ARRAY_BUFFER * (int) BUFFER_STORAGE_MODE_LAST + i;
68 pools[poolIndex]->set_alignment(4);
69
70
71 poolIndex = (int) TEXTURE_BUFFER * (int) BUFFER_STORAGE_MODE_LAST + i;
72#ifdef USE_SHARED_TBO_BUFFER
73 pools[poolIndex]->set_alignment(glGetInteger(GL_TEXTURE_BUFFER_OFFSET_ALIGNMENT));
74#else
75 pools[poolIndex]->set_minSize(1);
76#endif
77 // Meaning: Max number of texels, not bytes!
78 pools[poolIndex]->set_maxSize(glGetInteger(GL_MAX_TEXTURE_BUFFER_SIZE) * 16);
79 poolIndex = (int) UNIFORM_BUFFER * (int) BUFFER_STORAGE_MODE_LAST + i;
80#ifdef USE_SHARED_UBO_BUFFER
81 pools[poolIndex]->set_alignment(glGetInteger(GL_UNIFORM_BUFFER_OFFSET_ALIGNMENT));
82#else
83 pools[poolIndex]->set_minSize(1);
84#endif
85 // common: ~64KB
86 pools[poolIndex]->set_maxSize(glGetInteger(GL_MAX_UNIFORM_BLOCK_SIZE));
87 poolIndex = (int) SHADER_STORAGE_BUFFER * (int) BUFFER_STORAGE_MODE_LAST + i;
88#ifdef USE_SHARED_SSBO_BUFFER
89 pools[poolIndex]->set_alignment(glGetInteger(GL_SHADER_STORAGE_BUFFER_OFFSET_ALIGNMENT));
90#else
91 pools[poolIndex]->set_minSize(1);
92#endif
93 // common: ~2GB
94 pools[poolIndex]->set_maxSize(glGetInteger(GL_MAX_SHADER_STORAGE_BLOCK_SIZE));
95 // common: ~4KB
96 poolIndex = (int) ATOMIC_COUNTER_BUFFER * (int) BUFFER_STORAGE_MODE_LAST + i;
97 pools[poolIndex]->set_maxSize(glGetInteger(GL_MAX_ATOMIC_COUNTER_BUFFER_SIZE));
98
99 poolIndex = (int) DRAW_INDIRECT_BUFFER * (int) BUFFER_STORAGE_MODE_LAST + i;
100 // The Indirect draw buffers are rather small.
101 // One mesh has maybe 4 DrawCommands, each 32 bytes -> 128 bytes.
102 // Set let's say we allow 100 meshes to be drawn in one indirect draw call,
103 // so 100 * 128 = 12.5 KB.
104 pools[poolIndex]->set_minSize(12800); // 12800 bytes = 12.5 KB
105 }
106}

116ref_ptr<BufferReference> &BufferObject::adoptBufferRangeInPool(uint32_t numBytes, BufferPool *memoryPool) {
117 if (numBytes == 0) {
118 REGEN_WARN("Attempting to allocate buffer of 0 bytes.");
119 return BufferReference::nullReference();
120 }
121 // get an allocator
122 BufferPool::Node *allocator = memoryPool->chooseAllocator(numBytes);
123 if (allocator == nullptr) {
124 allocator = memoryPool->createAllocator(numBytes);
125 }
126 if (allocator == nullptr) {
127 REGEN_ERROR("BufferObject::createReference: no allocator found for " << numBytes/1024.0 << " KB for " <<
128 "buffer target " << flags_.target << ", access mode " << flags_.accessMode <<
129 " and map mode " << flags_.mapMode << ".");
130 return BufferReference::nullReference();
131 }
132
133 ref_ptr<BufferReference> ref = ref_ptr<BufferReference>::alloc();
134 ref->poolReference_ = memoryPool->alloc(allocator, numBytes);
135 if (ref->poolReference_.allocatorNode == nullptr) {
136 return BufferReference::nullReference();
137 }
138 if (allocator->mapped) {
139 // store persistent mapped data pointer with the reference
140 ref->mappedData_ = (((byte*)allocator->mapped) + ref->address());
141 }
142
143 allocations_.push_back(ref);
144 ref->allocatedSize_ = numBytes;
145 ref->bufferObject_ = this;
146
147 allocatedSize_ += numBytes;
148 return allocations_.back();
149}

185void BufferObject::orphanBufferRange(BufferReference *ref) {
186 if (ref->bufferObject_ != nullptr) {
187 auto *bo = (BufferObject *) ref->bufferObject_;
188 bo->allocatedSize_ -= ref->allocatedSize_;
189 for (auto it = bo->allocations_.begin(); it != bo->allocations_.end(); ++it) {
190 if (it->get() == ref) {

193 }

205void BufferObject::setBufferData(const void *data, const ref_ptr<BufferReference> &ref) {
206 if (!flags_.isWritable()) {
207 // CPU is not allowed to write to this buffer, so we cannot set data directly.
208 // But we can copy data to a temporary buffer and then copy it to the target buffer.
209 auto tempRef = BufferObject::adoptBufferRange(
210 ref->allocatedSize(),
211 bufferPool(flags_.target, BUFFER_MODE_STATIC_WRITE));
212 glNamedBufferSubData(
213 tempRef->bufferID(),
214 tempRef->address(),
215 tempRef->allocatedSize(),
216 data);
217 glCopyNamedBufferSubData(
218 tempRef->bufferID(),
219 ref->bufferID(),
220 tempRef->address(),
221 ref->address(),
222 tempRef->allocatedSize());
223 } else if(ref->mappedData()) {
224 // the buffer is write-mapped, so we can write directly to it.
225 // this might be the case for implicit staging buffers.
226 std::memcpy(ref->mappedData(), data, ref->allocatedSize());
227 if (flags_.mapMode == BUFFER_MAP_PERSISTENT_FLUSH) {
228 glFlushMappedNamedBufferRange(ref->bufferID(),
229 0, ref->allocatedSize());
230 }
231 } else {
232 // the buffer is writable, but not mapped persistently
233 glNamedBufferSubData(
234 ref->bufferID(),
235 ref->address(),
236 ref->allocatedSize(),
237 data);
238 }
