497 constexpr void rotate(float angle, float rx, float ry, float rz) {
498 float c = cosf(angle);
499 float s = sinf(angle);
500 x = (rx * rx * (1 - c) + c) * x
501 + (rx * ry * (1 - c) - rz * s) * y
502 + (rx * rz * (1 - c) + ry * s) * z;
503 y = (ry * rx * (1 - c) + rz * s) * x
504 + (ry * ry * (1 - c) + c) * y
505 + (ry * rz * (1 - c) - rx * s) * z;
506 z = (rx * rz * (1 - c) - ry * s) * x
507 + (ry * rz * (1 - c) + rx * s) * y
508 + (rz * rz * (1 - c) + c) * z;
509 }
