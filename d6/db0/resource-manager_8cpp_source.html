18template <typename T> bool tryGet(SceneLoader *parser, const std::string &id, T &out) {

23void ResourceManager::loadResources(SceneLoader *parser, const std::string &id) {
24 if (tryGet(parser, id, assets_)) return;

33 if (tryGet(parser, id, skies_)) return;
34 if (tryGet(parser, id, worldObjects_)) return;
