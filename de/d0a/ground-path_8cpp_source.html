32void GroundPath::generateLODLevel(
33 const Config &cfg,
34 uint32_t numCurveSamples,
35 uint32_t vertexOffset,
36 uint32_t indexOffset) {
37#define _VERTEX_DATA(T, ATTR) (T*) ATTR->clientBuffer()->clientData(0)
38 Vec3f* v_pos = _VERTEX_DATA(Vec3f, pos_);
39 Vec3f* v_nor = (cfg.isNormalRequired ? _VERTEX_DATA(Vec3f, nor_) : nullptr);
40 Vec4f* v_tan = (cfg.isTangentRequired ? _VERTEX_DATA(Vec4f, tan_) : nullptr);
41 Vec2f* v_texco = (cfg.isTexCoordRequired ? _VERTEX_DATA(Vec2f, texco_) : nullptr);
42 byte* indices = _VERTEX_DATA(byte, indices_);
43 auto indexType = indices_->baseType();
44#undef _VERTEX_DATA
45
46 uint32_t vertexIndex = vertexOffset;
47 for (auto &spline: cfg.splines) {
48 float lengthAccum = 0.0f;
49 Vec2f prev;
50
51 vertexOffset = vertexIndex;
52
53 for (uint32_t sampleIdx = 0; sampleIdx <= numCurveSamples; ++sampleIdx) {
54 float t = (float)sampleIdx / (float)numCurveSamples;
55 // point on the spline
56 Vec2f p = spline.sample(t);
57 // create two points at left and right of the spline
58 // (perpendicular to the tangent)
59 Vec2f tan = spline.tangent(t);
60 Vec2f right(-tan.y, tan.x); // perpendicular
61 right.normalize();
62 // push last/first sample a bit further along the tangent to avoid
63 // artifacts when two path segments meet at sharp angles
64 if (sampleIdx == numCurveSamples) {
65 p += tan * cfg.pathWidth * 0.25f;
66 }
67 if (sampleIdx == 0) {
68 p -= tan * cfg.pathWidth * 0.25f;
69 }
70
71 Vec2f leftPos = p - right * cfg.pathWidth * 0.5f;
72 Vec2f rightPos = p + right * cfg.pathWidth * 0.5f;
73
74 // left+right vertex
75 v_pos[vertexIndex+0] = Vec3f(leftPos.x, cfg.baseHeight, leftPos.y) * cfg.posScale;
76 v_pos[vertexIndex+1] = Vec3f(rightPos.x, cfg.baseHeight, rightPos.y) * cfg.posScale;
77
78 minPosition_.setMin(v_pos[vertexIndex+0]);
79 minPosition_.setMin(v_pos[vertexIndex+1]);
80 maxPosition_.setMax(v_pos[vertexIndex+0]);
81 maxPosition_.setMax(v_pos[vertexIndex+1]);
82 if (cfg.isNormalRequired) {
83 v_nor[vertexIndex+0] = Vec3f(0.0f, 1.0f, 0.0f);
84 v_nor[vertexIndex+1] = Vec3f(0.0f, 1.0f, 0.0f);
85 }
86 if (cfg.isTexCoordRequired) {
87 if (sampleIdx > 0) {
88 // accumulate length
89 lengthAccum += (p - prev).length();
90 }
91 v_texco[vertexIndex+0] = Vec2f(0.0f, lengthAccum) * cfg.texcoScale;
92 v_texco[vertexIndex+1] = Vec2f(1.0f, lengthAccum) * cfg.texcoScale;
93 }
94 if (cfg.isTangentRequired) {
95 Vec4f tan4(tan.x, 0.0f, tan.y, 1.0f);
96 v_tan[vertexIndex+0] = tan4;
97 v_tan[vertexIndex+1] = tan4;
98 }
99 vertexIndex += 2;
100
101 // Create two triangles per sample, connecting to previous sample
102 if (sampleIdx > 0) {
103 uint32_t baseIdx = vertexOffset + (sampleIdx - 1) * 2;
104 // triangle 1
105 setIndexValue(indices, indexType, indexOffset++, baseIdx + 0);
106 setIndexValue(indices, indexType, indexOffset++, baseIdx + 1);
107 setIndexValue(indices, indexType, indexOffset++, baseIdx + 2);
108 // triangle 2
109 setIndexValue(indices, indexType, indexOffset++, baseIdx + 1);
110 setIndexValue(indices, indexType, indexOffset++, baseIdx + 3);
111 setIndexValue(indices, indexType, indexOffset++, baseIdx + 2);
112 }
113 prev = p;
114 }
115 }

118void GroundPath::updateAttributes(const Config &cfg) {
119 std::vector<uint32_t> numSamplesPerLOD;
120 uint32_t vertexOffset = 0;
121 uint32_t indexOffset = 0;
122 uint32_t numCurves = cfg.splines.size();
123
124 for (auto &lod: cfg.levelOfDetails) {
125 uint32_t numCurveSamples = 4u * pow(2u, lod);
126 uint32_t numVerticesPerCurve = (numCurveSamples + 1) * 2;
127 uint32_t numLODVertices = numCurves * numVerticesPerCurve;
128 uint32_t numLODTris = numCurves * numCurveSamples * 2;
129 numSamplesPerLOD.push_back(numCurveSamples);
130 auto &x = meshLODs_.emplace_back();
131 // two vertices per sample, one left side, one right side
132 x.d->numVertices = numLODVertices;
133 // these two vertices form two triangles with vertices from the previous sample
134 x.d->numIndices = numLODTris * 3;
135 x.d->vertexOffset = vertexOffset;
136 x.d->indexOffset = indexOffset;
137 vertexOffset += x.d->numVertices;
138 indexOffset += x.d->numIndices;
139 }
140 uint32_t numVertices = vertexOffset;
141 uint32_t numIndices = indexOffset;
142
143 pos_->setVertexData(numVertices);
144 if (cfg.isNormalRequired) {
145 nor_->setVertexData(numVertices);
146 }
147 if (cfg.isTangentRequired) {
148 tan_->setVertexData(numVertices);
149 }
150 if (cfg.isTexCoordRequired) {
151 texco_ = ref_ptr<ShaderInput2f>::alloc("texco0");
152 texco_->setVertexData(numVertices);
153 }
154 indices_ = createIndexInput(numIndices, numVertices);
155
156 minPosition_ = Vec3f::posMax();
157 maxPosition_ = Vec3f::negMax();
158 for (auto i = 0u; i < numSamplesPerLOD.size(); ++i) {
159 generateLODLevel(cfg, numSamplesPerLOD[i],
160 meshLODs_[i].d->vertexOffset,
161 meshLODs_[i].d->indexOffset);
162 }
163
164 // Set up the vertex attributes
165 auto indexRef = setIndices(indices_, numVertices);
166 setInput(pos_);
167 if (cfg.isNormalRequired) {
168 setInput(nor_);
169 }
170 if (cfg.isTexCoordRequired) {
171 setInput(texco_);
172 }
173 if (cfg.isTangentRequired) {
174 setInput(tan_);
175 }
176 updateVertexData();
177
178 for (auto &x: meshLODs_) {
179 // add the index buffer offset (in number of bytes)
180 x.d->indexOffset = indexRef->address() + x.d->indexOffset * indices_->dataTypeBytes();
181 }
182 activateLOD(0);
183}

185static inline void addPathSpline(
186 GroundPath::Config &cfg,
187 const WayPoint *source,
188 const WayPoint *target,
189 const Vec2f &dir_a,
190 const Vec2f &dir_d) {
191 Vec2f p0 = source->position2D();
192 Vec2f p3 = target->position2D();
193
194 float handleScale = (p3 - p0).length() * 0.5f;
195 Vec2f p1 = p0 - dir_a * handleScale * 0.5f;
196 Vec2f p2 = p3 + dir_d * handleScale * 0.5f;
197
198 math::Bezier<Vec2f> &spline = cfg.splines.emplace_back();
199 spline.p0 = p0;
200 spline.p1 = p1;
201 spline.p2 = p2;
202 spline.p3 = p3;
203}

205static inline void addPathSpline(
206 GroundPath::Config &cfg,
207 const WayPoint *source,
208 const WayPoint *target,
209 const WayPoint *approachingFrom,
210 const WayPoint *departingTo) {
211 Vec2f p0 = source->position2D();
212 Vec2f p3 = target->position2D();
213
214 Vec2f dir_a = p0 - approachingFrom->position2D();
215 float len_a = dir_a.length();
216 if (len_a > 1e-4f) {
217 dir_a /= len_a;
218 } else {
219 dir_a = p3 - p0;
220 dir_a.normalize();
221 }
222
223 Vec2f dir_d = departingTo->position2D() - p3;
224 float len_d = dir_d.length();
225 if (len_d > 1e-4f) {
226 dir_d /= len_d;
227 } else {
228 dir_d = p3 - p0;
229 dir_d.normalize();
230 }
231
232 float handleScale = (p3 - p0).length() * 0.5f;
233 Vec2f p1 = p0 + dir_a * handleScale * 0.5f;
234 Vec2f p2 = p3 - dir_d * handleScale * 0.5f;
235
236 math::Bezier<Vec2f> &spline = cfg.splines.emplace_back();
237 spline.p0 = p0;
238 spline.p1 = p1;
239 spline.p2 = p2;
240 spline.p3 = p3;
241}

243static inline Vec2f computeArrivalDirection(
244 const WayPoint *from,
245 const WayPoint *to,
246 const WayPoint *next) {
247 Vec2f dir = (to->position2D() - next->position2D()) * 0.5f +
248 (from->position2D() - to->position2D()) * 0.5f;
249 float len = dir.length();
250 if (len > 1e-4f) {
251 dir /= len;
252 } else {
253 dir = to->position2D() - from->position2D();
254 dir.normalize();
255 }
256 return dir;
257}

259ref_ptr<GroundPath> GroundPath::load(LoadingContext &ctx, scene::SceneInputNode &input) {
260 auto scene = ctx.scene();
261 ref_ptr<WorldModel> world = scene->worldModel();
262
263 GroundPath::Config cfg;
264 if (input.hasAttribute("lod-levels")) {
265 auto lodVec = input.getValue<Vec4ui>("lod-levels", Vec4ui::create(0));
266 cfg.levelOfDetails.resize(4);
267 cfg.levelOfDetails[0] = lodVec.x;
268 cfg.levelOfDetails[1] = lodVec.y;
269 cfg.levelOfDetails[2] = lodVec.z;
270 cfg.levelOfDetails[3] = lodVec.w;
271 } else {
272 cfg.levelOfDetails.push_back(input.getValue<uint32_t>("lod-level", 0));
273 }
274 cfg.isTexCoordRequired = input.getValue<bool>("use-texco", false);
275 cfg.isNormalRequired = input.getValue<bool>("use-normal", false);
276 cfg.isTangentRequired = input.getValue<bool>("use-tangent", false);
277 cfg.posScale = input.getValue<Vec3f>("scaling", Vec3f::one());
278 cfg.rotation = input.getValue<Vec3f>("rotation", Vec3f::zero());
279 cfg.texcoScale = input.getValue<Vec2f>("texco-scaling", Vec2f::one());
280 cfg.baseHeight = input.getValue<float>("base-height", 0.0f);
281 cfg.pathWidth = input.getValue<float>("path-width", 4.0f);
282 cfg.updateHint.frequency = input.getValue<BufferUpdateFrequency>(
283 "update-frequency", BUFFER_UPDATE_NEVER);
284 cfg.updateHint.scope = input.getValue<BufferUpdateScope>(
285 "update-scope", BUFFER_UPDATE_FULLY);
286 cfg.accessMode = input.getValue<ClientAccessMode>(
287 "access-mode", BUFFER_CPU_WRITE);
288 cfg.mapMode = input.getValue<BufferMapMode>(
289 "map-mode", BUFFER_MAP_DISABLED);
290
291 for (auto &c : world->wayPointConnections) {
292 const WayPoint *wp0 = c.first.get();
293 const WayPoint *wp1 = c.second.get();
294 // We make an assumption here that character approach this wp0 to reach wp1 by approaching
295 // from wp closest to wp1, and vice versa.
296 const WayPoint *wp_a = world->getClosestTo(*wp0, {wp1});
297 const WayPoint *wp_d = world->getClosestTo(*wp1, {wp0});
298 if (!wp_a) wp_a = wp0;
299 if (!wp_d) wp_d = wp1;
300 addPathSpline(cfg, wp0, wp1, wp_a, wp_d);
301 }
302
303 for (auto &place : world->places) {
304 for (const auto &val: place->pathWays() | std::views::values) {
305 for (auto &typedPath : val) {
306 const WayPoint *wp0 = typedPath.front().get();
307 Vec2f startDir, stopDir;
308
309 startDir = computeArrivalDirection(
310 typedPath.back().get(),
311 wp0,
312 typedPath.size() > 1 ? typedPath[1].get() : typedPath[0].get());
313
314 for (uint32_t i = 1; i < typedPath.size(); ++i) {
315 const WayPoint *wp1 = typedPath[i].get();
316 const WayPoint *wp_d = (i + 1 < typedPath.size() ? typedPath[i + 1].get() : typedPath[0].get());
317 stopDir = computeArrivalDirection(wp0, wp1, wp_d);
318 addPathSpline(cfg, wp0, wp1, startDir, stopDir);
319 wp0 = wp1;
320 startDir = stopDir;
321 }
322 // add closing segment
323 if (typedPath.size() >= 2) {
324 const WayPoint *wp1 = typedPath[0].get();
325 stopDir = computeArrivalDirection(wp0, wp1, typedPath[1].get());
326 addPathSpline(cfg, wp0, wp1, startDir, stopDir);
327 }
328 }
329 }
330 }
331
332 return ref_ptr<GroundPath>::alloc(cfg);
