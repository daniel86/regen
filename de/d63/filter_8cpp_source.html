133ref_ptr<FilterSequence> FilterSequence::load(LoadingContext &ctx, scene::SceneInputNode &input) {
134 auto scene = ctx.scene();
135 if (!input.hasAttribute("texture") && !input.hasAttribute("fbo")) {
136 REGEN_WARN("Ignoring " << input.getDescription() << " without input texture.");
137 return {};
138 }
139
140 ref_ptr<Texture> tex;
141 if (input.hasAttribute("texture")) {
142 tex = scene->getResource<Texture>(input.getValue("texture"));
143 if (tex.get() == nullptr) {
144 REGEN_WARN("Unable to find Texture for " << input.getDescription() << ".");
145 return {};
146 }
147 } else {
148 ref_ptr<FBO> fbo = scene->getResource<FBO>(input.getValue("fbo"));
149 if (fbo.get() == nullptr) {
150 REGEN_WARN("Unable to find FBO for " << input.getDescription() << ".");
151 return {};
152 }
153 auto attachment = input.getValue<uint32_t>("attachment", 0);
154 std::vector<ref_ptr<Texture> > textures = fbo->colorTextures();
155 if (attachment >= textures.size()) {
156 REGEN_WARN("FBO " << input.getValue("fbo") <<
157 " has less then " << (attachment + 1) << " attachments.");
158 return {};
159 }
160 tex = textures[attachment];
161 }
162
163 ref_ptr<FilterSequence> filterSeq = ref_ptr<FilterSequence>::alloc(
164 tex,
165 input.getValue<bool>("bind-input", true));
166 filterSeq->set_format(glenum::textureFormat(
167 input.getValue<std::string>("format", "NONE")));
168 filterSeq->set_internalFormat(glenum::textureInternalFormat(
169 input.getValue<std::string>("internal-format", "NONE")));
170 filterSeq->set_pixelType(glenum::pixelType(
171 input.getValue<std::string>("pixel-type", "NONE")));
172 if (input.hasAttribute("clear-color")) {
173 filterSeq->setClearColor(
174 input.getValue<Vec4f>("clear-color", Vec4f::zero()));
175 }
176
177 const std::list<ref_ptr<scene::SceneInputNode> > &childs = input.getChildren();
178 for (auto it = childs.begin(); it != childs.end(); ++it) {
179 ref_ptr<scene::SceneInputNode> n = *it;
180 if (!n->hasAttribute("shader")) {
181 REGEN_WARN("Ignoring filter '" << n->getDescription() << "' without shader.");
182 continue;
183 }
184 filterSeq->addFilter(ref_ptr<Filter>::alloc(
185 n->getValue("shader"),
186 n->getValue<float>("scale", 1.0f)));
187 }
188
189 StateConfigurer shaderConfigurer;
190 shaderConfigurer.addNode(ctx.parent().get());
191 shaderConfigurer.addState(filterSeq.get());
192 filterSeq->createShader(shaderConfigurer.cfg());
193
194 // Make the filter output available to the Texture resource provider.
195 //scene->putTexture(input.getName(), filterSeq->output());
196 scene->putResource<Texture>(input.getName(), filterSeq->output());
197
198 return filterSeq;
199}

220void FilterSequence::addFilter(const ref_ptr<Filter> &f) {
221 if (filterSequence_.empty()) {
222 f->set_bindInput(bindInput_);
223 f->set_format(format_);
224 f->set_internalFormat(internalFormat_);
225 f->set_pixelType(pixelType_);
226 // no filter was added before, create a new framebuffer
227 f->setInput(input_);
228 } else {
229 const FilterData &lastFilter = *filterSequence_.rbegin();
230 // another filter was added before
231 if (math::isApprox(f->scaleFactor(), 1.0)) {
232 // filter does not apply scaling. We gonna ping pong render
233 // to the framebuffer provided by previous filter.
234 f->setInput(
235 lastFilter.filter->output(),
236 lastFilter.filter->outputAttachment());
237 } else {
238 // create a new framebuffer for this filter
239 GLenum last = lastFilter.filter->outputAttachment();
240 if (last == GL_COLOR_ATTACHMENT0) {
241 f->setInput(lastFilter.filter->output()->tex0_);
242 } else {
243 f->setInput(lastFilter.filter->output()->tex1_);
244 }
245 }
246 }
247
248 auto &data = filterSequence_.emplace_back();
249 data.filter = f;
250 data.viewport = ref_ptr<ShaderInput2f>::alloc("viewport");
251 data.viewport->setUniformData(Vec2f::zero());
252 data.inverseViewport = ref_ptr<ShaderInput2f>::alloc("inverseViewport");
253 data.inverseViewport->setUniformData(Vec2f::zero());
254}

266void FilterSequence::resize() {
267 //uint32_t size = min(input_->width(), input_->height());
268 uint32_t width = input_->width();
269 uint32_t height = input_->height();
270 if (width == lastWidth_ && height == lastHeight_) return;
271 lastWidth_ = width;
272 lastHeight_ = height;
273
274 FBO *last = nullptr;
275 for (auto &it: filterSequence_) {
276 auto *f = it.filter.get();
277 FBO *fbo = f->output()->fbo_.get();
278
279 if (last != fbo) {
280 width *= f->scaleFactor();
281 height *= f->scaleFactor();
282 fbo->resize(width, height, fbo->depth());
283 }
284
285 it.viewport->setVertex(0,
286 fbo->viewport()->getVertex(0).r);
287 it.inverseViewport->setVertex(0,
288 fbo->inverseViewport()->getVertex(0).r);
289
290 last = fbo;
291 }
