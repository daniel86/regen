48LODState::LODState(
49 const ref_ptr<Camera> &camera,
50 const ref_ptr<CullShape> &cullShape)
51 : State(),
52 camera_(camera),
53 cullShape_(cullShape) {
54 hasShadowTarget_ = dynamic_cast<LightCamera *>(camera_.get()) != nullptr;
55 if (!cullShape_->parts().empty()) {
56 mesh_ = cullShape_->parts().front();
57 } else {
58 REGEN_WARN("No mesh set for shape '" << cullShape_->shapeName() << "'.");
59 }
60 if (!cullShape_->boundingShape()->transform().get()) {
61 REGEN_WARN("No TF for cull shape '"
62 << cullShape_->shapeName()
63 << "'. LOD and culling will not work properly!");
64 }
65 if (camera_->hasFixedLOD()) {
66 if (camera_->fixedLODQuality() == LODQuality::HIGH) {
67 fixedLOD_ = 0u; // always use highest quality LOD
68 } else if (camera_->fixedLODQuality() == LODQuality::LOW) {
69 fixedLOD_ = std::max(1u,mesh_->numLODs()) - 1u; // always use lowest quality LOD
70 } else {
71 fixedLOD_ = 0u;
72 if (mesh_->numLODs() == 2) {
73 fixedLOD_ = 1u;
74 } else if (mesh_->numLODs() == 3) {
75 fixedLOD_ = 1u;
76 } else if (mesh_->numLODs() > 3) {
77 fixedLOD_ = 2u;
78 }
79 }
80 }
81 initLODState();

84void LODState::initLODState() {
85 numLODs_ = 1;
86 if (shapeIndex_.get()) {
87 numLODs_ = std::max(shapeIndex_->numLODs(), numLODs_);
88 }
89 if (cullShape_.get()) {
90 for (auto &part : cullShape_->parts()) {
91 numLODs_ = std::max(part->numLODs(), numLODs_);
92 }
93 }
94 if (mesh_.get()) {
95 numLODs_ = std::max(mesh_->numLODs(), numLODs_);
96 }
97 if (useCPUPath()) {
98 auto index = cullShape_->spatialIndex();
99 shapeIndex_ = index->getIndexedShape(camera_, cullShape_->shapeName());
100 }
101
102 if (cullShape_->numInstances() > 1) {
103 // create instance buffer for per-frame updates.
104 createInstanceBuffer();
105 // Create indirect draw buffers for each mesh part and LOD.
106 createIndirectDrawBuffers();
107 } else {
108 if constexpr(LOD_USE_DIBO_FOR_SINGLE_LOD) {
109 createIndirectDrawBuffers();
110 } else if (camera_->numLayer()>1) {
111 // even for single instance we need indirect draw buffers if multiple layers are used
112 createIndirectDrawBuffers();
113 }
114 }
115 if (!indirectDrawBuffers_.empty()) {
116 cullShape_->setIndirectDrawBuffers(indirectDrawBuffers_);
117 }
118 if (instanceBuffer_.get()) {
119 // Make sure that all meshes that share the cull shape also have access to the instance buffer.
120 cullShape_->setInstanceBuffer(instanceBuffer_);
121 cullShape_->setInstanceSortMode(instanceSortMode_);
122 }
123
124 if (useCPUPath()) {
125 if (!shapeIndex_.get()) {
126 REGEN_WARN("No indexed shape found for cull shape '" << cullShape_->shapeName() << "'.");
127 } else {
128 shapeIndex_->setInstanceSortMode(instanceSortMode_);
129 // Note: we do not update in state enable, but rather use a separate animation for this.
130 // This is done as the animations are dedicated place to write to client buffers,
131 // and we avoid computations in between draw calls, however would still be ok
132 // to update in state traverse.
133 lodAnim_ = ref_ptr<InstanceUpdater>::alloc(this);
134 lodAnim_->startAnimation();
135 }
136 } else {
137 createComputeShader();
138 }
139 REGEN_INFO("Created LOD state for cull shape '"
140 << cullShape_->shapeName()
141 << "' with " << cullShape_->numInstances() << " instances, "
142 << numLODs_ << " LODs, "
143 << (cullShape_->isIndexShape() ? "CPU" : "GPU") << " mode.");
144}

146void LODState::createInstanceBuffer() {
147 // Note: make enough space for case where all instances are visible in all layers.
148 // but usually we will write far less data.
149 const int32_t numIndices = cullShape_->numInstances() * camera_->numLayer();
150 std::vector<uint32_t> clearData(numIndices, 0);
151
152 if constexpr (LOD_SEPARATE_INSTANCE_BUFFERS) {
153 instanceData_ = ref_ptr<ShaderInput1ui>::alloc("instanceIDMap", numIndices);
154 if (cullShape_->isIndexShape()) {
155 instanceData_->setInstanceData(1, 1, (byte*)clearData.data());
156 }
157 } else {
158 if (shapeIndex_.get()) {
159 instanceData_ = shapeIndex_->instanceIDs();
160 } else {
161 instanceData_ = ref_ptr<ShaderInput1ui>::alloc("instanceIDMap", numIndices);
162 }
163 }
164 instanceBuffer_ = ref_ptr<SSBO>::alloc("InstanceIDs", BufferUpdateFlags::FULL_PER_FRAME);
165 instanceBuffer_->addStagedInput(instanceData_);
166 instanceBuffer_->update();
167 if (!cullShape_->isIndexShape()) {
168 instanceBuffer_->setBufferSubData(0, numIndices, clearData.data());
169 }
170 // Set the instance buffer as input of the LOD state.
171 // This should make it available for the mesh state.
172 setInput(instanceBuffer_);
173}

175ref_ptr<SSBO> LODState::createIndirectDrawBuffer(const std::vector<DrawCommand> &initialData) {
176 const uint32_t numLayer = camera_->numLayer();
177 auto buffer = ref_ptr<DrawIndirectBuffer>::alloc(
178 "IndirectDrawBuffer",
179 BufferUpdateFlags::FULL_PER_FRAME);
180 auto input = ref_ptr<ShaderInputStruct<DrawCommand>>::alloc(
181 "DrawCommand", "drawParams", numLODs_ * numLayer);
182 if (cullShape_->isIndexShape()) {
183 input->setInstanceData(1, 1, (byte*)initialData.data());

185 buffer->addStagedInput(input);
186 buffer->update();
187 if (!cullShape_->isIndexShape()) {
188 buffer->setBufferData((byte*)initialData.data());
189 }
190 return buffer;

193static inline uint32_t getPartLOD(uint32_t lodLevel, uint32_t numPartLevels) {
194 static constexpr uint32_t lodMappings[5][4] = {
195 {0, 0, 0, 0}, // 0 LOD
196 {0, 0, 0, 0}, // 1 LODs
197 {0, 0, 1, 1}, // 2 LODs
198 {0, 1, 2, 2}, // 3 LODs
199 {0, 1, 2, 3} // 4 LODs
200 };
201 return lodMappings[numPartLevels][lodLevel];
202}

204void LODState::createIndirectDrawBuffers() {
205 const uint32_t numParts = cullShape_->parts().size();
206 const uint32_t numLayers = camera_->numLayer();
207
208 indirectDrawBuffers_.resize(numParts);
209 indirectDrawData_.resize(numParts);
210
211 for (uint32_t partIdx=0; partIdx < numParts; ++partIdx) {
212 auto &part = cullShape_->parts()[partIdx];
213 auto &partLODs = part->meshLODs();
214 auto &drawData = indirectDrawData_[partIdx];
215
216 drawData.current.resize(numLODs_ * numLayers);
217 drawData.clear.resize(numLODs_ * numLayers);
218
219 // Create the indirect draw data for this part and the first layer.
220 // DrawID order: LOD0_layer0, LOD0_layer1, LOD0_layer2
221 // LOD1_layer0, LOD1_layer1, LOD1_layer2, ...
222 for (uint32_t lodIdx = 0; lodIdx < numLODs_; ++lodIdx) {
223 const uint32_t layer0_idx = lodIdx * numLayers;
224 const uint32_t partLOD_idx = std::min(lodIdx, std::max(1u,part->numLODs())-1);
225 const auto &lodData = partLODs[partLOD_idx];
226 DrawCommand &drawParams = drawData.current[layer0_idx];
227
228 Mesh *m = lodData.impostorMesh.get() ? lodData.impostorMesh.get() : part.get();
229 if (m->indices().get()) {
230 drawParams.mode = 1u; // 1=elements, 2=arrays
231 drawParams.setCount(lodData.d->numIndices);
232 drawParams.setFirstElement(lodData.d->indexOffset / m->indices()->dataTypeBytes());
233 drawParams.data[3] = 0; // base vertex
234 } else {
235 drawParams.mode = 2u; // 1=elements, 2=arrays
236 drawParams.setCount(lodData.d->numVertices);
237 drawParams.setFirstElement(lodData.d->vertexOffset);
238 }
239 drawParams.setInstanceCount(lodIdx==0 ? part->numInstances() : 0);
240 drawParams.setBaseInstance(0);
241
242 // Set the clear draw command to zero instance count.
243 std::memcpy(
244 &drawData.clear[layer0_idx],
245 &drawParams,
246 sizeof(DrawCommand));
247 drawData.clear[layer0_idx].setInstanceCount(0);
248
249 // Copy over the data for the remaining layers.
250 for (uint32_t layerIdx = 1; layerIdx < numLayers; ++layerIdx) {
251 const uint32_t lodLayerIdx = layer0_idx + layerIdx;
252 std::memcpy(
253 &drawData.current[lodLayerIdx],
254 &drawData.current[layer0_idx],
255 sizeof(DrawCommand));
256 std::memcpy(
257 &drawData.clear[lodLayerIdx],
258 &drawData.clear[layer0_idx],
259 sizeof(DrawCommand));
260 }
261 }
262
263 // finally create the indirect draw buffer for this part
264 indirectDrawBuffers_[partIdx] = createIndirectDrawBuffer(drawData.current);
265 }

268void LODState::updateVisibility(uint32_t layerIdx, uint32_t lodLevel, uint32_t numInstances, uint32_t baseInstance) {
269 const uint32_t numLayer = camera_->numLayer();
270
271 // set the LOD level
272 for (uint32_t partIdx = 0; partIdx < cullShape_->parts().size(); ++partIdx) {
273 if (indirectDrawBuffers_.empty()) {
274 auto &part = cullShape_->parts()[partIdx];
275 part->updateVisibility(
276 getPartLOD(lodLevel, part->numLODs()),
277 numInstances, baseInstance);
278 } else {
279 auto &current = indirectDrawData_[partIdx].current;
280 auto &drawParams = current[numLayer * lodLevel + layerIdx];
281 drawParams.setInstanceCount(numInstances);
282 drawParams.setBaseInstance(baseInstance);
283 }
284 }

289
290 for (uint32_t partIdx = 0; partIdx < parts.size(); ++partIdx) {
291 auto &part = parts[partIdx];
292
293 for (uint32_t lodLevel = 0; lodLevel < part->numLODs(); ++lodLevel) {

295 }

297 std::memcpy(
298 indirectDrawData_[partIdx].current.data(),

311
312 if (useGPUPath()) {
313#ifdef LOD_DEBUG_TIME
314 static ElapsedTimeDebugger elapsedTime("GPU LOD", 300);
315 elapsedTime.beginFrame();
316 elapsedTime.push("frame begin");

341 }

410void LODState::traverseCPU() {
411 const uint32_t numLayer = camera_->numLayer();
412 bool hasVisibleInstance = shapeIndex_->isVisibleInAnyLayer();
413 if (!hasVisibleInstance_ && !hasVisibleInstance) {
414 // No instance is visible, early exit.
415 // Note: the LOD state has been reset before, so nothing to do here.
416 return;
417 }
418 hasVisibleInstance_ = hasVisibleInstance;
419
420 if (hasVisibleInstance_) {
421 if (cullShape_->numInstances() == 1) {
422 auto count = shapeIndex_->mapInstanceCounts(BUFFER_CPU_READ);
423 auto base = shapeIndex_->mapBaseInstances(BUFFER_CPU_READ);
424
425 for (uint32_t lodLevel=0; lodLevel<numLODs_; ++lodLevel) {
426 for (uint32_t layerIdx=0; layerIdx<numLayer; ++layerIdx) {
427 const uint32_t binIdx = CullShape::binIdx(lodLevel, layerIdx, numLayer);
428 if (count.r[binIdx] != 0) {
429 updateVisibility(layerIdx, lodLevel, 1, 0);
430 }
431 }
432 }
433 } else if (camera_->hasFixedLOD()) {
434 auto count = shapeIndex_->mapInstanceCounts(BUFFER_CPU_READ);
435 auto base = shapeIndex_->mapBaseInstances(BUFFER_CPU_READ);
436
437 for (uint32_t lodLevel=0; lodLevel<numLODs_; ++lodLevel) {
438 for (uint32_t layerIdx=0; layerIdx<numLayer; ++layerIdx) {
439 const uint32_t binIdx = CullShape::binIdx(lodLevel, layerIdx, numLayer);
440 if (count.r[binIdx] != 0) {
441 updateVisibility(layerIdx, fixedLOD_, count.r[binIdx], base.r[binIdx]);
442 }
443 }}
444 } else {
445 auto count = shapeIndex_->mapInstanceCounts(BUFFER_CPU_READ);
446 auto base = shapeIndex_->mapBaseInstances(BUFFER_CPU_READ);
447
448 // update local data of indirect draw buffers
449 for (uint32_t lodLevel=0; lodLevel<numLODs_; ++lodLevel) {
450 for (uint32_t layer=0; layer<numLayer; ++layer) {
451 const uint32_t binIdx = CullShape::binIdx(lodLevel, layer, numLayer);
452 if (count.r[binIdx] != 0) {
453 updateVisibility(layer, lodLevel, count.r[binIdx], base.r[binIdx]);
454 }
455 }
456 }
457
458 if constexpr (LOD_SEPARATE_INSTANCE_BUFFERS) {
459 auto ids = shapeIndex_->mapInstanceIDs(BUFFER_CPU_READ);
460 if (tfStamp_ != cullShape_->boundingShape()->tfStamp() || cameraStamp_ != camera_->stamp()) {
461 tfStamp_ = cullShape_->boundingShape()->tfStamp();
462 cameraStamp_ = camera_->stamp();
463
464 const uint32_t lastBinIdx = (numLODs_ * numLayer) - 1;
465 const uint32_t numVisible = base.r[lastBinIdx] + count.r[lastBinIdx];
466 const uint32_t dataSize = numVisible * sizeof(uint32_t);
467
468 auto mappedClientData = instanceData_->mapClientData<uint32_t>(
469 BUFFER_GPU_WRITE, 0, dataSize);
470 std::memcpy(mappedClientData.w.data(), ids.r.data(), dataSize);
471 }
472 }
473 }
474 }
475
476 // update client buffer of indirect draw buffers
477 if (!indirectDrawBuffers_.empty()) {
478 for (uint32_t partIdx = 0; partIdx < cullShape_->parts().size(); ++partIdx) {
479 auto &indirectBuffer = indirectDrawBuffers_[partIdx];
480 auto &indirectData = indirectDrawData_[partIdx];
481
482 if (indirectBuffer->hasClientData()) {
483 // also reset the client data buffer.
484 // This is done in case not all draw buffers are updated this frame using updateVisibility.
485 auto mapped = indirectBuffer->mapClientData<DrawCommand>(
486 BUFFER_GPU_WRITE, 0, indirectBuffer->inputSize());
487 std::memcpy(
488 mapped.w.data(),
489 indirectData.current.data(),
490 indirectBuffer->inputSize());
491 }
492 }
493 }
494}

500void LODState::createComputeShader() {
501 { // Create a static indirect draw buffer, which is used for clearing the
502 // first indirect draw buffer each frame.
503 auto clearData = ref_ptr<ShaderInputStruct<DrawCommand>>::alloc(
504 "DrawCommand", "drawParams", numLODs_ * camera_->numLayer());
505 clearData->setUniformUntyped((byte*)indirectDrawData_[0].clear.data());
506 clearIndirectBuffer_ = ref_ptr<SSBO>::alloc(
507 "Clear_IndirectDrawBuffer",
508 BufferUpdateFlags::NEVER,
509 SSBO::RESTRICT);
510 clearIndirectBuffer_->addStagedInput(clearData);
511 clearIndirectBuffer_->update();
512 }
513 { // Create a static buffer with zeroes for resetting the number of visible instances.
514 std::vector<uint32_t> clearData(camera_->numLayer(), 0);
515 clearNumVisibleBuffer_ = ref_ptr<SSBO>::alloc(
516 "Clear_NumVisibleBuffer",
517 BufferUpdateFlags::NEVER,
518 SSBO::RESTRICT);
519 auto clearInput = ref_ptr<ShaderInput1ui>::alloc(
520 "numVisibleKeys", camera_->numLayer());
521 clearInput->setUniformUntyped((byte*)clearData.data());
522 clearNumVisibleBuffer_->addStagedInput(clearInput);
523 clearNumVisibleBuffer_->update();
524 }
525
526 { // radix sort
527 radixSort_ = ref_ptr<RadixSort_GPU>::alloc(cullShape_->numInstances(), camera_->numLayer());
528 radixSort_->setUseCompaction(useCompaction_);
529 radixSort_->setOutputBuffer(instanceBuffer_, false);
530 radixSort_->setRadixBits(RADIX_BITS_PER_PASS);
531 radixSort_->setSortGroupSize(RADIX_GROUP_SIZE);
532 radixSort_->setScanGroupSize(RADIX_OFFSET_GROUP_SIZE);
533 radixSort_->createResources();
534 }
535
536 { // cull
537 setInput(camera_->getFrustumBuffer());
538 StateConfigurer shaderCfg;
539 if (instanceSortMode_ == SortMode::BACK_TO_FRONT) {
540 shaderCfg.define("USE_REVERSE_SORT", "TRUE");
541 }
542 if (useCompaction_) {
543 shaderCfg.define("USE_COMPACTION", "TRUE");
544 }
545 cullPass_ = ref_ptr<ComputePass>::alloc("regen.shapes.lod.radix.cull");
546 cullPass_->computeState()->shaderDefine("LOD_NUM_INSTANCES", REGEN_STRING(cullShape_->numInstances()));
547 cullPass_->computeState()->shaderDefine("NUM_LAYERS", REGEN_STRING(camera_->frustum().size()));
548 // X-direction: one work unit per instance
549 // Y-direction: one work unit per layer
550 cullPass_->computeState()->setNumWorkUnits(
551 static_cast<int>(cullShape_->numInstances()),
552 static_cast<int>(camera_->frustum().size()), 1);
553 cullPass_->computeState()->setGroupSize(RADIX_GROUP_SIZE, 1, 1);
554 cullPass_->setInput(mesh_->u_lodThresholds());
555 cullPass_->setInput(camera_->getFrustumBuffer());
556 // Note: LOD pass only writes into first buffer, we need to copy into the other buffers
557 // in a separate pass.
558 cullPass_->setInput(indirectDrawBuffers_[0]);
559 cullPass_->setInput(radixSort_->keyBuffer());
560 cullPass_->setInput(instanceBuffer_);
561 auto boundingShape = mesh_->boundingShape();
562 if (boundingShape->shapeType() == BoundingShapeType::SPHERE) {
563 auto *sphere = static_cast<BoundingSphere*>(boundingShape.get());
564 cullPass_->setInput(createUniform<ShaderInput1f, float>(
565 "shapeRadius", sphere->radius()));
566 shaderCfg.define("SHAPE_TYPE", "SPHERE");
567 }
568 else if (boundingShape->shapeType() == BoundingShapeType::AABB) {
569 auto *box = static_cast<AABB*>(boundingShape.get());
570 cullPass_->setInput(createUniform<ShaderInput4f, Vec4f>(
571 "shapeAABBMin", Vec4f::create(box->baseBounds().min,0.0f)));
572 cullPass_->setInput(createUniform<ShaderInput4f, Vec4f>(
573 "shapeAABBMax", Vec4f::create(box->baseBounds().max,0.0f)));
574 shaderCfg.define("SHAPE_TYPE", "AABB");
575 }
576 else if (boundingShape->shapeType() == BoundingShapeType::OBB) {
577 auto *box = static_cast<OBB*>(boundingShape.get());
578 cullPass_->setInput(createUniform<ShaderInput4f, Vec4f>(
579 "shapeAABBMin", Vec4f::create(box->baseBounds().min,0.0f)));
580 cullPass_->setInput(createUniform<ShaderInput4f, Vec4f>(
581 "shapeAABBMax", Vec4f::create(box->baseBounds().max,0.0f)));
582 shaderCfg.define("SHAPE_TYPE", "OBB");
583 }
584 auto &tf = cullShape_->boundingShape()->transform();
585 if (tf.get()) {
586 cullPass_->joinStates(tf);
587 }
588 cullPass_->joinStates(camera_);
