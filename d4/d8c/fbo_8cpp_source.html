114void FBO::createDepthTexture(GLenum target, GLenum format, GLenum type, bool isStencil, uint32_t numSamples) {
115 depthAttachmentTarget_ = target;
116 depthAttachmentFormat_ = format;
117 depthAttachmentType_ = type;
118
119 ref_ptr<Texture> depth;
120 if (target == GL_TEXTURE_CUBE_MAP) {
121 depth = ref_ptr<TextureCubeDepth>::alloc();
122 } else if (target == GL_TEXTURE_2D_ARRAY) {
123 ref_ptr<Texture3D> depth3D;
124 if (numSamples > 1) {
125 depth3D = ref_ptr<Texture2DArrayMultisampleDepth>::alloc(numSamples);
126 } else {
127 depth3D = ref_ptr<Texture2DArrayDepth>::alloc();
128 }
129 depth3D->set_depth(depth_);
130 depth = depth3D;
131 } else if (depth_ > 1) {
132 ref_ptr<Texture3DDepth> depth3D = ref_ptr<Texture3DDepth>::alloc();
133 depth3D->set_depth(depth_);
134 depth = depth3D;
135 } else if (numSamples > 1) {
136 depth = ref_ptr<Texture2DMultisampleDepth>::alloc(numSamples);
137 } else {
138 depth = ref_ptr<Texture2DDepth>::alloc();
139 }
140 depth->set_rectangleSize(width(), height());
141 depth->set_internalFormat(glenum::textureInternalFormat(format));
142 depth->set_pixelType(type);
143 depth->allocTexture();
144 if (numSamples == 1) {
145 depth->set_wrapping(TextureWrapping::create(GL_REPEAT));
146 depth->set_filter(TextureFilter::create(GL_NEAREST));
147 depth->set_compare(TextureCompare(GL_NONE, GL_EQUAL));
148 }
149 if (isStencil) {
150 set_depthStencilTexture(depth);
151 } else {
152 set_depthAttachment(depth);
153 }
154}

194ref_ptr<Texture> FBO::createTexture(
195 uint32_t width,
196 uint32_t height,
197 uint32_t depth,
198 uint32_t count,
199 GLenum targetType,
200 GLenum format,
201 int internalFormat,
202 GLenum pixelType,
203 uint32_t numSamples) {
204 ref_ptr<Texture> tex;
205 ref_ptr<Texture3D> tex3d;
206
207 switch (targetType) {
208 case GL_TEXTURE_RECTANGLE:
209 tex = ref_ptr<TextureRectangle>::alloc(count);
210 break;
211
212 case GL_TEXTURE_2D:
213 if (numSamples > 1) {
214 tex = ref_ptr<Texture2DMultisample>::alloc(numSamples, count);
215 } else {
216 tex = ref_ptr<Texture2D>::alloc(GL_TEXTURE_2D, count);
217 }
218 break;
219
220 case GL_TEXTURE_2D_ARRAY:
221 if (numSamples > 1) {
222 tex3d = ref_ptr<Texture2DArrayMultisample>::alloc(numSamples, count);
223 } else {
224 tex3d = ref_ptr<Texture2DArray>::alloc(GL_TEXTURE_2D_ARRAY, count);
225 }
226 tex3d->set_depth(depth);
227 tex = tex3d;
228 break;
229
230 case GL_TEXTURE_CUBE_MAP:
231 tex = ref_ptr<TextureCube>::alloc(count);
232 if (numSamples > 1) {
233 REGEN_WARN("Multisample cube textures not supported. Using normal cube texture.");
234 }
235 break;
236
237 case GL_TEXTURE_3D:
238 tex3d = ref_ptr<Texture3D>::alloc(GL_TEXTURE_3D, count);
239 if (numSamples > 1) {
240 REGEN_WARN("Multisample 3D textures not supported. Using normal 3D texture.");
241 }
242 tex3d->set_depth(depth);
243 tex = tex3d;
244 break;
245
246 default: // GL_TEXTURE_2D:
247 REGEN_WARN("Unknown texture type " << targetType << ". Using 2D texture.");
248 tex = ref_ptr<Texture2D>::alloc(GL_TEXTURE_2D, count);
249 break;
250
251 }
252
253 tex->set_rectangleSize(width, height);
254 tex->set_format(format);
255 tex->set_internalFormat(internalFormat);
256 tex->set_pixelType(pixelType);
257 tex->allocTexture();
258
259 for (uint32_t j = 0; j < count; ++j) {
260 if (numSamples == 1) {
261 tex->set_wrapping(TextureWrapping::create(GL_CLAMP_TO_EDGE));
262 tex->set_filter(TextureFilter::create(GL_LINEAR));
263 }
264 tex->nextObject();
265 }
266
267 return tex;
268}

312void FBO::blitCopy(
313 FBO &dst,
314 GLenum readAttachment,
315 GLenum writeAttachment,
316 GLbitfield mask,
317 GLenum filter,
318 bool keepRatio) {
319 if (readAttachment != GL_DEPTH_ATTACHMENT) {
320 applyReadBuffer(readAttachment);
321 }
322 if (writeAttachment != GL_DEPTH_ATTACHMENT) {
323 dst.applyDrawBuffers(writeAttachment);
324 }
325 if (keepRatio) {
326 uint32_t dstWidth = dst.width();
327 uint32_t dstHeight = dst.width() * ((float) width() / height());
328 uint32_t offsetX, offsetY;
329 if (dstHeight > dst.height()) {
330 dstHeight = dst.height();
331 dstWidth = dst.height() * ((float) height() / width());
332 offsetX = (dst.width() - dstWidth) / 2;
333 offsetY = 0;
334 } else {
335 offsetX = 0;
336 offsetY = (dst.height() - dstHeight) / 2;
337 }
338 glBlitNamedFramebuffer(
339 id(),
340 dst.id(),
341 0, 0,
342 static_cast<int32_t>(width()),
343 static_cast<int32_t>(height()),
344 static_cast<int32_t>(offsetX),
345 static_cast<int32_t>(offsetY),
346 static_cast<int32_t>(offsetX + dstWidth),
347 static_cast<int32_t>(offsetY + dstHeight),
348 mask, filter);
349 } else {
350 glBlitNamedFramebuffer(
351 id(),
352 dst.id(),
353 0, 0,
354 static_cast<int32_t>(width()),
355 static_cast<int32_t>(height()),
356 0, 0,
357 static_cast<int32_t>(dst.width()),
358 static_cast<int32_t>(dst.height()),
359 mask, filter);
360 }
361}

444void FBO::resize(uint32_t w, uint32_t h, uint32_t depth) {
445 RenderState *rs = RenderState::get();
446 if (w == width() && h == height() && depth == depth_) {
447 return; // no resize needed
448 }
449 set_rectangleSize(w, h);
450 depth_ = depth;
451
452 viewport_->setVertex(0,
453 Vec2f((float) w, (float) h));
454 inverseViewport_->setVertex(0,
455 Vec2f(1.0f / (float) w, 1.0f / (float) h));
456 glViewport_ = Vec4ui(0, 0, w, h);
457
458 // resize depth attachment
459 if (depthTexture_.get() != nullptr) {
460 depthTexture_->set_rectangleSize(w, h);
461 auto *tex3D = dynamic_cast<Texture3D *>(depthTexture_.get());
462 if (tex3D != nullptr) { tex3D->set_depth(depth); }
463 depthTexture_->allocTexture();
464 set_depthAttachment(depthTexture_);
465 }
466
467 // resize stencil attachment
468 if (stencilTexture_.get() != nullptr) {
469 stencilTexture_->set_rectangleSize(w, h);
470 auto *tex3D = dynamic_cast<Texture3D *>(stencilTexture_.get());
471 if (tex3D != nullptr) { tex3D->set_depth(depth); }
472 stencilTexture_->allocTexture();
473 set_stencilTexture(stencilTexture_);
474 }
475
476 // resize depth stencil attachment
477 if (depthStencilTexture_.get() != nullptr) {
478 depthStencilTexture_->set_rectangleSize(w, h);
479 auto *tex3D = dynamic_cast<Texture3D *>(depthStencilTexture_.get());
480 if (tex3D != nullptr) { tex3D->set_depth(depth); }
481 depthStencilTexture_->allocTexture();
482 set_depthStencilTexture(depthStencilTexture_);
483 }
484
485 uint32_t attachmentIdx = 0;
486 // resize color attachments
487 for (uint32_t colorIdx = 0; colorIdx < colorTextures_.size(); ++colorIdx) {
488 auto &tex = colorTextures_[colorIdx];
489 tex->set_rectangleSize(w, h);
490 auto *tex3D = dynamic_cast<Texture3D *>(tex.get());
491 if (tex3D != nullptr) { tex3D->set_depth(depth); }
492 tex->allocTexture();
493
494 for (uint32_t i = 0; i < tex->numObjects(); ++i) {
495 attachTexture(tex, colorAttachments_.buffers_[attachmentIdx++]);
496 tex->nextObject();
497 }
498 }
499
500 // Resize rbo attachments.
501 // Note: RBOs are always mutable, so we do not need to re-create them,
502 // nor re-attach them, just calling storage() is enough.
503 for (auto & rbo : renderBuffers_) {
504 rbo->set_rectangleSize(w, h);
505 for (uint32_t i = 0; i < rbo->numObjects(); ++i) {
506 rs->renderBuffer().push(rbo->id());
507 rbo->storage();
508 rs->renderBuffer().pop();
509 rbo->nextObject();
510 }
511 }
512
513 // reset FBO state
514 drawBuffers_.buffers_.clear();
515 readBuffer_ = GL_NONE;
516}

558ref_ptr<FBO> FBO::load(LoadingContext &ctx, scene::SceneInputNode &input) {
559 auto sizeMode = input.getValue<std::string>("size-mode", "abs");
560 auto relSize = input.getValue<Vec3f>("size", Vec3f(256.0, 256.0, 1.0));
561 auto absSize = Texture::getSize(ctx.scene()->screen()->viewport().r, sizeMode, relSize);
562 absSize.z = std::max(absSize.z, 1);
563
564 ref_ptr<FBO> fbo = ref_ptr<FBO>::alloc(absSize.x, absSize.y, absSize.z);
565 if (sizeMode == "rel") {
566 ref_ptr<FBOResizer> resizer = ref_ptr<FBOResizer>::alloc(
567 ctx.scene()->application(),
568 fbo,
569 ctx.scene()->screen(),
570 relSize.x,
571 relSize.y);
572 ctx.scene()->addEventHandler(Scene::RESIZE_EVENT, resizer);
573 }
574
575 bool hasDepthAttachment = false;
576 bool hasStencilAttachment = false;
577
578 for (auto &n: input.getChildren()) {
579 bool isRenderBuffer = n->getValue<bool>("render-buffer", false);
580 if (n->getCategory() == "texture") {
581 LoadingContext texCfg(ctx.scene(), ctx.parent());
582 ref_ptr<Texture> tex = Texture::load(texCfg, *n.get());
583 for (uint32_t j = 0; j < tex->numObjects(); ++j) {
584 fbo->addTexture(tex);
585 tex->nextObject();
586 }
587 } else if (n->getCategory() == "depth") {
588 if (hasStencilAttachment) {
589 REGEN_WARN("Ignoring depth attachment in " << n->getDescription()
590 << ", stencil attachment already present.");
591 } else if (hasDepthAttachment) {
592 REGEN_WARN("Ignoring depth attachment in " << n->getDescription()
593 << ", depth attachment already present.");
594 } else {
595 hasDepthAttachment = true;
596 auto depthSize = n->getValue<int>("pixel-size", 16);
597 GLenum depthType = glenum::pixelType(
598 n->getValue<std::string>("pixel-type", "UNSIGNED_BYTE"));
599 GLenum textureTarget = glenum::textureTarget(
600 n->getValue<std::string>("target", "TEXTURE_2D"));
601 auto numSamples = n->getValue<uint32_t>("num-samples", 1);
602
603 GLenum depthFormat;
604 if (depthSize <= 16) depthFormat = GL_DEPTH_COMPONENT16;
605 else if (depthSize <= 24) depthFormat = GL_DEPTH_COMPONENT24;
606 else depthFormat = GL_DEPTH_COMPONENT32;
607
608 if (isRenderBuffer) {
609 auto rbo = ref_ptr<RenderBuffer>::alloc(
610 depthFormat, fbo->width(), fbo->height());
611 fbo->set_depthAttachment(rbo);
612 } else {
613 fbo->createDepthTexture(textureTarget, depthFormat, depthType, numSamples);
614 ref_ptr<Texture> tex = fbo->depthTexture();
615 Texture::configure(tex, *n.get());
616 }
617 }
618 } else if (n->getCategory() == "depth-stencil") {
619 if (hasStencilAttachment) {
620 REGEN_WARN("Ignoring depth-stencil attachment in " << n->getDescription()
621 << ", stencil attachment already present.");
622 } else if (hasDepthAttachment) {
623 REGEN_WARN("Ignoring depth-stencil attachment in " << n->getDescription()
624 << ", depth attachment already present.");
625 } else {
626 hasDepthAttachment = true;
627 hasStencilAttachment = true;
628
629 auto depthSize = n->getValue<uint32_t>("depth-size", 24);
630 auto stencilSize = n->getValue<uint32_t>("stencil-size", 8);
631 if (depthSize != 24) {
632 REGEN_WARN("Ignoring depth-size in " << n->getDescription()
633 << ", only 24 is supported for depth-stencil.");
634 }
635 if (stencilSize != 8) {
636 REGEN_WARN("Ignoring stencil-size in " << n->getDescription()
637 << ", only 8 is supported for depth-stencil.");
638 }
639
640 GLenum textureTarget = glenum::textureTarget(
641 n->getValue<std::string>("target", "TEXTURE_2D"));
642 GLenum depthFormat = GL_DEPTH24_STENCIL8;
643 GLenum depthType = GL_UNSIGNED_INT_24_8;
644
645 if (isRenderBuffer) {
646 auto rbo = ref_ptr<RenderBuffer>::alloc(
647 depthFormat, fbo->width(), fbo->height());
648 fbo->set_depthStencilTexture(rbo);
649 } else {
650 auto tex = fbo->createTexture(
651 fbo->width(), fbo->height(), fbo->depth(),
652 1, textureTarget,
653 GL_DEPTH_STENCIL, depthFormat, depthType);
654 fbo->set_depthStencilTexture(tex);
655 }
656 }
657 } else if (n->getCategory() == "stencil") {
658 if (hasStencilAttachment) {
659 REGEN_WARN("Ignoring stencil attachment in " << n->getDescription()
660 << ", stencil attachment already present.");
661 } else if (hasDepthAttachment) {
662 REGEN_WARN("Ignoring stencil attachment in " << n->getDescription()
663 << ", depth attachment already present.");
664 } else {
665 hasStencilAttachment = true;
666 GLenum pixelType = glenum::pixelType(
667 n->getValue<std::string>("pixel-type", "UNSIGNED_BYTE"));
668 GLenum textureTarget = glenum::textureTarget(
669 n->getValue<std::string>("target", "TEXTURE_2D"));
670
671 GLenum stencilFormat;
672 auto stencilSize = n->getValue<int>("pixel-size", 8);
673 if (stencilSize < 8) stencilFormat = GL_STENCIL_INDEX1;
674 else if (stencilSize < 16) stencilFormat = GL_STENCIL_INDEX4;
675 else stencilFormat = GL_STENCIL_INDEX8;
676
677 if (isRenderBuffer) {
678 auto rbo = ref_ptr<RenderBuffer>::alloc(
679 stencilFormat, fbo->width(), fbo->height());
680 fbo->set_stencilTexture(rbo);
681 } else {
682 auto stencilTexture = FBO::createTexture(
683 fbo->width(), fbo->height(), fbo->depth(),
684 1, textureTarget,
685 GL_STENCIL_INDEX,
686 stencilFormat,
687 pixelType);
688 fbo->set_stencilTexture(stencilTexture);
689 }
690 }
691 } else {
692 REGEN_WARN("No processor registered for '" << n->getDescription() << "'.");
693 }
694 }
695
696 if (input.hasAttribute("clear-color")) {
697 auto c = input.getValue<Vec4f>("clear-color", Vec4f::zero());
698 fbo->clearAllColorAttachments(c);
