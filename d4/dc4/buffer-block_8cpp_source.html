36void BufferBlock::enableBufferBlock(int loc) {
37 if (!isBufferValid_) return;
38 auto *rs = RenderState::get();
39
40 prepareRebind(loc);
41 if (!shared_->isGloballyStaged_) {
42 update();
43 }
44 rs->bufferRange(glTarget_).apply(loc, *drawBufferRange_.get());
45 if (shared_->stagingBuffer_.get()) {
46 // mark the point of accessing a mapped buffer segment for draw operation,
47 // which is needed to avoid writing to the buffer while it is being read.
48 // only used in implicit staging mode, as the fence point is set after the
49 // staging-to-main copy in explicit staging mode.
50 // note: this is used in global and local staging modes.
51 shared_->stagingBuffer_->markDrawAccessed(*drawBufferRange_.get());
52 }
53 bindingIndex_ = loc;
54}

63void BufferBlock::prepareRebind(int loc) {
64 if (bindingIndex_ != loc && bindingIndex_ != -1) {
65 // seems the buffer switched to another index!
66 // this is something the buffer manager should try to avoid, but there are some situations
67 // where it might be difficult.
68 // In case of doing the switch, we need to unbind the old binding index.
69 auto *rs = RenderState::get();
70 auto &actual = rs->bufferRange(glTarget_).value(bindingIndex_);
71 if (actual.buffer_ == drawBufferRef_->bufferID() &&
72 actual.offset_ == drawBufferRef_->address() &&
73 actual.size_ == drawBufferRef_->allocatedSize()) {
74 //REGEN_INFO("Rebinding buffer block " << name()
75 // << " from binding index " << bindingIndex_ << " to " << loc);
76 rs->bufferRange(glTarget_).apply(bindingIndex_, BufferRange::nullReference());
77 bindingIndex_ = -1;
78 }
79 }

83 auto blockType = input.getValue<std::string>("type", "ubo");

85 BufferUpdateFlags updateFlags;

89
90 ref_ptr<BufferBlock> block;
91 if (blockType == "ubo") {
92 block = ref_ptr<UBO>::alloc(input.getName(), updateFlags);
93 } else if (blockType == "ssbo") {
94 block = ref_ptr<SSBO>::alloc(input.getName(), updateFlags);
95 } else {
96 REGEN_WARN("Unknown buffer block type '" << blockType << "'. Using UBO.");
97 block = ref_ptr<UBO>::alloc(input.getName(), updateFlags);
98 }
99 if (input.hasAttribute("access-mode")) {
100 block->setClientAccessMode(
101 input.getValue<ClientAccessMode>("access-mode", BUFFER_CPU_WRITE));
102 }
103 if (input.hasAttribute("map-mode")) {
104 block->setBufferMapMode(
105 input.getValue<BufferMapMode>("map-mode", BUFFER_MAP_DISABLED));
106 }
107 auto dummyState = ref_ptr<State>::alloc();
108
109 for (auto &n: input.getChildren()) {
110 if (n->getCategory() == "uniform" || n->getCategory() == "input") {
111 auto uniform = scene::ShaderInputProcessor::createShaderInput(
112 ctx.scene(), *n.get(), dummyState);
113 if (uniform->isVertexAttribute()) {
114 REGEN_WARN("UBO cannot contain vertex attributes. In node '" << n->getDescription() << "'.");
115 continue;
116 }
117 auto name = n->getValue("name");
118 block->addStagedInput(uniform, name);
119 } else {
120 REGEN_WARN("Unknown UBO child category '" << n->getCategory() << "'.");
121 }
122 }
