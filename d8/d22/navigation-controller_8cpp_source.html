96void NavigationController::startApproaching(const Vec2f &source, const Vec2f &target) {
97 // Direction from source to target
98 Vec2f dir = target - source;
99 float len = dir.length();
100 if (len < 1e-6f) {
101 dir = Vec2f(cos(currentDir_.x + baseOrientation_), sin(currentDir_.x + baseOrientation_));
102 } else {
103 dir /= len;
104 }
105
106 // Mix with orientation of next waypoint, if any.
107 if (currentPathIndex_ + 2 <= currentPath_.size()) {
108 Vec2f nextPos = currentPath_[currentPathIndex_ + 1]->position2D();
109 Vec2f nextDir = nextPos - target;
110 float nextLen = nextDir.length();
111 if (nextLen > 1e-6f) {
112 nextDir /= nextLen;
113 // blend directions
114 nextDir = math::lerp(nextDir, nextDir, 0.25f);
115 float nextDirLen = nextDir.length();
116 if (nextDirLen > 1e-6f) {
117 dir = nextDir / nextDirLen;
118 }
119 }
120 }
121
122 float angle = atan2(dir.y, dir.x) - baseOrientation_;
123 startApproaching(source, target, Vec3f(angle, 0.0f, 0.0f));
124}

126void NavigationController::startApproaching(
127 const Vec2f &source,
128 const Vec2f &target,
129 const Vec3f &desiredDir) {
130 curvePath_.p0 = source;
131 curvePath_.p3 = target;
132 // compute control points using Euler angles
133 float directDistance = (curvePath_.p0 - curvePath_.p3).length();
134 // add base orientation
135 float angle_rad1 = currentDir_.x + baseOrientation_;
136 float angle_rad2 = desiredDir.x + baseOrientation_;
137 float angleDiff = wrapPi(angle_rad2 - angle_rad1);
138 // shorter handles for sharper turns
139 float handleScale = directDistance * 0.5f * (1.0f - 0.5f * std::abs(angleDiff) / M_PI);
140 //float handleScale = directDistance * 0.25f * (1.0f - 0.5f * std::abs(angleDiff) / M_PI);
141 // p1: in direction of start orientation
142 // - Allow sharper turns by using a shorter handle. e.g. in case NPC makes a 180deg turn.
143 curvePath_.p1 = curvePath_.p0 + Vec2f(cos(angle_rad1), sin(angle_rad1)) * handleScale * 0.25f;
144 // p2: in direction of target orientation
145 const Vec2f approachDir = Vec2f(cos(angle_rad2), sin(angle_rad2));
146 curvePath_.p2 = curvePath_.p3 - approachDir * handleScale;
147 curveLUT_ = curvePath_.buildArcLengthLUT(numLUTEntries_);
148 curveEndDir_.x = approachDir.x;
149 curveEndDir_.y = 0.0f;

153 // reset approach time

156 approachDuration_ = curveLUT_.totalLength / (isWalking_ ? walkSpeed_ : runSpeed_);

198void NavigationController::handleCharacterCollision(const CollisionData &percept) {
199 // Note: percept.distance measures center-to-center distance.
200 if (percept.distance < personalSpace_ * 3.0f) {
201 // The other character is close by, steer away.
202 // Use inverse square law for strength falloff.
203 float distSqInv = 1.0f / (percept.distance * percept.distance);
204 // scale to reasonable values
205 distSqInv *= personalSpace_ * characterAvoidance_;
206 // Scale by motion activity -> less avoidance when standing still, but at least 10% influence
207 // such that small corrections can be done when standing still.
208 distSqInv *= std::clamp(currentSpeed_ / walkSpeed_, 0.1f, 1.0f);
209 navCollision_.x += percept.dir.x * distSqInv;
210 navCollision_.z += percept.dir.z * distSqInv;
211 navCollisionCount_++;
212 }

215void NavigationController::handleStaticCollision(const CollisionData &percept) {
216#ifdef USE_DYNAMIC_LOOKAHEAD
217 float influenceRadius = dynLookAheadDistance_;
218#else
219 float influenceRadius = lookAheadDistance_;
220#endif
221 const Vec3f &otherCenter3D = percept.other->tfOrigin();
222 const Vec3f &selfCenter3D = percept.self->tfOrigin();
223
224 // Find the closest point on the surface of the other shape.
225 // It could be a huge shape, so we can't just use center-to-center direction.
226 // Note: closestPointOnSurface() returns a point in world space.
227 // Note: If we are very close to the wall, closestDir might be numerically unstable,
228 // so we need to handle this case.
229 const Vec3f closest = percept.other->closestPointOnSurface(selfCenter3D);
230 Vec3f closestDir = selfCenter3D - closest;
231 const float closestDistance = closestDir.length();
232 if (closestDistance > influenceRadius) return;
233 if (closestDistance < 1e-4f) {
234 // We are extremely close to the wall, use center-to-center direction instead.
235 closestDir = percept.dir;
236 } else {
237 closestDir /= closestDistance;
238 }
239
240 // Compute direction from the closest point on the surface to the shape center.
241 Vec3f innerWallDir = closest - otherCenter3D;
242 innerWallDir.y = 0.0f;
243 innerWallDir.normalize();
244 float innerAngle = closestDir.dot(innerWallDir);
245 if (innerAngle < 0.0f) {
246 // We are inside the wall, use center direction instead.
247 closestDir = percept.dir;
248 innerAngle *= -1.0f;
249 }
250
251 // Compute tangent blend factor based on angle between delta and innerWallDelta. Here we set
252 // tangent max influence if the approach direction coincides with dir between the closest
253 // surface point and shape origin.
254 const float tanBlend = std::clamp(
255 wallTangentWeight_ + 0.5f*innerAngle, 0.0f, 1.0f);
256 Vec3f avoidance = Vec3f(-closestDir.z, 0.0f, closestDir.x);
257 if (avoidance.dot(currentVel_) < 0.0f) {
258 // Pick the tangent handedness that goes in the direction of current velocity
259 avoidance = -avoidance;
260 }
261 avoidance = closestDir * (1.0f - tanBlend) + avoidance * tanBlend;
262 avoidance.normalize();
263 // Scale avoidance strength based on distance.
264 // Here we use a simple linear falloff.
265 avoidance *= influenceRadius / (0.1f + closestDistance);
266 // Other falloff options:
267 //avoidance *= std::clamp(influenceRadius / (0.3f + dist), 0.0f, maxAvoidance_);
268 //avoidance *= std::max(0.0f, 1.0f - dist / influenceRadius) * 10.0f;
269 //avoidance *= expf(-dist * dist / (2.0f * sigma * sigma));
270
271 // Accumulate avoidance vector.
272 navCollision_ += avoidance * wallAvoidance_;
273 navCollisionCount_++;
274}

321void NavigationController::updateNavVelocity(float desiredSpeed, const Vec3f &avoidance, float avoidanceStrength) {
322 // Compute desired velocity by blending approach-dir with avoidance.
323 if (avoidanceStrength > 1e-4f) {
324 // Blend desired with avoidance vector to avoid collisions.
325 desiredVel_ = (approachDir_ + avoidance);
326 float d = desiredVel_.length();
327 if (d < 1e-4f) {
328 desiredVel_ = approachDir_;
329 } else {
330 desiredVel_ /= d;
331 }
332 // Slow down when avoidance is strong.
333 desiredVel_ *= desiredSpeed * std::clamp(1.0f - avoidanceStrength*0.05f, 0.5f, 1.0f);
334 } else {
335 desiredVel_ = approachDir_ * desiredSpeed;
336 }
337
338 // First assume current vel, then blend toward desired vel
339 currentVelDir_ = currentVel_;
340 currentSpeed_ = currentVelDir_.length();
341 if (currentSpeed_ > 1e-4f) currentVelDir_ /= currentSpeed_;
342
343 // Update position by following the desired velocity
344 float angleDiff = acos(std::clamp(desiredVel_.dot(currentVelDir_), -1.0f, 1.0f));
345 float blendFactor = std::clamp(0.1f + (angleDiff / M_PIf) * 0.4f, 0.1f, 0.5f);
346 // Note: avoid rapid changes in velocity by blending with current vel
347 currentVel_ = math::lerp(currentVel_, desiredVel_, blendFactor);
348}

350void NavigationController::updateNavOrientation(const Vec2f &currentPos2D, float desiredSpeed) {
351 // Blend velocity direction and path tangent to form a stable target direction.
352 // Idea: when moving fast, prefer velocity; when slow, prefer path tangent.
353 float vWeight = std::clamp(currentSpeed_ / (desiredSpeed + 1e-6f), 0.0f, 1.0f); // 0..1
354 float velBlend = vWeight * velOrientationWeight_; // in [0,1]
355 // clamp rotation speed (radians per second)
356 float maxDeltaThisFrame = maxTurn_ * lastDT_;
357
358 // Reduce influence of velocity when close to target, and
359 // when we are approaching the last waypoint of the current path.
360 if (currentPathIndex_ + 1 == currentPath_.size() || currentPath_.empty()) {
361 const float distToGoal = (curvePath_.p3 - currentPos2D).lengthSquared();
362 const float goalProximity = std::clamp(1.0f - distToGoal / personalSpaceSq_, 0.0f, 1.0f);
363 if (goalProximity > 1e-4f) {
364 velBlend *= (1.0f - goalProximity);
365 // allow faster turning when close to goal
366 maxDeltaThisFrame += turnFactorPersonalSpace_ * goalProximity * maxDeltaThisFrame;
367 // Final orientation blending toward goal
368 desiredDir_ = math::lerp(desiredDir_, curveEndDir_, goalProximity);
369 const float tmpLen = desiredDir_.length();
370 if (tmpLen > 1e-6f) desiredDir_ /= tmpLen;
371 }
372 }
373
374 // compute blended direction
375 Vec3f blendedDir = desiredDir_;
376 if (currentSpeed_ > 1e-4f) {
377 blendedDir *= (1.0f - velBlend);
378 blendedDir += currentVelDir_ * velBlend;
379 // safety: fallback to tangent if blended is too small
380 float tmpLen = blendedDir.length();
381 if (tmpLen > 1e-4f) blendedDir /= tmpLen;
382 }
383
384 // compute target yaw (world yaw)
385 float targetYaw = atan2(blendedDir.z, blendedDir.x);
386 // retrieve current stored yaw (undo baseOrientation_ offset that is stored in currentDir_.x)
387 float prevYaw = currentDir_.x + baseOrientation_;
388 // shortest angular difference
389 float deltaYaw = wrapPi(targetYaw - prevYaw);
390 if (deltaYaw > maxDeltaThisFrame) deltaYaw = maxDeltaThisFrame;
391 else if (deltaYaw < -maxDeltaThisFrame) deltaYaw = -maxDeltaThisFrame;
392 // apply the clamped rotation
393 currentDir_.x = prevYaw + deltaYaw - baseOrientation_;
394}

396void NavigationController::updateNavController() {
397 const Vec2f currentPos2D(currentPos_.x, currentPos_.z);
398 float desiredSpeed = (isWalking_ ? walkSpeed_ : runSpeed_);
399 float tmpLen;
400
401 if (navModeMask_ & APPROACHING) {
402 // Compute next step along the Bezier curve, and the approach direction
403 // from current position to that point.
404 auto bezierSample = curvePath_.sample(curveTime_);
405 approachDir_.x = bezierSample.x - currentPos_.x;
406 approachDir_.y = 0.0f;
407 approachDir_.z = bezierSample.y - currentPos_.z;
408 tmpLen = approachDir_.length();
409 if (tmpLen < 1e-4f) { return; }
410 approachDir_ /= tmpLen;
411
412 // Compute curve tangent, this is the desired direction along the path.
413 const Vec2f bezierTan2 = curvePath_.tangent(curveTime_);
414 desiredDir_.x = bezierTan2.x;
415 desiredDir_.y = 0.0f;
416 desiredDir_.z = bezierTan2.y;
417 tmpLen = desiredDir_.length();
418 if (tmpLen > 1e-6f) desiredDir_ /= tmpLen;
419 } else if (navModeMask_ & FLOCKING) {
420 approachDir_ = Vec3f::zero();
421 // Orient in direction of group center.
422 Vec2f groupDir = navGroup_->groupCenter2D() - currentPos2D;
423 desiredDir_.x = groupDir.x;
424 desiredDir_.y = 0.0f;
425 desiredDir_.z = groupDir.y;
426 tmpLen = desiredDir_.length();
427 if (tmpLen > 1e-6f) desiredDir_ /= tmpLen;
428 } else {
429 approachDir_ = Vec3f::zero();
430 desiredDir_ = Vec3f::zero();
431 }
432 currentForce_ = Vec3f::zero();
433
434 // Compute avoidance vector
435 Vec3f avoidance = navCollision_ * collisionWeight_;
436 float avoidanceStrength = collisionStrength_;
437 if (!(navModeMask_ & APPROACHING)) {
438 // Reduce avoidance influence when not approaching.
439 avoidance *= 0.05f;
440 avoidanceStrength *= 0.05f;
441 }
442 updateNavFlocking();
443 avoidance += navFlocking_;
444 avoidanceStrength += flockingStrength_;
445
446 // Compute new velocity
447 updateNavVelocity(desiredSpeed, avoidance, avoidanceStrength);
448
449 // Advance the position
450 currentPos_ += currentVel_ * lastDT_;
451 float desiredHeight = getHeight(currentPos2D);
452#ifdef USE_HEIGHT_SMOOTHING
453 // Smooth height changes over time.
454 currentPos_.y = math::lerp(currentPos_.y, desiredHeight,
455 std::clamp(lastDT_ * heightSmoothFactor_, 0.0, 1.0));
456#else
457 currentPos_.y = desiredHeight;
458#endif
459
460 // Update orientation
461 updateNavOrientation(currentPos2D, desiredSpeed);
462}
