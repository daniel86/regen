72 BufferStorageMode getBufferStorageMode(
73 ClientAccessMode accessMode,
74 BufferMapMode mapMode,
75 BufferUpdateFlags /*updateHints*/) {
76 if (accessMode == BUFFER_GPU_ONLY) {
77 if (mapMode != BUFFER_MAP_DISABLED) {
78 REGEN_WARN("Buffer access mode is GPU_ONLY, but map mode is not DISABLED. "
79 "Using GPU_ONLY.");
80 }
81 return BUFFER_MODE_GPU_ONLY;
82 } else if (accessMode == BUFFER_CPU_READ) {
83 if (mapMode == BUFFER_MAP_DISABLED) {
84 REGEN_WARN("Buffer access mode is CPU_READ, but map mode is DISABLED. "
85 "Using CPU_R_MAP_TEMPORARY.");
86 return BUFFER_MODE_CPU_R_MAP_TEMPORARY;
87 } else if (mapMode == BUFFER_MAP_TEMPORARY) {
88 return BUFFER_MODE_CPU_R_MAP_TEMPORARY;
89 } else if (mapMode == BUFFER_MAP_PERSISTENT_COHERENT) {
90 return BUFFER_MODE_CPU_R_MAP_PERSISTENT_COHERENT;
91 } else if (mapMode == BUFFER_MAP_PERSISTENT_FLUSH) {
92 return BUFFER_MODE_CPU_R_MAP_PERSISTENT_FLUSH;
93 }
94 } else if (accessMode == BUFFER_CPU_WRITE) {
95 if (mapMode == BUFFER_MAP_DISABLED) {
96 return BUFFER_MODE_CPU_W_MAP_DISABLED;
97 } else if (mapMode == BUFFER_MAP_TEMPORARY) {
98 return BUFFER_MODE_CPU_W_MAP_TEMPORARY;
99 } else if (mapMode == BUFFER_MAP_PERSISTENT_COHERENT) {
100 return BUFFER_MODE_CPU_W_MAP_PERSISTENT_COHERENT;
101 } else if (mapMode == BUFFER_MAP_PERSISTENT_FLUSH) {
102 return BUFFER_MODE_CPU_W_MAP_PERSISTENT_FLUSH;
103 }
104 }
105 REGEN_WARN("Unknown buffer access mode or map mode. "
106 "Using BUFFER_MODE_CPU_W_MAP_DISABLED.");
107 return BUFFER_MODE_CPU_W_MAP_DISABLED;
108 }
