186void Mesh::addShaderInput(const std::string &name, const ref_ptr<ShaderInput> &in) {
187 if (!meshShader_.get()) return;
188
189 if (in->isStagedBuffer()) {
190 auto *bo = dynamic_cast<StagedBuffer*>(in.get());
191 if (!bo) {
192 REGEN_ERROR("Shader input '" << name << "' is not a StagedBuffer.");
193 return;
194 }
195 for (auto &blockUniform: bo->stagedInputs()) {
196 if (blockUniform.in_->numInstances() > 1) {
197 set_numInstances(blockUniform.in_->numInstances());
198 set_numVisibleInstances(blockUniform.in_->numInstances());
199 hasInstances_ = true;
200 }
201 }
202 }
203 if (in->numInstances() > 1) {
204 set_numInstances(in->numInstances());
205 set_numVisibleInstances(in->numInstances());
206 hasInstances_ = true;
207 }
208
209 if (in->isVertexAttribute()) {
210 int loc = meshShader_->attributeLocation(name);
211 if (loc == -1) {
212 // not used in shader
213 return;
214 }
215
216 auto needle = vaoLocations_.find(loc);
217 if (needle == vaoLocations_.end()) {
218 vaoAttributes_.emplace_back(in, loc);
219 vaoLocations_[loc] = static_cast<uint32_t>(vaoAttributes_.size() - 1);
220 } else {
221 const auto attIdx = needle->second;
222 vaoAttributes_[attIdx].input = in;
223 }
224 }

227void Mesh::createShader(const ref_ptr<StateNode> &parentNode) {
228 StateConfigurer shaderConfigurer;
229 if (material_.get()) {
230 // Add material first, this is needed to add textures before the parent node
231 // such that parent node textures are applied after material textures.
232 // This is done such that the material textures act as base textures.
233 // Note: we need to disjoin to avoid that material state overwrites again
234 // the parent node state.
235 shaderConfigurer.addState(material_.get());
236 disjoinStates(material_);
237 }
238 shaderConfigurer.addNode(parentNode.get());
239 shaderConfigurer.addState(sharedState_.get());
240 shaderConfigurer.addState(this);
241 if (meshLODs_.size()>1) {
242 shaderConfigurer.define("HAS_LOD", "TRUE");
243 }
244 if (!indirectDrawBuffer_.get()) {
245 // If we do not have an indirect draw buffer,
246 // then multi-layered rendering must be done with GS.
247 shaderConfigurer.define("USE_GS_LAYERED_RENDERING", "TRUE");
248 }
249 if (cullShape_.get()) {
250 shaderConfigurer.addState(cullShape_.get());
251 }
252 createShader(parentNode, shaderConfigurer.cfg());
253 if (material_.get()) {
254 // re-join material state
255 joinStates(material_);
256 }
257}

259void Mesh::createShader(const ref_ptr<StateNode> &parentNode, StateConfig &shaderConfig) {
260 auto shaderState = ref_ptr<ShaderState>::alloc();
261 for (auto &joinedState : *joined().get()) {
262 // disjoin any ShaderState that might be joined
263 auto *shaderStateJoined = dynamic_cast<ShaderState *>(joinedState.get());
264 if (shaderStateJoined != nullptr) {
265 disjoinStates(joinedState);
266 break;
267 }
268 }
269 joinStates(shaderState);
270
271 if(shaderKey_.empty()) {
272 shaderKey_ = "regen.objects.mesh";
273 }
274 bool hasValidShader = false;
275 if(shaderStageKeys_.empty()) {
276 hasValidShader = shaderState->createShader(shaderConfig, shaderKey_);
277 }
278 else {
279 std::vector<std::string> stageKeys(glenum::glslStageCount());
280 for (int i = 0; i < glenum::glslStageCount(); ++i) {
281 stageKeys[i] = shaderKey_;
282 }
283 for (int i = 0; i < glenum::glslStageCount(); ++i) {
284 auto stage = glenum::glslStages()[i];
285 auto it = shaderStageKeys_.find(stage);
286 if (it != shaderStageKeys_.end()) {
287 stageKeys[i] = it->second;
288 }
289 }
290 hasValidShader = shaderState->createShader(shaderConfig, stageKeys);
291 }
292
293 if (!hasValidShader) {
294 // Hide the mesh if no valid shader could be created.
295 set_isHidden(true);
296 } else {
297 updateVAO(shaderConfig, shaderState->shader());
298 }
299
300 // create shader of lod meshes
301 bool hasImpostor = false;
302 for (auto &lod : meshLODs_) {
303 if (!hasImpostor && lod.impostorMesh.get()) {
304 lod.impostorMesh->setIndirectDrawBuffer(
305 indirectDrawBuffer_, baseDrawIdx_, numDrawLayers_);
306 lod.impostorMesh->numDrawLODs_ = 1;
307 lod.impostorMesh->createShader(parentNode);
308 hasImpostor = true;
309 }
310 }

313void Mesh::updateVAO(const StateConfig &cfg, const ref_ptr<Shader> &meshShader) {
314 // remember the shader
315 meshShader_ = meshShader;
316 hasInstances_ = cfg.numInstances_ > 1;
317 if (cfg.numInstances_ > 1) {
318 set_numInstances(cfg.numInstances_);
319 set_numVisibleInstances(cfg.numInstances_);
320 }
321
322 // reset attribute list
323 vaoAttributes_.clear();
324 vaoLocations_.clear();
325 // and load from Config
326 for (const auto & input : cfg.inputs_) { addShaderInput(input.name_, input.in_); }
327 // Get input from mesh and joined states (might be handled by StateConfig allready)
328 ShaderInputList localInputs;
329 collectShaderInput(localInputs);
330 for (auto & localInput : localInputs) {
331 addShaderInput(localInput.name_, localInput.in_);
332 }
333 // Add Textures
334 for (const auto & texture : cfg.textures_) {
335 addShaderInput(texture.first, texture.second.first);
336 }
337
338 // In most cases the VBO is already allocated in updateVertexData(),
339 // but in case some attributes were not used in the shader,
340 // we need to ensure the VBO is allocated here.
341 bool allAttributesAllocated = true;
342 ref_ptr<BufferReference> vertexBufferRef;
343 for (const auto &vaoAttribute : vaoAttributes_) {
344 auto &attributeRef = vaoAttribute.input->mainBufferRef();
345 if (!attributeRef) {
346 allAttributesAllocated = false;
347 break;
348 } else {
349 if (!vertexBufferRef) {
350 vertexBufferRef = attributeRef;
351 } else if (vertexBufferRef->bufferID() != attributeRef->bufferID()) {
352 // different buffer IDs, cannot handle this case here
353 allAttributesAllocated = false;
354 break;
355 }
356 }
357 }
358 if (!allAttributesAllocated) {
359 REGEN_WARN("Not all mesh attributes were allocated in VBO, reallocating VBO.");
360 std::vector<ref_ptr<ShaderInput>> attributes(vaoAttributes_.size());
361 for (size_t i = 0; i < vaoAttributes_.size(); ++i) {
362 attributes[i] = vaoAttributes_[i].input;
363 }
364 vertexBuffer_->alloc(attributes);
365 }
366
367 updateVAO();
368 updateDrawFunction();
369}

396void Mesh::updateVAO_() {
397 if (vaoAttributes_.empty()) return;
398 updateVAO(vaoAttributes_.front().input->mainBufferName());
399
400 // group together LODs that can be drawn with multi draw calls,
401 // i.e. those that do not have impostor meshes.
402 indirectDrawGroups_.clear();
403
404 if (meshLODs_.empty()) {
405 ensureLOD();
406 } else {
407#ifndef REGEN_MESH_DISABLE_MULTI_DRAW
408 uint32_t drawGroupIdx = 0;
409 bool isLastImpostor = false;
410#endif
411
412 if (numDrawLODs_ > meshLODs_.size() && meshLODs_.size() == 2) {
413 // duplicate the first LOD once and insert it as a new second LOD
414 meshLODs_.insert(meshLODs_.begin() + 1, meshLODs_.front());
415 }
416
417 for (auto &lodData : meshLODs_) {
418 if (lodData.d->numVertices == lastNumVertices_ && !lodData.impostorMesh.get()) {
419 // update the full LOD data if it is not an impostor mesh
420 lodData.d->numVertices = numVertices();
421 lodData.d->vertexOffset = vertexOffset();
422 lodData.d->numIndices = numIndices();
423 lodData.d->indexOffset = indexOffset();
424 }
425#ifndef REGEN_MESH_DISABLE_MULTI_DRAW
426 if (indirectDrawGroups_.size() <= drawGroupIdx) {
427 indirectDrawGroups_.emplace_back(0);
428 }
429 if (lodData.impostorMesh.get()) {
430 if (isLastImpostor) {
431 // NOTE assume we can join two impostor LODs into one draw group.
432 if(indirectDrawGroups_.back()==0) {
433 indirectDrawGroups_[drawGroupIdx-1] += 1;
434 } else {
435 indirectDrawGroups_.back() += 1;
436 }
437 } else if(indirectDrawGroups_.back()==0) {
438 indirectDrawGroups_.back() = 1;
439 drawGroupIdx += 1;
440 } else {
441 indirectDrawGroups_.emplace_back(1);
442 drawGroupIdx += 2;
443 }
444 isLastImpostor = true;
445 } else {
446 indirectDrawGroups_[drawGroupIdx] += 1;
447 isLastImpostor = false;
448 }
449#endif
450 }
451 if(!indirectDrawGroups_.empty() && indirectDrawGroups_.back()==0) {
452 indirectDrawGroups_.pop_back();
453 }
454 if (numDrawLODs_ < meshLODs_.size()) {
455 numDrawLODs_ = static_cast<uint32_t>(meshLODs_.size());
456 } else if (numDrawLODs_ > meshLODs_.size()) {
457 const uint32_t numAdditionalLODs = numDrawLODs_ - static_cast<uint32_t>(meshLODs_.size());
458 const auto lastLOD = meshLODs_.back();
459#ifndef REGEN_MESH_DISABLE_MULTI_DRAW
460 if(indirectDrawGroups_.empty()) {
461 indirectDrawGroups_.emplace_back(0);
462 }
463 // enlarge last draw group. This is fine as additional LODs are copies of the last LOD.
464 indirectDrawGroups_.back() += numAdditionalLODs;
465#endif
466 for (uint32_t i = 0; i < numAdditionalLODs; ++i) {
467 meshLODs_.emplace_back(
468 lastLOD.d->numVertices,
469 lastLOD.d->vertexOffset,
470 lastLOD.d->numIndices,
471 lastLOD.d->indexOffset);
472 if (lastLOD.impostorMesh.get()) {
473 meshLODs_.back().impostorMesh = ref_ptr<Mesh>::alloc(lastLOD.impostorMesh);
474 }
475 }
476 }
477 }
478#ifdef REGEN_MESH_DEBUG_DRAW_GROUPS
479 std::stringstream drawGroupsStr;
480 for (size_t i = 0; i < indirectDrawGroups_.size(); ++i) {
481 if (i > 0) drawGroupsStr << ", ";
482 drawGroupsStr << indirectDrawGroups_[i];
483 }
484 REGEN_INFO("Mesh has " << meshLODs_.size() << " LODs, grouped into "
485 << indirectDrawGroups_.size() << " draw groups: [" << drawGroupsStr.str() << "]");
486#endif
487#ifndef REGEN_MESH_DISABLE_MULTI_DRAW
488 if (indirectDrawGroups_.size() == meshLODs_.size()) {
489 // seems nothing was joined...
490 indirectDrawGroups_.clear();
491 }
492#endif
493 lastNumVertices_ = numVertices();
494 numLODs_ = static_cast<uint32_t>(meshLODs_.size());
495
496 // initialize num visible instances and offsets for all LODs
497 resetVisibility();
498}

528uint32_t Mesh::getLODLevel(float distanceSquared, const Vec4i &lodShift) const {
529 int32_t lod = (distanceSquared >= v_lodThresholds_.x)
530 + (distanceSquared >= v_lodThresholds_.y)
531 + (distanceSquared >= v_lodThresholds_.z);
532 return std::min(
533 static_cast<uint32_t>(std::max(lod+lodShift[lod], 0)),
534 numLODs() - 1);
535}

684void Mesh::createIndirectDrawBuffer(uint32_t numDrawLayers) {
685 ensureLOD();
686 numLODs_ = meshLODs_.size();
687 std::vector<DrawCommand> drawData(numLODs_ * numDrawLayers);
688
689 // Create the indirect draw data for this part and the first layer.
690 // DrawID order: LOD0_layer0, LOD0_layer1, LOD0_layer2, ...
691 // LOD1_layer0, LOD1_layer1, LOD1_layer2, ...
692 for (uint32_t lodIdx = 0; lodIdx < numLODs_; ++lodIdx) {
693 const uint32_t lodStartIdx = lodIdx * numDrawLayers;
694 DrawCommand &drawParams = drawData[lodStartIdx];
695 auto &lodData = meshLODs_[lodIdx];
696
697 Mesh *m = lodData.impostorMesh.get() ? lodData.impostorMesh.get() : this;
698 if (m->indices().get()) {
699 drawParams.mode = 1u; // 1=elements, 2=arrays
700 drawParams.setCount(lodData.d->numIndices);
701 drawParams.setFirstElement(lodData.d->indexOffset / m->indices()->dataTypeBytes());
702 drawParams.data[3] = 0; // base vertex
703 } else {
704 drawParams.mode = 2u; // 1=elements, 2=arrays
705 drawParams.setCount(lodData.d->numVertices);
706 drawParams.setFirstElement(lodData.d->vertexOffset);
707 }
708 drawParams.setInstanceCount(lodIdx==0 ? numInstances() : 0);
709 drawParams.setBaseInstance(0);
710
711 // Copy over the data for the remaining layers.
712 for (uint32_t layerIdx = 1; layerIdx < numDrawLayers; ++layerIdx) {
713 const uint32_t lodLayerIdx = lodStartIdx + layerIdx;
714 std::memcpy(
715 &drawData[lodLayerIdx],
716 &drawData[lodStartIdx],
717 sizeof(DrawCommand));
718 }
719 }
720
721 // finally create the indirect draw buffer for this part
722 indirectDrawBuffer_ = ref_ptr<DrawIndirectBuffer>::alloc(
723 "IndirectDrawBuffer", BufferUpdateFlags::FULL_RARELY);
724 auto input = ref_ptr<ShaderInputStruct<DrawCommand>>::alloc(
725 "DrawCommand", "drawParams", numLODs_ * numDrawLayers);
726 input->setInstanceData(1, 1, (byte*)drawData.data());
727 indirectDrawBuffer_->addStagedInput(input);
728 indirectDrawBuffer_->update();
729 numDrawLayers_ = numDrawLayers;
730 updateDrawFunction();
731}

816void Mesh::drawMeshLOD(RenderState *rs, uint32_t lodLevel, uint32_t drawIdx, int32_t multiDrawCount) {
817 auto &lod = meshLODs_[lodLevel];
818 if (!hasIndirectDrawBuffer() && lod.d->numVisibleInstances == 0) {
819 // no instances to draw, skip
820 // Note: In case of indirect draw buffers, these could be updated GPU-side
821 // then we wouldn't know the instance count here.
822 // Also for the other case we currently skip setting the instance count
823 // on the LOD object, and only write to the buffer -- but we could skip
824 // the raw for this case too.
825 return;
826 }
827 // set the LOD level vertex meta data
828 activateLOD_(lodLevel);
829
830 if (lod.impostorMesh.get()) {
831 // let the LOD mesh do the draw call.
832 if (hasIndirectDrawBuffer()) {
833 lod.impostorMesh->setIndirectDrawBuffer(indirectDrawBuffer_, drawIdx, multiDrawCount);
834 lod.impostorMesh->numDrawLODs_ = 1;
835 lod.impostorMesh->draw(rs);
836 } else {
837 lod.impostorMesh->resetVisibility(true);
838 lod.impostorMesh->updateVisibility(0,
839 lod.d->numVisibleInstances, lod.d->instanceOffset);
840 lod.impostorMesh->draw(rs);
841 }
842 }
843 else if (hasIndirectDrawBuffer()) {
844 set_indirectOffset(
845 indirectDrawBuffer_->drawBufferRef()->address() +
846 drawIdx * sizeof(DrawCommand));
847 set_multiDrawCount(multiDrawCount);
848 drawMesh(rs);
849 set_indirectOffset(0);
850 set_multiDrawCount(1);
851 } else {
852 set_numVisibleInstances(lod.d->numVisibleInstances);
853 set_baseInstance(lod.d->instanceOffset);
854 drawMesh(rs);
855 set_numVisibleInstances(numInstances());
856 set_baseInstance(0);
857 }
858}

860void Mesh::drawMesh(RenderState *rs) {
861 const bool useTransformFeedback = (feedbackRange_ != nullptr);
862 if (useTransformFeedback) {
863 rs->feedbackBufferRange().push(0, *feedbackRange_);
864 rs->beginTransformFeedback(GL_POINTS);
865 }
866
867 rs->vao().apply(vao_->id());
868 if (hasIndirectDrawBuffer()) {
869 rs->drawIndirectBuffer().apply(indirectDrawBuffer()->drawBufferRef()->bufferID());
870 }
871 (this->*draw_)(primitive_);
872
873 if (useTransformFeedback) {
874 rs->endTransformFeedback();
875 rs->feedbackBufferRange().pop(0);
876 }

879void Mesh::enable(RenderState *rs) {
880 State::enable(rs);
881
882 if (!indirectDrawBuffer_.get() || indirectDrawGroups_.empty()) {
883 uint32_t drawIdx = baseDrawIdx_;
884 if (lodSortMode_ == SortMode::BACK_TO_FRONT) {
885 for (uint32_t lodLevel = meshLODs_.size(); lodLevel > 0; --lodLevel) {
886 drawMeshLOD(rs, lodLevel - 1, drawIdx, numDrawLayers_);
887 drawIdx += numDrawLayers_;
888 }
889 }
890 else {
891 for (uint32_t lodLevel = 0; lodLevel < meshLODs_.size(); ++lodLevel) {
892 drawMeshLOD(rs, lodLevel, drawIdx, numDrawLayers_);
893 drawIdx += numDrawLayers_;
894 }
895 }
896 activateLOD_(0);
897 } else {
898 uint32_t lodLevel = 0u;
899 uint32_t drawIdx = baseDrawIdx_;
900 for (int32_t groupSize : indirectDrawGroups_) {
901 const uint32_t numDraws = groupSize * numDrawLayers_;
902 drawMeshLOD(rs, lodLevel, drawIdx, numDraws);
903 lodLevel += groupSize;
904 drawIdx += numDraws;
905 }
906 }

1017void Mesh::loadShaderConfig(LoadingContext &ctx, scene::SceneInputNode &input) {
1018 if (input.hasAttribute("shader")) {
1019 setShaderKey(input.getValue<std::string>("shader", shaderKey_));
1020 } else if (input.hasAttribute("shader-key")) {
1021 setShaderKey(input.getValue<std::string>("shader-key", shaderKey_));
1022 } else if (input.hasAttribute("key")) {
1023 setShaderKey(input.getValue<std::string>("key", shaderKey_));
1024 }
1025 if (input.hasAttribute("num-instances")) {
1026 set_numInstances(input.getValue<int32_t>("num-instances", numInstances()));
1027 set_numVisibleInstances(numInstances());
1028 }
1029 for (int i = 0; i < glenum::glslStageCount(); ++i) {
1030 auto stage = glenum::glslStages()[i];
1031 auto keyName = glenum::glslStagePrefix(stage);
1032 if (input.hasAttribute(keyName)) {
1033 setShaderKey(input.getValue(keyName), stage);
1034 continue;
1035 }
1036 keyName = REGEN_STRING("shader-" << keyName);
1037 if (input.hasAttribute(keyName)) {
1038 setShaderKey(input.getValue(keyName), stage);
1039 continue;
1040 }
1041 }
1042 // read child nodes into shared state
1043 for (auto &child : input.getChildren()) {
1044 auto processor = ctx.scene()->getStateProcessor(child->getCategory());
1045 if (processor.get() == nullptr) {
1046 REGEN_WARN("No processor registered for '" << child->getDescription() << "'.");
1047 } else {
1048 processor->processInput(ctx.scene(), *child.get(), ctx.parent(), sharedState_);
1049 }
1050 }
