32 // map client data for writing
33 auto v_pos = (Vec3f*) pos_->clientBuffer()->clientData(0);
34 auto v_nor = (cfg.isNormalRequired ?
35 (Vec3f*) nor_->clientBuffer()->clientData(0) : nullptr);
36
37 float phi = acos(cfg.cosAngle);
38 float radius = tan(phi) * cfg.height;
39 float angle = 0.0f;
40 float angleStep = 2.0f * M_PI / (float) lodLevel;
41 uint32_t i = vertexOffset;
42
43 v_pos[i] = Vec3f::zero();
44 if (cfg.isNormalRequired) {
45 v_nor[i] = Vec3f(0.0f, -1.0f, 0.0f);
46 }
47
48 for (; i < lodLevel + 1; ++i) {
49 angle += angleStep;
50 float s = sin(angle) * radius;
51 float c = cos(angle) * radius;
52 Vec3f pos(c, s, cfg.height);
53 v_pos[i + 1] = pos;
54 minPosition_.setMin(pos);
55 maxPosition_.setMax(pos);
56 if (cfg.isNormalRequired) {
57 Vec3f n(c, 0.0, s);
58 n.normalize();
59 v_nor[i + 1] = n;
60 }

172 const Config &cfg,

174 uint32_t vertexOffset,
175 uint32_t indexOffset) {
176 // map client data for writing
177 auto v_pos = (Vec3f*) pos_->clientBuffer()->clientData(0);
178 auto v_nor = (cfg.isNormalRequired ?
179 (Vec3f*) nor_->clientBuffer()->clientData(0) : nullptr);
180 auto indices = (byte*)indices_->clientBuffer()->clientData(0);
181 auto indexType = indices_->baseType();
182
183 // create cone vertex data
184 loadConeData(
185 v_pos+vertexOffset,
186 (v_nor ? v_nor+vertexOffset : v_nor),
187 minPosition_, maxPosition_,
188 cfg.isBaseRequired, lodLevel,
189 cfg.radius, cfg.height);
190
191 // create cone index data
192 const uint32_t apexIndex = vertexOffset;
193 const uint32_t baseCenterIndex = vertexOffset + 1;
194 uint32_t faceIndex = indexOffset;
195 int vIndex = vertexOffset + cfg.isBaseRequired ? 2 : 1;
196 // cone
197 for (uint32_t i = 0; i < lodLevel; ++i) {
198 setIndexValue(indices, indexType, faceIndex++, apexIndex);
199 setIndexValue(indices, indexType, faceIndex++, (i + 1 == lodLevel ? vIndex : vIndex + i + 1));
200 setIndexValue(indices, indexType, faceIndex++, vIndex + i);
201 }
202 // base
203 if (cfg.isBaseRequired) {
204 for (uint32_t i = 0; i < lodLevel; ++i) {
205 setIndexValue(indices, indexType, faceIndex++, baseCenterIndex);
206 setIndexValue(indices, indexType, faceIndex++, vIndex + i);
207 setIndexValue(indices, indexType, faceIndex++, (i + 1 == lodLevel ? vIndex : vIndex + i + 1));
208 }
209 }
