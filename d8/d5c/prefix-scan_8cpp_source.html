21PrefixScan::Mode PrefixScan::getScanMode(uint32_t numDesiredInvocations) {
22 // we need to check if the currently selected scan mode hits some limits,
23 // and if so, we need to switch to a different mode.
24 if (scanMode_ == Mode::SERIAL) return scanMode_;
25 uint32_t maxSharedMem = glParam<int>(GL_MAX_COMPUTE_SHARED_MEMORY_SIZE);
26 uint32_t maxWorkGroupInvocations = glParam<int>(GL_MAX_COMPUTE_WORK_GROUP_INVOCATIONS);
27 uint32_t parallelScanMemory = numDesiredInvocations * 4u; // 4 bytes per uint32
28 bool isParallelScanPossible = (
29 parallelScanMemory <= maxSharedMem &&
30 numDesiredInvocations <= maxWorkGroupInvocations);
31 if (isParallelScanPossible) {
32 return Mode::PARALLEL;
33 } else {
34 return Mode::HIERARCHICAL;
35 }
36}

103void PrefixScan::createHierarchicalPass() {
104 // divide the histogram into "blocks" of scanGroupSize_
105 int32_t numBlocks = ceil(static_cast<float>(currentHistogramSize_) / static_cast<float>(scanGroupSize_));
106 // need to enforce power of two below
107 auto numBlocks2 = static_cast<int32_t>(math::nextPow2(numBlocks));
108
109 { // global memory for the offsets
110 blockOffsetsBuffer_ = ref_ptr<SSBO>::alloc(
111 "BlockOffsetsBuffer",
112 BufferUpdateFlags::FULL_PER_FRAME,
113 SSBO::RESTRICT);
114 blockOffsetsBuffer_->addStagedInput(ref_ptr<ShaderInput1ui>::alloc(
115 "blockOffsets", numBlocks * numLayers_));
116 blockOffsetsBuffer_->stagedInputs()[0].in_->set_forceArray(true);
117 blockOffsetsBuffer_->update();
118 }
119 { // pass 1: local offsets
120 computeLocalOffsets_ = ref_ptr<ComputePass>::alloc("regen.compute.prefix-scan.local");
121 computeLocalOffsets_->computeState()->setGroupSize(scanGroupSize_, 1, 1);
122 computeLocalOffsets_->computeState()->setNumWorkUnits(currentHistogramSize_, numLayers_, 1);
123 computeLocalOffsets_->setInput(globalHistogramBuffer_);
124 computeLocalOffsets_->setInput(blockOffsetsBuffer_);
125
126 StateConfigurer shaderCfg;
127 if (hasHistogramConstantSize_) {
128 shaderCfg.define("SCAN_HISTOGRAM_SIZE", REGEN_STRING(currentHistogramSize_));
129 shaderCfg.define("SCAN_NUM_BLOCKS", REGEN_STRING(numBlocks));
130 } else {
131 shaderCfg.define("SCAN_DYNAMIC_HISTOGRAM_SIZE", "TRUE");
132 }
133 shaderCfg.addState(computeLocalOffsets_.get());
134 computeLocalOffsets_->createShader(shaderCfg.cfg());
135 if (!hasHistogramConstantSize_) {
136 histogramSizeLoc_h_l_ = computeLocalOffsets_->shaderState()->shader()->uniformLocation("histogramSize");
137 numBlocksLoc_h_l_ = computeLocalOffsets_->shaderState()->shader()->uniformLocation("numBlocks");
138 }
139 }
140 { // pass 2: global offsets
141 computeGlobalOffsets_h_ = ref_ptr<ComputePass>::alloc("regen.compute.prefix-scan.global");
142 computeGlobalOffsets_h_->computeState()->setGroupSize(numBlocks2, 1, 1);
143 computeGlobalOffsets_h_->computeState()->setNumWorkUnits(numBlocks2, numLayers_, 1);
144 computeGlobalOffsets_h_->setInput(blockOffsetsBuffer_);
145 if (hasHistogramConstantSize_) {
146 computeGlobalOffsets_h_cfg_.define("SCAN_NUM_BLOCKS", REGEN_STRING(numBlocks));
147 } else {
148 computeGlobalOffsets_h_cfg_.define("SCAN_DYNAMIC_HISTOGRAM_SIZE", "TRUE");
149 }
150 computeGlobalOffsets_h_cfg_.addState(computeGlobalOffsets_h_.get());
151 computeGlobalOffsets_h_->createShader(computeGlobalOffsets_h_cfg_.cfg());
152 if (!hasHistogramConstantSize_) {
153 numBlocksLoc_h_g_ = distributeOffsets_->shaderState()->shader()->uniformLocation("numBlocks");
154 }
155 }
156 { // pass 3: distribute offsets
157 distributeOffsets_ = ref_ptr<ComputePass>::alloc("regen.compute.prefix-scan.distribute");
158 distributeOffsets_->computeState()->setGroupSize(scanGroupSize_, 1, 1);
159 distributeOffsets_->computeState()->setNumWorkUnits(currentHistogramSize_, numLayers_, 1);
160 distributeOffsets_->setInput(globalHistogramBuffer_);
161 distributeOffsets_->setInput(blockOffsetsBuffer_);
162
163 StateConfigurer shaderCfg;
164 if (hasHistogramConstantSize_) {
165 shaderCfg.define("SCAN_HISTOGRAM_SIZE", REGEN_STRING(currentHistogramSize_));
166 shaderCfg.define("SCAN_NUM_BLOCKS", REGEN_STRING(numBlocks));
167 } else {
168 shaderCfg.define("SCAN_DYNAMIC_HISTOGRAM_SIZE", "TRUE");
169 }
170 shaderCfg.addState(distributeOffsets_.get());
171 distributeOffsets_->createShader(shaderCfg.cfg());
172 if (!hasHistogramConstantSize_) {
173 histogramSizeLoc_h_d_ = distributeOffsets_->shaderState()->shader()->uniformLocation("histogramSize");
174 numBlocksLoc_h_d_ = distributeOffsets_->shaderState()->shader()->uniformLocation("numBlocks");
175 }
176 }
177 currentNumBlocks_ = numBlocks;
178 REGEN_DEBUG("Using hierarchical scan for histogram with " << currentHistogramSize_
179 << " elements, " << numBlocks << " blocks, wg size "
180 << scanGroupSize_
181 << ", " << numBlocks2 << " work groups.");

184void PrefixScan::updateHierarchicalPass() {
185 if (!computeGlobalOffsets_h_.get()) {
186 createHierarchicalPass();
187 return;
188 }
189 // divide the histogram into "blocks" of scanGroupSize_
190 uint32_t numBlocks = ceil(static_cast<float>(currentHistogramSize_) / static_cast<float>(scanGroupSize_));
191 // need to enforce power of two below
192 auto numBlocks2 = math::nextPow2(numBlocks);
193 { // global memory for the offsets -> resize to numBlocks * numLayers_
194 auto &offsets = blockOffsetsBuffer_->stagedInputs()[0].in_;
195 auto oldNumBlocks = offsets->numArrayElements();
196 if (oldNumBlocks != numBlocks * numLayers_) {
197 offsets->set_numArrayElements(numBlocks * numLayers_);
198 blockOffsetsBuffer_->update();
199 }
200 }
201 { // pass 1: local offsets -> change numWorkUnits
202 computeLocalOffsets_->computeState()->setNumWorkUnits(currentHistogramSize_, numLayers_, 1);
203 }
204 { // pass 3: distribute offsets
205 distributeOffsets_->computeState()->setNumWorkUnits(currentHistogramSize_, numLayers_, 1);
206 }
207 { // pass 2: global offsets
208 auto oldNumBlocks2 = computeGlobalOffsets_h_->computeState()->workGroupSize().x;
209 if (oldNumBlocks2 != numBlocks2) {
210 computeGlobalOffsets_h_->computeState()->setGroupSize(numBlocks2, 1, 1);
211 computeGlobalOffsets_h_->computeState()->setNumWorkUnits(numBlocks2, numLayers_, 1);
212 computeGlobalOffsets_h_cfg_.define("CS_LOCAL_SIZE_X", REGEN_STRING(numBlocks2));
213 computeGlobalOffsets_h_cfg_.define("SCAN_NUM_BLOCKS", REGEN_STRING(numBlocks));
214 computeGlobalOffsets_h_->createShader(computeGlobalOffsets_h_cfg_.cfg());
215 REGEN_DEBUG("PrefixScan: Number of global scan invocations changed from "
216 << oldNumBlocks2 << " to " << numBlocks2);
217 }
218 }
219 currentNumBlocks_ = numBlocks;
220}

222void PrefixScan::createResources() {
223 currentHistogramSize_ = globalHistogramBuffer_->allocatedSize() / (sizeof(uint32_t) * numLayers_);
224 // compute offsets by scanning. We prefer here to do a single-pass parallel scan, if possible.
225 // But we need to check if the histogram fits into shared memory and if the number of
226 // work group invocations is not too high. Else we need to do a hierarchical scan.
227 auto numDesiredInvocations = getNumDesiredInvocations();
228 scanMode_ = getScanMode(numDesiredInvocations);
229 switch (scanMode_) {
230 case Mode::HIERARCHICAL:
231 createHierarchicalPass();
232 break;
233 case Mode::PARALLEL:
234 updateParallelPass(numDesiredInvocations);
235 break;
236 case Mode::SERIAL:
237 createSerialPass();
238 break;
239 }

264void PrefixScan::scan1(RenderState *rs) {
265 switch (scanMode_) {
266 case Mode::HIERARCHICAL:
267 // Each workgroup scans WG_SIZE elements of in_globalHistogram and:
268 // - Writes scanned local offset to in_globalHistogram.
269 // - Writes total size of its block to in_blockOffsets[] array.
270 computeLocalOffsets_->enable(rs);
271 computeLocalOffsets_->disable(rs);
272 // Perform parallel scan on the block offsets in order to
273 // compute the global offsets for each work group.
274 computeGlobalOffsets_h_->enable(rs);
275 computeGlobalOffsets_h_->disable(rs);
276 // Each workgroup distributes WG_SIZE elements from the
277 // in_globalHistogram to the in_blockOffsets[] array.
278 distributeOffsets_->enable(rs);
279 distributeOffsets_->disable(rs);
280 break;
281 case Mode::PARALLEL:
282 computeGlobalOffsets_p_->enable(rs);
283 computeGlobalOffsets_p_->disable(rs);
284 break;
285 case Mode::SERIAL:
286 computeGlobalOffsets_s_->enable(rs);
287 computeGlobalOffsets_s_->disable(rs);
288 break;
289 }
290}

292void PrefixScan::scan2(RenderState *rs) {
293 switch (scanMode_) {

302 computeGlobalOffsets_h_->disable(rs);

306 glUniform1ui(histogramSizeLoc_h_d_, currentHistogramSize_);
307 distributeOffsets_->disable(rs);

322void PrefixScan::scan(RenderState *rs) {
323 // Run offsets pass. As a result we will have the offsets for each work group
324 // in the globalHistogramBuffer_.
325 if (hasHistogramConstantSize_) {
326 scan1(rs);
327 } else {
328 auto newHistogramSize = globalHistogramBuffer_->allocatedSize() / (sizeof(uint32_t) * numLayers_);
329 auto hasHistogramSizeChanged = (newHistogramSize != currentHistogramSize_);
330 if (hasHistogramSizeChanged) {
331 updateHistogramSize(newHistogramSize);
332 scan2(rs);
333 }
334 else if (hasInitializedUniforms_) {
335 scan1(rs);
336 }
337 else {
338 // make sure to set uniforms in first frame
339 scan2(rs);
340 hasInitializedUniforms_ = true;
341 }
342 }
343#ifdef RADIX_DEBUG_HISTOGRAM
344 printHistogram(rs);
345#endif
346}
