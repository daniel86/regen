123const float Noise::noise2(
124 float s
125, float t
126, const unsigned int r) const
127{
128 s += m_xoff;
129 t += m_yoff;
130
131 s *= 1 << r;
132 t *= 1 << r;
133
134 const float is = floor(s);
135 const float it = floor(t);
136
137 const regen::Vec2f f(_frac(s), _frac(t));
138
139 // range [-1;+1]
140
141 const float aa = dot(grad2(is + 0, it + 0, r), (f ));
142 const float ba = dot(grad2(is + 1, it + 0, r), (f - regen::Vec2f(1.f, 0.f)));
143 const float ab = dot(grad2(is + 0, it + 1, r), (f - regen::Vec2f(0.f, 1.f)));
144 const float bb = dot(grad2(is + 1, it + 1, r), (f - regen::Vec2f(1.f, 1.f)));
145
146 const regen::Vec2f i = mix(regen::Vec2f(aa, ab), regen::Vec2f(ba, bb), fade(f.x));
147
148 return mix(i.x, i.y, fade(f.y));
149}
