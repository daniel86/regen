35int32_t BindingManager::request_(
36 BlockType blockType,
37 std::uintptr_t bufferID,
38 std::string_view bufferName,
39 const std::set<int32_t> &avoidBindingPoints) {
40 auto &maxBindingPoints = maxBindings_[blockType];
41 auto &nextBindingPoint = nextBindingPoint_[blockType];
42
43 // check if the same buffer range was used before, and if so try to reuse the binding point.
44 if (bufferID != 0) {
45 auto it1 = bufferBindings_.find(bufferID);
46 if (it1 != bufferBindings_.end()) {
47 auto &bindingPoint = it1->second;
48 if (avoidBindingPoints.find(bindingPoint) == avoidBindingPoints.end()) {
49 // binding point is not in the list of avoided binding points
50 bindingPointCounter_[blockType][bindingPoint]++;
51 return bindingPoint;
52 }
53 }
54 }
55
56 int32_t bindingPoint;
57 // no binding point was found, create a new one if possible
58 if (nextBindingPoint < maxBindingPoints &&
59 avoidBindingPoints.find(nextBindingPoint) == avoidBindingPoints.end()) {
60 // raise the global binding point counter
61 bindingPoint = nextBindingPoint++;
62 } else {
63 // all global binding points are used, try to reuse one
64 auto minIt = std::min_element(
65 bindingPointCounter_[blockType].begin(),
66 bindingPointCounter_[blockType].end());
67 bindingPoint = static_cast<int32_t>(std::distance(
68 bindingPointCounter_[blockType].begin(), minIt));
69 while (avoidBindingPoints.find(bindingPoint) != avoidBindingPoints.end()) {
70 bindingPoint++;
71 if (bindingPoint >= maxBindingPoints) {
72 bindingPoint = 0;
73 }
74 }
75 }
76
77 bindingPointCounter_[blockType][bindingPoint]++;
78 if (bufferID == 0) {
79 // the block input was not added to the scene when the shader was created,
80 // so we cannot ensure that once resolve the BO can be used across different shader programs
81 // with same binding point.
82 // This is usually only the case if there is some ping-ponging of the buffer objects involved,
83 // i.e. if BO used by the program changes e.g. per frame.
84 REGEN_DEBUG("Allocating unresolved "
85 << (blockType == UBO ? "UBO" : "SSBO") << " \""
86 << bufferName << "\" to binding point "
87 << bindingPoint << " (next=" << nextBindingPoint << ").");
88 } else {
89 bufferBindings_[bufferID] = bindingPoint;
90 REGEN_DEBUG("Allocating "
91 << (blockType == UBO ? "UBO" : "SSBO") << " \""
92 << bufferName << "\" (" << bufferID << ") to binding point "
93 << bindingPoint << " (next=" << nextBindingPoint << ").");
94 }
95 return bindingPoint;
96}
