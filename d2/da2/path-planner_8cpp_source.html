40std::vector<ref_ptr<WayPoint>> PathPlanner::findPath(const Vec2f &from, const Vec2f &to) const {
41 // Find a waypoint near the target first.
42 // But only consider waypoints that are closer to the source than the target.
43 float maxDistanceSq = (from - to).lengthSquared();
44 auto closest = getClosest(to, from, maxDistanceSq);
45 if (!closest) {
46 // Directly go to target if no waypoint could be found
47 return {};
48 } else {
49 return findPath(from, closest);
50 }
51}

86 const std::unordered_map<WayPoint *, NodeRecord> &records,
87 const NodeRecord &goalRec) {
88 std::vector<ref_ptr<WayPoint> > path;

99std::vector<ref_ptr<WayPoint> > PathPlanner::findPath(
100 const ref_ptr<WayPoint> &start,
101 const ref_ptr<WayPoint> &goal) const {
102 if (!start || !goal) return {};
103 if (start == goal) return {start};
104
105 std::priority_queue<NodeRecord, std::vector<NodeRecord>, NodeRecordCompare> open;
106 std::unordered_map<WayPoint *, NodeRecord> allRecords;
107 std::unordered_map<WayPoint *, bool> closed;
108
109 // Init start
110 NodeRecord startRec{
111 start, {}, 0.0f,
112 heuristic(start, goal)
113 };
114 open.push(startRec);
115 allRecords[start.get()] = startRec;
116
117 while (!open.empty()) {
118 NodeRecord current = open.top();
119 open.pop();
120
121 if (current.node == goal) {
122 return reconstructPath(allRecords, current);
123 }
124
125 closed[current.node.get()] = true;
126
127 // Expand neighbors
128 for (auto &neighbor: connections_.at(current.node.get())) {
129 if (closed[neighbor.get()]) continue;
130
131 float gNew = current.gCost + distance(current.node, neighbor);
132 auto it = allRecords.find(neighbor.get());
133
134 if (it == allRecords.end() || gNew < it->second.gCost) {
135 NodeRecord rec;
136 rec.node = neighbor;
137 rec.parent = current.node;
138 rec.gCost = gNew;
139 rec.fCost = gNew + heuristic(neighbor, goal);
140
141 allRecords[neighbor.get()] = rec;
142 open.push(rec);
143 }
144 }
145 }
146
147 // No path, return target only
148 return {goal};
149}
