175void AnimationManager::gpuUpdateStep(double dt) {
176 RenderState *rs = RenderState::get();
177 // Remember the GPU thread ID
178 gpu_threadID_ = boost::this_thread::get_id();
179
180 // Set processing flags, so that other threads can wait for the completion of this loop
181 gpu_isUpdateActive_.value.test_and_set(std::memory_order_acquire);
182
183 // Avoid race condition with close waiting for gpu_isUpdateActive_ to clear
184 if (closeFlag_.value.test(std::memory_order_acquire) ||
185 pauseFlag_.value.test(std::memory_order_acquire)) {
186 gpu_isUpdateActive_.value.clear(std::memory_order_release);
187 return;
188 }
189
190 // Advance each active GPU animation.
191 for (const auto &anim : gpuAnimations_) {
192 if (!anim->isRunning()) { continue; }
193 auto animState = anim->animationState();
194 animState->enable(rs);
195 anim->gpuUpdate(rs, dt);
196 animState->disable(rs);
197 }
198
199 // Clear processing flags
200 gpu_isUpdateActive_.value.clear(std::memory_order_release);
201
202 // Perform pending add/remove operations
203 while (!gpu_commandQueue_.empty()) {
204 Command &cmd = gpu_commandQueue_.front();
205 switch (cmd.type) {
206 case ADD:
207 gpuAnimations_.push_back(cmd.animation);
208 break;
209 case REMOVE:
210 gpuAnimations_.erase(std::ranges::remove(
211 gpuAnimations_, cmd.animation).begin(), gpuAnimations_.end());
212 break;
213 }
214 gpu_commandQueue_.pop();
215 }

280void AnimationManager::unsyncedUpdate(Animation *animation) {
281 using Clock = std::chrono::steady_clock;
282 using ms = std::chrono::duration<double, std::milli>;
283
284 const double targetMs = 1000.0 / animation->desiredFrameRate();
285 const auto d_frameDuration = ms(targetMs);
286 const auto frameDuration = std::chrono::duration_cast<Clock::duration>(d_frameDuration);
287 auto nextFrame = Clock::now();
288
289 while (closeFlag_.value.test(std::memory_order_acquire) == false && animation->isRunning()) {
290 // Increase update counter atomic
291 unsynced_numActiveUpdates_.fetch_add(1, std::memory_order_release);
292
293 // Avoid race condition on close
294 if (closeFlag_.value.test(std::memory_order_acquire)) {
295 unsynced_numActiveUpdates_.fetch_sub(1, std::memory_order_release);
296 break;
297 }
298
299 // Spin until we can continue
300 if (pauseFlag_.value.test(std::memory_order_acquire)) {
301 unsynced_numActiveUpdates_.fetch_sub(1, std::memory_order_release);
302 CPU_PAUSE();
303 continue;
304 }
305
306 auto frameStart = Clock::now();
307 double dt = std::chrono::duration<double, std::milli>(frameStart - nextFrame + frameDuration).count();
308
309 // Run the animation logic
310 animation->cpuUpdate(dt);
311
312 // decrease update counter atomic
313 unsynced_numActiveUpdates_.fetch_sub(1, std::memory_order_release);
314
315 // Schedule next frame
316 nextFrame += frameDuration;
317
318 auto now = Clock::now();
319 if (now < nextFrame) {
320 std::this_thread::sleep_until(nextFrame);
321 } else {
322 // Missed the frame deadline: resync
323 nextFrame = now;
324 }
325 }
326}
