252void Texture::setTextureData(const ref_ptr<ImageData> &textureData) {
253 textureData_ = textureData;
254 if (!textureData_->floatPixels) {
255 if (textureData_->pixelType == GL_FLOAT) {
256 // already float data
257 textureData_->floatPixels = reinterpret_cast<float *>(textureData_->pixels);
258 } else if (textureData_->pixelType == GL_UNSIGNED_BYTE) {
259 // convert to float data
260 const uint32_t numPixels = static_cast<uint32_t>(textureData_->width) *
261 static_cast<uint32_t>(textureData_->height) *
262 static_cast<uint32_t>(textureData_->depth);
263 const uint32_t numComponents = static_cast<uint32_t>(textureData_->bpp);
264 auto *floatPixels = new float[numPixels * numComponents];
265 for (uint32_t i = 0; i < numPixels * numComponents; ++i) {
266 floatPixels[i] = static_cast<float>(textureData_->pixels[i]) / 255.0f;
267 }
268 textureData_->floatPixels = floatPixels;
269 }
270 }

301void Texture::allocTexture() {
302 Vec3ui size(width(), height(), depth());
303 if (size == allocatedSize_) {
304 // already allocated
305 return;
306 }
307 bool isReAlloc = (
308 allocatedSize_.x > 0 &&
309 allocatedSize_.y > 0 &&
310 allocatedSize_.z > 0);
311 allocatedSize_ = size;
312 numTexel_ = size.x * size.y * size.z;
313 if (isReAlloc) {
314 // NOTE: texture objects must be destroyed and re-created
315 glDeleteTextures(numObjects_, ids_);
316 glCreateTextures(texBind_.target_, numObjects_, ids_);
317 }
318 texBind_.id_ = ids_[0];
319 objectIndex_ = 0;
320 for (uint32_t j = 0; j < numObjects_; ++j) {
321 (this->*(this->allocTexture_))();
322 if (isReAlloc) {
323 set_wrapping(wrappingMode_);
324 set_filter(texFilter_);
325 if (texLoD_) { set_lod(*texLoD_); }
326 if (texSwizzle_) { set_swizzle(*texSwizzle_); }
327 if (texCompare_) { set_compare(*texCompare_); }
328 if (texMaxLevel_) { set_maxLevel(*texMaxLevel_); }
329 if (texAniso_) { set_aniso(*texAniso_); }
330 }
331 nextObject();
332 }
333 if (isReAlloc) {
334 TextureBinder::rebind(this);
335 }

374Bounds<Vec2ui> Texture::getRegion(const Vec2f &texco, const Vec2f &regionTS) const {
375 auto w = static_cast<float>(width());
376 auto h = static_cast<float>(height());
377 auto startX = static_cast<unsigned int>(std::floor(texco.x * w));
378 auto startY = static_cast<unsigned int>(std::floor(texco.y * h));
379 auto endX = static_cast<unsigned int>(std::ceil((texco.x + regionTS.x) * w));
380 auto endY = static_cast<unsigned int>(std::ceil((texco.y + regionTS.y) * h));
381 if (endX >= width()) endX = width() - 1;
382 if (endY >= height()) endY = height() - 1;
383 return Bounds<Vec2ui>::create(
384 Vec2ui(startX, startY),
385 Vec2ui(endX, endY));

388unsigned int Texture::texelIndex(const Vec2f &texco) const {
389 auto w = width();
390 auto h = height();
391 //auto x = static_cast<unsigned int>(std::round(texco.x * static_cast<float>(w)));
392 //auto y = static_cast<unsigned int>(std::round(texco.y * static_cast<float>(h)));
393 auto x = static_cast<unsigned int>(texco.x * static_cast<float>(w));
394 auto y = static_cast<unsigned int>(texco.y * static_cast<float>(h));
395 // clamp to texture size
396 switch (wrappingMode_.x) {
397 case GL_REPEAT:
398 x = x % w;
399 y = y % h;
400 break;
401 case GL_MIRRORED_REPEAT:
402 x = x % (2 * w);
403 y = y % (2 * h);
404 if (x >= w) x = 2 * w - x - 1;
405 if (y >= h) y = 2 * h - y - 1;
406 break;
407 default: // GL_CLAMP_TO_EDGE:
408 if (x >= w) x = w - 1;
409 if (y >= h) y = h - 1;
410 break;
411 }
412 return (y * w + x);
413}

431static std::vector<GLubyte> readTextureData_cfg(LoadingContext&, scene::SceneInputNode &input, GLenum format) {
432 std::vector<GLubyte> data;
433 auto numPixelComponents = glenum::pixelComponents(format);
434 // iterate over all "texel" children
435 for (auto &child: input.getChildren("texel")) {
436 if (!child->hasAttribute("v")) {
437 REGEN_WARN("No 'v' attribute found for texel child '" << child->getName() << "'.");
438 continue;
439 }
440 int texelWidth = child->getValue<int>("width", 1);
441 for (int i = 0; i < texelWidth; ++i) {
442 if (numPixelComponents == 1) {
443 data.push_back(child->getValue<uint32_t>("v", 0u));
444 } else if (numPixelComponents == 2) {
445 auto v = child->getValue<Vec2ui>("v", Vec2ui::zero());
446 data.push_back(v.x);
447 data.push_back(v.y);
448 } else if (numPixelComponents == 3) {
449 auto v = child->getValue<Vec3ui>("v", Vec3ui::zero());
450 data.push_back(v.x);
451 data.push_back(v.y);
452 data.push_back(v.z);
453 } else {
454 auto v = child->getValue<Vec4ui>("v", Vec4ui::zero());
455 data.push_back(v.x);
456 data.push_back(v.y);
457 data.push_back(v.z);
458 data.push_back(v.w);
459 }
460 }
461 }
462 return data;
463}

510ref_ptr<Texture> Texture::load(LoadingContext &ctx, scene::SceneInputNode &input) {
511 ref_ptr<Texture> tex;
512 auto &screen = ctx.scene()->screen();
513 const std::string typeName = input.getValue("type");
514
515 if (input.hasAttribute("file")) {
516 TextureConfig texConfig;
517 texConfig.forcedInternalFormat = glenum::textureInternalFormat(
518 input.getValue<std::string>("internal-format", "NONE"));
519 texConfig.forcedFormat = glenum::textureFormat(
520 input.getValue<std::string>("forced-format", "NONE"));
521 texConfig.forcedSize =
522 input.getValue<Vec3ui>("forced-size", Vec3ui::zero());
523 texConfig.keepData = input.getValue<bool>("keep-data", false);
524 texConfig.useMipmaps = input.getValue<bool>("mipmap", false);
525 const std::string filePath =
526 resourcePath(input.getValue("file"));
527
528 REGEN_DEBUG("Loading texture from file " << filePath << " with config: "
529 << " internal-format=0x" << std::hex << texConfig.forcedInternalFormat << std::dec
530 << " format=0x" << std::hex << texConfig.forcedFormat << std::dec
531 << " size=" << texConfig.forcedSize
532 << " keep-data=" << texConfig.keepData
533 << " use-mipmaps=" << texConfig.useMipmaps);
534
535 try {
536 if (input.getValue<bool>("is-cube", false)) {
537 tex = textures::loadCube(
538 filePath,
539 input.getValue<bool>("cube-flip-back", false),
540 texConfig);
541 } else if (input.getValue<bool>("is-array", false)) {
542 tex = textures::loadArray(
543 filePath,
544 input.getValue<std::string>("name-pattern", ".*"),
545 texConfig);
546 } else if (input.getValue<bool>("is-raw", false)) {
547 tex = textures::loadRAW(
548 filePath,
549 input.getValue<Vec3ui>("raw-size", Vec3ui::create(256u)),
550 input.getValue<uint32_t>("raw-components", 3u),
551 input.getValue<uint32_t>("raw-bytes", 4u));
552 } else {
553 tex = textures::load(filePath, texConfig);
554 }
555 }
556 catch (textures::Error &ie) {
557 REGEN_ERROR("Failed to load Texture at " << filePath << ". " << ie.what());
558 }
559 }
560#ifdef HAS_AV_LIBS
561 else if (input.hasAttribute("video")) {
562 const std::string filePath = resourcePath(input.getValue("video"));
563 ref_ptr<VideoTexture> video = ref_ptr<VideoTexture>::alloc();
564 try {
565 video->set_file(filePath);
566 auto filename = filePath.substr(filePath.find_last_of('/') + 1);
567 video->setAnimationName(REGEN_STRING("Video" << filename));
568 video->demuxer()->set_repeat(
569 input.getValue<bool>("repeat", true));
570 tex = video;
571 video->play();
572 video->startAnimation();
573 }
574 catch (VideoTexture::Error &ve) {
575 REGEN_ERROR("Failed to load Video at " << filePath << ". " << ve.what());
576 }
577 }
578#endif
579 else if (typeName == "noise") {
580 auto sizeMode = input.getValue<std::string>("size-mode", "abs");
581 auto sizeRel = input.getValue<Vec3f>("size", Vec3f(256.0, 256.0, 1.0));
582 auto sizeAbs = getSize(screen->viewport().r, sizeMode, sizeRel);
583 auto isSeamless = input.getValue<bool>("is-seamless", false);
584 auto generator = NoiseGenerator::load(ctx, input);
585 if (generator.get()) {
586 auto noise = ref_ptr<NoiseTexture2D>::alloc(sizeAbs.x, sizeAbs.y, isSeamless);
587 noise->setNoiseScale(input.getValue<float>("noise-scale", 1.0f));
588 noise->setNoiseGenerator(generator);
589 tex = noise;
590 } else {
591 REGEN_WARN("No noise generator found for " << input.getDescription() << ".");
592 }
593 } else if (input.hasAttribute("ramp")) {
594 auto ramp = input.getValue("ramp");
595 if (ramp == "dark-white") {
596 tex = RampTexture::darkWhite();
597 } else if (ramp == "dark-white-skewed") {
598 tex = RampTexture::darkWhiteSkewed();
599 } else if (ramp == "normal") {
600 tex = RampTexture::normal();
601 } else if (ramp == "three-step") {
602 tex = RampTexture::threeStep();
603 } else if (ramp == "four-step") {
604 tex = RampTexture::fourStep();
605 } else if (ramp == "four-step-skewed") {
606 tex = RampTexture::fourStepSkewed();
607 } else if (ramp == "black-white-black") {
608 tex = RampTexture::blackWhiteBlack();
609 } else if (ramp == "stripes") {
610 tex = RampTexture::stripes();
611 } else if (ramp == "stripe") {
612 tex = RampTexture::stripe();
613 } else if (ramp == "rgb") {
614 tex = RampTexture::rgb();
615 } else if (ramp == "inline") {
616 auto format = glenum::textureFormat(
617 input.getValue<std::string>("format", "LUMINANCE"));
618 auto internalFormat = glenum::textureInternalFormat(format);
619 if (input.hasAttribute("internal-format")) {
620 internalFormat = glenum::textureInternalFormat(
621 input.getValue<std::string>("internal-format", "LUMINANCE"));
622 }
623 auto data = readTextureData_cfg(ctx, input, format);
624 tex = ref_ptr<RampTexture>::alloc(format, internalFormat, data);
625 } else {
626 REGEN_WARN("Unknown ramp type '" << ramp << "'.");
627 }
628 } else if (input.hasAttribute("spectrum")) {
629 auto spectrum = input.getValue<Vec2d>("spectrum", Vec2d(0.0, 1.0));
630 auto numTexels = input.getValue<int>("num-texels", 256u);
631 tex = regen::textures::loadSpectrum(spectrum.x, spectrum.y, numTexels);
632 } else if (typeName == "bloom") {
633 auto numMips = input.getValue<uint32_t>("num-mips", 5u);
634 auto bloomTexture = ref_ptr<BloomTexture>::alloc(numMips);
635 auto inputFBO = ctx.scene()->getResource<FBO>(input.getValue("input-fbo"));
636 if (inputFBO.get() == nullptr) {
637 REGEN_WARN("Unable to find FBO for '" << input.getDescription() << "'.");
638 } else {
639 auto resizer = ref_ptr<TextureResizer>::alloc(
640 bloomTexture, screen, 1.0, 1.0);
641 ctx.scene()->addEventHandler(Scene::RESIZE_EVENT, resizer);
642 tex = bloomTexture;
643 bloomTexture->resize(inputFBO->width(), inputFBO->height());
644 }
645 } else {
646 auto sizeMode = input.getValue<std::string>("size-mode", "abs");
647 auto sizeRel = input.getValue<Vec3f>("size", Vec3f(256.0, 256.0, 1.0));
648 Vec3i sizeAbs = getSize(screen->viewport().r, sizeMode, sizeRel);
649
650 auto texCount = input.getValue<uint32_t>("count", 1);
651 auto pixelComponents = input.getValue<uint32_t>("pixel-components", 4);
652 auto pixelType = glenum::pixelType(
653 input.getValue<std::string>("pixel-type", "UNSIGNED_BYTE"));
654 auto textureTarget = glenum::textureTarget(
655 input.getValue<std::string>("target", sizeAbs.z > 1 ? "TEXTURE_3D" : "TEXTURE_2D"));
656 auto numSamples = input.getValue<uint32_t>("num-samples", 1);
657
658 GLenum internalFormat;
659 if (input.hasAttribute("internal-format")) {
660 internalFormat = glenum::textureInternalFormat(
661 input.getValue<std::string>("internal-format", "RGBA8"));
662 } else {
663 auto pixelSize = input.getValue<uint32_t>("pixel-size", 16);
664 internalFormat = glenum::textureInternalFormat(pixelType,
665 pixelComponents, pixelSize);
666 }
667
668 auto textureFormat = glenum::textureFormat(pixelComponents);
669 REGEN_DEBUG("Creating texture with "
670 << sizeAbs.x << "x" << sizeAbs.y << "x" << sizeAbs.z
671 << ", " << texCount << " textures, "
672 << "target: 0x" << std::hex << textureTarget << ", "
673 << "format: 0x" << std::hex << textureFormat << ", "
674 << "internal-format: 0x" << std::hex << internalFormat << ", "
675 << "pixel-type: 0x" << std::hex << pixelType << std::dec << ", "
676 << numSamples << " samples.");
677 tex = FBO::createTexture(
678 sizeAbs.x, sizeAbs.y, sizeAbs.z,
679 texCount,
680 textureTarget,
681 textureFormat,
682 internalFormat,
683 pixelType,
684 numSamples);
685
686 if (input.hasAttribute("size-mode") && sizeMode == "rel") {
687 auto resizer = ref_ptr<TextureResizer>::alloc(
688 tex, screen,
689 sizeRel.x, sizeRel.y);
690 ctx.scene()->addEventHandler(Scene::RESIZE_EVENT, resizer);
691 }
692 }
693
694 if (tex.get() == nullptr) {
695 REGEN_WARN("Failed to create Texture for '" << input.getDescription() << ".");
696 return tex;
697 }
698 tex->set_name(input.getName());
699 configure(tex, input);
700
701 if (typeName == "height") {
702 auto tex2d = dynamic_cast<Texture2D *>(tex.get());
703 if (tex2d) {
704 auto heightMap = ref_ptr<HeightMap>::alloc(*tex2d);
705 heightMap->setMapCenter(input.getValue<Vec2f>(
706 "map-center", Vec2f::zero()));
707 heightMap->setMapFactor(input.getValue<float>(
708 "map-factor", 8.0f));
709 heightMap->setMapSize(input.getValue<Vec2f>(
710 "map-size", Vec2f::create(10.0)));
711 tex = heightMap;
712 }
713 }
714
715 return tex;
716}

718void Texture::configure(ref_ptr<Texture> &tex, scene::SceneInputNode &input) {
719 if (!input.getValue("sampler-type").empty()) {
720 tex->set_samplerType(input.getValue("sampler-type"));
721 }
722 if (tex->numSamples() > 1) {
723 // NOTE: glTexParameter* not allowed for multi-sampled textures
724 return;
725 }
726 if (!input.getValue("wrapping").empty()) {
727 tex->set_wrapping(TextureWrapping::create(glenum::wrappingMode(
728 input.getValue<std::string>("wrapping", "CLAMP_TO_EDGE"))));
729 }
730 if (!input.getValue("aniso").empty()) {
731 tex->set_aniso(input.getValue<float>("aniso", 2.0f));
732 }
733 if (!input.getValue("lod").empty()) {
734 tex->set_lod(input.getValue<Vec2f>("lod", Vec2f::create(1.0f)));
735 }
736 if (!input.getValue("swizzle-r").empty() ||
737 !input.getValue("swizzle-g").empty() ||
738 !input.getValue("swizzle-b").empty() ||
739 !input.getValue("swizzle-a").empty()) {
740 auto swizzleR = static_cast<int>(glenum::textureSwizzle(
741 input.getValue<std::string>("swizzle-r", "RED")));
742 auto swizzleG = static_cast<int>(glenum::textureSwizzle(
743 input.getValue<std::string>("swizzle-g", "GREEN")));
744 auto swizzleB = static_cast<int>(glenum::textureSwizzle(
745 input.getValue<std::string>("swizzle-b", "BLUE")));
746 auto swizzleA = static_cast<int>(glenum::textureSwizzle(
747 input.getValue<std::string>("swizzle-a", "ALPHA")));
748 tex->set_swizzle(Vec4i(swizzleR, swizzleG, swizzleB, swizzleA));
749 }
750 if (!input.getValue("compare-mode").empty()) {
751 auto function = static_cast<int>(glenum::compareFunction(
752 input.getValue<std::string>("compare-function", "LEQUAL")));
753 auto mode = static_cast<int>(glenum::compareMode(
754 input.getValue<std::string>("compare-mode", "NONE")));
755 tex->set_compare(TextureCompare(mode, function));
756 }
757 if (!input.getValue("max-level").empty()) {
758 tex->set_maxLevel(input.getValue<int>("max-level", 1000));
759 }
760
761 if (!input.getValue("min-filter").empty() &&
762 !input.getValue("mag-filter").empty()) {
763 auto min = static_cast<int>(glenum::filterMode(input.getValue("min-filter")));
764 auto mag = static_cast<int>(glenum::filterMode(input.getValue("mag-filter")));
765 tex->set_filter(TextureFilter(min, mag));
766 } else if (!input.getValue("min-filter").empty() ||
767 !input.getValue("mag-filter").empty()) {
768 REGEN_WARN("Minification and magnification filters must be specified both." <<
769 " One missing for '"
770 << input.getDescription()
771 << "'.");
772 }
773 GL_ERROR_LOG();
774}
