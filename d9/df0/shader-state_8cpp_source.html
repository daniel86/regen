29void ShaderState::loadStage(
30 const std::map<std::string, std::string> &shaderConfig,
31 const std::string &effectName,
32 std::map<GLenum, std::string> &code,
33 GLenum stage) {
34 std::string stageName = glenum::glslStageName(stage);
35 std::string effectKey = REGEN_STRING(effectName << "." << glenum::glslStagePrefix(stage));
36 std::string ignoreKey = REGEN_STRING("IGNORE_" << stageName);
37
38 auto it = shaderConfig.find(ignoreKey);
39 if (it != shaderConfig.end() && it->second == "TRUE") { return; }
40
41 code[stage] = Includer::get().include(effectKey);
42 // failed to include ?
43 if (code[stage].empty()) { code.erase(stage); }
44}

46bool ShaderState::createShader(const StateConfig &cfg, const std::string &shaderKey) {
47 std::map<GLenum, std::string> unprocessedCode;
48 for (int i = 0; i < glenum::glslStageCount(); ++i) {
49 loadStage(cfg.defines_, shaderKey, unprocessedCode, glenum::glslStages()[i]);
50 }
51 if (unprocessedCode.empty()) {
52 REGEN_ERROR("Failed to load shader with key '" << shaderKey << "'");
53 return false;
54 }
55 if(createShader(cfg, unprocessedCode)) {
56 REGEN_INFO("Shader [" << shader_->id() << "] successfully loaded from '" << shaderKey << "'.");
57 return true;
58 } else {
59 return false;
60 }

83bool ShaderState::createShader(const StateConfig &cfg, const std::map<GLenum, std::string> &unprocessedCode) {
84 std::map<GLenum, std::string> processedCode;
85
86 PreProcessorConfig preProcessCfg(
87 cfg.version(),
88 unprocessedCode,
89 cfg.defines_,
90 cfg.functions_,
91 cfg.inputs_,
92 cfg.includes_);
93 Shader::preProcess(processedCode, preProcessCfg);
94 shader_ = ref_ptr<Shader>::alloc(processedCode);
95 // setup transform feedback attributes
96 shader_->setTransformFeedback(cfg.feedbackAttributes_, cfg.feedbackMode_, cfg.feedbackStage_);
97 shader_->setInputs(cfg.inputs_);
98
99 if (!shader_->compile()) {
100 REGEN_ERROR("Shader failed to compiled.");
101 for (auto it = processedCode.begin(); it != processedCode.end(); ++it) {
102 REGEN_DEBUG("Shader code failed to compile:\n" << it->second);
103 }
104 return false;
105 }
106 if (!shader_->link()) {
107 REGEN_ERROR("Shader failed to link.");
108 for (auto it = processedCode.begin(); it != processedCode.end(); ++it) {
109 REGEN_DEBUG("Shader code failed to link:\n" << it->second);
110 }
111 } else {
112 set_isHidden(false);
113 }
114
115 for (const auto & texture : cfg.textures_) {
116 shader_->setTexture(texture.second.first, texture.first);
117 }
118
119 return true;
120}

132ref_ptr<Shader> ShaderState::findShader(State *s) {
133 auto joined = s->joined();
134 for (auto it = joined->rbegin(); it != joined->rend(); ++it) {
135 ref_ptr<Shader> out = findShader((*it).get());
136 if (out.get() != nullptr) return out;
137 }
138
139 auto *shaderState = dynamic_cast<ShaderState *>(s);
140 if (shaderState != nullptr) return shaderState->shader();
141
142 auto *hasShader = dynamic_cast<HasShader *>(s);
143 if (hasShader != nullptr) return hasShader->shaderState()->shader();
144
145 return {};
146}

157ref_ptr<ShaderState> ShaderState::load(LoadingContext &ctx, scene::SceneInputNode &input) {
158 if (!input.hasAttribute("key") && !input.hasAttribute("code")) {
159 REGEN_WARN("Ignoring " << input.getDescription() << " without shader input.");
160 return {};
161 }
162 ref_ptr<ShaderState> shaderState = ref_ptr<ShaderState>::alloc();
163
164 const std::string shaderKey = input.hasAttribute("key") ?
165 input.getValue("key") : input.getValue("code");
166 StateConfigurer stateConfigurer;
167 stateConfigurer.addNode(ctx.parent().get());
168 stateConfigurer.addState(shaderState.get());
169 shaderState->createShader(stateConfigurer.cfg(), shaderKey);
170
171 return shaderState;
172}
