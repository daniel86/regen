97 btMesh.m_vertexStride = static_cast<int>(pos->vertexSize());

111 break;

161 mass = 0.0;

172 shape = createCone(input);

174 shape = createConvexHull(input, mesh);

262 if (input.hasAttribute("radius")) {

264 } else {

266 }

275 sphere = ref_ptr<BoundingSphere>::alloc(center, radius_opt.value());
276 for (auto &part: parts) { sphere->addPart(part); }

327 parts.push_back(part);

365void ShapeProcessor::processInput(
366 scene::SceneLoader *scene,
367 SceneInputNode &input,
368 const ref_ptr<StateNode> &parentNode,
369 const ref_ptr<State> &parentState) {
370 auto transformID = input.getValue("transform-id");
371 auto mesh = getMesh(scene, input);
372 auto parts = getParts(scene, input);
373 // read flags for target of shape
374 auto isMeshShape = input.getValue<uint32_t>("mesh", 0u);
375 auto isIndexShape = input.getValue<uint32_t>("index", 0u);
376 auto isGPUShape = input.getValue<uint32_t>("gpu", 0u);
377 auto isPhysicalShape = input.getValue<uint32_t>("physics", 0u);
378 auto isCullShape = input.getValue<uint32_t>("cull", 0u);
379 auto useLocalStamp = input.getValue<bool>("local-stamp", false);
380 if (!isIndexShape && !isGPUShape && !isPhysicalShape) {
381 isMeshShape = 1;
382 }
383 auto numInstances = 1u;
384 auto shapeName = input.getName();
385
386 ref_ptr<HeightMap> heightMap;
387 if (input.hasAttribute("height-map")) {
388 auto heightMap2D = scene->getResources()->getTexture2D(scene, input.getValue("height-map"));
389 if (heightMap2D.get()) {
390 heightMap = ref_ptr<HeightMap>::dynamicCast(heightMap2D);
391 if (!heightMap) {
392 REGEN_WARN("Texture for height map is not a height map for " << input.getDescription() << ".");
393 } else {
394 heightMap->ensureTextureData();
395 }
396 } else if (input.hasAttribute("height-map")) {
397 REGEN_WARN("Ignoring height map for " << input.getDescription() << " without texture.");
398 }
399 }
400
401 uint32_t traversalMask = 1;
402 for (auto &child: input.getChildren("traversal-bit")) {
403 auto bit = child->getValue<uint32_t>("value", 0u);
404 if (child->getValue<bool>("enabled", true)) {
405 traversalMask |= (1 << bit);
406 } else {
407 traversalMask &= ~(1 << bit);
408 }
409 }
410
411 auto transform = scene->getResource<ModelTransformation>(transformID);
412 if (transform.get()) {
413 numInstances = transform->numInstances();
414 }
415 if (input.hasAttribute("num-instances")) {
416 numInstances = input.getValue<uint32_t>("num-instances", numInstances);
417 }
418 numInstances = std::max(numInstances, 1u);
419
420 if (isPhysicalShape) {
421 // add shape to physics engine
422 if (numInstances == 1) {
423 auto motion = ref_ptr<ModelMatrixMotion>::alloc(transform, 0);
424 auto physicalProps = createPhysicalProps(input, mesh, motion);
425 auto physicalObject = ref_ptr<PhysicalObject>::alloc(physicalProps);
426 mesh->addPhysicalObject(physicalObject);
427 scene->getPhysics()->addObject(physicalObject);
428 } else {
429 auto motionAnim = ref_ptr<ModelMatrixUpdater>::alloc(transform);
430 for (uint32_t i = 0; i < numInstances; ++i) {
431 auto motion = ref_ptr<Mat4fMotion>::alloc(motionAnim, i);
432 auto physicalProps = createPhysicalProps(input, mesh, motion);
433 auto physicalObject = ref_ptr<PhysicalObject>::alloc(physicalProps);
434 mesh->addPhysicalObject(physicalObject);
435 scene->getPhysics()->addObject(physicalObject);
436 }
437 motionAnim->startAnimation();
438 mesh->addAnimation(motionAnim);
439 }
440 }
441
442 if (isIndexShape) {
443 // add shape to spatial index
444 auto spatialIndex = getSpatialIndex(scene, input);
445 if (spatialIndex.get()) {
446 for (uint32_t i = 0; i < numInstances; ++i) {
447 auto shape = createShape(input, mesh, parts);
448 if (!shape.get()) {
449 REGEN_WARN("Skipping shape node " << input.getDescription() << " without shape.");
450 continue;
451 }
452 shape->setUseLocalStamp(useLocalStamp);
453 shape->setName(shapeName);
454 shape->setInstanceID(i);
455 shape->setTraversalMask(traversalMask);
456 if (transform.get()) {
457 shape->setTransform(transform, transform->numInstances() > 1 ? i : 0);
458 }
459 if (heightMap.get()) {
460 // translate to half height of height map
461 shape->setBaseOffset(Vec3f(0.0f, 0.5f*heightMap->mapFactor(), 0.0f));
462 }
463 spatialIndex->insert(shape);
464 }
465
466 auto indexedShapes = spatialIndex->getShapes(shapeName);
467 if (indexedShapes.get()) {
468 if (mesh.get()) {
469 mesh->setIndexedShapes(indexedShapes);
470 }
471 for (auto &part: parts) {
472 part->setIndexedShapes(indexedShapes);
473 }
474 }
475
476 if (isCullShape) {
477 REGEN_DEBUG("Creating CPU cull shape for " << input.getDescription() << ".");
478 auto cullShape = ref_ptr<CullShape>::alloc(spatialIndex, input.getName());
479 if (mesh.get()) {
480 mesh->setCullShape(cullShape);
481 }
482 for (auto &part: parts) {
483 part->setCullShape(cullShape);
484 }
485 }
486 } else {
487 REGEN_WARN("Skipping shape node " << input.getDescription() << " without spatial index.");
488 }
489 if (mesh.get() && !mesh->hasBoundingShape()) {
490 // also create a shape without instances for the mesh bounding box
491 isMeshShape = 1;
492 }
493 }
494
495 if (isGPUShape || isMeshShape) {
496 auto shape = createShape(input, mesh, parts);
497 if (shape.get()) {
498 shape->setName(shapeName);
499 if (mesh.get()) {
500 mesh->setBoundingShape(shape);
501 }
502 for (auto &part: parts) {
503 part->setBoundingShape(shape);
504 }
505 } else {
506 REGEN_WARN("Skipping shape node " << input.getDescription() << " without shape.");
507 }
508 if (transform.get()) {
509 shape->setTransform(transform);
510 }
511 if (isGPUShape && isCullShape) {
512 REGEN_DEBUG("Creating GPU cull shape for " << input.getDescription() << ".");
513 auto cullShape = ref_ptr<CullShape>::alloc(shape, input.getName());
514 if (mesh.get()) {
515 mesh->setCullShape(cullShape);
516 }
517 for (auto &part: parts) {
518 part->setCullShape(cullShape);
519 }
520 }
521 }
