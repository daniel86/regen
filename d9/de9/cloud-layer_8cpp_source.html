9static float *createNoiseSlice(uint32_t texSize, uint32_t octave) {
10 uint32_t size2 = texSize * texSize;
11 float oneOverTexSize = 1.f / static_cast<float>(texSize);
12 osgHimmel::Noise n(1 << (octave + 2),
13 math::random<float>(0.f, 1.f),
14 math::random<float>(0.f, 1.f));
15
16 auto *noise = new float[size2];
17 uint32_t o;
18 for (uint32_t s = 0; s < texSize; ++s)
19 for (uint32_t t = 0; t < texSize; ++t) {
20 o = t * texSize + s;
21 noise[o] = n.noise2(
22 static_cast<float>(s) * oneOverTexSize,
23 static_cast<float>(t) * oneOverTexSize, octave) * 0.5f + 0.5f;
24 }
25
26 return noise;
27}
