26 enum ArenaType {
27 // Note: keep per-frame modes first!
28 // An arena for per-frame updates using persistent mapping of a small adaptive ring buffer.
29 // Small meaning that the maximum number of ring segments is capped to some rather small value.
30 WRITE_PER_FRAME_LARGE_DATA = 0,
31 // An arena for per-frame updates using persistent mapping of a large adaptive ring buffer.
32 // Large meaning that the maximum number of ring segments is capped to some rather high value.
33 WRITE_PER_FRAME_SMALL_DATA,
34 // An arena for per-frame updates where ring buffer is not feasible, e.g. due to excessive size.
35 // In this arena mode, only a single buffer is used in staging, and data is copied directly
36 // into this staging buffer (without any mapping).
37 WRITE_PER_FRAME_HUGE_DATA,
38 // An arena for reading data per frame. The arena uses persistent mapping
39 // with an adaptive ring buffer.
40 READ_PER_FRAME,
41 // An arena for rare reading of data. It uses temporary mapping with a single buffer
42 // in staging and polls the main buffer at regular intervals.
43 READ_RARELY,
44 // An arena for rare updates which are performed via a single buffer in staging which is
45 // temporary mapped with range invalidation.
46 WRITE_RARELY,
47 // An arena for static data which is only updated very rarely.
48 // This is using implicit staging without multi-buffering, the most appropriate
49 // mode of uploading data to the GPU is determined by storage flags of the main buffer.
50 WRITE_ALMOST_NEVER,
51 ARENA_TYPE_LAST // keep last
52 };
