11float Earth::orbitEccentricity() {

26float Earth::longitudeNutation(const osgHimmel::t_julianDay &t) {
27 const float sM = _rad(Sun::meanAnomaly(t));
28 const float mM = _rad(Moon::meanAnomaly(t));
29 const float O = _rad(Moon::meanOrbitLongitude(t));
30 // (AA.21 p132)
31 const float r =
32 -_decimal(0, 0, 17.20f) * sinf(O)
33 - _decimal(0, 0, 1.32f) * sinf(2.0f * sM)
34 - _decimal(0, 0, 0.23f) * sinf(2.0f * mM)
35 + _decimal(0, 0, 0.21f) * sinf(2.0f * O);
36
37 return r;
38}

40float Earth::obliquityNutation(const osgHimmel::t_julianDay &t) {
41 const float O = _rad(Moon::meanOrbitLongitude(t));
42 const float Ls = _rad(Sun::meanAnomaly(t));
43 const float Lm = _rad(Moon::meanAnomaly(t));
44 // (AA.21 p132)
45 const float e =
46 +_decimal(0, 0, 9.20f) * cosf(O)
47 + _decimal(0, 0, 0.57f) * cosf(2.0f * Ls)
48 + _decimal(0, 0, 0.10f) * cosf(2.0f * Lm)
49 - _decimal(0, 0, 0.09f) * cosf(2.0f * O);
50
51 return e;
52}

85float Earth::atmosphericRefraction(float altitude) {
86 // Effect of refraction for true altitudes (AA.15.4).
87 // G.G. Bennet, "The Calculation of the Astronomical Refraction in marine Navigation", 1982
88 // and Porsteinn Saemundsson, "Sky and Telescope" 1982
89 osgHimmel::t_longf R = 1.02 /
90 tan(_rad(altitude + 10.3 / (altitude + 5.11))) + 0.0019279;
91 return _decimal(0, R, 0); // (since R is in minutes)
92}
