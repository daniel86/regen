156void Scene::mouseMove(const Vec2i &pos) {
157 boost::posix_time::ptime time(
158 boost::posix_time::microsec_clock::local_time());
159 int dx, dy;
160 {
161 auto mouse_m = mousePosition_->getVertex(0);
162 dx = pos.x - static_cast<int>(mouse_m.r.x);
163 dy = pos.y - static_cast<int>(mouse_m.r.y);
164 }
165 mousePosition_->setVertex(0, Vec2f(pos.x, pos.y));
166 updateMousePosition();
167
168 ref_ptr<MouseMotionEvent> event = ref_ptr<MouseMotionEvent>::alloc();
169 event->dt = ((double) (time - lastMotionTime_).total_microseconds()) / 1000.0;
170 event->dx = dx;
171 event->dy = dy;
172 queueEmit(MOUSE_MOTION_EVENT, event);

406void Scene::updateBOs() {
407 std::stack<const StateNode*> nodeQueue;
408 std::stack<const State*> stateQueue;
409 std::set<const ShaderInput*> visited;
410 nodeQueue.push(renderTree_.get());
411
412 for (auto &anim : AnimationManager::get().gpuAnimations()) {
413 if(anim->animationState().get() != nullptr)
414 stateQueue.push(anim->animationState().get());
415 }
416
417 while (!nodeQueue.empty()) {
418 const StateNode *node = nodeQueue.top();
419 nodeQueue.pop();
420 stateQueue.push(node->state().get());
421
422 // push node children to the queue
423 for (const auto &child: node->childs()) {
424 nodeQueue.push(child.get());
425 }
426
427 // process state of node
428 while (!stateQueue.empty()) {
429 const State *state = stateQueue.top();
430 stateQueue.pop();
431
432 const auto *lp = dynamic_cast<const LightPass*>(state);
433 if (lp) {
434 for (const auto &light: lp->lights()) {
435 stateQueue.push(light.light.get());
436 }
437 }
438
439 for (const auto &ni: state->inputs()) {
440 auto input = ni.in_;
441 if (visited.find(input.get()) != visited.end()) {
442 continue; // already processed
443 }
444 visited.insert(input.get());
445
446 ref_ptr<BufferBlock> bufferBlock = ref_ptr<BufferBlock>::dynamicCast(input);
447 if (bufferBlock.get() && bufferBlock->stagingUpdateHint().frequency <= BUFFER_UPDATE_PER_FRAME) {
448 bufferBlock->update();
449 }
450 }
451
452 auto joined = state->joined();
453 for (const auto &x: *joined.get()) {
454 stateQueue.push(x.get());
455 }
456 }
457 }
458}
