308 int32_t queuedIdx = 0;
309 for (; queuedIdx + simd::RegisterWidth <= numQueued; queuedIdx += simd::RegisterWidth) {
310 // Load the sphere data into SIMD registers, we need 4 registers.
311 const BatchOf_float spherePosX = BatchOf_float::loadAligned(d_spherePosX + queuedIdx);
312 const BatchOf_float spherePosY = BatchOf_float::loadAligned(d_spherePosY + queuedIdx);
313 const BatchOf_float spherePosZ = BatchOf_float::loadAligned(d_spherePosZ + queuedIdx);
314 BatchOf_float sphereRadius = BatchOf_float::loadAligned(d_sphereRadius + queuedIdx);
315 // r = r * r
316 sphereRadius *= sphereRadius;
317
318 // Find closest point on OBB to sphere center, start with:
319 // closest = obbCenter
320 BatchOf_float closestX = BatchOf_float::fromScalar(obbCenter.x);
321 BatchOf_float closestY = BatchOf_float::fromScalar(obbCenter.y);
322 BatchOf_float closestZ = BatchOf_float::fromScalar(obbCenter.z);
323 // delta = p1 - obbCenter
324 BatchOf_float deltaX = spherePosX - closestX;
325 BatchOf_float deltaY = spherePosY - closestY;
326 BatchOf_float deltaZ = spherePosZ - closestZ;
327
328 // We need to project delta onto each OBB axis, clamp to halfSize, and accumulate
329 for (int i = 0; i < 3; ++i) {
330 // axis = obbAxes[i]
331 BatchOf_float halfSizeOnAxis = BatchOf_float::fromScalar(g_halfSize[i][g_idx]);
332 // dist = delta.dot(axis)
333 BatchOf_float dist =
334 (deltaX * g_ax[i][g_idx]) +
335 (deltaY * g_ay[i][g_idx]) +
336 (deltaZ * g_az[i][g_idx]);
337 // dist = clamp(dist, -halfSize, halfSize)
338 dist.c = simd::min_ps(simd::max_ps(dist.c, -halfSizeOnAxis.c), halfSizeOnAxis.c);
339 // closest += axis * clamped, use fused-multiply-add
340 closestX.c = simd::mul_add_ps(dist.c, simd::set1_ps(g_ax[i][g_idx]), closestX.c);
341 closestY.c = simd::mul_add_ps(dist.c, simd::set1_ps(g_ay[i][g_idx]), closestY.c);
342 closestZ.c = simd::mul_add_ps(dist.c, simd::set1_ps(g_az[i][g_idx]), closestZ.c);
343 }
344 // delta = closest - p1
345 deltaX = closestX - spherePosX;
346 deltaY = closestY - spherePosY;
347 deltaZ = closestZ - spherePosZ;
348 BatchOf_float tmp = (deltaX * deltaX + deltaY * deltaY + deltaZ * deltaZ);
349 // tmp = (delta(dot)delta) < r_sum*r_sum
350 tmp = (tmp < sphereRadius);
351
352 // Convert lanes (1/0) to bitmask value
353 uint8_t mask = tmp.toBitmask8();
354 while (mask) {
355 int bitIndex = simd::nextBitIndex<uint8_t>(mask);
356 td.hits->push(td.queuedIndices[queuedIdx + bitIndex]);
357 }
358 }
359
360 // Scalar fallback for remaining items
361 for (; queuedIdx < numQueued; queuedIdx++) {
362 const auto itemIdx = td.queuedIndices[queuedIdx];
363 const Vec3f p1(
364 d_spherePosX[queuedIdx],
365 d_spherePosY[queuedIdx],
366 d_spherePosZ[queuedIdx]);
367 const float radiusSq = d_sphereRadius[queuedIdx] * d_sphereRadius[queuedIdx];
368 // compute the closest point on OBB to sphere center
369 const Vec3f closest = testShape.closestPointOnSurface(p1);
370 if ((closest - p1).lengthSquared() <= radiusSq) {
371 td.hits->push(itemIdx);
372 }
373 }
