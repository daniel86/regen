26void ImpostorBillboard::createShader(const ref_ptr<StateNode> &parentNode) {
27 StateConfigurer shaderConfigurer;
28 shaderConfigurer.addNode(parentNode.get());
29
30 // join states of the input mesh, besides its material texture state which we
31 // want to bake into the array textures.
32 // NOTE: might very well be that some texture mapping techniques will cause problems here.
33 // effectively we loose UV coordinates, so we cannot apply any (uv-mapped) textures from
34 // the original mesh to the impostor.
35 for (auto &mesh: meshes_) {
36 std::stack<ref_ptr<State>> stateStack;
37 for (auto &state: *mesh.meshOrig->joined().get()) {
38 stateStack.push(state);
39 }
40 if (mesh.meshOrig->material().get()) {
41 stateStack.push(mesh.meshOrig->material());
42 }
43 while (!stateStack.empty()) {
44 auto state = stateStack.top();
45 stateStack.pop();
46 if (auto *textureState = dynamic_cast<TextureState*>(state.get())) {
47 if (textureState->texture()->targetType() == GL_TEXTURE_BUFFER) {
48 // TBOs must be joined, they could be used for instancing of uniforms
49 joinStates(state);
50 } else {
51 continue; // skip (material) texture states, we will bake them into the snapshot textures
52 }
53 }
54 else {
55 for (auto &input: state->inputs()) {
56 if (dynamic_cast<Texture*>(input.in_.get()) != nullptr) {
57 continue; // skip texture inputs, we will bake them into the snapshot textures
58 }
59 if (!input.in_->isVertexAttribute()) {
60 setInput(input.in_, input.name_);
61 }
62 }
63 }
64 for (auto &joined: *state->joined().get()) {
65 stateStack.push(joined);
66 }
67 }
68 }
69
70 shaderConfigurer.addState(sharedState().get());
71 shaderConfigurer.addState(this);
72 Mesh::createShader(parentNode, shaderConfigurer.cfg());
73}

115void ImpostorBillboard::updateNumberOfViews() {
116 // compute the number of snapshots.
117 numSnapshotViews_ = 0u;
118 if (numLatitudeSteps_ == 0) {
119 numSnapshotViews_ = numLongitudeSteps_;
120 } else {
121 if (isHemispherical_) {
122 numSnapshotViews_ = numLatitudeSteps_ * numLongitudeSteps_;
123 } else {
124 numSnapshotViews_ = (numLatitudeSteps_ * 2 - 1) * numLongitudeSteps_;
125 }
126 }
127 if (hasTopView_) numSnapshotViews_++;
128 if (hasBottomView_ && !isHemispherical_) numSnapshotViews_++;
129 shaderDefine("NUM_IMPOSTOR_VIEWS", REGEN_STRING(numSnapshotViews_));
130 if (isHemispherical_) {
131 shaderDefine("IS_HEMISPHERICAL", "TRUE");
132 }

134 REGEN_DEBUG("impostor num snapshots: " << numSnapshotViews_);

268void ImpostorBillboard::addSnapshotView(uint32_t viewIdx, const Vec3f &dir, const Vec3f &up) {
269 // this is an offset of the mesh that translates it to origin.
270 // in many cases this will be (0,0,0).
271 auto eye = meshCenterPoint_ - dir * meshBoundsRadius_ * 1.5f;
272 snapshotCamera_->setView(viewIdx, Mat4f::lookAtMatrix(eye, dir, up));
273 auto &view = snapshotCamera_->view(viewIdx);
274 snapshotCamera_->setViewInverse(viewIdx, view.lookAtInverse());
275 Vec4f defaultNormal = view ^ Vec4f::create(up, 0.0f);
276 defaultNormals_[viewIdx] = defaultNormal.xyz() * 0.5f + Vec3f::create(0.5f);
277
278 float minX = +FLT_MAX, maxX = -FLT_MAX;
279 float minY = +FLT_MAX, maxY = -FLT_MAX;
280 float minZ = +FLT_MAX, maxZ = -FLT_MAX;
281
282 for (const auto &corner: meshCornerPoints_) {
283 auto viewSpace = view ^ Vec4f::create(corner, 1.0f);
284 minX = std::min(minX, viewSpace.x);
285 maxX = std::max(maxX, viewSpace.x);
286 minY = std::min(minY, viewSpace.y);
287 maxY = std::max(maxY, viewSpace.y);
288 minZ = std::min(minZ, -viewSpace.z);
289 maxZ = std::max(maxZ, -viewSpace.z);
290 }
291 const float zPadding = 0.1f * (maxZ - minZ);
292 minZ -= zPadding;
293 maxZ += zPadding;
294
295 m_viewDir_[viewIdx].xyz() = -dir;
296 m_viewDir_[viewIdx].w = 0.0f; // no w-component, this is a direction vector
297 m_viewBounds_[viewIdx] = Vec4f(minX, maxX, minY, maxY);
298 m_viewDepth_[viewIdx] = Vec2f(minZ, maxZ);
299 if constexpr(IMPOSTOR_DEBUG_SNAPSHOT_VIEWS) {
300 REGEN_INFO("Snapshot view " << viewIdx << ":"
301 << "\n\tmesh-origin=" << meshCenterPoint_
302 << "\n\teye=" << eye
303 << "\n\tdir=" << -dir
304 << "\n\tbounds=" << Vec4f(minX, maxX, minY, maxY)
305 << "\n\tdepth=" << Vec2f(minZ, maxZ));
306 }
307
308 snapshotCamera_->setProjection(viewIdx,
309 Mat4f::orthogonalMatrix(minX, maxX, minY, maxY, minZ, maxZ));
310 snapshotCamera_->setProjectionInverse(viewIdx,
311 snapshotCamera_->projection(viewIdx).orthogonalInverse());
312 ProjectionParams params = snapshotCamera_->projParams(viewIdx);
313 params.near = minZ;
314 params.far = maxZ;
315 params.aspect = abs((maxX - minX) / (maxY - minY));
316 params.fov = 0.0f; // orthographic projection, no fov
317 snapshotCamera_->setProjParams(viewIdx, params);
318 snapshotCamera_->setPosition(viewIdx, eye);
319}

321int ImpostorBillboard::getViewIdx(const Vec3f &dir) {
322 // Compute the longitude segment
323 float lonAngle = atan2(dir.z, dir.x); // returns [-π, π]
324 lonAngle = fmod((lonAngle + math::twoPi<float>()), math::twoPi<float>());
325 int lonIdx = int((lonAngle + 0.5 * longitudeStepSize_) / longitudeStepSize_);
326 lonIdx = math::clamp<int>(lonIdx, 0, numLongitudeSteps_ - 1);
327 // Compute the latitude segment
328 float latAngle = atan2(-dir.y, Vec2f(dir.x, dir.z).length());
329 auto latIdx = static_cast<int>(((latAngle + latitudeStepSize_ * 0.5f) / latitudeStepSize_));
330 if (isHemispherical_) {
331 // map southern hemisphere to the equator
332 latIdx = math::clamp<int>(latIdx, 0, static_cast<int>(numLatitudeSteps_)-1);
333 } else if (latIdx < 0) {
334 // southern hemisphere
335 latIdx = (numLatitudeSteps_-1) + std::min(-latIdx, static_cast<int>(numLatitudeSteps_-1));
336 }
337 // Finally compute view index
338 return latIdx * numLongitudeSteps_ + lonIdx;
339}

341void ImpostorBillboard::updateSnapshotViews() {
342 // make sure resources were created
343 ensureResourcesExist();
344 uint32_t viewIdx = 0u;
345
346 // Latitude steps in radians
347 std::vector<float> latAngles;
348 if (numLatitudeSteps_ == 0) {
349 latAngles.push_back(0.0f);
350 numLatitudeSteps_ = 1;
351 } else {
352 for (uint32_t i = 0; i < numLatitudeSteps_; ++i) {
353 float frac = static_cast<float>(i) / static_cast<float>(numLatitudeSteps_);
354 latAngles.push_back(-frac * math::halfPi<float>());
355 }
356 if (!isHemispherical_) {
357 for (uint32_t i = 0; i < numLatitudeSteps_; ++i) {
358 if (latAngles[i] < 0.0f) {
359 latAngles.push_back(-latAngles[i]);
360 }
361 }
362 }
363 }
364
365 // map data pointers
366 auto mappedBuffer1 = snapshotDirs_->mapClientData<Vec4f>(BUFFER_GPU_WRITE);
367 auto mappedBuffer2 = snapshotOrthoBounds_->mapClientData<Vec4f>(BUFFER_GPU_WRITE);
368 auto mappedBuffer3 = snapshotDepthRanges_->mapClientData<Vec2f>(BUFFER_GPU_WRITE);
369 m_viewDir_ = mappedBuffer1.w.data();
370 m_viewBounds_ = mappedBuffer2.w.data();
371 m_viewDepth_ = mappedBuffer3.w.data();
372
373 for (float lat: latAngles) {
374 float y = sin(lat);
375 float horizontalRadius = cos(lat); // radius on equator ring
376
377 for (uint32_t i = 0; i < numLongitudeSteps_; ++i) {
378 float lon = static_cast<float>(i) /
379 static_cast<float>(numLongitudeSteps_) * math::twoPi<float>();
380 Vec3f dir(
381 horizontalRadius * cos(lon),
382 y,
383 horizontalRadius * sin(lon));
384 dir.normalize();
385 if (isHemispherical_ && dir.y > 0.0f) {
386 // skip southern hemisphere
387 continue;
388 }
389 addSnapshotView(viewIdx++, dir);
390 }
391 }
392
393 if (hasTopView_) {
394 addSnapshotView(viewIdx++, Vec3f::up(), Vec3f::right());
395 }
396 if (hasBottomView_ && !isHemispherical_) {
397 addSnapshotView(viewIdx++, Vec3f::down(), Vec3f::right());
398 }
399
400 mappedBuffer1.unmap();
401 mappedBuffer2.unmap();
402 mappedBuffer3.unmap();
403 impostorBuffer_->clientBuffer()->swapData();
404 impostorBuffer_->update();
405
406 snapshotCamera_->updateViewProjection1();
407 snapshotCamera_->updateShaderData(0.0f);
408 snapshotCamera_->updateBuffers();
409
410 createUpdateShader();
411}

413void ImpostorBillboard::createSnapshot() {
414 auto rs = RenderState::get();
415 // make sure resources were created
416 ensureResourcesExist();
417 snapshotFBO_->enable(rs);
418
419 if(snapshotNormal_.get()) {
420 // Clear the normal textures individually, as each of them may have a different
421 // background color it requires for interpolation.
422 for (uint32_t arrayIndex = 0; arrayIndex < numSnapshotViews_; ++arrayIndex) {
423 Vec3f &f_clearNormal = defaultNormals_[arrayIndex];
424 glClearTexSubImage(snapshotNormal_->textureBind().id_,
425 0, 0, 0, arrayIndex, // level, x, y, z offset
426 snapshotNormal_->width(),
427 snapshotNormal_->height(), 1, // width, height, depth
428 GL_RGB, GL_FLOAT,
429 &f_clearNormal.x);
430 }
431 }
432
433 // render all meshes into the snapshot FBO
434 for (auto &view: meshes_) {
435 auto oldNumInstances = view.meshCopy->numVisibleInstances();
436 // make sure only one instance is rendered
437 view.meshCopy->set_numVisibleInstances(1);
438 view.shaderState->enable(rs);
439 view.meshCopy->draw(rs);
440 view.shaderState->disable(rs);
441 view.meshCopy->set_numVisibleInstances(oldNumInstances);
442 }
443 snapshotFBO_->disable(rs);
444 if constexpr(IMPOSTOR_MIPMAPS) {
445 snapshotAlbedo_->updateMipmaps();
446 }

450 auto parser = ctx.scene();
451 auto impostor = ref_ptr<ImpostorBillboard>::alloc();
452 // find the original mesh
453 auto originalMeshVec = parser->getResources()->getMesh(
454 parser,
455 input.getValue("base-mesh"));
456 if (originalMeshVec.get() == nullptr || originalMeshVec->meshes().empty()) {
457 REGEN_WARN("Ignoring " << input.getDescription() << ", failed to load base mesh.");
458 return {};
459 }
460 if (input.hasAttribute("depth-offset")) {
461 impostor->depthOffset_->setVertex(0, input.getValue<float>("depth-offset", 0.0f));
462 }
463 if (input.hasAttribute("texture-size")) {

465 impostor->setSnapshotTextureSize(textureSize.x, textureSize.y);
466 }
467 if (input.hasAttribute("longitude-steps")) {
468 impostor->numLongitudeSteps_ = input.getValue<uint32_t>("longitude-steps", 8u);
469 }
470 if (input.hasAttribute("latitude-steps")) {
471 impostor->numLatitudeSteps_ = input.getValue<uint32_t>("latitude-steps", 0u);
472 }
473 if (input.hasAttribute("hemispherical")) {
474 impostor->isHemispherical_ = input.getValue<bool>("hemispherical", true);
475 }
476 if (input.hasAttribute("top-view")) {
477 impostor->hasTopView_ = input.getValue<bool>("top-view", false);
478 }
479 if (input.hasAttribute("bottom-view")) {
480 impostor->hasBottomView_ = input.getValue<bool>("bottom-view", false);
481 }
482 if (input.hasAttribute("normal-correction")) {
483 impostor->useNormalCorrection_ = input.getValue<bool>("normal-correction", true);
484 }
485 if (input.hasAttribute("snapshot-shader")) {
486 impostor->snapshotShaderKey_ = input.getValue<std::string>("snapshot-shader", "regen.objects.lod.impostor.update");
487 }
488
489 // load update state
490 auto updateStateNode = input.getFirstChild("update-state");
491 if(updateStateNode.get()) {
492 for (auto &child: updateStateNode->getChildren()) {
493 auto processor = ctx.scene()->getStateProcessor(child->getCategory());
494 if (processor.get() == nullptr) {
495 REGEN_WARN("No processor registered for '" << child->getDescription() << "'.");
496 } else {
497 processor->processInput(ctx.scene(), *child.get(), ctx.parent(), impostor->snapshotState());
498 }
499 }
500 input.removeChild(updateStateNode);
501 }
502
503 // Add meshes to the impostor
504 auto indexRange = CompositeMesh::loadIndexRange(input, "base-mesh");
505 if (indexRange.empty()) { indexRange.push_back(0); }
506 for (auto &index: indexRange) {
507 if (index >= originalMeshVec->meshes().size()) {
508 REGEN_WARN("Invalid mesh index '" << index << "' for '" << input.getDescription() << "'.");
509 } else {
510 auto originalMesh = originalMeshVec->meshes()[index];
511 impostor->addMesh(originalMesh);
512 }
513 }
514
515 impostor->updateSnapshotViews();
516 impostor->createSnapshot();
517 return impostor;
