94 const Mat4f &M = tf.r;
95
96 // Load base corners
97 const auto* __restrict cx = static_cast<float*>(__builtin_assume_aligned(baseCornerX_.data(), 32));
98 const auto* __restrict cy = static_cast<float*>(__builtin_assume_aligned(baseCornerY_.data(), 32));
99 const auto* __restrict cz = static_cast<float*>(__builtin_assume_aligned(baseCornerZ_.data(), 32));
100 const BatchOf_float xs = BatchOf_float::loadAligned(cx);
101 const BatchOf_float ys = BatchOf_float::loadAligned(cy);
102 const BatchOf_float zs = BatchOf_float::loadAligned(cz);
103
104 // Transform all 8 corners
105 BatchOf_float x_prime, y_prime, z_prime;
106 {
107 const BatchOf_float m00 = BatchOf_float::fromScalar(M.x[0]);
108 const BatchOf_float m01 = BatchOf_float::fromScalar(M.x[4]);
109 const BatchOf_float m02 = BatchOf_float::fromScalar(M.x[8]);
110 const BatchOf_float m03 = BatchOf_float::fromScalar(M.x[12]);
111 x_prime = xs*m00 + ys*m01 + zs*m02 + m03;
112 }
113 {
114 const BatchOf_float m10 = BatchOf_float::fromScalar(M.x[1]);
115 const BatchOf_float m11 = BatchOf_float::fromScalar(M.x[5]);
116 const BatchOf_float m12 = BatchOf_float::fromScalar(M.x[9]);
117 const BatchOf_float m13 = BatchOf_float::fromScalar(M.x[13]);
118 y_prime = xs*m10 + ys*m11 + zs*m12 + m13;
119 }
120 {
121 const BatchOf_float m20 = BatchOf_float::fromScalar(M.x[2]);
122 const BatchOf_float m21 = BatchOf_float::fromScalar(M.x[6]);
123 const BatchOf_float m22 = BatchOf_float::fromScalar(M.x[10]);
124 const BatchOf_float m23 = BatchOf_float::fromScalar(M.x[14]);
125 z_prime = xs*m20 + ys*m21 + zs*m22 + m23;
126 }
127
128 // Extract min/max
129 g_minX[globalIndex_] = x_prime.horizontalMin();
130 g_maxX[globalIndex_] = x_prime.horizontalMax();
131 g_minY[globalIndex_] = y_prime.horizontalMin();
132 g_maxY[globalIndex_] = y_prime.horizontalMax();
133 g_minZ[globalIndex_] = z_prime.horizontalMin();
134 g_maxZ[globalIndex_] = z_prime.horizontalMax();
135
136 // tfOrigin = (tfBounds.min + tfBounds.max) * 0.5f;
137 tfOrigin_.x = (g_minX[globalIndex_] + g_maxX[globalIndex_]) * 0.5f;
138 tfOrigin_.y = (g_minY[globalIndex_] + g_maxY[globalIndex_]) * 0.5f;
139 tfOrigin_.z = (g_minZ[globalIndex_] + g_maxZ[globalIndex_]) * 0.5f;
140 } else {
141 _set_min(baseBounds_.min);
142 _set_max(baseBounds_.max);
143 tfOrigin_ = basePosition_;
144 }
145
146 // apply model offset if available
147 if (transform_->hasModelOffset()) {
148 auto &modelOffset = transform_->modelOffset();
149 auto offset = modelOffset->getVertexClamped(transformIndex_);
150 const Vec3f &o = offset.r.xyz();
151 // tfBounds += o;
152 g_minX[globalIndex_] += o.x;
153 g_minY[globalIndex_] += o.y;
154 g_minZ[globalIndex_] += o.z;
155 g_maxX[globalIndex_] += o.x;
156 g_maxY[globalIndex_] += o.y;
157 g_maxZ[globalIndex_] += o.z;
158 tfOrigin_ += o;
159 }
160 } else {
161 _set_min(baseBounds_.min);
162 _set_max(baseBounds_.max);
163 tfOrigin_ = basePosition_;
164 }
165#undef _set_min
166#undef _set_max

207 const BatchOf_float spherePosY = BatchOf_float::loadAligned(d_spherePosY + queuedIdx);

324void AABB::batchTest_OBBs(BatchedIntersectionCase &td) {
325 const auto numQueued = static_cast<int32_t>(td.numQueued);
326 auto *testShape = static_cast<const AABB *>(td.testShape);
327
328 REGEN_AABB_BATCH_DATA(global, testShape->globalBatchData());

342 const BatchOf_float aabbMinZ = BatchOf_float::fromScalar(global_minZ[globalIdx]);
343 const BatchOf_float aabbMaxX = BatchOf_float::fromScalar(global_maxX[globalIdx]);
344 const BatchOf_float aabbMaxY = BatchOf_float::fromScalar(global_maxY[globalIdx]);
345 const BatchOf_float aabbMaxZ = BatchOf_float::fromScalar(global_maxZ[globalIdx]);
346
347 int32_t queuedIdx = 0;
348 for (; queuedIdx + simd::RegisterWidth <= numQueued; queuedIdx += simd::RegisterWidth) {
349 // Accumulate projected radius from each OBB axis
350 BatchOf_float obbRadiusX = BatchOf_float::allZeros();
351 BatchOf_float obbRadiusY = BatchOf_float::allZeros();;
352 BatchOf_float obbRadiusZ = BatchOf_float::allZeros();
353 {
354 BatchOf_float objHalfSize[3] = {
355 BatchOf_float::loadAligned(d_obbHalfSizeX + queuedIdx),
356 BatchOf_float::loadAligned(d_obbHalfSizeY + queuedIdx),
357 BatchOf_float::loadAligned(d_obbHalfSizeZ + queuedIdx)};
358
359 for (uint32_t axisIdx=0u; axisIdx < 3u; ++axisIdx) {
360 // Load axis into SIMD registers
361 auto &axisBatch = d_axes[axisIdx];
362 BatchOf_float axis;
363 // X axis
364 axis = BatchOf_float::loadAligned(axisBatch.x.data() + queuedIdx);
365 obbRadiusX += objHalfSize[axisIdx] * axis.abs();
366 // Y axis
367 axis = BatchOf_float::loadAligned(axisBatch.y.data() + queuedIdx);
368 obbRadiusY += objHalfSize[axisIdx] * axis.abs();
369 // Z axis
370 axis = BatchOf_float::loadAligned(axisBatch.z.data() + queuedIdx);
371 obbRadiusZ += objHalfSize[axisIdx] * axis.abs();
372 }
373 }
374
375 // Intersection test:
376 // ((obbCenter - r) < aabbMax) &&
377 // ((obbCenter + r) > aabbMin)
378 const BatchOf_float obbCenterX = BatchOf_float::loadAligned(d_obbCenterX + queuedIdx);
379 const BatchOf_float obbCenterY = BatchOf_float::loadAligned(d_obbCenterY + queuedIdx);
380 const BatchOf_float obbCenterZ = BatchOf_float::loadAligned(d_obbCenterZ + queuedIdx);
381 const BatchOf_float hasOverlap =
382 // X axis overlap test
383 (obbCenterX - obbRadiusX < aabbMaxX) &&
384 (obbCenterX + obbRadiusX > aabbMinX) &&
385 // Y axis overlap test
386 (obbCenterY - obbRadiusY < aabbMaxY) &&
387 (obbCenterY + obbRadiusY > aabbMinY) &&
388 // Z axis overlap test
389 (obbCenterZ - obbRadiusZ < aabbMaxZ) &&
390 (obbCenterZ + obbRadiusZ > aabbMinZ);
391
392 // Convert lanes (1/0) to bitmask value
393 uint8_t mask = hasOverlap.toBitmask8();
394 while (mask) {
395 int bitIndex = simd::nextBitIndex<uint8_t>(mask);
396 td.hits->push(td.queuedIndices[queuedIdx + bitIndex]);
397 }
398 }
399
400 // Scalar fallback for remaining items
401 for (; queuedIdx < numQueued; queuedIdx++) {
402 // The OBB's projected radius
403 float obbRadiusX =
404 d_obbHalfSizeX[queuedIdx] * std::abs(d_axes[0].x[queuedIdx]) +
405 d_obbHalfSizeY[queuedIdx] * std::abs(d_axes[1].x[queuedIdx]) +
406 d_obbHalfSizeZ[queuedIdx] * std::abs(d_axes[2].x[queuedIdx]);

408 d_obbHalfSizeX[queuedIdx] * std::abs(d_axes[0].y[queuedIdx]) +
409 d_obbHalfSizeY[queuedIdx] * std::abs(d_axes[1].y[queuedIdx]) +
410 d_obbHalfSizeZ[queuedIdx] * std::abs(d_axes[2].y[queuedIdx]);
411 float obbRadiusZ =
412 d_obbHalfSizeX[queuedIdx] * std::abs(d_axes[0].z[queuedIdx]) +
413 d_obbHalfSizeY[queuedIdx] * std::abs(d_axes[1].z[queuedIdx]) +
414 d_obbHalfSizeZ[queuedIdx] * std::abs(d_axes[2].z[queuedIdx]);
415 // Intersection test:
416 // ((obbCenter - r) < aabbMax) &&
417 // ((obbCenter + r) > aabbMin)
418 if (d_obbCenterX[queuedIdx] - obbRadiusX < global_maxX[globalIdx] &&
419 d_obbCenterX[queuedIdx] + obbRadiusX > global_minX[globalIdx] &&
420 d_obbCenterY[queuedIdx] - obbRadiusY < global_maxY[globalIdx] &&
421 d_obbCenterY[queuedIdx] + obbRadiusY > global_minY[globalIdx] &&
422 d_obbCenterZ[queuedIdx] - obbRadiusZ < global_maxZ[globalIdx] &&
423 d_obbCenterZ[queuedIdx] + obbRadiusZ > global_minZ[globalIdx])
424 {
425 auto itemIdx = td.queuedIndices[queuedIdx];
426 td.hits->push(itemIdx);
