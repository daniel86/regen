100 constexpr void setEuler(float fPitch, float fYaw, float fRoll) {
101 const float fSinPitch(sinf(fPitch * 0.5f));
102 const float fCosPitch(cosf(fPitch * 0.5f));
103 const float fSinYaw(sinf(fYaw * 0.5f));
104 const float fCosYaw(cosf(fYaw * 0.5f));
105 const float fSinRoll(sinf(fRoll * 0.5f));
106 const float fCosRoll(cosf(fRoll * 0.5f));
107 const float fCosPitchCosYaw(fCosPitch * fCosYaw);
108 const float fSinPitchSinYaw(fSinPitch * fSinYaw);
109 x = fSinRoll * fCosPitchCosYaw - fCosRoll * fSinPitchSinYaw;
110 y = fCosRoll * fSinPitch * fCosYaw + fSinRoll * fCosPitch * fSinYaw;
111 z = fCosRoll * fCosPitch * fSinYaw - fSinRoll * fSinPitch * fCosYaw;
112 w = fCosRoll * fCosPitchCosYaw + fSinRoll * fSinPitchSinYaw;
113 }

134 constexpr void setLookRotation(const Vec3f& forward, const Vec3f& up = Vec3f::up()) {
135 Vec3f right = up.cross(forward);
136 right.normalize();
137 Vec3f newUp = forward.cross(right);
138
139 // Construct rotation matrix (column-major)
140 float m00 = right.x, m01 = right.y, m02 = right.z;
141 float m10 = newUp.x, m11 = newUp.y, m12 = newUp.z;
142 float m20 = forward.x, m21 = forward.y, m22 = forward.z;
143
144 // Convert rotation matrix to quaternion
145 float trace = m00 + m11 + m22;
146 if (trace > 0.0f) {
147 float s = 0.5f / sqrtf(trace + 1.0f);
148 w = 0.25f / s;
149 x = (m21 - m12) * s;
150 y = (m02 - m20) * s;
151 z = (m10 - m01) * s;
152 } else {
153 if (m00 > m11 && m00 > m22) {
154 float s = 2.0f * sqrtf(1.0f + m00 - m11 - m22);
155 w = (m21 - m12) / s;
156 x = 0.25f * s;
157 y = (m01 + m10) / s;
158 z = (m02 + m20) / s;
159 } else if (m11 > m22) {
160 float s = 2.0f * sqrtf(1.0f + m11 - m00 - m22);
161 w = (m02 - m20) / s;
162 x = (m01 + m10) / s;
163 y = 0.25f * s;
164 z = (m12 + m21) / s;
165 } else {
166 float s = 2.0f * sqrtf(1.0f + m22 - m00 - m11);
167 w = (m10 - m01) / s;
168 x = (m02 + m20) / s;
169 y = (m12 + m21) / s;
170 z = 0.25f * s;
171 }
172 }
173 normalize();
174 }

193 constexpr Mat4f calculateMatrix() const {
194 Mat4f mat4x4 = Mat4f::identity();
195 const float xx = x * x;
196 const float yy = y * y;
197 const float zz = z * z;
198 mat4x4.x[0] = -2.0f * (yy + zz) + 1.0f;
199 mat4x4.x[5] = -2.0f * (xx + zz) + 1.0f;
200 mat4x4.x[10] = -2.0f * (xx + yy) + 1.0f;
201
202 const float xy = x * y;
203 const float zw = z * w;
204 mat4x4.x[1] = 2.0f * (xy - zw);
205 mat4x4.x[4] = 2.0f * (xy + zw);
206
207 const float xz = x * z;
208 const float yw = y * w;
209 mat4x4.x[2] = 2.0f * (xz + yw);
210 mat4x4.x[8] = 2.0f * (xz - yw);
211
212 const float yz = y * z;
213 const float xw = x * w;
214 mat4x4.x[6] = 2.0f * (yz - xw);
215 mat4x4.x[9] = 2.0f * (yz + xw);
216 return mat4x4;
217 }

222 constexpr Vec3f calculateDirection() const {
223 // Rotate the forward vector (0, 0, 1) using this Quaternion
224 const float tx = 2.0f * x;
225 const float ty = 2.0f * y;
226 const float tz = 2.0f * z;
227 const float twx = tx * w;
228 const float twz = tz * w;
229 const float txx = tx * x;
230 const float txy = ty * x;
231 const float tyy = ty * y;
232 const float tyz = tz * y;
233 return { txy + twz, tyz - twx, 1.0f - (txx + tyy) };
234 }

243 constexpr void interpolate(
244 const Quaternion &pStart,
245 const Quaternion &pEnd,
246 float pFactor) {
247 // calc cosine theta
248 float cosom =
249 pStart.x * pEnd.x +
250 pStart.y * pEnd.y +
251 pStart.z * pEnd.z +
252 pStart.w * pEnd.w;
253
254 // adjust signs (if necessary)
255 Quaternion end = pEnd;
256 if (cosom < 0.0f) {
257 cosom = -cosom;
258 end.x = -end.x; // Reverse all signs
259 end.y = -end.y;
260 end.z = -end.z;
261 end.w = -end.w;
262 }
263
264 // Calculate coefficients
265 float sclp, sclq;
266 if ((1.0f - cosom) > 0.0001f) // 0.0001 -> some epsillon
267 {
268 // Standard case (slerp)
269 float omega = acosf(cosom); // extract theta from dot product's cos theta
270 float sinom = sinf(omega);
271 sclp = sinf((1.0f - pFactor) * omega) / sinom;
272 sclq = sinf(pFactor * omega) / sinom;
273 } else {
274 // Very close, do linear interp (because it's faster)
275 sclp = 1.0f - pFactor;
276 sclq = pFactor;
277 }
278
279 x = sclp * pStart.x + sclq * end.x;
280 y = sclp * pStart.y + sclq * end.y;
281 z = sclp * pStart.z + sclq * end.z;
282 w = sclp * pStart.w + sclq * end.w;
283 }

357 constexpr Vec3f rotate(const Vec3f &v) const {
358 Quaternion q2(0.0f, v.x, v.y, v.z);
359 Quaternion q = *this;
360 q.conjugate();
361 q = q * q2 * (*this);
362 return {q.x, q.y, q.z};
