69void BufferContainer::createTBO(const NamedShaderInput &namedInput) {
70 // create a TBO for the input

74 bufferUpdateHints_);
75 tbos_.push_back(tbo);

84 texState->set_mapTo(TextureState::MAP_TO_CUSTOM);
85 joinStates(texState);
86 // add shader defines for accessing the buffer
87 auto shaderType = glenum::glslDataType(

90 texState->shaderInclude(REGEN_STRING("regen.memory.buffer.tbo." << shaderType));
91 texState->shaderDefine(

93 REGEN_STRING("tboRead_" << shaderType << "(tbo_" << namedInput.name_ << ", int(regen_InstanceID))"));
94 texState->shaderDefine(

96 REGEN_STRING("tboRead_" << shaderType << "(tbo_" << namedInput.name_ << ", int(i))"));

103void BufferContainer::updateBuffer() {
104 if (isAllocated_) return;
105 isAllocated_ = true;
106
107 static auto maxUBOSize = static_cast<uint32_t>(glGetInteger(GL_MAX_UNIFORM_BLOCK_SIZE));
108 static auto maxTBOSize = static_cast<uint32_t>(glGetInteger(GL_MAX_TEXTURE_BUFFER_SIZE)) * 16;
109 unsigned int uboSize = 0u;
110 std::vector<NamedShaderInput> nextUBOInputs;
111 std::vector<NamedShaderInput> nextSSBOInputs;
112
113 for (auto &namedInput: namedInputs_) {
114 auto inputSize = namedInput.in_->inputSize();
115 if (namedInput.in_->serverAccessMode() == BUFFER_GPU_WRITE || inputSize > maxTBOSize) {
116 // create SSBO for large inputs, or if usage is WRITE (i.e. the buffer is written to from a shader).
117 nextSSBOInputs.push_back(namedInput);
118 }
119 else if (inputSize > maxUBOSize) {
120 createTBO(namedInput);
121 if (namedInput.in_->numInstances() > 1) {
122 set_numInstances(namedInput.in_->numInstances());
123 }
124 }
125 else {
126 if (uboSize + inputSize > maxUBOSize) {
127 createUBO(nextUBOInputs);
128 nextUBOInputs.clear();
129 uboSize = 0;
130 }
131 nextUBOInputs.push_back(namedInput);
132 uboSize += inputSize;
133 }
134 }
135 if (!nextUBOInputs.empty()) {
136 createUBO(nextUBOInputs);
137 }
138 if (!nextSSBOInputs.empty()) {
139 createSSBO(nextSSBOInputs);
140 }
141}
