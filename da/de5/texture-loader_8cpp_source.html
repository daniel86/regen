66ref_ptr<Texture> textures::load(std::string_view file, const TextureConfig &texCfg) {
67 ImageDataArray imgData = TextureLoaderRegistry::load(file, texCfg);
68 if (imgData.empty()) {
69 throw Error("No image data loaded");
70 }
71 auto &firstImage = imgData[0];
72 const uint32_t depth = firstImage->depth;
73 uint32_t numImages = imgData.size();
74
75 ref_ptr<Texture> tex;
76 if (depth > 1) {
77 ref_ptr<Texture3D> tex3D = ref_ptr<Texture3D>::alloc();
78 tex3D->set_depth(depth);
79 tex = tex3D;
80 numImages = 1;
81 } else if (numImages > 1) {
82 ref_ptr<Texture2DArray> tex2DArray = ref_ptr<Texture2DArray>::alloc();
83 tex2DArray->set_depth(numImages);
84 tex = tex2DArray;
85 } else {
86 tex = ref_ptr<Texture2D>::alloc();
87 }
88 tex->set_textureFile(file);
89 tex->set_rectangleSize(firstImage->width, firstImage->height);
90 tex->set_pixelType(firstImage->pixelType);
91 tex->set_format(regenImageFormat(firstImage->format));
92 if (texCfg.forcedInternalFormat == GL_NONE) {
93 tex->set_internalFormat(glenum::textureInternalFormat(tex->format()));
94 } else if (texCfg.forcedInternalFormat == GL_BGRA) {
95 tex->set_internalFormat(GL_RGBA8);
96 } else if (texCfg.forcedInternalFormat == GL_BGR) {
97 tex->set_internalFormat(GL_RGB8);
98 } else {
99 tex->set_internalFormat(texCfg.forcedInternalFormat);
100 }
101 tex->allocTexture();
102 tex->set_wrapping(TextureWrapping::create(GL_REPEAT));
103 tex->set_filter(TextureFilter::create(GL_LINEAR));
104 if (numImages > 1) {
105 auto *tex3d = dynamic_cast<Texture3D *>(tex.get());
106 for (uint32_t i = 0u; i < numImages; ++i) {
107 tex3d->updateSubImage(i, imgData[i]->pixels);
108 }
109 } else {
110 tex->updateImage(firstImage->pixels);
111 }
112 if (texCfg.useMipmaps) {
113 tex->updateMipmaps();
114 }
115 unsetData(imgData, tex, texCfg.keepData);
116 GL_ERROR_LOG();
117

163void textures::reload(const ref_ptr<Texture> &tex, std::string_view file) {
164 bool keepData = false;
165 unsigned int textureDepth = 1;
166 if (tex->textureBind().target_ == GL_TEXTURE_3D) {
167 auto tex3D = dynamic_cast<Texture3D *>(tex.get());
168 if (tex3D) {
169 textureDepth = tex3D->depth();
170 }
171 }
172 else if (tex->textureBind().target_ == GL_TEXTURE_2D_ARRAY) {
173 auto tex2DArray = dynamic_cast<Texture2DArray *>(tex.get());
174 if (tex2DArray) {
175 textureDepth = tex2DArray->depth();
176 }
177 }
178 else if (tex->textureBind().target_ == GL_TEXTURE_CUBE_MAP) {
179 textureDepth = 6;
180 }
181
182 TextureConfig texCfg;
183 texCfg.forcedSize = Vec3ui(tex->width(), tex->height(), textureDepth);
184 texCfg.forcedFormat = tex->format();
185 texCfg.forcedInternalFormat = tex->internalFormat();
186 texCfg.useMipmaps = (tex->getNumMipmaps() > 1);
187 texCfg.keepData = keepData;
188 ImageDataArray imgData = TextureLoaderRegistry::load(file, texCfg);
189 if (imgData.empty()) {
190 throw Error("No image data loaded");
191 }
192 auto &firstImage = imgData[0];
193
194 tex->set_textureFile(file);
195 tex->set_rectangleSize(firstImage->width, firstImage->height);
196 tex->set_pixelType(firstImage->pixelType);
197 tex->allocTexture();
198 if (imgData.size() > 1) {
199 auto *tex3d = dynamic_cast<Texture3D *>(tex.get());
200 for (uint32_t i = 0u; i < imgData.size(); ++i) {
201 tex3d->updateSubImage(i, imgData[i]->pixels);
202 }
203 }
204 else if (!imgData.empty()) {
205 tex->updateImage(imgData[0]->pixels);
206 }
207 if (tex->getNumMipmaps() > 1) {
208 tex->updateMipmaps();
209 }
210 unsetData(imgData, tex, keepData);
211}

213ref_ptr<Texture2DArray> textures::loadArray(
214 const std::string &textureDirectory,
215 const std::string &textureNamePattern,
216 const TextureConfig &texCfg) {
217 std::filesystem::path dir(textureDirectory);
218 boost::regex pattern(textureNamePattern);
219
220 std::vector<TextureDescription> accumulator;
221 std::filesystem::directory_iterator it(dir), eod;
222 BOOST_FOREACH(const std::filesystem::path &filePath, std::make_pair(it, eod)) {
223 std::string name = filePath.filename().string();
224 if (boost::regex_match(name, pattern)) {
225 accumulator.emplace_back(filePath.string());
226 }
227 }
228 std::sort(accumulator.begin(), accumulator.end());
229
230 return loadArray(accumulator, texCfg);
231}

286ref_ptr<TextureCube> textures::loadCube(
287 std::string_view file,
288 bool flipBackFace,
289 const TextureConfig &texCfg) {
290 ImageDataArray imgData = TextureLoaderRegistry::load(file, texCfg);
291 if (imgData.empty()) {
292 throw Error("No image data loaded");
293 }
294 auto &firstImage = imgData[0];
295
296 int faceWidth, faceHeight;
297 int faces[12];
298 // guess layout
299 if (firstImage->width > firstImage->height) {
300 faceWidth = firstImage->width / 4;
301 faceHeight = firstImage->height / 3;
302 int faces_[12] = {
303 -1, TextureCube::TOP, -1, -1,
304 TextureCube::LEFT, TextureCube::BACK, TextureCube::RIGHT, TextureCube::FRONT,
305 -1, TextureCube::BOTTOM, -1, -1
306 };
307 for (int i = 0; i < 12; ++i) faces[i] = faces_[i];
308 } else {
309 faceWidth = firstImage->width / 3;
310 faceHeight = firstImage->height / 4;
311 int faces_[12] = {
312 -1, TextureCube::TOP, -1,
313 TextureCube::LEFT, TextureCube::BACK, TextureCube::RIGHT,
314 -1, TextureCube::BOTTOM, -1,
315 -1, TextureCube::FRONT, -1
316 };
317 for (int i = 0; i < 12; ++i) faces[i] = faces_[i];
318 }
319 const auto numRows = firstImage->height / faceHeight;
320 const auto numCols = firstImage->width / faceWidth;
321 const auto bpp = firstImage->bpp;
322 const auto faceBytes = bpp * faceWidth * faceHeight;
323 const auto rowBytes = faceBytes * numCols;
324
325 auto tex = ref_ptr<TextureCube>::alloc();
326 tex->set_textureFile(file);
327 tex->set_rectangleSize(faceWidth, faceHeight);
328 tex->set_pixelType(firstImage->pixelType);
329 tex->set_format(regenImageFormat(firstImage->format));
330 tex->set_internalFormat(texCfg.forcedInternalFormat == GL_NONE ?
331 glenum::textureInternalFormat(tex->format()) :
332 texCfg.forcedInternalFormat);
333 tex->allocTexture();
334 tex->set_filter(TextureFilter::create(GL_LINEAR));
335
336 std::vector<byte> tmpFace(faceBytes);
337 int index = 0;
338 for (int row = 0; row < numRows; ++row) {
339 int flippedRow = numRows - 1 - row; // account for bottom-left origin
340 byte *rowPtr = firstImage->pixels + flippedRow * rowBytes;
341
342 byte *colData = rowPtr;
343 for (int col = 0; col < numCols; ++col) {
344 int mappedFace = faces[index];
345 if (mappedFace != -1) {
346 auto nextFace = (TextureCube::CubeSide) mappedFace;
347
348 if (nextFace == TextureCube::TOP || nextFace == TextureCube::BOTTOM) {
349 glPixelStorei(GL_UNPACK_ROW_LENGTH, faceWidth * numCols);
350 tex->updateSubImage(nextFace, colData);
351 }
352 else if (flipBackFace && nextFace == TextureCube::FRONT) {
353 glPixelStorei(GL_UNPACK_ROW_LENGTH, faceWidth * numCols);
354 tex->updateSubImage(nextFace, colData);
355 } else {
356 glPixelStorei(GL_UNPACK_ROW_LENGTH, 0);
357 auto *dst = tmpFace.data();
358 for (int y = faceHeight - 1; y >= 0; --y) {
359 auto *rowData = colData + y * faceWidth * bpp * numCols;
360 for (int x = faceWidth - 1; x >= 0; --x) {
361 byte *pixelData = rowData + x * bpp;
362 memcpy(dst, pixelData, bpp);
363 dst += bpp;
364 }
365 }
366 tex->updateSubImage(nextFace, tmpFace.data());
367 }
368 }
369 index += 1;
370 colData += bpp * faceWidth;
371 }
372 }
373
374 if (texCfg.useMipmaps) {
375 tex->updateMipmaps();
376 }
377 glPixelStorei(GL_UNPACK_ROW_LENGTH, 0);
378
379 return tex;
380}
