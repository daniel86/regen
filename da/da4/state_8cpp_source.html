133void State::setInput(const ref_ptr<ShaderInput> &in,
134 const std::string &name, const std::string &memberSuffix) {
135 const std::string &inputName = (name.empty() ? in->name() : name);
136
137 if (in->isVertexAttribute() && in->numVertices() != static_cast<uint32_t>(shared_->numVertices_)) {
138 shared_->numVertices_ = static_cast<int>(in->numVertices());
139 }
140 if (in->numInstances() > 1) {
141 shared_->numInstances_ = static_cast<int>(in->numInstances());
142 }
143 // check for instances of attributes within UBO
144 if (in->isStagedBuffer()) {
145 auto *bo = dynamic_cast<StagedBuffer *>(in.get());
146 for (auto &namedInput: bo->stagedInputs()) {
147 if (namedInput.in_->isVertexAttribute() &&
148 namedInput.in_->numVertices() > static_cast<uint32_t>(shared_->numVertices_)) {
149 shared_->numVertices_ = static_cast<int>(namedInput.in_->numVertices());
150 }
151 if (namedInput.in_->numInstances() > 1) {
152 shared_->numInstances_ = static_cast<int>(namedInput.in_->numInstances());
153 }
154 }
155 }
156
157 if (inputMap_.contains(inputName)) {
158 removeInput(inputName);
159 } else {
160 // insert into map of known attributes
161 inputMap_.insert(inputName);
162 }
163
164 inputs_.emplace_back(NamedShaderInput{
165 in, inputName, "", memberSuffix});
166}

189std::optional<StateInput> State::findShaderInput(const std::string &name) {
190 StateInput ret;
191
192 auto &l = inputs();
193 for (const auto &inNamed: l) {
194 if (name == inNamed.name_ || name == inNamed.in_->name()) {
195 ret.in = inNamed.in_;
196 ret.bo = {};
197 return ret;
198 }
199 if (inNamed.in_->isStagedBuffer()) {
200 auto bo = ref_ptr<StagedBuffer>::dynamicCast(inNamed.in_);
201 for (auto &blockUniform: bo->stagedInputs()) {
202 if (name == blockUniform.name_ || name == blockUniform.in_->name()) {
203 ret.bo = bo;
204 ret.in = blockUniform.in_;
205 return ret;
206 }
207 }
208 }
209 }
210
211 auto local = joined_;
212 for (auto &joined: *local.get()) {
213 auto joinedRet = joined->findShaderInput(name);
214 if (joinedRet.has_value()) {
215 return joinedRet.value();
216 }
217 auto *bufferContainer = dynamic_cast<BufferContainer *>(joined.get());
218 if (bufferContainer != nullptr) {
219 for (auto &stagedBuffer: bufferContainer->stagedBuffers()) {
220 if (boost::starts_with(stagedBuffer->name(), name)) {
221 ret.bo = {};
222 ret.in = stagedBuffer;
223 return ret;
224 }
225
226 for (auto &blockUniform: stagedBuffer->stagedInputs()) {
227 if (name == blockUniform.name_ || name == blockUniform.in_->name()) {
228 ret.bo = stagedBuffer;
229 ret.in = blockUniform.in_;
230 return ret;
231 }
232 }
233 }
234 }
235 }
236
237 return std::nullopt;
238}
