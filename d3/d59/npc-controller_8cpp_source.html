77static ref_ptr<WayPoint> pickArrivalWP(
78 const ref_ptr<Place> &target, const Vec2f &fromPos) {
79 const float desiredDistanceToCenter = target->radius() * 0.75f;
80 Vec2f pos = target->position2D();
81 // Direction from place center to source position
82 Vec2f dir = fromPos - pos;
83 dir.y = 0.0f;
84 float l = dir.length();
85 if (l < desiredDistanceToCenter) {
86 // The source position is already within the desired distance to the center.
87 return {};
88 } else {
89 dir /= l;
90 }
91 // move from center towards source position
92 pos += dir * desiredDistanceToCenter;
93
94 auto arrivalWP = ref_ptr<WayPoint>::alloc(target->name(), target->position3D());
95 arrivalWP->setRadius(target->radius() * 0.5f);
96 return arrivalWP;
97}

99uint32_t NonPlayerCharacterController::findClosestWP(const std::vector<ref_ptr<WayPoint> > &wps) const {
100 if (wps.empty()) return 0;
101 float bestDist = std::numeric_limits<float>::max();
102 uint32_t bestIdx = 0;
103 for (uint32_t i = 0; i < wps.size(); i++) {
104 auto &wp = wps[i];
105 float dist = (wp->position2D() - Vec2f(currentPos_.x, currentPos_.z)).lengthSquared();
106 if (dist < bestDist) {
107 bestDist = dist;
108 bestIdx = i;
109 }
110 }
111 return bestIdx;

114bool NonPlayerCharacterController::startNavigate(bool loopPath, bool advancePath) {
115 auto &kb = knowledgeBase_;
116 if (!currentPath_.empty()) {
117 // Reached target point
118 if (advancePath) {
119 currentPathIndex_++;
120 }
121 if (loopPath && currentPathIndex_ >= currentPath_.size()) {
122 currentPathIndex_ = 0;
123 }
124 if (currentPathIndex_ < currentPath_.size()) {
125 // Move to next point on path
126#ifdef NAV_CTRL_DEBUG_WPS
127 REGEN_INFO("[" << tfIdx_ << "] Navigating to waypoint " <<
128 currentPath_[currentPathIndex_]->name() <<
129 " (" << (currentPathIndex_ + 1) << "/" << currentPath_.size() << ")");
130#endif
131 return updatePathNPC();
132 } else if (advancePath) {
133 currentPathIndex_ = 0;
134 currentPath_.clear();
135 return false;
136 }
137 }
138 auto &currentPlace = kb.currentPlace();
139 auto &targetPlace = kb.targetPlace();
140 if (currentPlace.get() && currentPlace.get() == targetPlace.get()) {
141 // Already at target place.
142 // For now we don't do local navigation planning within a place,
143 // but just go directly to the target.
144 return updatePathNPC();
145 }
146 if (!targetPlace) {
147 if (!kb.hasNavigationTarget()) {
148 REGEN_WARN("No navigation target set for NPC.");
149 return false;
150 } else {
151 return updatePathNPC();
152 }
153 }
154 auto currentPos2D = Vec2f(currentPos_.x, currentPos_.z);
155 // Compute a path to the target.
156 currentPath_ = pathPlanner_->findPath(currentPos2D, targetPlace->position2D());
157 // Add a waypoint within the target place which is close to the last waypoint.
158 if (currentPath_.empty()) {
159 if (currentPlace.get() != targetPlace.get()) {
160 // We are not at the target place yet, so add the arrival waypoint.
161 auto arrivalWP = pickArrivalWP(targetPlace, currentPos2D);
162 if (arrivalWP.get()) currentPath_ = {arrivalWP};
163 }
164 } else {
165 auto &lastWP = currentPath_.back();
166 auto arrivalWP = pickArrivalWP(targetPlace, lastWP->position2D());
167 if (arrivalWP.get()) currentPath_.push_back(arrivalWP);
168 }
169 currentPathIndex_ = 0;
170#ifdef NAV_CTRL_DEBUG_WPS
171 REGEN_INFO("[" << tfIdx_ << "] Planned path to place " << targetPlace->name()
172 << " num WPs: " << currentPath_.size());
173#endif
174 return updatePathNPC();
175}

177Vec2f NonPlayerCharacterController::pickTravelPosition(const WorldObject &wp) const {
178 // Base target position (waypoint center)
179 Vec2f baseTarget = wp.position2D();
180 if (wp.radius() < 0.1f) {
181 return baseTarget;
182 } else {
183 // Pick an angle roughly facing *towards* the waypoint from the NPC's current position
184 Vec2f toWP = baseTarget - Vec2f(currentPos_.x, currentPos_.z);
185 float angle = atan2(toWP.y, toWP.x) + M_PI;
186 // Allow a small angular variation (+-45°)
187 angle += (math::random<float>() - 0.5f) * (M_PI / 2.0f);
188 const float distanceToWP = wp.radius() * (0.7f * math::random<float>() + 0.1f);
189 // Offset final target
190 return baseTarget + Vec2f(cos(angle), sin(angle)) * distanceToWP;
191 }
192}

204bool NonPlayerCharacterController::updatePathNPC() {
205 Vec2f source(currentPos_.x, currentPos_.z);
206
207 if (currentPath_.size() < 2 || currentPathIndex_ + 1 >= currentPath_.size()) {
208 // Navigation actions should have a navigation target set.
209 auto &navTarget = knowledgeBase_.hasNavigationTarget()
210 ? knowledgeBase_.navigationTarget()
211 : knowledgeBase_.interactionTarget();
212
213 if (!currentPath_.empty()) {
214 auto &lastWP = currentPath_.back();
215 startApproaching(source, pickTravelPosition(*lastWP.get()));
216 } else if (navTarget.object->objectType() == ObjectType::PLACE) {
217 // Travel to place, prefer to use pre-computed waypoint but fallback to
218 // going to a computed arrival position within the place.
219 startApproaching(source, pickTravelPosition(*navTarget.object.get()));
220 } else {
221 // Travel to object (with affordance slot if any)
222 Vec2f navPos = pickTargetPosition(navTarget);
223#ifdef NAV_CTRL_DEBUG_WPS
224 REGEN_INFO("[" << tfIdx_ << "] Navigating to object " << navTarget.object->name() << ".");
225#endif
226
227 // NPC shall look from affordance slot to center of attention
228 // (i.e. affordance target, or owner of affordance)
229 Vec2f dir = (navTarget.object->position2D() - navPos);
230 float dist = dir.length();
231 if (dist < affordanceSlotRadius_) {
232 dir = Vec2f(cos(currentDir_.x + baseOrientation_), sin(currentDir_.x + baseOrientation_));
233 } else {
234 dir /= dist;
235 }
236
237 float o = atan2(dir.y, dir.x) - baseOrientation_;
238 startApproaching(source, navPos, Vec3f(o, 0.0f, 0.0f));
239 }
240 } else {
241 auto &nextWP = currentPath_[currentPathIndex_];
242 startApproaching(source, pickTravelPosition(*nextWP.get()));
243 }
244 return true;
245}

247void NonPlayerCharacterController::updateNavigationBehavior() {
248 auto &kb = knowledgeBase_;
249
250 bool hasNavigatingAction = kb.isCurrentAction(ActionType::NAVIGATING);
251 bool hasPatrollingAction = kb.isCurrentAction(ActionType::PATROLLING);
252 bool hasStrollingAction = kb.isCurrentAction(ActionType::STROLLING);
253 bool hasFlockingAction = kb.isCurrentAction(ActionType::FLOCKING);
254 bool hasApproachLoop = (hasPatrollingAction || hasStrollingAction);
255 bool hasApproachAction = (hasNavigatingAction || hasApproachLoop);
256 bool hasAnyNavAction = hasApproachAction || hasFlockingAction;
257
258 if (!hasAnyNavAction) {
259 // No navigation action active, nothing to do.
260 currentPath_.clear();
261 stopNavigation();
262 return;
263 }
264
265 bool hasApproachChanged = (
266 hasNavigatingAction != hasNavigatingAction_ ||
267 hasPatrollingAction != hasPatrollingAction_ ||
268 hasStrollingAction != hasStrollingAction_
269 );
270 if (hasApproachChanged && hasApproachAction) {
271 // Make sure to reset path in case the navigation mode changed.
272 // e.g. when NAVIGATE is still active when PATROL is selected,
273 // we might end up with continuing with NAVIGATE path is we don't clear
274 // here.
275 currentPath_.clear();
276 // Also unset the navigation target in case of looped navigation
277 if (hasApproachLoop) {
278 kb.unsetNavigationTarget();
279 }
280 }
281 hasNavigatingAction_ = hasNavigatingAction;
282 hasPatrollingAction_ = hasPatrollingAction;
283 hasStrollingAction_ = hasStrollingAction;
284
285 if (kb.isCurrentAction(ActionType::FLOCKING)) {
286 auto &group = kb.currentGroup();
287 if (!group) {
288 REGEN_WARN("No group for flocking action.");
289 } else if (!isNavigationFlocking()) {
290 startFlocking(group);
291 }
292 } else if (isNavigationFlocking()) {
293 stopFlocking();
294 }
295
296 if (!hasApproachAction || isNavigationApproaching()) {
297 } else if (hasApproachLoop && currentPath_.empty() && kb.currentPlace().get()) {
298 // We are patrolling or strolling, but have no path yet, so start one.
299 auto &currentPlace = kb.currentPlace();
300 auto loopPath = currentPlace->getPathWays(
301 hasPatrollingAction ? PathwayType::PATROL : PathwayType::STROLL);
302 // Pick a random path
303 if (!loopPath.empty()) {
304 currentPath_ = loopPath[math::randomInt() % loopPath.size()];
305 currentPathIndex_ = findClosestWP(currentPath_);
306 startNavigate(true, false);
307 } else {
308 REGEN_WARN("No pathways of type " << (hasPatrollingAction ? "PATROL" : "STROLL") <<
309 " at place " << currentPlace->name());
310 setIdle();
311 }
312 } else {
313 // The TF animation has no active frame, advance to next waypoint if any.
314 if (!currentPath_.empty() && (hasApproachLoop || currentPathIndex_ < currentPath_.size())) {
315 // Set next TF waypoint for movement
316 startNavigate(hasApproachLoop, true);
317 } else {
318 // No waypoint remaining, but navigation action is still active,
319 // so it seems we need to re-plan.
320 currentPath_.clear();
321 startNavigate(false, false);
322 }
323 }

326void NonPlayerCharacterController::updateKnowledgeBase(float dt_s) {
327 auto &kb = knowledgeBase_;
328 kb.advanceTime(dt_s);
329
330 // Re-compute distance to patient and target
331 if (kb.hasInteractionTarget()) {
332 // Update distance to patient
333 auto &target = kb.interactionTarget();
334 Vec2f delta;
335 if (!target.affordance || target.affordanceSlot < 0) {
336 // No affordance, so just go to the center of the object
337 delta = target.object->position2D() - Vec2f(currentPos_.x, currentPos_.z);
338 } else {
339 // Go to the affordance slot position
340 auto &slotPos = target.affordance->slotPosition(target.affordanceSlot);
341 delta = Vec2f(slotPos.x, slotPos.z) - Vec2f(currentPos_.x, currentPos_.z);
342 }
343 kb.setDistanceToPatient(delta.length());
344 // Also let motion controller know the shape of the patient, such that it can
345 // ignore collisions with it.
346 if (patientShape_.get() != target.object->shape().get()) {
347 patientShape_ = target.object->shape();
348 }
349 } else if (patientShape_.get()) {
350 // No patient anymore
351 patientShape_ = {};
352 kb.setDistanceToPatient(std::numeric_limits<float>::max());
353 }
354 if (kb.hasNavigationTarget()) {
355 auto &navTarget = kb.navigationTarget();
356 if (navTarget.object.get() == kb.interactionTarget().object.get() &&
357 navTarget.affordance.get() == kb.interactionTarget().affordance.get() &&
358 navTarget.affordanceSlot == kb.interactionTarget().affordanceSlot) {
359 // Navigation target is the same as the patient, so reuse distance to patient.
360 kb.setDistanceToTarget(kb.distanceToPatient());
361 } else {
362 auto delta = navTarget.object->position2D() - Vec2f(currentPos_.x, currentPos_.z);
363 kb.setDistanceToTarget(delta.length());
364 }
365 // Also make the distance accessible to motion controller, as it does not have access to the blackboard.
366 distanceToTarget_ = kb.distanceToTarget();
367 }

378void NonPlayerCharacterController::updateController(double dt) {
379 auto &kb = knowledgeBase_;
380 float dt_s = dt / 1000.0f;
381 lastDT_ = dt_s;
382 timeSinceLastDecision_s_ += dt_s;
383 timeSinceLastPerception_s_ += dt_s;
384
385 if (perceptionSystem_.get() && timeSinceLastPerception_s_ > perceptionInterval_s_) {
386 // Insert new percepts into the knowledge base, and update existing ones.
387 perceptionSystem_->update(knowledgeBase_, timeSinceLastPerception_s_);
388 timeSinceLastPerception_s_ = 0.0f;
389 }
390 // Do some controller specific updates to the knowledge base.
391 updateKnowledgeBase(dt_s);
392
393 if (timeSinceLastDecision_s_ > decisionInterval_s_) {
394 // Reset current action state in the blackboard.
395 kb.unsetCurrentActions();
396 // Update blackboard and make some actions current.
397 auto behaviorStatus = behaviorTree_->tick(kb, timeSinceLastDecision_s_);
398 if (behaviorStatus == BehaviorStatus::FAILURE) {
399 REGEN_WARN("[" << tfIdx_ << "] Behavior tree returned FAILURE " <<
400 " linger time: " << kb.lingerTime() <<
401 " activity time: " << kb.activityTime());
402 //setIdle();
403 }
404 timeSinceLastDecision_s_ = 0.0f;
405
406 bool hasNavigatingAction = kb.isCurrentAction(ActionType::NAVIGATING);
407 bool hasPatrollingAction = kb.isCurrentAction(ActionType::PATROLLING);
408 bool hasStrollingAction = kb.isCurrentAction(ActionType::STROLLING);
409 bool hasApproachAction = (hasNavigatingAction || hasPatrollingAction || hasStrollingAction);
410 if (!hasApproachAction && !isStandingStill()) {
411 kb.setCurrentAction(ActionType::WALKING);
412 }
413 }
414
415 // Update the state of current bone animations based on current actions, intends etc.
416 boneController_->updateBoneController(dt_s, knowledgeBase_);
417 // Update path etc. for navigation action.
418 updateNavigationBehavior();
419
420 // Set flags for current bone animations
421 if (boneController_->isCurrentBoneAnimation(MotionType::RUN)) {
422 isWalking_ = false;
423 isLastAnimationMovement_ = true;
424 } else if (boneController_->isCurrentBoneAnimation(MotionType::WALK)) {
425 isWalking_ = true;
426 isLastAnimationMovement_ = true;
427 } else {
428 isLastAnimationMovement_ = false;
429 }
430}
