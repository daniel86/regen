127void SpatialIndex::createIndexShape(IndexCamera &ic, const ref_ptr<BoundingShape> &shape) {
128 auto needle = ic.nameToShape_.find(shape->name());
129 if (needle != ic.nameToShape_.end()) {
130 // Remember bounding-shape to indexed-shape mapping
131 // Note: important to insert this exactly in item-order
132 ic.itemToIndexedShape_.push_back(needle->second->indexedShapeIdx_);
133 return;
134 }
135 const uint32_t numLayer = ic.cullCamera->numLayer();
136 const uint32_t numInstances = shape->numInstances();
137 const uint32_t numIndices = numInstances * numLayer;
138
139 auto is = ref_ptr<IndexedShape>::alloc(ic.cullCamera, ic.lodCamera, ic.lodShift, shape);
140 const uint32_t numLOD = std::max(1u, is->numLODs());
141 is->instanceIDs_ = ref_ptr<ShaderInput1ui>::alloc("instanceIDs", 1);
142 is->instanceIDs_->setInstanceData(numIndices, 1, nullptr);
143 is->drawBinCount_ = ref_ptr<ShaderInput1ui>::alloc("instanceCounts", numLayer * numLOD);
144 is->drawBinCount_->setInstanceData(1, 1, nullptr);
145 is->drawBinBase_ = ref_ptr<ShaderInput1ui>::alloc("baseInstances", numLayer * numLOD);
146 is->drawBinBase_->setInstanceData(1, 1, nullptr);
147
148 auto mapped_ids = is->instanceIDs_->mapClientData<uint32_t>(BUFFER_GPU_WRITE);
149 auto mapped_count = is->drawBinCount_->mapClientData<uint32_t>(BUFFER_GPU_WRITE);
150 auto mapped_base = is->drawBinBase_->mapClientData<uint32_t>(BUFFER_GPU_WRITE);
151 for (unsigned int instanceIdx = 0; instanceIdx < numInstances; ++instanceIdx) {
152 // write instance data for every layer
153 for (unsigned int layerIdx = 0; layerIdx < numLayer; ++layerIdx) {
154 mapped_ids.w[instanceIdx + layerIdx * numInstances] = instanceIdx;
155 }
156 }
157 for (unsigned int layerIdx = 0; layerIdx < numLayer; ++layerIdx) {
158 for (unsigned int lodLevel = 0; lodLevel < numLOD; ++lodLevel) {
159 uint32_t drawBin = lodLevel * numLayer + layerIdx;
160 if (lodLevel==0) {
161 mapped_count.w[drawBin] = numInstances;
162 mapped_base.w[drawBin] = layerIdx * numInstances;
163 } else {
164 mapped_count.w[drawBin] = 0;
165 mapped_base.w[drawBin] = (layerIdx+1) * numInstances;
166 }
167 }
168 }
169 mapped_base.unmap();
170 mapped_count.unmap();
171 mapped_ids.unmap();
172
173 is->indexedShapeIdx_ = static_cast<uint16_t>(ic.indexShapes_.size());
174 ic.nameToShape_[shape->name()] = is;
175 ic.indexShapes_.push_back(is.get());
176 ic.itemToIndexedShape_.push_back(is->indexedShapeIdx_);
177 // mark camera as dirty as the camera buffers must be resized.
178 // We do this lazy to avoid multiple resizes when adding many shapes.
179 ic.isDirty = true;
180}

238void SpatialIndex::Private::gatherVisibleHitSOA(IndexCamera &ic, VisibleHitSOA &hitSoA, uint32_t layerIdx) {
239 const auto &itemShapes = ic.index->itemBoundingShapes_;
240 const std::vector<uint32_t> &itemToGlobalID = ic.itemToGlobalID_[layerIdx];
241 const HitBuffer &hits = *hitSoA.hits;
242
243 for (uint32_t hitIdx=0; hitIdx < hits.count; ++hitIdx) {
244 // Gather data for this shape
245 const uint32_t itemIdx = hits.data[hitIdx];
246 const uint32_t shapeIdx = ic.itemToIndexedShape_[itemIdx];
247 const Vec3f &shapeOrigin = itemShapes[itemIdx]->tfOrigin();
248 const Vec3f &lodThresholds = ic.indexShapes_[shapeIdx]->lodThresholds();
249 // Store in SOA
250 hitSoA.ox[hitIdx] = shapeOrigin.x;
251 hitSoA.oy[hitIdx] = shapeOrigin.y;;
252 hitSoA.oz[hitIdx] = shapeOrigin.z;
253 hitSoA.t0[hitIdx] = lodThresholds.x;
254 hitSoA.t1[hitIdx] = lodThresholds.y;
255 hitSoA.t2[hitIdx] = lodThresholds.z;
256 hitSoA.globalID[hitIdx] = itemToGlobalID[itemIdx];
257 hitSoA.shapeIdx[hitIdx] = shapeIdx;

261template<SpatialIndex::DistanceKeySize DistanceType>
262static uint32_t computeDistanceKey(float lodDistance, const uint32_t flip) {
263 //const uint32_t xf = conversion::floatBitsToUint(lodDistance);
264 const uint32_t xf = std::bit_cast<uint32_t>(lodDistance);
265 if constexpr (DistanceType == SpatialIndex::DISTANCE_KEY_16) {
266 const uint16_t q = ((xf>>16)&0x8000) // sign bit
267 | ((((xf&0x7f800000)-0x38000000)>>13)&0x7c00) // exponent bits
268 | ((xf>>13)&0x03ff); // mantissa bits
269 return (q ^ (flip & 0xFFFF));
270 } else if constexpr (DistanceType == SpatialIndex::DISTANCE_KEY_24) {
271 const uint32_t q = ((xf >> 8) & 0x800000) // sign bit
272 | ((((xf & 0x7f800000) - 0x3f800000) >> 7) & 0x7f0000) // exponent bits
273 | ((xf >> 8) & 0x00ffff); // mantissa bits
274 return (q ^ (flip & 0xFFFFFF));
275 } else {
276 return (xf ^ flip);
277 }
278}

288template <typename KeyType, SpatialIndex::DistanceKeySize DistanceType>
289void SpatialIndex::Private::pushVisibleShapes(IndexCamera &ic, uint32_t layerIdx, HitBuffer &hitBuffer) {
290 static constexpr bool UseSoA = SPATIAL_INDEX_USE_SIMD_BINNING;
291 static constexpr int SortKeyBits = (sizeof(KeyType) * 8);
292 static constexpr uint8_t bitOffset_layer = static_cast<uint8_t>(DistanceType);
293 static constexpr uint32_t lodLevelMask = (1u << LOD_LEVEL_BITS) - 1u;
294
295 const uint32_t numLayer = ic.cullCamera->numLayer();
296 const uint32_t numHits = hitBuffer.count;
297 const uint32_t flip = -(ic.sortMode == BACK_TO_FRONT);
298 const uint8_t bitOffset_lodLevel = bitOffset_layer + ic.layerBits;
299 const uint8_t bitOffset_shape = bitOffset_lodLevel + LOD_LEVEL_BITS;
300 const Vec3f &camPos = getCameraPosition(ic, layerIdx);
301
302 // Compute sort keys, a key is composed of:
303 // [ shapeIdx | lodLevel | layerIdx | distance ]
304 // where the number of bits for each field is determined by:
305 // shapeIdx: determined by number of shapes in the camera (shapeBits)
306 // lodLevel: fixed number of bits (LOD_LEVEL_BITS)
307 // layerIdx: determined by number of layers in the camera (layerBits)
308 // distance: user defined either 16, 24 or 32 bits
309 // Note: It seems a bit redundant having lodLevel + distance as distance establishes
310 // LOD-ordering. However, we need to creat LOD-major layout across layers,
311 // as all meshes of one LOD-level are drawn in one draw call.
312 KeyType *sortKeys;
313 if constexpr (SortKeyBits == 32) {
314 sortKeys = ic.sortKeys32_.data();
315 } else {
316 sortKeys = ic.sortKeys64_.data();
317 }
318
319 if constexpr (UseSoA) {
320 thread_local VisibleHitSOA hitSoA;
321 uint32_t hitIdx = 0;
322 hitSoA.hits = &hitBuffer;
323 hitSoA.resize(hitBuffer.count);
324
325 const float *d_ox = hitSoA.ox.data();
326 const float *d_oy = hitSoA.oy.data();
327 const float *d_oz = hitSoA.oz.data();
328 const float *d_t0 = hitSoA.t0.data();
329 const float *d_t1 = hitSoA.t1.data();
330 const float *d_t2 = hitSoA.t2.data();
331
332 float *d_sortDistance = hitSoA.sortDistance.data();
333 uint32_t *d_lodLevel = hitSoA.lodLevel.data();
334 uint32_t *d_drawBin = hitSoA.drawBin.data();
335
336 // First pass: gather all data into SOA arrays
337 gatherVisibleHitSOA(ic, hitSoA, layerIdx);
338
339 // Second pass: process all hits, write output: lodDistance & drawBin
340 {
341 const BatchOf_float cx = BatchOf_float::fromScalar(camPos.x);
342 const BatchOf_float cy = BatchOf_float::fromScalar(camPos.y);
343 const BatchOf_float cz = BatchOf_float::fromScalar(camPos.z);
344 const BatchOf_int32 nl = BatchOf_int32::fromScalar(static_cast<int32_t>(numLayer));
345 const BatchOf_int32 layer = BatchOf_int32::fromScalar(static_cast<int32_t>(layerIdx));
346 const BatchOf_int32 one = BatchOf_int32::fromScalar(1);
347
348 for (; hitIdx + simd::RegisterWidth <= numHits; hitIdx += simd::RegisterWidth) {
349 // delta = shape-pos - cam-pos
350 const BatchOf_float dx = BatchOf_float::loadAligned(d_ox + hitIdx) - cx;
351 const BatchOf_float dy = BatchOf_float::loadAligned(d_oy + hitIdx) - cy;
352 const BatchOf_float dz = BatchOf_float::loadAligned(d_oz + hitIdx) - cz;
353 const BatchOf_float lodDistance = (dx*dx) + (dy*dy) + (dz*dz);
354
355 // comparisons -> mask (true -> 0xFFFFFFFF)
356 const BatchOf_float c0 = (lodDistance >= BatchOf_float::loadAligned(d_t0 + hitIdx));
357 const BatchOf_float c1 = (lodDistance >= BatchOf_float::loadAligned(d_t1 + hitIdx));
358 const BatchOf_float c2 = (lodDistance >= BatchOf_float::loadAligned(d_t2 + hitIdx));
359
360 // convert masks to integer 0/1 per lane, and sum to get LOD level
361 const BatchOf_int32 lodLevel =
362 (BatchOf_int32::castFloatBatch(c0) & one) +
363 (BatchOf_int32::castFloatBatch(c1) & one) +
364 (BatchOf_int32::castFloatBatch(c2) & one);
365
366 // compute drawBin = lodLevel * numLayer + layerIdx
367 const BatchOf_int32 drawBin = (lodLevel * nl) + layer;
368
369 // Finally store results
370 drawBin.storeAligned(d_drawBin + hitIdx);
371 lodDistance.storeAligned(d_sortDistance + hitIdx);
372 lodLevel.storeAligned(d_lodLevel + hitIdx);
373 }
374
375 // process remaining hits scalar-wise
376 for (; hitIdx < numHits; ++hitIdx) {
377 const float dx = d_ox[hitIdx] - camPos.x;
378 const float dy = d_oy[hitIdx] - camPos.y;
379 const float dz = d_oz[hitIdx] - camPos.z;
380 // Compute squared distance from shape to camera
381 const float lodDistance = (dx * dx) + (dy * dy) + (dz * dz);
382 // Compute LOD level for this shape by distance to camera.
383 const uint32_t lodLevel =
384 static_cast<uint32_t>(lodDistance >= d_t0[hitIdx]) +
385 static_cast<uint32_t>(lodDistance >= d_t1[hitIdx]) +
386 static_cast<uint32_t>(lodDistance >= d_t2[hitIdx]);
387 // compute bin and item indices
388 d_sortDistance[hitIdx] = lodDistance;
389 d_lodLevel[hitIdx] = lodLevel;
390 d_drawBin[hitIdx] = lodLevel * numLayer + layerIdx;
391 }
392 }
393
394 // Third pass: compute sort keys and scatter
395 for (hitIdx=0; hitIdx < numHits; ++hitIdx) {
396 const uint32_t globalID = hitSoA.globalID[hitIdx];
397 const uint32_t shapeIdx = hitSoA.shapeIdx[hitIdx];
398 const float sortDistance = d_sortDistance[hitIdx];
399 const uint32_t lodLevel = d_lodLevel[hitIdx];
400 const uint32_t drawBin = d_drawBin[hitIdx];
401
402 // Compute sort key for this shape.
403 const KeyType sortKey =
404 // pack distance (lower bits; sort within LOD level + layer)
405 computeDistanceKey<DistanceType>(sortDistance, flip) |
406 // pack layer
407 ((static_cast<KeyType>(layerIdx) & ic.layerMask) << bitOffset_layer) |
408 // pack LOD level (lod-major ordering per shape)
409 ((static_cast<KeyType>(lodLevel) & lodLevelMask) << bitOffset_lodLevel) |
410 // pack shape (upper bits)
411 ((static_cast<KeyType>(shapeIdx) & ic.shapeMask) << bitOffset_shape);
412
413 // Build a mapping from global ID to sort key
414 sortKeys[globalID] = sortKey;
415 // Push global ID in hit-ordering, this will be used for sorting.
416 ic.globalQueue_.push_back(globalID);
417 // Bin the shape into the (lod, layer) bin.
418 // This is used to fill the indirect draw buffer.
419 IndexedShape &is = *ic.indexShapes_[shapeIdx];
420 is.mapped_drawBinCount_[drawBin] += 1;
421 }
422 } else {
423 const auto &itemShapes = ic.index->itemBoundingShapes_;
424 const uint32_t* itemToGlobalID = ic.itemToGlobalID_[layerIdx].data();
425
426 for (uint32_t hitIdx=0; hitIdx < hitBuffer.count; ++hitIdx) {
427 // Gather shape data for this hit.
428 const uint32_t itemIdx = hitBuffer.data[hitIdx];
429 const uint32_t globalID = itemToGlobalID[itemIdx];
430 const uint32_t shapeIdx = ic.itemToIndexedShape_[itemIdx];
431
432 BoundingShape &bs = *itemShapes[itemIdx].get();
433 IndexedShape &is = *ic.indexShapes_[shapeIdx];
434 const Vec3f &shapeOrigin = bs.tfOrigin();
435 const Vec3f &lodThresholds = is.lodThresholds();
436
437 // Compute squared distance from shape to camera
438 const float lodDistance = (shapeOrigin - camPos).lengthSquared();
439
440 // Compute LOD level for this shape by distance to camera.
441 const uint32_t lodLevel =
442 static_cast<uint32_t>(lodDistance >= lodThresholds.x) +
443 static_cast<uint32_t>(lodDistance >= lodThresholds.y) +
444 static_cast<uint32_t>(lodDistance >= lodThresholds.z);
445 // compute draw bin and item indices
446 const uint32_t drawBin = lodLevel * numLayer + layerIdx;
447
448 // Compute sort key for this shape.
449 KeyType sortKey =
450 // pack distance (lower bits; sort within LOD level + layer)
451 computeDistanceKey<DistanceType>(lodDistance, flip) |
452 // pack layer
453 ((static_cast<KeyType>(layerIdx) & ic.layerMask) << bitOffset_layer) |
454 // pack LOD level (lod-major ordering per shape)
455 ((static_cast<KeyType>(lodLevel) & lodLevelMask) << bitOffset_lodLevel) |
456 // pack shape (upper bits)
457 ((static_cast<KeyType>(shapeIdx) & ic.shapeMask) << bitOffset_shape);
458
459 // Build a mapping from global ID to sort key
460 sortKeys[globalID] = sortKey;
461 // Push global ID in hit-ordering, this will be used for sorting.
462 ic.globalQueue_.push_back(globalID);
463 // Bin the shape into the (lod, layer) bin.
464 // This is used to fill the indirect draw buffer.
465 is.mapped_drawBinCount_[drawBin] += 1;
466 }
467 }
468}

472#define _push(KT,DT) pushVisibleShapes<KT,DT>(ic, layerIdx, hitBuffer)

488 _push(KeyType, DISTANCE_KEY_32);

525void SpatialIndex::resetCamera(IndexCamera *indexCamera, DistanceKeySize distanceBits, uint32_t traversalMask) {
526 const uint32_t numLayer = indexCamera->cullCamera->numLayer();
527 const uint32_t numItems = static_cast<uint32_t>(itemBoundingShapes_.size());
528
529 if (indexCamera->isDirty) {
530 // Compute the number of keys which is the sum of L * I for all shapes
531 // assigned to this camera.
532 indexCamera->numKeys = 0;
533
534 uint32_t shapeBase = 0;
535 for (auto &indexShape: indexCamera->indexShapes_) {
536 const uint32_t numInstances = indexShape->shape().numInstances();
537 const uint32_t numKeys = numInstances * numLayer;
538 indexCamera->numKeys += numKeys;
539 indexShape->shapeBase_ = shapeBase;
540 shapeBase += numKeys;
541 }
542 indexCamera->globalQueue_.reserve(indexCamera->numKeys);
543 indexCamera->globalToInstance_.resize(indexCamera->numKeys, 0);
544
545 // resize indexCamera->globalInstanceIDs_ arrays
546 indexCamera->itemToGlobalID_.resize(numLayer);
547 for (auto &layerVec: indexCamera->itemToGlobalID_) {
548 layerVec.resize(numItems, 0);
549 }
550
551 // Compute the global index for all shapes assigned to this camera.
552 // Note that we cannot insert instance IDs directly as we need to distinguish
553 // between instances of different shapes and layers.
554 for (uint32_t itemIdx=0; itemIdx<numItems; ++itemIdx) {
555 BoundingShape &bs = *itemBoundingShapes_[itemIdx].get();
556 IndexedShape &is = *indexCamera->indexShapes_[indexCamera->itemToIndexedShape_[itemIdx]];
557 const uint32_t numInstances = bs.numInstances();
558 const uint32_t instanceID = bs.instanceID();
559 for (uint32_t layerIdx = 0; layerIdx < numLayer; ++layerIdx) {
560 const uint32_t globalID =
561 is.shapeBase_ + // shape base
562 layerIdx * numInstances + // layer base
563 instanceID;
564 indexCamera->itemToGlobalID_[layerIdx][itemIdx] = globalID;
565 indexCamera->globalToInstance_[globalID] = instanceID;
566 }
567 }
568
569 indexCamera->layerBits = getMinBits(numLayer);
570 indexCamera->layerMask = (1u << indexCamera->layerBits) - 1u;
571 indexCamera->shapeBits = getMinBits(indexCamera->indexShapes_.size());
572 indexCamera->shapeMask = (1u << indexCamera->shapeBits) - 1u;
573 // Note: We need to add lod level bits, plus distance bits. Distance
574 // Is only used for sorting within a LOD level in a layer.
575 indexCamera->keyBits = indexCamera->shapeBits + LOD_LEVEL_BITS +
576 indexCamera->layerBits + static_cast<uint8_t>(distanceBits);
577
578 if (indexCamera->keyBits <= 32) {
579 using KeyType = uint32_t;
580 indexCamera->sortKeys32_.resize(indexCamera->numKeys);
581 indexCamera->sortKeys = static_cast<void*>(&indexCamera->sortKeys32_);
582 indexCamera->sortFun[IndexCamera::SORT_FUN_SMALL] = smallSortFun<KeyType>;
583 if (indexCamera->keyBits <= 24) {
584 // use 24-bit radix sort (uint32_t keys)
585 using RadixType = RadixSort_CPU_seq<uint32_t, KeyType, 8, 24>;
586 indexCamera->radixSort = IndexCamera::createRadix<RadixType>(indexCamera->numKeys);
587 indexCamera->sortFun[IndexCamera::SORT_FUN_LARGE] = radixSortFun<RadixType,KeyType>;
588 } else {
589 // use 32-bit radix sort (uint32_t keys)
590 using RadixType = RadixSort_CPU_seq<uint32_t, KeyType, 8, 32>;
591 indexCamera->radixSort = IndexCamera::createRadix<RadixType>(indexCamera->numKeys);
592 indexCamera->sortFun[IndexCamera::SORT_FUN_LARGE] = radixSortFun<RadixType,KeyType>;
593 }
594 } else {
595 using KeyType = uint64_t;
596 indexCamera->sortKeys64_.resize(indexCamera->numKeys);
597 indexCamera->sortKeys = static_cast<void*>(&indexCamera->sortKeys64_);
598 indexCamera->sortFun[IndexCamera::SORT_FUN_SMALL] = smallSortFun<KeyType>;
599 if (indexCamera->keyBits <= 40) {
600 // use 40-bit radix sort (uint64_t keys)
601 using RadixType = RadixSort_CPU_seq<uint32_t, KeyType, 8, 40>;
602 indexCamera->radixSort = IndexCamera::createRadix<RadixType>(indexCamera->numKeys);
603 indexCamera->sortFun[IndexCamera::SORT_FUN_LARGE] = radixSortFun<RadixType,KeyType>;
604 } else if (indexCamera->keyBits <= 48) {
605 // use 48-bit radix sort (uint64_t keys)
606 using RadixType = RadixSort_CPU_seq<uint32_t, KeyType, 8, 48>;
607 indexCamera->radixSort = IndexCamera::createRadix<RadixType>(indexCamera->numKeys);
608 indexCamera->sortFun[IndexCamera::SORT_FUN_LARGE] = radixSortFun<RadixType,KeyType>;
609 } else {
610 // use 64-bit radix sort (uint64_t keys)
611 using RadixType = RadixSort_CPU_seq<uint32_t, KeyType, 8, 64>;
612 indexCamera->radixSort = IndexCamera::createRadix<RadixType>(indexCamera->numKeys);
613 indexCamera->sortFun[IndexCamera::SORT_FUN_LARGE] = radixSortFun<RadixType,KeyType>;
614 }
615 }
616 indexCamera->isDirty = false;
617 }
618 indexCamera->traversalMask = traversalMask;
619}

621void SpatialIndex::updateVisibility(uint32_t traversalMask) {
622 if constexpr (SPATIAL_INDEX_USE_MULTITHREADING) {
623 JobPool& pool = threading::getJobPool();
624 const uint32_t numIndexedCameras = indexCameras_.size();
625
626 jobFrame_.numPushed = 0;
627
628 // Schedule jobs for all index cameras, but keep the one with the most keys for the local thread.
629 IndexCamera *localIndexCamera = &indexCameras_.front();
630 resetCamera(localIndexCamera, distanceBits_, traversalMask);
631 for (uint32_t i = 1; i < numIndexedCameras; ++i) {
632 IndexCamera *nextIndexCamera = &indexCameras_[i];
633 resetCamera(nextIndexCamera, distanceBits_, traversalMask);
634 if (nextIndexCamera->numKeys > localIndexCamera->numKeys) {
635 std::swap(localIndexCamera, nextIndexCamera);
636 }
637 pool.addJobPreFrame(jobFrame_,
638 Job{ .fn = visibilityJobFunc, .arg = nextIndexCamera });
639 }
640
641 // Execute jobs
642 pool.beginFrame(jobFrame_, 1u); // one local job
643 FramedJob localJob { { .fn = visibilityJobFunc, .arg = localIndexCamera }, &jobFrame_ };
644 do {
645 pool.performJob(localJob);
646 } while (pool.stealJob(localJob));
647 pool.endFrame(jobFrame_);
648 } else { // no multithreading
649 for (auto &indexCamera: indexCameras_) {
650 resetCamera(&indexCamera, distanceBits_, traversalMask);
651 updateVisibility(&indexCamera);
652 }
653 }
654}

656void SpatialIndex::updateVisibility(IndexCamera *indexCamera) {
657 const uint32_t L = indexCamera->cullCamera->numLayer();
658
659 // Clear the global queue
660 indexCamera->globalQueue_.clear();
661
662 for (auto &indexShape: indexCamera->indexShapes_) {
663 // Map instance data for this shape, and reset the bin counts
664 indexShape->mapInstanceData_internal();
665 indexShape->mapped_drawBinBase_ = indexShape->mappedBaseInstance();
666 indexShape->mapped_drawBinCount_ = indexShape->mappedInstanceCounts();
667 // Reset the per-bin counts. We accumulate them during traversal, so they need to be reset first.
668 std::memset(indexShape->mapped_drawBinCount_, 0,
669 sizeof(uint32_t) * indexShape->numLODs() * L);
670 }
671
672 // Process each layer
673 auto &frustumShapes = indexCamera->cullCamera->frustum();
674 for (uint32_t layerIdx = 0; layerIdx < frustumShapes.size(); ++layerIdx) {
675 auto &shape = frustumShapes[layerIdx];
676 shape.updateOrthogonalProjection();
677 updateLayerVisibility(*indexCamera, layerIdx, shape);
678 }
679
680 // Sort the visible instances according to their sort keys
681 std::vector<uint32_t>& queuedGlobalIDs = indexCamera->globalQueue_;
682 indexCamera->sortFun[static_cast<int>(queuedGlobalIDs.size()<SMALL_ARRAY_SIZE)]
683 (indexCamera->radixSort.get(), indexCamera->sortKeys, queuedGlobalIDs);
684
685 uint32_t shapeBase = 0;
686 for (auto &indexShape: indexCamera->indexShapes_) {
687 // Compute base offsets for all bins in LOD-major order.
688 const uint32_t numBins = indexShape->numLODs() * L;
689 uint32_t numVisibleInstances = 0;
690 for (uint32_t b = 0; b < numBins; ++b) {
691 const uint32_t count = indexShape->mapped_drawBinCount_[b];
692 indexShape->mapped_drawBinBase_[b] = numVisibleInstances;
693 numVisibleInstances += count;
694 }
695 const uint8_t isVisible = static_cast<uint8_t>(numVisibleInstances > 0);
696 indexShape->isVisibleInAnyLayer_.store(isVisible, std::memory_order_relaxed);
697
698 if (isVisible) {
699 // Write IDs to mapped buffer
700 // Each shape has a fixed contiguous region in tmp_layerShapes_ starting at some offset.
701 auto mapped_ids = indexShape->mappedInstanceIDs();
702 const auto &globalToInstance = indexCamera->globalToInstance_;
703 std::vector<uint32_t>::iterator vecBegin = queuedGlobalIDs.begin() + shapeBase;
704 std::vector<uint32_t>::iterator vecEnd = vecBegin + numVisibleInstances;
705 std::transform(vecBegin, vecEnd, mapped_ids,
706 [&globalToInstance](uint32_t globalID) {
707 return globalToInstance[globalID];
708 });
709 }
710
711 indexShape->unmapInstanceData_internal();
712 indexShape->mapped_drawBinBase_ = nullptr;
713 indexShape->mapped_drawBinCount_ = nullptr;
714 shapeBase += numVisibleInstances;
715 }
716}

775 auto indexType = input.getValue<std::string>("type", "quadtree");

777
778 if (indexType == "quadtree") {
779 auto quadTree = ref_ptr<QuadTree>::alloc();
