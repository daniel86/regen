30static void processMeshChildren(LoadingContext &ctx, scene::SceneInputNode &input, CompositeMesh &x) {
31 auto parser = ctx.scene();
32
33 for (auto &child: input.getChildren()) {
34 auto processor = parser->getStateProcessor(child->getCategory());
35 if (processor.get() == nullptr) {
36 auto res = parser->createResource(*child.get());
37 if (!res) {
38 REGEN_WARN("No processor registered for '" << child->getDescription() << "'.");
39 }
40 continue;
41 }
42 if (x.meshes().size() > 1) {
43 auto state = ref_ptr<State>::alloc();
44 processor->processInput(parser, *child.get(), ctx.parent(), state);
45 for (auto &mesh: x.meshes()) {
46 mesh->joinStates(state);
47 }
48 } else {
49 processor->processInput(parser, *child.get(), ctx.parent(), x.meshes()[0]);
50 }
51 }
52}

490 }

496 mesh->updateVertexData();
497 }
498 }

503ref_ptr<CompositeMesh> CompositeMesh::createCompositeMesh(
504 LoadingContext &ctx,
505 scene::SceneInputNode &input,
506 const ref_ptr<AssetImporter> &importer) {
507 const auto scaling = input.getValue<Vec3f>("scaling", Vec3f::one());
508 const auto rotation = input.getValue<Vec3f>("rotation", Vec3f::zero());
509 const auto translation = input.getValue<Vec3f>("translation", Vec3f::zero());
510 const auto assetIndices = input.getValue<std::string>("asset-indices", "*");
511 bool useAnimation = input.getValue<bool>("asset-animation", false);
512 BufferUpdateFlags updateFlags;
513 updateFlags.frequency = input.getValue<BufferUpdateFrequency>("update-frequency", BUFFER_UPDATE_NEVER);
514 updateFlags.scope = input.getValue<BufferUpdateScope>("update-scope", BUFFER_UPDATE_FULLY);
515 updateFlags.compute = input.getValue<BufferComputeMode>("compute-mode", BUFFER_NO_COMPUTE);
516
517 BufferFlags bufferConfig(ARRAY_BUFFER, updateFlags);
518 if (input.hasAttribute("access-mode")) {
519 bufferConfig.accessMode = input.getValue<ClientAccessMode>("access-mode", BUFFER_CPU_WRITE);
520 }
521 if (input.hasAttribute("map-mode")) {
522 bufferConfig.mapMode = input.getValue<BufferMapMode>("map-mode", BUFFER_MAP_DISABLED);
523 }
524
525 ref_ptr<CompositeMesh> out_ = ref_ptr<CompositeMesh>::alloc();
526 CompositeMesh &out = *out_.get();
527
528 ref_ptr<BoneTree> nodeAnim = importer->getNodeAnimation();
529 if (useAnimation && !nodeAnim) {
530 REGEN_WARN(input.getDescription() << " has use-animation=1 but Asset '" <<
531 input.getValue("asset") << "' has not.");
532 useAnimation = false;
533 }
534
535 // Create import model transform
536 Mat4f transform = Mat4f::scaleMatrix(scaling);
537 Quaternion q(0.0, 0.0, 0.0, 1.0);
538 q.setEuler(rotation.x, rotation.y, rotation.z);
539 transform *= q.calculateMatrix();
540 transform.translate(translation);
541
542 // Asset can contain multiple meshes, they can be accessed by index.
543 // Parse user specified indices ...
544 std::vector<std::string> indicesStr;
545 boost::split(indicesStr, assetIndices, boost::is_any_of(","));
546 std::vector<uint32_t> indices(indicesStr.size());
547 bool useAllIndices = false;
548 for (uint32_t i = 0u; i < indices.size(); ++i) {
549 if (indicesStr[i] == "*") {
550 useAllIndices = true;
551 break;
552 } else {
553 std::stringstream ss(indicesStr[i]);
554 ss >> indices[i];
555 }
556 }
557
558 if (useAllIndices) {
559 for (auto &mesh : importer->loadAllMeshes(transform, bufferConfig)) {
560 out.addMesh(mesh);
561 }
562 } else {
563 for (auto &mesh : importer->loadMeshes(transform, bufferConfig, indices)) {
564 out.addMesh(mesh);
565 }
566 }
567 for (uint32_t i = 0u; i < out.meshes().size(); ++i) {
568 ref_ptr<Mesh> mesh = out.meshes()[i];
569 if (mesh.get() == nullptr) continue;
570
571 // Join in material state.
572 // Can be set to false to allow overwriting material stuff.
573 if (input.getValue<bool>("asset-material", true)) {
574 ref_ptr<Material> material =
575 importer->getMeshMaterial(mesh.get());
576 if (material.get() != nullptr) {
577 mesh->setMaterial(material);
578 }
579 }
580
581 if (useAnimation) {
582 std::vector<ref_ptr<BoneNode>> meshBones;
583 uint32_t numBoneWeights = importer->numBoneWeights(mesh.get());
584
585 // Find bones influencing this mesh
586 if (nodeAnim.get()) {
587 auto ibonNodes = importer->loadMeshBones(mesh.get(), nodeAnim.get());
588 meshBones.insert(meshBones.end(), ibonNodes.begin(), ibonNodes.end());
589 }

592 // animation data to GL.
593 if (!meshBones.empty()) {
594 ref_ptr<Bones> bonesState = ref_ptr<Bones>::alloc(nodeAnim, meshBones, numBoneWeights);
595 bonesState->setAnimationName(REGEN_STRING("bones-" << input.getName()));
596 bonesState->startAnimation();
597 mesh->joinStates(bonesState);
598 }
599 }
600 }
601
602 return out_;

670ref_ptr<Particles> CompositeMesh::createParticleMesh(LoadingContext &ctx, scene::SceneInputNode &input,
671 const uint32_t numParticles) {
672 ref_ptr<Particles> particles;
673 auto parser = ctx.scene();
674 if (input.hasAttribute("update-shader")) {
675 const auto updateShader = input.getValue("update-shader");
676 particles = ref_ptr<Particles>::alloc(numParticles, updateShader);
677 } else {
678 particles = ref_ptr<Particles>::alloc(numParticles);
679 }
680 particles->setAnimationName(input.getName());
681 particles->startAnimation();
682
683 if (input.hasAttribute("max-emits")) {
684 particles->setMaxEmits(input.getValue<uint32_t>("max-emits", 100u));
685 }
686 if (input.hasAttribute("animation-state")) {
687 auto animNodeName = input.getValue("animation-state");
688 auto animState = parser->getState(animNodeName);
689 if (animState.get() == nullptr) {
690 // Try to load the animation node
691 auto animNodeInput = parser->getRoot()->getFirstChild("node", animNodeName);
692 if (animNodeInput.get() != nullptr) {
693 auto animNode = ref_ptr<StateNode>::alloc();
694 animNode->set_name(animNodeName);
695 parser->processNode(animNode, animNodeName);
696 if (animNode->childs().empty()) {
697 animState = animNode->state();
698 } else {
699 animState = animNode->childs().front()->state();
700 }
701 parser->putState(animNodeName, animState);
702 }
703 }
704 if (animState.get() == nullptr) {
705 REGEN_WARN("Ignoring " << input.getDescription() << ", unknown animation-node.");
706 } else {
707 particles->joinAnimationState(animState);
708 }
709 }
710
711 // the attributes of particles may have special attributes in the scene file that determine default, variance, etc.
712 // these are ignored by processMeshChildren, so we need to process them here.
713 for (auto &child: input.getChildren()) {
714 if (child->getCategory() == "input" && child->getValue<bool>("is-attribute", false)) {
715 const std::string type = child->getValue("type");
716 if (type == "float") {
717 configureParticleAttribute<ShaderInput1f, float>(parser, particles, *child.get());
718 } else if (type == "vec2") {
719 configureParticleAttribute<ShaderInput2f, Vec2f>(parser, particles, *child.get());
720 } else if (type == "vec3") {
721 configureParticleAttribute<ShaderInput3f, Vec3f>(parser, particles, *child.get());
722 } else if (type == "vec4") {
723 configureParticleAttribute<ShaderInput4f, Vec4f>(parser, particles, *child.get());
724 } else if (type == "int") {
725 configureParticleAttribute<ShaderInput1i, int>(parser, particles, *child.get());
726 } else if (type == "ivec2") {
727 configureParticleAttribute<ShaderInput2i, Vec2i>(parser, particles, *child.get());
728 } else if (type == "ivec3") {
729 configureParticleAttribute<ShaderInput3i, Vec3i>(parser, particles, *child.get());
730 } else if (type == "ivec4") {
731 configureParticleAttribute<ShaderInput4i, Vec4i>(parser, particles, *child.get());
732 } else {
733 REGEN_WARN("Ignoring " << child->getDescription() << ", unknown attribute type '" << type << "'.");
734 }
735 }
736 }
737
738 particles->begin();
739 // Mesh resources can have State children
740 auto x = ref_ptr<CompositeMesh>::alloc();
741 x->addMesh(particles);
742 // Make the mesh available for child nodes.
743 parser->putResource<CompositeMesh>(input.getName(), x);
744 processMeshChildren(ctx, input, *x.get());
745 particles->end();
746
747 return particles;
748}

783std::vector<uint32_t> CompositeMesh::loadIndexRange(scene::SceneInputNode &input, const std::string &prefix) {
784 std::vector<uint32_t> out;
785 auto meshIndex = input.getValue<int>(REGEN_STRING(prefix << "-index"), -1);
786 if (meshIndex >= 0) {
787 out.push_back(meshIndex);
788 } else if (input.hasAttribute(REGEN_STRING(prefix << "-indices"))) {
789 out = input.getArray<uint32_t>(REGEN_STRING(prefix << "-indices"));
790 } else if (input.hasAttribute(REGEN_STRING(prefix << "-index-range"))) {
791 auto meshIndexRange = input.getValue(REGEN_STRING(prefix << "-index-range"));
792 std::vector<std::string> range;
793 boost::split(range, meshIndexRange, boost::is_any_of("-"));
794 if (range.size() == 2 && !range[0].empty() && !range[1].empty()) {
795 int start = std::stoi(range[0]);
796 int end = std::stoi(range[1]);
797 for (int i = start; i <= end; ++i) {
798 if (i >= 0) {
799 out.push_back(i);
800 } else {
801 REGEN_WARN("Ignoring " << input.getDescription() << ", invalid mesh index '" << i << "'.");
802 }
803 }
804 }
805 }
806
807 return out;
808}

826 REGEN_WARN("Ignoring " << input.getDescription() << ", invalid mesh index '" << index << "'.");
827 }
828 }
