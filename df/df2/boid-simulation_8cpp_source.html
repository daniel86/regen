139void BoidSimulation::updateGridSize() {
140 auto cs = cellSize_->getVertex(0).r;
141 // ensure we stay in simulation bounds
142 auto simMin = simulationBoundsMin_->getVertex(0);
143 auto simMax = simulationBoundsMax_->getVertex(0);
144#ifdef BOID_USE_FIXED_GRID
145 gridBounds_.min = simMin.r;
146 gridBounds_.max = simMax.r;
147#else
148 gridBounds_.min = Vec3f::min(gridBounds_.min, Vec3f::max(boidBounds_.min, simMin.r));
149 gridBounds_.max = Vec3f::max(gridBounds_.max, Vec3f::min(boidBounds_.max, simMax.r));
150#endif
151 // Here we compute the grid based on cell size which is twice the visual range.
152 // The idea is that we only need to consider neighbors in one direction per dimension.
153 // Here, we define neighbourhood based on distance, and also compute avoidance influence based on distance.
154 // Alternatively, it would be possible to skip distance computation entirely, and define both via cell adjacency.
155 auto gridSize = (gridBounds_.max - gridBounds_.min) / cs;
156 Vec3ui v_gridSize(
157 static_cast<uint32_t>(ceil(gridSize.x)),
158 static_cast<uint32_t>(ceil(gridSize.y)),
159 static_cast<uint32_t>(ceil(gridSize.z)));
160 uint32_t newNumCells = v_gridSize.x * v_gridSize.y * v_gridSize.z;
161 // makes ure we will get at least one cell
162 if (newNumCells == 0) {
163 newNumCells = 1;
164 v_gridSize.x = 1;
165 v_gridSize.y = 1;
166 v_gridSize.z = 1;
167 gridBounds_.min -= Vec3f::create(cs) * 0.5f;
168 gridBounds_.max += Vec3f::create(cs) * 0.5f;
169 }
170 if (newNumCells != numCells_) {
171 numCells_ = newNumCells;
172 gridSize_->setVertex(0, v_gridSize);
173 }
174}
