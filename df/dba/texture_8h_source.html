326 template<class T>
327 T sampleAverage(const Vec2f &texco, const Vec2f &regionTS, const ref_ptr<ImageData> &textureData) const {
328 auto bounds = getRegion(texco, regionTS);
329 T avg = T(0.0f);
330 int numSamples = 0;
331 for (unsigned int y = bounds.min.y; y <= bounds.max.y; ++y) {
332 for (unsigned int x = bounds.min.x; x <= bounds.max.x; ++x) {
333 unsigned int index = (y * width() + x);
334 avg += sample<T>(index, textureData);
335 numSamples++;
336 }
337 }
338 if (numSamples > 0) {
339 return avg / static_cast<float>(numSamples);
340 } else {
341 return avg;
342 }
343 }

414 template<class T, uint32_t NumComponents>
415 T sampleLinear(const Vec2f &uv, const ref_ptr<ImageData> &textureData) const {
416 const auto i_w = static_cast<int32_t>(width());
417 const auto i_h = static_cast<int32_t>(height());
418 float f_x = std::clamp(uv.x,0.0f,1.0f) * static_cast<float>(i_w);
419 float f_y = std::clamp(uv.y,0.0f,1.0f) * static_cast<float>(i_h);
420
421 const auto i_x0 = std::min(static_cast<int32_t>(f_x), i_w - 1);
422 const auto i_y0 = std::min(static_cast<int32_t>(f_y), i_h - 1);
423 const int32_t i_x1 = std::min(i_x0 + 1, i_w - 1);
424 const int32_t i_y1 = std::min(i_y0 + 1, i_h - 1);
425 const int32_t indices[4] = {
426 i_y0 * i_w + i_x0,
427 i_y1 * i_w + i_x0,
428 i_y0 * i_w + i_x1,
429 i_y1 * i_w + i_x1 };
430
431 T vals[4] = {
432 Vec::create<T>(0.0f),
433 Vec::create<T>(0.0f),
434 Vec::create<T>(0.0f),
435 Vec::create<T>(0.0f)
436 };
437 auto *vals_f = reinterpret_cast<float *>(&vals[0]);
438 for (uint32_t valIdx=0; valIdx < 4; ++valIdx) {
439 float *vals_ff = vals_f + valIdx * NumComponents;
440 auto *textureData_ff = textureData->floatPixels + indices[valIdx] * NumComponents;
441 for (unsigned int i = 0; i < NumComponents; ++i) {
442 vals_ff[i] = textureData_ff[i];
443 }
444 }
445
446 f_x -= static_cast<float>(i_x0);
447 f_y -= static_cast<float>(i_y0);
448 return
449 (vals[0] * (1.0f - f_x) + vals[2] * f_x) * (1.0f - f_y) +
450 (vals[1] * (1.0f - f_x) + vals[3] * f_x) * f_y;
451 }

459 template<class T>
460 T sample(unsigned int texelIndex, const ref_ptr<ImageData> &textureData) const {
461 T v = Vec::create<T>(0.0f);
462 auto *typedData = (float *) &v;
463 auto *floatData = textureData->floatPixels + texelIndex * numComponents_;
464 for (unsigned int i = 0; i < numComponents_; ++i) {
465 typedData[i] = floatData[i];
466 }
467 return v;
468 }
