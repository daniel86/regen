138void StagedBuffer::updateStorageFlags() {
139 if (!hasClientData_) return;
140 if (flags_.updateHints.frequency == BUFFER_UPDATE_NEVER) return;
141
142 // update the storage flags based on added inputs
143 auto sizeClass = StagingBuffer::getBufferSizeClass(estimatedSize_);
144 // input has client data, so we need to set the access mode such that the CPU can write to it.
145 enableWriteAccess();
146 setStagingBuffering(SINGLE_BUFFER);
147 if constexpr(REGEN_FORCE_CLIENT_DOUBLE_BUFFER) {
148 clientBuffer_->setClientBufferMode(ClientBuffer::DoubleBuffer);
149 }
150
151 // NOTE: in local staging we avoid persistent mapping the buffer to CPU memory to avoid performance issues
152 // with fencing, as currently local staging uses per-BO and per-segment fences which is overkill
153 // for most cases.
154 if (sizeClass == BUFFER_SIZE_SMALL) {
155 // If the buffer is small (e.g. < 512 Byte), then ..
156 setStagingMapMode(BUFFER_MAP_TEMPORARY);
157 if (stagingFlags_.updateHints.frequency >= BUFFER_UPDATE_PER_FRAME) {
158 setStagingBuffering(DOUBLE_BUFFER);
159 }
160 } else if (sizeClass == BUFFER_SIZE_MEDIUM) {
161 // If the buffer is medium-sized (e.g. < 64KB), then ...
162 if (stagingFlags_.areUpdatesFrequent()) {
163 setStagingMapMode(BUFFER_MAP_TEMPORARY);
164 setStagingBuffering(DOUBLE_BUFFER);
165 } else if (stagingFlags_.areUpdatesVeryFrequent()) {
166 // The current local fencing would not work well with very frequent updates!
167 // So better use temporary mapping in this case.
168 setStagingMapMode(BUFFER_MAP_TEMPORARY);
169 setStagingBuffering(TRIPLE_BUFFER);
170 } else {
171 // Use single-buffering in staging with unmapped copy or temporary mapping for infrequent updates.
172 if (stagingFlags_.areUpdatesPartial()) {
173 setStagingMapMode(BUFFER_MAP_DISABLED);
174 } else {
175 setStagingMapMode(BUFFER_MAP_TEMPORARY);
176 }
177 }
178 } else if (sizeClass == BUFFER_SIZE_LARGE) {
179 // If the buffer is large (e.g. < 1MB)
180 if (stagingFlags_.areUpdatesFrequent()) {
181 // If updates are frequent, then use range invalidation.
182 setStagingBuffering(DOUBLE_BUFFER);
183 setStagingMapMode(BUFFER_MAP_TEMPORARY);
184 } else if (stagingFlags_.areUpdatesVeryFrequent()) {
185 setStagingMapMode(BUFFER_MAP_TEMPORARY);
186 } else {
187 // If updates are infrequent, then use single-buffering in implicit staging and avoid mapping
188 // the buffer to CPU memory.
189 setStagingMapMode(BUFFER_MAP_DISABLED);
190 setSyncFlag(BUFFER_SYNC_IMPLICIT_STAGING);
191 }
192 } else { // sizeClass == BUFFER_SIZE_VERY_LARGE
193 // If the buffer is very large (e.g. > 1MB), avoid mapping it to CPU memory.
194 setStagingMapMode(BUFFER_MAP_DISABLED);
195 if (stagingFlags_.areUpdatesRare()) {
196 setSyncFlag(BUFFER_SYNC_IMPLICIT_STAGING);
197 }
198 }

261uint32_t StagedBuffer::updateStagedInputs() {
262 const uint32_t numStagedInputs = static_cast<uint32_t>(stagedInputs_.size());
263 const StagedInput* stagedInputs = stagedInputs_.data();
264
265 // whether the size of the block has changed
266 bool hasNewSize = (requiredSize_ == 0);
267 // whether all inputs have client data, and we have at least one input.
268 bool hasClientData = (numStagedInputs != 0);
269 for (uint32_t i = 0; i < numStagedInputs; ++i) {
270 const ShaderInput &in = *stagedInputs[i].input;
271 hasNewSize |= (stagedInputs[i].inputSize != in.alignedInputSize());
272 hasClientData &= in.hasClientData();
273 }
274 hasClientData_ = hasClientData;
275
276 // Initialize the client buffer here lazily.
277 if (!clientBuffer_->hasSegments() && hasClientData) [[unlikely]] {
278 std::vector<ref_ptr<ClientBuffer>> segments(numStagedInputs);
279 for (uint32_t i = 0; i < numStagedInputs; ++i) {
280 segments[i] = stagedInputs[i].input->clientBuffer();
281 }
282 clientBuffer_->setSegments(segments);
283 clientBuffer_->swapData();
284 }
285 // Update required size if needed.
286 if (hasNewSize) [[unlikely]] {
287 updateRequiredSize();
288 }
289
290 // Update dirty segments.
291 const bool isDirty = updateDirtySegments();
292 // Remember if we had a dirty segment in this frame for computing the update rate.
293 // this is useful for detecting stalls in the staging system, for adaptive ring buffering.
294 setUpdatedFrame(isDirty);
295
296 return requiredSize_;
297}

299void StagedBuffer::updateRequiredSize() {
300 const uint32_t numStagedInputs = static_cast<uint32_t>(stagedInputs_.size());
301 StagedInput* stagedInputs = stagedInputs_.data();
302
303 requiredSize_ = 0;
304 for (size_t i = 0; i < numStagedInputs; ++i) {
305 StagedInput &blockInput = stagedInputs[i];
306 const ShaderInput &in = *blockInput.input;
307 // Align the offset to the required alignment
308 // baseAlignment is always a power of two, so we can use bitwise AND
309 requiredSize_ = (requiredSize_ + in.baseAlignment() - 1) & ~(in.baseAlignment() - 1);
310 blockInput.offset = requiredSize_;
311 blockInput.inputSize = in.alignedInputSize();
312 requiredSize_ += blockInput.inputSize;
313 }
314 // Round total size up to next multiple of 16 (vec4 alignment for std140)
315 if (memoryLayout_ == BUFFER_MEMORY_STD140) {
316 static constexpr size_t std140AlignmentMinOne = 15; // 16 - 1
317 requiredSize_ = (requiredSize_ + std140AlignmentMinOne) & ~(std140AlignmentMinOne);
318 }
319}

321bool StagedBuffer::updateDirtySegments() {
322 const uint32_t numStagedInputs = static_cast<uint32_t>(stagedInputs_.size());
323 auto* stagedInputs = stagedInputs_.data();
324
325 if (dirtySegmentRanges_.size() < numStagedInputs) [[unlikely]] {
326 // resize the dirty arrays if needed
327 dirtySegmentRanges_.resize(numStagedInputs + 4);
328 dirtyBufferRanges_.resize(numStagedInputs + 4);
329 }
330
331 auto* __restrict bufferRanges = dirtyBufferRanges_.data();
332 auto* __restrict segmentRanges = dirtySegmentRanges_.data();
333
334 // number of dirty segments found
335 uint32_t numDirtySegments = 0u;
336 uint32_t dirtyBytes = 0u;
337 // whether the last input changed or not
338 bool lastChanged = false;
339
340 for (uint32_t inputIdx = 0u; inputIdx < numStagedInputs; ++inputIdx) {
341 StagedInput &blockInput = stagedInputs[inputIdx];
342 const bool thisChanged = blockInput.input->stampOfReadData() != getLastInputStamp(blockInput);
343 if (thisChanged && lastChanged) {
344 // this input adds to the current segment
345 const uint32_t dirtyIdx = numDirtySegments - 1;
346 auto &dirty_s = segmentRanges[dirtyIdx];
347 auto &dirty_b = bufferRanges[dirtyIdx];
348 dirty_b.size = blockInput.offset - dirty_b.offset + blockInput.inputSize;
349 dirty_s.endIdx = inputIdx+1;
350 dirtyBytes += blockInput.inputSize;
351 }
352 else if (thisChanged) {
353 // this input starts a new segment
354 const uint32_t dirtyIdx = numDirtySegments++;
355 auto &dirty_s = segmentRanges[dirtyIdx];
356 auto &dirty_b = bufferRanges[dirtyIdx];
357 dirty_b.offset = blockInput.offset;
358 dirty_b.size = blockInput.inputSize;
359 dirty_s.startIdx = inputIdx;
360 dirty_s.endIdx = inputIdx+1;
361 dirtyBytes += blockInput.inputSize;
362 }
363 lastChanged = thisChanged;
364 }
365
366 numDirtySegments_ = numDirtySegments;
367 updatedSize_ = dirtyBytes;
368 return numDirtySegments > 0;
369}

371void StagedBuffer::copyDirtyData(byte* __restrict dstData, uint32_t localMapOffset) {
372 const uint32_t fullSize = clientBuffer_->dataSize();
373 const uint32_t numDirtySegments = numDirtySegments_;
374
375 auto* __restrict stagedInputs = stagedInputs_.data();
376 const auto* __restrict bufferRanges = dirtyBufferRanges_.data();
377 const auto* __restrict segmentRanges = dirtySegmentRanges_.data();
378
379 auto mapped = clientBuffer_->mapRange(BUFFER_GPU_READ, 0u, fullSize);
380 const byte* __restrict srcData = mapped.r;
381
382 for (uint32_t segmentIdx = 0; segmentIdx < numDirtySegments; ++segmentIdx) {
383 // Copy the whole segment range at once.
384 const auto &dirtyRange = bufferRanges[segmentIdx];
385 // NOTE: The main buffer maybe is not mapped from the start if the adopted buffer range, e.g.
386 // in case starts at first dirt segment. However, the block input offsets are always
387 // relative to the start of the buffer, so we need to adjust the offset accordingly...
388 const uint32_t dstOffset = dirtyRange.offset - localMapOffset;
389 memcpy(
390 dstData + dstOffset,
391 srcData + dirtyRange.offset,
392 dirtyRange.size);
393
394 const auto &segmentRange = segmentRanges[segmentIdx];
395 for (uint32_t inputIdx = segmentRange.startIdx; inputIdx < segmentRange.endIdx; ++inputIdx) {
396 setLastInputStamp(stagedInputs[inputIdx]);
397 }
398 }
399
400 clientBuffer_->unmapRange(BUFFER_GPU_READ, 0u, fullSize, mapped.r_index);
401}

403void StagedBuffer::copyFullData(byte *mappedBufferData, uint32_t localMapOffset) {
404 // full write of mapped range
405 // get start and end indices from first and last segment
406 uint32_t startIdx = dirtySegmentRanges_[0].startIdx;
407 uint32_t endIdx = dirtySegmentRanges_[numDirtySegments_ - 1].endIdx;
408 auto &firstSegment = stagedInputs_[startIdx];
409 auto &lastSegment = stagedInputs_[endIdx - 1];
410
411 // copy the whole range of block inputs.
412 const auto fullSize = clientBuffer_->dataSize();
413 auto mapped = clientBuffer_->mapRange(BUFFER_GPU_READ, 0u, fullSize);
414
415 const uint32_t offset = firstSegment.offset - localMapOffset;
416 memcpy(mappedBufferData + offset,
417 mapped.r + firstSegment.offset,
418 lastSegment.offset + lastSegment.inputSize - firstSegment.offset);
419
420 // update the last stamps for all inputs in the dirty range
421 for (uint32_t inputIdx = startIdx; inputIdx < endIdx; ++inputIdx) {
422 setLastInputStamp(stagedInputs_[inputIdx]);
423 }
424 clientBuffer_->unmapRange(BUFFER_GPU_READ, 0u, fullSize, mapped.r_index);
425}

454void StagedBuffer::updateDrawBuffer() {
455 if (adoptedSize_ == requiredSize_) {
456 // nothing to do, the draw buffer is already up-to-date
457 return;
458 }
459 if (drawBufferRef_.get()) {
460 orphanBufferRange(drawBufferRef_.get());
461 }
462
463 // if neither read nor write access is requested, we can use implicit staging.
464 if (!stagingFlags_.isWritable() && !stagingFlags_.isReadable()) {
465 setSyncFlag(BUFFER_SYNC_IMPLICIT_STAGING);
466 }
467
468 if (flags_.useExplicitStaging()) {
469 drawBufferRef_ = adoptBufferRange_(requiredSize_);
470 } else {
471 // note: in case of implicit staging with multi-buffering, we need to allocate space for each segment.
472 if (flags_.bufferingMode == RING_BUFFER) {
473 REGEN_WARN("Ring buffer is not supported with implicit staging. "
474 "Switching to TRIPLE_BUFFER.");
475 flags_.bufferingMode = TRIPLE_BUFFER;
476 }
477 drawBufferRef_ = adoptBufferRange_(requiredSize_ * (int) flags_.bufferingMode);
478 }
479
480 // validate the allocation
481 if (!drawBufferRef_.get()) {
482 REGEN_ERROR("failed to allocate buffer for buffer flags " << flags_);
483 isBufferValid_ = false;
484 return;
485 }
486 if (isMapModePersistent(flags_.mapMode) && !drawBufferRef_->mappedData()) {
487 REGEN_WARN("something went wrong with persistent mapping for buffer flags " << flags_);
488 isBufferValid_ = false;
489 return;
490 }
491 isBufferValid_ = true;
492
493 adoptedSize_ = requiredSize_;
494 inputSize_ = requiredSize_;
495 // set draw buffer range to first segment in the ring buffer
496 drawBufferRange_->buffer_ = drawBufferRef_->bufferID();
497 drawBufferRange_->size_ = requiredSize_;
498 drawBufferRange_->offset_ = drawBufferRef_->address();
499 queueStagingUpdate();
500
501 if (hasClientData()) {
502 // Copy over client data initially into the main buffer.
503 // This is done to ensure that the draw buffer has some initial data
504 // that can be drawn before the staging buffer is filled.
505 auto mapped = mapClientDataRaw(BUFFER_GPU_READ);
506 setBufferData(mapped.r);
507 }
508
509 // set up the inputs to point to the correct offsets in the draw buffer
510 const uint32_t drawBufferOffset = drawBufferRef_->address();
511 uint32_t bufferByteOffset = 0u;
512 for (auto &staged: stagedInputs_) {
513 ShaderInput &att = *staged.input;
514 // Fulfill requirement for alignment of starting offset of attribute data,
515 // i.e. the start byte of the data must be a multiple of the base alignment.
516 const uint32_t nextAlignment = att.baseAlignment() - 1;
517 bufferByteOffset = (bufferByteOffset + nextAlignment) & ~nextAlignment;
518 // set the main buffer reference for the input
519 att.setMainBuffer(drawBufferRef_, drawBufferOffset + bufferByteOffset);
520 bufferByteOffset += att.alignedInputSize();
521 }
522
523 REGEN_DEBUG("Created "
524 << StagingBuffer::getBufferSizeClass(requiredSize_)
525 << " " << stagingFlags_.target
526 << " \"" << name() << "\" with"
527 << " " << requiredSize_ / 1024.0 << " Kib"
528 << " BO: " << drawBufferRef_->bufferID()
529 << " at: " << drawBufferRef_->address());
530}

546 ref_ptr<StagingBuffer> buf;
547 if constexpr(!REGEN_DISABLE_GLOBAL_STAGING) {
548 buf = StagingSystem::instance().addBufferBlock(this);
549 }

575void StagedBuffer::copyStagingData(bool forceUpdate) {
576 if (forceUpdate) { markBufferDirty(); }
577 bool needsUpdate = (numDirtySegments_ > 0);
578 // advance update history
579 setUpdatedFrame(needsUpdate);
580 if (!needsUpdate) { return; }
581
582 // lazy initialization of the staging buffer
583 if (shared_->stagingBuffer_.get() == nullptr) {
584 createStagingBuffer();
585 } else if (!shared_->isGloballyStaged_ &&
586 isMapModePersistent(stagingFlags_.mapMode) &&
587 stagingFlags_.bufferingMode == RING_BUFFER) {
588 // increase number of ring buffer segments if fence is stalled too much,
589 // and we are not globally staged. in case of global staging, the system
590 // will handle the resizing of the staging buffer.
591 if (shared_->stagingBuffer_->getStallRate() > StagingBuffer::MAX_ACCEPTABLE_STALL_RATE) {
592 const uint32_t currentNumSegments = shared_->stagingBuffer_->numBufferSegments();
593 const uint32_t desiredNumSegments = currentNumSegments + 1;
594 REGEN_INFO("Resizing local staging buffer for block \""
595 << name() << "\" to " << desiredNumSegments
596 << " segments due to high stall rate.");
597 if (desiredNumSegments < shared_->stagingBuffer_->maxRingSegments()) {
598 shared_->stagingBuffer_->resizeBuffer(
599 requiredSize_,
600 desiredNumSegments);
601 shared_->stagingBuffer_->resetStallRate();
602 }
603 }
604 }
605 if (!shared_->stagingBuffer_->hasAdoptedRange() && stagingFlags_.useExplicitStaging()) {
606 // the staging buffer is not initialized yet, probably globally managed by the staging system,
607 // but the system is not ready yet. Or implicit staging is used.
608 // So we need to wait until the system is ready.
609 return;
610 }
611
612 const uint32_t numStagingSegments = shared_->stagingBuffer_->numBufferSegments();
613 if (shared_->numBufferSegments_ != numStagingSegments) {
614 // the number of segments in the staging buffer has changed, so we need to resize the
615 // last stamp vector for each input.
616 // we reset the stamps causing a re-load of all segments.
617 for (auto &input: stagedInputs_) {
618 input.lastStamp.resize(numStagingSegments);
619 }
620 resetDataStamps();
621 shared_->numBufferSegments_ = numStagingSegments;
622 }
623
624 if (shared_->stagingBuffer_->stagingFlags().isReadable()) {
625 // Copy from draw buffer to the staging buffer, then read from the staging buffer into CPU memory.
626 if (!updateReadBuffer()) {
627 REGEN_WARN("Failed to update read buffer for block \""
628 << name() << "\". This is likely a bug, buffer object will be disabled."
629 << " Staging flags: " << stagingFlags_ << ".");
630 isBufferValid_ = false;
631 }
632 } else if (hasClientData_) {
633 // Write from CPU memory to the staging buffer, and then copy to the draw buffer.
634 // Note: this is even done in case the staging storage is not writable, in which
635 // case a temporary writable buffer range is adopted.
636 if (isMapModePersistent(stagingFlags_.mapMode)) {
637 updatePersistentMapped();
638 } else if (stagingFlags_.mapMode == BUFFER_MAP_TEMPORARY) {
639 updateTemporaryMapped();
640 } else {
641 updateNonMapped();
642 }
643 } else if (stagingFlags_.useExplicitStaging()) {
644 REGEN_WARN("No client data to update BO \""
645 << name() << "\". This is likely a bug, buffer object will be disabled."
646 << " Staging flags: " << stagingFlags_ << ".");
647 isBufferValid_ = false;
648 }
649}

651void StagedBuffer::updateNonMapped() {
652 // iterate over the changed segments and copy only those into the staging buffer.
653 shared_->stagingBuffer_->beginNonMappedWrite();
654
655 const auto dataSize = clientBuffer_->dataSize();
656 auto mapped = clientBuffer_->mapRange(BUFFER_GPU_READ, 0u, dataSize);
657
658 for (uint32_t dirtyIdx = 0; dirtyIdx < numDirtySegments_; ++dirtyIdx) {
659 auto &dirtyRange_s = dirtySegmentRanges_[dirtyIdx];
660 auto &dirtyRange_b = dirtyBufferRanges_[dirtyIdx];
661 const uint32_t localOffset = shared_->stagingOffset_ + dirtyRange_b.offset;
662
663 shared_->stagingBuffer_->setSubData(
664 drawBufferRef_,
665 localOffset,
666 dirtyRange_b.size,
667 mapped.r + dirtyRange_b.offset);
668
669 // update the last stamps for all inputs in the dirty range
670 for (uint32_t inputIdx = dirtyRange_s.startIdx; inputIdx < dirtyRange_s.endIdx; ++inputIdx) {
671 setLastInputStamp(stagedInputs_[inputIdx]);
672 }
673 }
674
675 clientBuffer_->unmapRange(BUFFER_GPU_READ, 0u, dataSize, mapped.r_index);
676 shared_->stagingBuffer_->endNonMappedWrite(
677 drawBufferRef_,
678 *drawBufferRange_.get(),
679 shared_->stagingOffset_);
680 stamp_ += 1;
681}

683void StagedBuffer::updateTemporaryMapped() {
684 // Selectively enable partial updates.
685 // However, note that we need to do multiple mappings in case of partial updates,
686 // as we need to always should use range invalidation for the mapped range.
687 // In case of full updates, we can map the whole buffer range at once with invalidation.
688 const bool doPartialUpdate = (numDirtySegments_ > 1) &&
689 // disable partial updates for small buffers, as they are fast to update anyway.
690 (StagingBuffer::getBufferSizeClass(estimatedSize_) > BUFFER_SIZE_SMALL) &&
691 // disable partial updates in case the update covers a large part of the buffer.
692 (updatedSize_ / static_cast<float>(requiredSize_) >
693 BufferBlock::MAX_UPDATE_RATIO_PARTIAL_TEMPORARY) &&
694 // disable partial updates for larger number of dirty segments.
695 (numDirtySegments_ <= BufferBlock::MIN_SEGMENTS_PARTIAL_TEMPORARY);
696
697 if (doPartialUpdate) {
698 for (uint32_t dirtyIdx = 0; dirtyIdx < numDirtySegments_; ++dirtyIdx) {
699 auto &dirtyRange_s = dirtySegmentRanges_[dirtyIdx];
700 auto &dirtyRange_b = dirtyBufferRanges_[dirtyIdx];
701 const uint32_t localOffset = shared_->stagingOffset_ + dirtyRange_b.offset;
702 byte *bufferData = shared_->stagingBuffer_->beginMappedWrite(
703 drawBufferRef_,
704 false,
705 localOffset,
706 dirtyRange_b.size);
707 if (bufferData) {
708 const auto dataSize = clientBuffer_->dataSize();
709 auto mapped = clientBuffer_->mapRange(BUFFER_GPU_READ, 0u, dataSize);
710
711 const uint32_t offset = dirtyRange_b.offset - localOffset;
712 memcpy(bufferData + offset,
713 mapped.r + dirtyRange_b.offset,
714 dirtyRange_b.size);
715
716 for (uint32_t inputIdx = dirtyRange_s.startIdx; inputIdx < dirtyRange_s.endIdx; ++inputIdx) {
717 setLastInputStamp(stagedInputs_[inputIdx]);
718 }
719
720 clientBuffer_->unmapRange(BUFFER_GPU_READ,
721 0u, dataSize, mapped.r_index);
722 shared_->stagingBuffer_->endMappedWrite(
723 drawBufferRef_,
724 *drawBufferRange_.get(),
725 shared_->stagingOffset_);
726 } // else: frame was dropped
727 }
728 stamp_ += 1;
729 } else if (numDirtySegments_ > 0) {
730 // full update: map the whole range between the first and last dirty segment.
731 auto &firstSegment = dirtyBufferRanges_[0];
732 auto &lastSegment = dirtyBufferRanges_[numDirtySegments_ - 1];
733 const uint32_t mapRangeSize = lastSegment.offset - firstSegment.offset + lastSegment.size;
734 const uint32_t localOffset = shared_->stagingOffset_ + firstSegment.offset;
735
736 byte *bufferData = shared_->stagingBuffer_->beginMappedWrite(
737 drawBufferRef_,
738 // partial writing not ok, as we invalidate the whole range
739 false,
740 localOffset,
741 mapRangeSize);
742 if (bufferData) {
743 copyFullData(bufferData, firstSegment.offset);
744 shared_->stagingBuffer_->endMappedWrite(
745 drawBufferRef_,
746 *drawBufferRange_.get(),
747 shared_->stagingOffset_);
748 stamp_ += 1;
749 } // else: frame was dropped
750 }
751}
752
753void StagedBuffer::updatePersistentMapped() {
754 // note: partial writing is ok with persistent mapping.
755 static constexpr bool usePersistentPartialUpdate = true;
756
757 if (numDirtySegments_ == 0) { return; }
758 // copy first to last dirty segments to the mapped buffer.
759 auto &firstSegment = dirtyBufferRanges_[0];
760 auto &lastSegment = dirtyBufferRanges_[numDirtySegments_ - 1];
761 auto &sb = shared_->stagingBuffer_;
762 const uint32_t localOffset = shared_->stagingOffset_ + firstSegment.offset;
763
764 byte *bufferData = sb->beginMappedWrite(
765 drawBufferRef_,
766 usePersistentPartialUpdate,
767 localOffset,
768 lastSegment.offset - firstSegment.offset + lastSegment.size);
769 if (bufferData) {
770 // only copy the dirty segments to the mapped buffer.
771 copyDirtyData(
772 bufferData,
773 firstSegment.offset);
774 // push the dirty segments to the flush queue for just-in-time flushing.
775 if (stagingFlags_.useExplicitFlushing()) {
776 auto dirtySegments = (BufferRange2ui *) (&dirtyBufferRanges_.data()[0].offset);
777 sb->pushToFlushQueue(
778 dirtySegments + shared_->stagingOffset_,
779 numDirtySegments_);
780 }
781 sb->endMappedWrite(
782 drawBufferRef_,
783 *drawBufferRange_.get(),
784 shared_->stagingOffset_);
785 stamp_ += 1;
786 } // else: frame was dropped
787}

807 bool &wasUpdated = updatedFrames_[updateIdx_++];

809 updateCount_ += (wasUpdated != isUpdated) * (isUpdated*2 - 1);

821template<typename T, bool ExportFromGPU>
822void exportToBinary(ShaderInput &input, const std::filesystem::path &exportPath) {
823 const uint32_t numElements = input.numElements();
824 const uint32_t numComponents = input.valsPerElement();
825
826 std::vector<T> values(numElements * numComponents, T(0));
827 T* dstPtr = values.data();
828
829 if constexpr (ExportFromGPU) {
830 byte *gpuData = new byte[input.alignedInputSize()];
831 input.readServerData(gpuData);
832
833 for (uint32_t i = 0; i < numElements; ++i) {
834 const size_t offset = i * input.alignedElementSize();
835 std::memcpy(dstPtr, gpuData + offset, numComponents * sizeof(T));
836 dstPtr += numComponents;
837 }
838
839 delete[] gpuData;
840 } else {
841 auto mapped = input.mapClientDataRaw(BUFFER_GPU_READ);
842 auto* gpuData = mapped.r;
843
844 for (uint32_t i = 0; i < numElements; ++i) {
845 const size_t offset = i * input.alignedElementSize();
846 std::memcpy(dstPtr, gpuData + offset, numComponents * sizeof(T));
847 dstPtr += numComponents;
848 }
849 }
850
851 std::string binFile = (exportPath / input.name()).string() + ".bin";
852 std::ofstream bin(binFile, std::ios::binary);
853 bin.write(reinterpret_cast<const char*>(values.data()),
854 values.size() * sizeof(T));
855 bin.close();
856}

858template<bool ExportFromGPU>
859void exportToJSON(const std::filesystem::path &exportPath,
860 std::string_view bufferName,
861 std::vector<ShaderInput*> &inputs) {
862
863 // Create JSON meta data dictionary.
864 nlohmann::json j;
865 j["name"] = bufferName;
866 j["inputs"] = nlohmann::json::array();
867
868 // we create a subdirectory under the given export path for storing binary data files.
869 std::filesystem::path subPath = exportPath;
870 if (!std::filesystem::exists(subPath) && !std::filesystem::create_directories(subPath)) {
871 REGEN_ERROR("Failed to create export directory '"
872 << subPath.string() << "' for staged buffer JSON export.");
873 return;
874 }
875
876 for (auto &input : inputs) {
877 nlohmann::json &jInput = j["inputs"].emplace_back();
878 jInput["name"] = input->name();
879 jInput["binary"] = REGEN_STRING(input->name() << ".bin");
880 jInput["components"] = input->valsPerElement();
881 jInput["count"] = input->numElements();
882 jInput["layout"] = "packed";
883 jInput["originalStrideBytes"] = input->alignedElementSize();
884
885 // Read data from the main buffer to CPU memory for exporting.
886 // Write base type arrays as JSON arrays.
887 switch (input->baseType()) {
888 case GL_FLOAT:
889 exportToBinary<float,ExportFromGPU>(*input, subPath);
890 jInput["dtype"] = "float32";
891 break;
892 case GL_DOUBLE:
893 exportToBinary<double,ExportFromGPU>(*input, subPath);
894 jInput["dtype"] = "float64";
895 break;
896 case GL_INT:
897 exportToBinary<int32_t,ExportFromGPU>(*input, subPath);
898 jInput["dtype"] = "int32";
899 break;
900 case GL_UNSIGNED_INT:
901 exportToBinary<uint32_t,ExportFromGPU>(*input, subPath);
902 jInput["dtype"] = "uint32";
903 break;
904 default:
905 REGEN_WARN("Unsupported base type " << input->baseType()
906 << " for exporting staged buffer input '" << input->name() << "' to JSON.");
907 break;
908 }
909 }
910
911 // write JSON meta data file
912 std::filesystem::path jsonPath = subPath / "meta.json";
913 std::ofstream file(jsonPath);
914 if (!file.is_open()) {
915 REGEN_ERROR("Failed to open file '" << jsonPath.string() <<
916 "' for writing staged buffer JSON export.");
917 return;
918 }
919 file << j.dump(4);
920 file.close();
921}
