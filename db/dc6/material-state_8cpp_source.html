114void Material::set_textureState(const ref_ptr<TextureState> &texState, TextureState::MapTo mapTo) {
115 texState->set_mapping(TextureState::MAPPING_TEXCO);
116 texState->set_mapTo(TextureState::MAP_TO_CUSTOM);
117 switch (mapTo) {
118 case TextureState::MAP_TO_HEIGHT:
119 // add the offset to the original position
120 texState->set_blendMode(BLEND_MODE_ADD);
121 // map height map values from [0,1] to [0,maxOffset]
122 texState->set_blendFactor(maxOffset_);
123 if (heightMapMode_ == HEIGHT_MAP_VERTEX) {
124 // only modify vertex position in case no height mapping via UV transfer is used
125 texState->set_mapTo(mapTo);
126 }
127 break;
128 case TextureState::MAP_TO_DISPLACEMENT:
129 // add the offset to the original position
130 texState->set_blendMode(BLEND_MODE_ADD);
131 // map height map values from [0,1] to [0,maxOffset]
132 texState->set_blendFactor(maxOffset_);
133 break;
134 case TextureState::MAP_TO_COLOR:
135 case TextureState::MAP_TO_DIFFUSE:
136 texState->set_blendMode(colorBlendMode_);
137 texState->set_blendFactor(colorBlendFactor_);
138 texState->set_mapTo(mapTo);
139 break;
140 default:
141 texState->set_blendMode(BLEND_MODE_SRC);
142 texState->set_blendFactor(1.0f);
143 texState->set_mapTo(mapTo);
144 // if the material has a height map which is mapped to UV transfer, then every
145 // texture should use the same UV transfer
146 if (textures_.find(TextureState::MAP_TO_HEIGHT) != textures_.end()) {
147 if (heightMapMode_ == HEIGHT_MAP_RELIEF) {
148 texState->set_texcoTransfer(TextureState::TRANSFER_TEXCO_RELIEF);
149 } else if (heightMapMode_ == HEIGHT_MAP_PARALLAX) {
150 texState->set_texcoTransfer(TextureState::TRANSFER_TEXCO_PARALLAX);
151 } else if (heightMapMode_ == HEIGHT_MAP_PARALLAX_OCCLUSION) {
152 texState->set_texcoTransfer(TextureState::TRANSFER_TEXCO_PARALLAX_OCC);
153 }
154 }
155 break;
156 }
157 joinStates(texState);
158}

160static bool getMapTo(std::string_view fileName, TextureState::MapTo &mapTo) {
161 if (fileName.find("diffuse") != std::string::npos) {
162 mapTo = TextureState::MAP_TO_DIFFUSE;
163 } else if (fileName.find("color") != std::string::npos) {
164 mapTo = TextureState::MAP_TO_COLOR;
165 } else if (fileName.find("ambient") != std::string::npos) {
166 mapTo = TextureState::MAP_TO_AMBIENT;
167 } else if (fileName.find("specular") != std::string::npos) {
168 mapTo = TextureState::MAP_TO_SPECULAR;
169 } else if (fileName.find("emission") != std::string::npos) {
170 mapTo = TextureState::MAP_TO_EMISSION;
171 } else if (fileName.find("normal") != std::string::npos) {
172 mapTo = TextureState::MAP_TO_NORMAL;
173 } else if (fileName.find("height") != std::string::npos) {
174 mapTo = TextureState::MAP_TO_HEIGHT;
175 } else if (fileName.find("displacement") != std::string::npos) {
176 mapTo = TextureState::MAP_TO_DISPLACEMENT;
177 } else if (fileName.find("displace") != std::string::npos) {
178 mapTo = TextureState::MAP_TO_DISPLACEMENT;
179 } else if (fileName.find("occlusion") != std::string::npos) {
180 mapTo = TextureState::MAP_TO_LIGHT;
181 } else {
182 return false;
183 }
184 return true;

188MaterialDescription::MaterialDescription(std::string_view materialName, std::string_view variant) {
189 // find the base path with the textures
190 auto basePath0 = REGEN_STRING("res/textures/materials/" << materialName << "/" << variant);
191 basePath0 = resourcePath(basePath0);
192 if (!boost::filesystem::exists(basePath0)) {
193 basePath0 = REGEN_STRING("res/textures/" << materialName << "/" << variant);
194 basePath0 = resourcePath(basePath0);
195 }
196 if (!boost::filesystem::exists(basePath0)) {
197 return;
198 }
199
200 bool hasTextures = false;
201 for (auto &entry: boost::filesystem::directory_iterator(basePath0)) {
202 auto &path = entry.path();
203 if (entry.is_directory()) {
204 if (variant.empty() && !hasTextures) {
205 // look for textures in the subdirectory
206 for (auto &subEntry: boost::filesystem::directory_iterator(path)) {
207 if (subEntry.is_regular_file()) {
208 addTexture(subEntry.path());
209 hasTextures = true;
210 }
211 }
212 if (hasTextures) {
213 break;
214 }
215 }
216 } else {
217 addTexture(entry.path());
218 hasTextures = true;
219 }
220 }

232bool Material::set_textures(std::string_view materialName, std::string_view variant) {
233 // find the base path with the textures
234 MaterialDescription materialDescr(materialName, variant);
235
236 // remove any non-alphanumeric characters from material name
237 std::string materialNameStr(materialName);
238 materialNameStr.erase(std::remove_if(materialNameStr.begin(), materialNameStr.end(),
239 [](char c) { return !std::isalnum(c); }), materialNameStr.end());
240
241 // iterate over the texture files in the directory
242 for (auto &entry: materialDescr.textureFiles) {
243 for (auto &filePath: entry.second) {
244 auto tex = textures::load(filePath, texConfig_);
245 if (tex.get() != nullptr) {
246 // extract file name without extension
247 auto fileName = boost::filesystem::path(filePath).stem().string();
248 // remove any non-alphanumeric characters
249 fileName.erase(std::remove_if(fileName.begin(), fileName.end(),
250 [](char c) { return !std::isalnum(c); }), fileName.end());
251 auto texName = REGEN_STRING("tex_" << materialNameStr << "_" << fileName << textures_.size());
252 auto texState = ref_ptr<TextureState>::alloc(tex, texName);
253 textures_[entry.first].push_back(texState);
254
255 if (wrapping_.has_value()) {
256 tex->set_wrapping(TextureWrapping::create(wrapping_.value()));
257 }
258 } else {
259 REGEN_WARN("Failed to load texture '" << filePath << "'.");
260 }
261 }
262 }
263
264 for (auto &pair: textures_) {
265 for (auto &tex: pair.second) {
266 set_textureState(tex, pair.first);
267 }
268 }
269 return true;
270}

449ref_ptr<Material> Material::load(LoadingContext &ctx, scene::SceneInputNode &input) {
450 BufferUpdateFlags updateFlags;
451 updateFlags.frequency = input.getValue<BufferUpdateFrequency>(
452 "update-frequency", BUFFER_UPDATE_PER_FRAME);
453 updateFlags.scope = input.getValue<BufferUpdateScope>(
454 "update-scope", BUFFER_UPDATE_FULLY);
455 ref_ptr<Material> mat = ref_ptr<Material>::alloc(updateFlags);
456
457 if (input.hasAttribute("max-offset")) {
458 mat->set_maxOffset(input.getValue<float>("max-offset", 0.1f));
459 }
460 if (input.hasAttribute("height-map-mode")) {
461 mat->set_heightMapMode(input.getValue<Material::HeightMapMode>("height-map-mode",
462 Material::HEIGHT_MAP_VERTEX));
463 }
464 if (input.hasAttribute("color-blend-mode")) {
465 mat->set_colorBlendMode(input.getValue<BlendMode>("color-blend-mode", BLEND_MODE_SRC));
466 }
467 if (input.hasAttribute("color-blend-factor")) {
468 mat->set_colorBlendFactor(input.getValue<float>("color-blending-factor", 1.0f));
469 }
470
471 if (input.hasAttribute("asset")) {
472 ref_ptr<AssetImporter> assetLoader =
473 ctx.scene()->getResource<AssetImporter>(input.getValue("asset"));
474 if (assetLoader.get() == nullptr) {
475 REGEN_WARN("Skipping unknown Asset for '" << input.getDescription() << "'.");
476 } else {
477 const std::vector<ref_ptr<Material> > materials = assetLoader->materials();
478 auto materialIndex = input.getValue<uint32_t>("asset-index", 0u);
479 if (materialIndex >= materials.size()) {
480 REGEN_WARN("Invalid Material index '" << materialIndex <<
481 "' for Asset '" << input.getValue("asset") << "'.");
482 } else {
483 mat = materials[materialIndex];
484 }
485 }
486 } else if (input.hasAttribute("preset")) {
487 std::string presetVal(input.getValue("preset"));
488 auto variant = input.getValue<std::string>("variant", "");
489 if (presetVal == "jade") mat->set_jade(variant);
490 else if (presetVal == "ruby") mat->set_ruby(variant);
491 else if (presetVal == "chrome") mat->set_chrome(variant);
492 else if (presetVal == "gold") mat->set_gold(variant);
493 else if (presetVal == "copper") mat->set_copper(variant);
494 else if (presetVal == "silver") mat->set_silver(variant);
495 else if (presetVal == "pewter") mat->set_pewter(variant);
496 else if (presetVal == "iron") mat->set_iron(variant);
497 else if (presetVal == "steel") mat->set_steel(variant);
498 else if (presetVal == "metal") mat->set_metal(variant);
499 else if (presetVal == "leather") mat->set_leather(variant);
500 else if (presetVal == "stone") mat->set_stone(variant);
501 else if (presetVal == "wood") mat->set_wood(variant);
502 else if (presetVal == "marble") mat->set_marble(variant);
503 else {
504 mat->set_textures(presetVal, variant);
505 }
506 }
507 if (input.hasAttribute("diffuse"))
508 mat->diffuse()->setVertex(0,
509 input.getValue<Vec3f>("diffuse", Vec3f::one()));
510 if (input.hasAttribute("specular"))
511 mat->specular()->setVertex(0,
512 input.getValue<Vec3f>("specular", Vec3f::zero()));
513 if (input.hasAttribute("shininess"))
514 mat->shininess()->setVertex(0,
515 input.getValue<float>("shininess", 1.0f));
516 if (input.hasAttribute("emission"))
517 mat->set_emission(input.getValue<Vec3f>("emission", Vec3f::zero()));
518 if (input.hasAttribute("textures")) {
519 mat->set_textures(
520 input.getValue("textures"),
521 input.getValue<std::string>("variant", ""));
522 }
523
524 mat->alpha()->setVertex(0,
525 input.getValue<float>("alpha", 1.0f));
526 mat->refractionIndex()->setVertex(0,
527 input.getValue<float>("refractionIndex", 0.95f));
528 mat->set_fillMode(glenum::fillMode(
529 input.getValue<std::string>("fill-mode", "FILL")));
530 if (input.getValue<bool>("two-sided", false)) {
531 // this conflicts with shadow mapping front face culling.
532 mat->set_twoSided(true);
533 }
534
535 return mat;
