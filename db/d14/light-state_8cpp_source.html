77bool Light::updateConeMatrix() {
78 const uint32_t stamp = lightBuffer_->stampOfReadData();
79 if (lightConeStamp_ == stamp) return false; // no update needed
80
81 // Note: cone opens in positive z direction.
82 auto numInstances = static_cast<uint32_t>(this->numInstances());
83 if (coneMatrix_->numInstances() != numInstances) {
84 // ensure cone matrix has numInstances
85 coneMatrix_->setInstanceData(numInstances, 1, nullptr);
86 }
87 auto m_coneMatrix = coneMatrix_->mapClientData<Mat4f>(BUFFER_GPU_WRITE);
88
89 for (unsigned int i = 0; i < numInstances; ++i) {
90 auto dir = lightDirection_->getVertexClamped(i).r;
91 dir.normalize();
92 auto angleCos = dir.dot(Vec3f(0.0, 0.0, 1.0));
93
94 if (math::isApprox(abs(angleCos), 1.0)) {
95 m_coneMatrix.w[i] = Mat4f::identity();
96 } else {
97 auto radius = lightRadius_->getVertexClamped(i).r.y;
98 auto coneAngle = lightConeAngles_->getVertexClamped(i).r.y;
99
100 // Quaternion rotates view to light direction
101 Quaternion q;
102 auto axis = dir.cross(Vec3f(0.0, 0.0, 1.0));
103 axis.normalize();
104 q.setAxisAngle(axis, acos(angleCos));
105
106 // scale `unit`-cone, rotate to light direction and finally translate to light position
107 auto x = 2.0f * radius * tan(acos(coneAngle));
108 auto val = q.calculateMatrix();
109 val.scale(Vec3f(x, x, radius));
110 val.translate(lightPosition_->getVertexClamped(i).r.xyz());
111 m_coneMatrix.w[i] = val;
112 }
113 }
114
115 lightConeStamp_ = stamp;
116 return true;
117}

173ref_ptr<Light> Light::load(LoadingContext &ctx, scene::SceneInputNode &input) {
174 BufferUpdateFlags updateFlags;
175 updateFlags.frequency = input.getValue<BufferUpdateFrequency>(
176 "update-frequency", BUFFER_UPDATE_PER_FRAME);
177 updateFlags.scope = input.getValue<BufferUpdateScope>(
178 "update-scope", BUFFER_UPDATE_FULLY);
179
180 uint32_t numInstances = input.getValue<uint32_t>("num-instances", 1u);
181
182 auto lightType = input.getValue<Light::Type>("type", Light::SPOT);
183 ref_ptr<Light> light = ref_ptr<Light>::alloc(lightType, updateFlags);
184 light->set_isAttenuated(
185 input.getValue<bool>("is-attenuated", lightType != Light::DIRECTIONAL));
186
187 auto dir = input.getValue<Vec3f>("direction", Vec3f(0.0f, 0.0f, 1.0f));
188 dir.normalize();
189 light->setDirection(0, dir);
190 light->setPosition(0, input.getValue<Vec3f>("position", Vec3f::zero()));
191 light->setDiffuse(0, input.getValue<Vec3f>("diffuse", Vec3f::one()));
192 light->setSpecular(0, input.getValue<Vec3f>("specular", Vec3f::one()));
193 light->setRadius(0, input.getValue<Vec2f>("radius", Vec2f(50.0f, 50.0f)));
194
195 auto angles = input.getValue<Vec2f>("cone-angles", Vec2f(50.0f, 55.0f));
196 light->setConeAngles(angles.x, angles.y);
197 ctx.scene()->putState(input.getName(), light);
198
199 std::vector<ref_ptr<scene::SceneInputNode>> visited;
200 // process light node children
201 for (auto &child: input.getChildren()) {
202 if (child->getCategory() == "set") {
203 visited.push_back(child);
204 // set a given light input. The input key is given by the "target" attribute.
205 auto targetName = child->getValue("target");
206 // find the shader input in the light state
207 auto target_opt = light->findShaderInput(targetName);
208 if (!target_opt) {
209 REGEN_WARN("Cannot find light input for set in node " << child->getDescription());
210 continue;
211 }
212 auto setTarget = target_opt.value().in;
213 auto numInstances0 = std::max(
214 child->getValue<uint32_t>("num-instances", 1u),
215 setTarget->numInstances());
216 numInstances = std::max(numInstances, numInstances0);
217 // allocate memory for the shader input
218 setTarget->setInstanceData(numInstances0, 1, nullptr);
219 scene::ShaderInputProcessor::setInput(*child.get(), setTarget.get(), numInstances0);
220 }
221 if (child->getCategory() == "animation") {
222 auto animationType = child->getValue("type");
223 visited.push_back(child);
224 if (animationType == "boids") {
225 // let a boid simulation change the light positions
226 LoadingContext boidsConfig(ctx.scene(), ctx.parent());
227 // Note: In case of spotlights, SpotConeAnimation is used to update
228 // the cone matrix each frame.
229 ref_ptr<BoidsCPU> boidSimulation;
230 if (lightType == Light::SPOT) {
231 boidSimulation = ref_ptr<BoidsCPU>::alloc(light->position(), light->direction());
232 } else {
233 boidSimulation = ref_ptr<BoidsCPU>::alloc(light->position());
234 }
235 boidSimulation->loadSettings(ctx, *child.get());
236 light->attach(boidSimulation);
237 boidSimulation->startAnimation();
238 } else {
239 REGEN_WARN("Unknown animation type '" << animationType << "' in node " << child->getDescription());
240 }
241 }
242 }
243 light->set_numInstances(numInstances);
244
245 // remove visited children
246 for (auto &child : visited) {
247 input.removeChild(child);
248 }
249 if (lightType == Light::SPOT) {
250 light->updateConeMatrix();
251 }
252
253 auto parser = ctx.scene();
254 for (auto &child: input.getChildren()) {
255 auto processor = parser->getStateProcessor(child->getCategory());
256 if (processor.get() == nullptr) {
257 REGEN_WARN("No processor registered for '" << child->getDescription() << "'.");
258 continue;
259 }
260 processor->processInput(parser, *child.get(), ctx.parent(), light);
261 }
262
263 return light;
264}
