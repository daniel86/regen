17static ref_ptr<LODState> createCullState(
18 scene::SceneLoader *parser,
19 SceneInputNode &input,
20 const ref_ptr<StateNode> &parent,
21 const ref_ptr<CullShape> &cullShape) {
22 // get the parent camera. Note that this will be the light camera in case
23 // updating the shadow map.
24 auto cam = ref_ptr<Camera>::dynamicCast(parent->getParentCamera());
25 if (cam.get() == nullptr) {
26 REGEN_WARN("No Camera can be found for '" << input.getDescription() << "'.");
27 return {};
28 }
29 auto spatialIndex = cullShape->spatialIndex();
30 if (spatialIndex.get() && !spatialIndex->hasCamera(*cam.get())) {
31 // no need to create LOD state if the camera is not used by the spatial index
32 return {};
33 }
34 auto lodState = ref_ptr<LODState>::alloc(cam, cullShape);
35 if (input.hasAttribute("sort-mode")) {
36 lodState->setInstanceSortMode(input.getValue<SortMode>("sort-mode", SortMode::FRONT_TO_BACK));
37 }
38 if (input.hasAttribute("use-compaction")) {
39 lodState->setUseCompaction(input.getValue<bool>("use-compaction", true));
40 }
41
42 parser->putState(input.getName(), lodState);
43
44 return lodState;
45}

48 scene::SceneLoader *scene,

50 const ref_ptr<StateNode> &parent) {
51 ref_ptr<CompositeMesh> meshes = scene->getResource<CompositeMesh>(input.getName());
52 if (meshes.get() == nullptr) {
53 REGEN_WARN("Unable to load Mesh for '" << input.getDescription() << "'.");
54 return;
55 }
56 std::queue<std::pair<ref_ptr<Mesh>,uint32_t>> meshQueue;

58
59 while (!meshQueue.empty()) {
60 auto [meshOriginal,partIdx] = meshQueue.front();
61 meshQueue.pop();
62 auto meshCopy = getMeshCopy(meshOriginal);
63 if (input.hasAttribute("primitive")) {
64 meshCopy->set_primitive(glenum::primitive(input.getValue("primitive")));
65 }
66 meshCopy->setSortMode(SortMode::FRONT_TO_BACK);
67 meshCopy->ensureLOD();
68 meshCopy->resetVisibility(false);
69 StateConfigurer meshConfigurer;
70 auto meshNode = ref_ptr<StateNode>::alloc();
71 meshNode->set_name("base-mesh");
72 parent->addChild(meshNode);
73
74 auto baseStateInput = input.getFirstChild("base-state");
75 if (baseStateInput.get() != nullptr) {
76 // parse state from input
77 auto baseState = ref_ptr<State>::alloc();
78 for (auto &child: baseStateInput->getChildren()) {
79 auto processor = scene->getStateProcessor(child->getCategory());
80 if (processor.get() == nullptr) {
81 REGEN_WARN("No processor registered for '" << child->getDescription() << "'.");
82 continue;
83 }
84 processor->processInput(scene, *child.get(), parent, baseState);
85 }
86 meshConfigurer.addState(baseState.get());
87 input.removeChild(baseStateInput);
88 }
89
90 // load LOD state in case mesh has a cull shape + update-visibility="1"
91 if (meshCopy->hasCullShape()) {
92 auto updateVisibility = input.getValue<uint32_t>("update-visibility", 1u);
93 auto cullShape = ref_ptr<CullShape>::dynamicCast(meshCopy->cullShape());
94 if (updateVisibility) {
95 if (cullShape.get()) {
96 auto lodState = createCullState(scene, input, parent, cullShape);
97 if (lodState.get()) {
98 meshNode->state()->joinStates(lodState);
99 meshCopy->setInstanceBuffer(lodState->instanceBuffer());
100 if (lodState->hasIndirectDrawBuffers()) {
101 meshCopy->setIndirectDrawBuffer(
102 lodState->getIndirectDrawBuffer(meshOriginal),
103 0u,
104 lodState->numDrawLayers());
105 }
106 // set sorting mode
107 meshCopy->setSortMode(lodState->instanceSortMode());
108 } else {
109 auto setBaseLOD = ref_ptr<SetBaseLOD>::alloc(meshCopy);
110 meshNode->state()->joinStates(setBaseLOD);
111 }
112 } else {
113 REGEN_WARN("Mesh '" << input.getDescription() << "' has no cull shape.");
114 }
115 } else {
116 // try to get an instance buffer
117 auto cam = ref_ptr<Camera>::dynamicCast(parent->getParentCamera());
118 if (cam.get() && cullShape.get()) {
119 if (cullShape->hasIndirectDrawBuffers()) {
120 auto dibo = cullShape->getIndirectDrawBuffer(meshOriginal);
121 meshCopy->setIndirectDrawBuffer(dibo, 0u, cam->numLayer());
122 }
123 if (cullShape->hasInstanceBuffer()) {
124 meshCopy->setInstanceBuffer(cullShape->instanceBuffer());
125 }
126 meshCopy->setSortMode(cullShape->instanceSortMode());
127 }
128 }
129 } else {
130 if (input.hasAttribute("update-visibility")) {
131 REGEN_WARN("Mesh '" << input.getDescription() << "' has no cull shape, but update-visibility is set.");
132 }
133 // If multi-layer rendering is used, we need to create an indirect draw buffer
134 // to replicate each LOD level for each layer.
135 auto cam = ref_ptr<Camera>::dynamicCast(parent->getParentCamera());
136 if (cam.get() && cam->numLayer() > 1) {
137 meshCopy->createIndirectDrawBuffer(cam->numLayer());
138 }
139 }
140
141 LoadingContext ctx(scene, parent);
142 meshCopy->loadShaderConfig(ctx, input);
143 bool hasShader = false;
144 if (!meshCopy->hasShaderKey()) {
145 // Try to find a shader in the parent node.
146 // NOTE: This will only work in a limited fashion, and is at the moment not super useful.
147 // - The shader is once configured in the scope where it appears in the scene graph,
148 // including all defines etc. Note that eg. if a node is embedded in a sub-graph
149 // with eg. shadow mapping camera, there will be specific code inserted for that case (eg layer selection)
150 // So each shader is very specialized and cannot easily be shared in another context at the moment.
151 // - The shader will only have the uniforms which are defined above it in the scene graph.
152 // So the mesh cannot simply overwrite uniforms etc.
153 // - The shader has a fixed list of uniforms that it enables. A copy would need to be created
154 // such that the shader can be "re-configured", but this might a bit more bookkeeping.
155 // - Also the active render target in the scene graph has influence on shader code.
156 // In the long term, a proper mechanism for shader sharing and re-use should be implemented,
157 // as a kind of trade-off between re-use and specialization instead of just going for full specialization.
158 //
159 auto meshShader = ShaderState::findShader(parent.get());
160 if (meshShader.get() != nullptr) {
161 StateConfigurer stateConfigurer;
162 stateConfigurer.addNode(parent.get());
163 stateConfigurer.addState(meshCopy.get());
164 meshCopy->updateVAO(stateConfigurer.cfg(), meshShader);
165 hasShader = true;
166 }
167 }
168 if (!hasShader) {
169 meshCopy->createShader(parent);
170 }
171 meshNode->state()->joinStates(meshCopy);
172
173 // add hidden nodes for LOD meshes to show up in the GUI
174 for (auto &lodLevel: meshCopy->meshLODs()) {
175 if (lodLevel.impostorMesh.get()) {
176 auto lodNode = ref_ptr<StateNode>::alloc(lodLevel.impostorMesh);
177 lodNode->set_name("lod-impostor");
178 lodNode->set_isHidden(true);
179 meshNode->addChild(lodNode);
180 }
181 }
182 }
183}
