24void Particles::begin() {
25 uint32_t numParticles = numVertices();
26
27 // Initialize the random number generator and distribution
28 std::random_device rd;
29 std::mt19937 gen(rd());
30 std::uniform_int_distribution<> dis(0, std::numeric_limits<int>::max());
31
32 // get a random seed for each particle
33 ref_ptr<ShaderInput1ui> randomSeed_ = ref_ptr<ShaderInput1ui>::alloc("randomSeed");
34 randomSeed_->setVertexData(numParticles, nullptr);
35 for (uint32_t i = 0u; i < numParticles; ++i) {
36 randomSeed_->setVertex(i, dis(gen));
37 }
38 setInput(randomSeed_);
39
40 // initially set lifetime to zero so that particles
41 // get emitted in the first step
42 ref_ptr<ShaderInput1f> lifetimeInput_ = ref_ptr<ShaderInput1f>::alloc("lifetime");
43 lifetimeInput_->setVertexData(numParticles, nullptr);
44 for (uint32_t i = 0u; i < numParticles; ++i) {
45 lifetimeInput_->setVertex(i, -1.0);
46 }
47 setInput(lifetimeInput_);
48}

50ref_ptr<BufferReference> Particles::end() {
51 vboRef_ = Mesh::updateVertexData();
52 particleBuffer_ = ref_ptr<SSBO>::alloc(*vertexBuffer_.get(), "ParticleBlock");
53 bufferRange_.size_ = vboRef_->allocatedSize();
54
55 // Create shader defines.
56 uint32_t counter = 0;
57 for (auto it = inputs().begin(); it != inputs().end(); ++it) {
58 if (!it->in_->isVertexAttribute()) continue;
59 shaderDefine(

61 glenum::glslDataType(it->in_->baseType(), it->in_->valsPerElement()));
62 shaderDefine(

64 it->in_->name());

66 configureAdvancing(it->in_, counter, advanceModes_[it->in_->name()]);
67 }
68 counter += 1;
69 REGEN_DEBUG("Particle attribute '" << it->in_->name() << "' added.");
70 }
71 shaderDefine("NUM_PARTICLE_ATTRIBUTES", REGEN_STRING(counter));
72 shaderDefine("NUM_PARTICLES", REGEN_STRING(numParticles_));
73 createUpdateShader();
74
75 // start with zero emitted particles
76 //set_numVertices(0);
77
78 // create bounding box compute pass
79 bboxBuffer_ = ref_ptr<BBoxBuffer>::alloc(Bounds<Vec3f>());
80 bboxPass_ = ref_ptr<ComputePass>::alloc("regen.compute.bbox");

102void Particles::configureAdvancing(
103 const ref_ptr<ShaderInput> &in,
104 uint32_t counter,
105 AdvanceMode mode) {
106 std::string advanceImportKey;
107 std::string advanceFunction;
108 float advanceFactor = 1.0f;
109 switch (mode) {
110 case ADVANCE_MODE_CUSTOM: {
111 auto needle = advanceFunctions_.find(in->name());
112 if (needle == advanceFunctions_.end()) {
113 REGEN_WARN("No custom advance function found for '" << in->name() << "'.");
114 return;
115 }
116 advanceImportKey = needle->second;
117 // get string after the last dot, convention is that this is the function name
118 advanceFunction = advanceImportKey.substr(advanceImportKey.find_last_of('.') + 1);
119 // get the factor for this attribute
120 auto factorNeedle = advanceFactors_.find(in->name());
121 if (factorNeedle != advanceFactors_.end()) {
122 advanceFactor = factorNeedle->second;
123 }
124 break;
125 }
126 case ADVANCE_MODE_SRC:
127 advanceImportKey = "regen.colors.src";
128 advanceFunction = "blend_src";
129 break;
130 case ADVANCE_MODE_MULTIPLY:
131 advanceImportKey = "regen.colors.multiply";
132 advanceFunction = "blend_multiply";
133 break;
134 case ADVANCE_MODE_ADD:
135 advanceImportKey = "regen.colors.add";
136 advanceFunction = "blend_add";
137 break;
138 case ADVANCE_MODE_SMOOTH_ADD:
139 advanceImportKey = "regen.colors.smooth_add";
140 advanceFunction = "blend_smooth_add";
141 break;
142 case ADVANCE_MODE_SUBTRACT:
143 advanceImportKey = "regen.colors.sub";
144 advanceFunction = "blend_sub";
145 break;
146 case ADVANCE_MODE_REVERSE_SUBTRACT:
147 advanceImportKey = "regen.colors.reverse_sub";
148 advanceFunction = "blend_reverse_sub";
149 break;
150 case ADVANCE_MODE_DIFFERENCE:
151 advanceImportKey = "regen.colors.difference";
152 advanceFunction = "blend_difference";
153 break;
154 case ADVANCE_MODE_LIGHTEN:
155 advanceImportKey = "regen.colors.lighten";
156 advanceFunction = "blend_lighten";
157 break;
158 case ADVANCE_MODE_DARKEN:
159 advanceImportKey = "regen.colors.darken";
160 advanceFunction = "blend_darken";
161 break;
162 case ADVANCE_MODE_MIX:
163 advanceImportKey = "regen.colors.mix";
164 advanceFunction = "blend_mix";
165 break;
166 }
167 // create a new shader input for the advance function
168 shaderDefine(
169 REGEN_STRING("PARTICLE_ATTRIBUTE" << counter << "_ADVANCE_FUNCTION"),
170 advanceFunction);
171 shaderDefine(
172 REGEN_STRING("PARTICLE_ATTRIBUTE" << counter << "_ADVANCE_KEY"),
173 advanceImportKey);
174 shaderDefine(
175 REGEN_STRING("PARTICLE_ATTRIBUTE" << counter << "_ADVANCE_FACTOR"),
176 REGEN_STRING(advanceFactor));
177
178 // configure the ramp texture
179 auto rampNeedle = ramps_.find(in->name());
180 if (rampNeedle != ramps_.end()) {
181 auto &rampTexture = rampNeedle->second.texture;
182 auto &rampMode = rampNeedle->second.mode;
183 std::string rampImportKey, rampFunction;
184 switch (rampMode) {
185 case RAMP_MODE_CUSTOM: {
186 auto needle = rampFunctions_.find(in->name());
187 if (needle == rampFunctions_.end()) {
188 REGEN_WARN("No custom ramp function found for '" << in->name() << "'.");
189 return;
190 }
191 rampImportKey = needle->second;
192 rampFunction = advanceImportKey.substr(rampImportKey.find_last_of('.') + 1);
193 break;
194 }
195 case RAMP_MODE_TIME:
196 rampImportKey = "regen.particles.ramp.time";
197 rampFunction = "ramp_time";
198 break;
199 case RAMP_MODE_LIFETIME:
200 rampImportKey = "regen.particles.ramp.lifetime";
201 rampFunction = "ramp_lifetime";
202 break;
203 case RAMP_MODE_EMITTER_DISTANCE:
204 rampImportKey = "regen.particles.ramp.emitter_distance";
205 rampFunction = "ramp_emitter_distance";
206 break;
207 case RAMP_MODE_CAMERA_DISTANCE:
208 rampImportKey = "regen.particles.ramp.camera_distance";
209 rampFunction = "ramp_camera_distance";
210 break;
211 case RAMP_MODE_VELOCITY:
212 rampImportKey = "regen.particles.ramp.velocity";
213 rampFunction = "ramp_velocity";
214 break;
215 }
216 shaderDefine(
217 REGEN_STRING("PARTICLE_ATTRIBUTE" << counter << "_RAMP_KEY"),
218 rampImportKey);
219 shaderDefine(
220 REGEN_STRING("PARTICLE_ATTRIBUTE" << counter << "_RAMP_FUNCTION"),
221 rampFunction);
222 // finally create a texture state for the ramp texture
223 auto rampTextureState = ref_ptr<TextureState>::alloc();
224 rampTextureState->set_texture(rampTexture);
225 rampTextureState->set_name(REGEN_STRING(in->name() << "Ramp"));
226 joinStates(rampTextureState);
227 shaderDefine(
228 REGEN_STRING("PARTICLE_ATTRIBUTE" << counter << "_RAMP_TEXEL_SIZE"),
229 REGEN_STRING(1.0 / rampTexture->width()));
230 }

250void Particles::gpuUpdate(RenderState *rs, double dt) {
251 updateState_->enable(rs);
252 updateState_->disable(rs);
253
254 if (useGPUBoundingBox_) {
255 // Update bounding box every ~166 ms.
256 bbox_time_ += dt;
257 if (bbox_time_ > 166.0) {
258 bboxBuffer_->clear();
259 bboxPass_->enable(rs);
260 bboxPass_->disable(rs);
261 // update the grid in case the bounding box around the boids changed.
262 if(bboxBuffer_->updateBoundingBox()) {
263 auto &newBounds = bboxBuffer_->bbox();
264 if (newBounds.max != newBounds.min) {
265 set_bounds(newBounds.min, newBounds.max);
266 }
267 }
268 bbox_time_ = 0.0;
269 }
270 }
