31void MaskMesh::updateMask() {
32 unsigned int quadCountX = std::ceil(maskMeshCfg_.meshSize.x / rectangleConfig_.posScale.x);
33 unsigned int quadCountY = std::ceil(maskMeshCfg_.meshSize.y / rectangleConfig_.posScale.z);
34 Vec2f quadSize_ts = Vec2f(
35 rectangleConfig_.posScale.x / maskMeshCfg_.meshSize.x,
36 rectangleConfig_.posScale.z / maskMeshCfg_.meshSize.y);
37 Vec2f quadHalfSize = Vec2f(rectangleConfig_.posScale.x, rectangleConfig_.posScale.z) * 0.5f;
38 std::vector<Vec4f> instanceData(quadCountX * quadCountY);
39
40 unsigned int numInstances = 0;
41
42 if (maskTexture_.get()) {
43 maskTexture_->ensureTextureData();
44 }
45
46 Vec2f maskUV = quadSize_ts * 0.5f;
47
48 auto baseOffset = tf_->modelOffset()->getVertex(0).r;
49
50 for (unsigned int y = 0; y < quadCountY; ++y) {
51 for (unsigned int x = 0; x < quadCountX; ++x) {
52 float maskDensity = 0.0f;
53 if (!maskTexture_.get()) {
54 maskDensity = 1.0f;
55 } else if (maskTexture_->format()==GL_RGBA) {
56 maskDensity = maskTexture_->sampleMax<Vec4f>(maskUV,
57 quadSize_ts, maskTexture_->textureData(), maskIndex_);
58 } else if (maskTexture_->format()==GL_RGB) {
59 maskDensity = maskTexture_->sampleMax<Vec3f>(maskUV,
60 quadSize_ts, maskTexture_->textureData(), maskIndex_);
61 } else if (maskTexture_->format()==GL_RG) {
62 maskDensity = maskTexture_->sampleMax<Vec2f>(maskUV,
63 quadSize_ts, maskTexture_->textureData(), maskIndex_);
64 } else {
65 maskDensity = maskTexture_->sampleMax<float>(maskUV,
66 quadSize_ts, maskTexture_->textureData());
67 }
68 maskUV.x += quadSize_ts.x;
69 if (maskDensity > 0.1) {
70 // Note: The quads are no perfect fit, as we use instancing and cannot change the geometry per instance.
71 instanceData[numInstances++] = baseOffset + Vec4f(
72 static_cast<float>( x ) * rectangleConfig_.posScale.x + quadHalfSize.x - maskMeshCfg_.meshSize.x * 0.5f,
73 maskMeshCfg_.height,
74 static_cast<float>( y ) * rectangleConfig_.posScale.z + quadHalfSize.y - maskMeshCfg_.meshSize.y * 0.5f,
75 0.0f);
76 }
77 }
78 maskUV.x = quadSize_ts.x * 0.5f;
79 maskUV.y += quadSize_ts.y;
80 }
81
82 if (numInstances==0) {
83 REGEN_WARN("No mask instances generated, creating one default instance.");
84 numInstances = 1; // at least one instance to avoid issues with zero-sized buffers
85 }
86
87 // update the model offset attribute
88 instanceData.resize(numInstances);
89 tf_->set_numInstances(numInstances);
90 tf_->modelOffset()->setInstanceData(numInstances, 1,
91 (byte*)instanceData.data());
92
93 disjoinStates(tf_);
94
95 tf_->tfBuffer()->updateBuffer();
96 joinStates(tf_);
97}
