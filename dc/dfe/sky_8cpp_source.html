160float Sky::computeHorizonExtinction(const Vec3f &position, const Vec3f &dir, float planetRadius) {
161 float u = dir.dot(-position);
162 if (u < 0.0) {
163 return 1.0;
164 }
165 Vec3f near = position + dir * u;
166 if (near.length() < planetRadius) {
167 return 0.0;
168 } else {
169 near.normalize();
170 Vec3f v2 = near * planetRadius - position;
171 v2.normalize();
172 float diff = acos(v2.dot(dir));
173 return math::smoothstep(0.0f, 1.0f, powf(diff * 2.0f, 3.0f));
174 }
175}

178 const Vec3f eyePosition(0.0, planetRadius, 0.0);

190void Sky::cpuUpdate(double dt) {
191 if (worldTime_) {
192 time_osg_.sett(boost::posix_time::to_time_t(worldTime_->p_time));
193 time_osg_.setUtcOffset(14400); // UTC+4, Berlin time
194 astro_->update(osgHimmel::t_aTime::fromTimeF(time_osg_));
195 }
196
197 const float planetRadius = Earth::meanRadius();
198 // Compute sun/moon directions
199 Vec3f moon = astro_->getMoonPosition(false);
200 Vec3f sun = astro_->getSunPosition(false);
201 sun_->setDirection(0, sun);
202 moon_->setDirection(0, moon);
203 // Compute sun/moon diffuse color
204 float sunExt = computeEyeExtinction(sun, planetRadius);
205 Vec3f sunColor = math::mix(dawnColor_, noonColor_, abs(sunExt));
206 sun_->setDiffuse(0, computeColor(
207 sunColor,
208 sunExt));
209 moon_->setDiffuse(0, computeColor(
210 sunColor * moonSunLightReflectance_,
211 computeEyeExtinction(moon, planetRadius)));
212 R_->setVertex(0, astro().getEquToHorTransform());
213
214 if (camStamp_ != cam_->stamp() || viewportStamp_ != screen_->stampOfWriteData()) {
215 const float fovHalf = camera()->projParams()[0].fov * 0.5f * DEGREE_TO_RAD;
216 const float height = static_cast<float>(screen_->viewport().r.y);
217 const float q = 2.8284271247461903f // = sqrt(2.0f) * 2.0f
218 * tan(fovHalf) / height; // q is the distance from the camera to the sky quad
219 q_->setVertex(0, q);
220 sqrt_q_->setVertex(0, sqrt(q));

523ref_ptr<Sky> Sky::load(LoadingContext &ctx, scene::SceneInputNode &input) {
524 auto scene = ctx.scene();
525
526 if (!input.hasAttribute("camera")) {
527 REGEN_WARN("No user-camera attribute specified for '" << input.getDescription() << "'.");
528 return {};
529 }
530 ref_ptr<Camera> cam = scene->getResource<Camera>(input.getValue("camera"));
531 if (cam.get() == nullptr) {
532 REGEN_WARN("No Camera can be found for '" << input.getDescription() << "'.");
533 return {};
534 }
535
536 ref_ptr<Sky> sky = ref_ptr<Sky>::alloc(cam, scene->screen());
537
538 sky->setWorldTime(&scene->application()->worldTime());
539
540 if (input.hasAttribute("noon-color"))
541 sky->set_noonColor(input.getValue<Vec3f>("noon-color", Vec3f::create(0.5)));
542 if (input.hasAttribute("dawn-color"))
543 sky->set_dawnColor(input.getValue<Vec3f>("dawn-color", Vec3f(0.2, 0.15, 0.15)));
544
545 if (input.hasAttribute("moon-reflectance"))
546 sky->set_moonSunLightReflectance(input.getValue<float>("moon-reflectance", 0.4));
547
548 sky->set_altitude(input.getValue<float>("altitude", 0.043));
549 sky->set_longitude(input.getValue<float>("longitude", 13.3611));
550 sky->set_latitude(input.getValue<float>("latitude", 52.5491));
551
552 for (auto &n: input.getChildren()) {
553 ref_ptr<SkyLayer> layer;
554 if (n->getCategory() == "atmosphere") {
555 layer = createAtmosphereLayer(sky, scene, *n.get(), input.getName());
556 } else if (n->getCategory() == "cloud-layer") {
557 layer = createCloudLayer(sky, scene, *n.get());
558 } else if (n->getCategory() == "moon") {
559 layer = createMoonLayer(sky, *n.get());
560 } else if (n->getCategory() == "star-map") {
561 layer = createStarMapLayer(sky, *n.get());
562 } else if (n->getCategory() == "stars") {
563 layer = createStarsLayer(sky, *n.get());
564 } else if (n->getCategory() == "darkness") {
565 layer = createDarknessLayer(sky, *n.get());
566 }
567 if (!layer.get()) {
568 REGEN_WARN("No layer created for '" << n->getDescription() << "'.");
569 continue;
570 }
571 for (auto &layerChild: n->getChildren()) {
572 auto processor = scene->getStateProcessor(layerChild->getCategory());
573 if (processor.get() == nullptr) {
574 REGEN_WARN("No processor registered for '" << layerChild->getDescription() << "'.");
575 continue;
576 }
577 processor->processInput(scene, *layerChild.get(), sky, layer->updateState());
578 }
579 }
580 sky->createShader();
581
582 // The Sky also exposes a Light (the sun) and a Texture (the cube map)
583 scene->putResource<Light>(input.getName() + "-sun", sky->sun());
584 scene->putResource<Light>(input.getName() + "-moon", sky->moon());
585 scene->putState(input.getName() + "-sun", sky->sun());
586 scene->putState(input.getName() + "-moon", sky->moon());
587
588 sky->startAnimation();
589
590 return sky;
591}
