12static inline void project(const std::vector<Vec2f> &points, OrthogonalProjection::Axis &axis) {
13 float mn = std::numeric_limits<float>::infinity();
14 float mx = -std::numeric_limits<float>::infinity();
15 for (const auto &p : points) {
16 float v = p.dot(axis.dir);
17 if (v < mn) mn = v;
18 if (v > mx) mx = v;
19 }
20 axis.min = mn;
21 axis.max = mx;
22}

46void OrthogonalProjection::update(const BoundingShape &shape) {
47 switch (shape.shapeType()) {
48 case BoundingShapeType::SPHERE: {
49 // sphere projection is a circle
50 type = OrthogonalProjection::Type::CIRCLE;
51 auto *sphere = static_cast<const BoundingSphere *>(&shape);
52 auto &batchData = sphere->globalBatchData_t();
53 const uint32_t batchIdx = sphere->globalIndex();
54 const float radius = batchData.radius()[batchIdx];
55 points[0].x = batchData.posX()[batchIdx];;
56 points[0].y = batchData.posZ()[batchIdx];
57 // note: second point stores the squared radius
58 points[1].x = radius*radius;
59 break;
60 }
61 case BoundingShapeType::AABB: {
62 createRectangle_AABB(shape);
63 break;
64 }
65 case BoundingShapeType::OBB: {
66 createRectangle_OBB(shape);
67 break;
68 }
69 case BoundingShapeType::FRUSTUM: {
70 auto *frustum = static_cast<const Frustum *>(&shape);
71 // use the convex hull of the frustum points as projection
72 createConvexHull(frustum->points, 8);
73 break;
74 }
75 case BoundingShapeType::LAST: {
76 return;
77 }
78 }
79
80 // compute the bounds of the projection
81 bounds.min = points[0];
82 bounds.max = points[0];
83 if (type == OrthogonalProjection::Type::CIRCLE) {
84 auto &center = points[0];
85 auto radius = std::sqrt(points[1].x);
86 bounds.min.x = center.x - radius;
87 bounds.min.y = center.y - radius;
88 bounds.max.x = center.x + radius;
89 bounds.max.y = center.y + radius;
90 } else {
91 for (size_t i = 1; i < points.size(); i++) {
92 bounds.min.x = std::min(bounds.min.x, points[i].x);
93 bounds.min.y = std::min(bounds.min.y, points[i].y);
94 bounds.max.x = std::max(bounds.max.x, points[i].x);
95 bounds.max.y = std::max(bounds.max.y, points[i].y);
96 }
97 }

138 auto *vertices_3D = static_cast<const OBB&>(obb).boxVertices();
139 type = RECTANGLE;
140 // generate the 4 corners of the box in the xz-plane.
141 // for this just take min/max of the box vertices as bounds,
142 // this is not a perfect fit but fast.
143 Vec2f min(vertices_3D[0].x, vertices_3D[0].z);

169void OrthogonalProjection::createConvexHull(const Vec3f *inputPoints, uint32_t numPoints) {
170 type = OrthogonalProjection::Type::CONVEX_HULL;
171
172 // Project into 2D (xz-plane) and sort lexicographically (x, then y)
173 if (tmpPoints.size() != numPoints) {
174 tmpPoints.resize(numPoints);
175 }
176 for (uint32_t i = 0; i < numPoints; ++i) {
177 tmpPoints[i].x = inputPoints[i].x;
178 tmpPoints[i].y = inputPoints[i].z;
179 }
180 std::sort(tmpPoints.begin(), tmpPoints.end(), [](const Vec2f& a, const Vec2f& b) {
181 return (a.x < b.x) || (a.x == b.x && a.y < b.y);
182 });
183
184 std::vector<Vec2f> hull;
185 hull.reserve(numPoints * 2);
186 // Build lower hull
187 for (const auto& p : tmpPoints) {
188 while (hull.size() >= 2 && cross(hull[hull.size()-2], hull.back(), p) <= 0) {
189 hull.pop_back();
190 }
191 hull.push_back(p);
192 }
193 size_t lowerSize = hull.size();
194 // Build upper hull (skip first and last point)
195 for (int i = int(tmpPoints.size()) - 2; i >= 0; --i) {
196 const auto& p = tmpPoints[i];
197 while (hull.size() > lowerSize && cross(hull[hull.size()-2], hull.back(), p) <= 0) {
198 hull.pop_back();
199 }
200 hull.push_back(p);
201 }
202 if (!hull.empty()) hull.pop_back();
203
204 // Save result + Compute SAT axes from polygon edges
205 points = hull;
206 axes.clear();
207 axes.reserve(points.size() + 2); // +2 for world axes
208 size_t N = points.size();
209 for (size_t i = 0; i < N; ++i) {
210 Vec2f edge = points[(i + 1) % N] - points[i];
211 if (edge.lengthSquared() < 1e-8f) continue; // skip degenerate edges
212 // Note: normalization is not strictly necessary for SAT
213 axes.emplace_back(Vec2f(-edge.y, edge.x));
214 }
215
216 // add world X/Y axes for fast early rejection
217 // note: this is not strictly necessary for correctness.
218 axes.emplace_back(Vec2f(1.0f, 0.0f));
219 axes.emplace_back(Vec2f(0.0f, 1.0f));
220
221 // Precompute projections along each axis for SAT
222 for (auto& axis : axes) {
223 project(points, axis);
224 }
