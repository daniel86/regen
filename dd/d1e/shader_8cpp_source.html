48void Shader::preProcess(
49 std::map<GLenum, std::string> &ret,
50 const PreProcessorConfig &cfg,
51 const ref_ptr<PreProcessor> &preProcessor) {
52 // configure shader using macros
53 std::stringstream header;
54 header << "#version " << cfg.version << std::endl;
55 for (const auto &define: cfg.defines) {
56 const std::string &name = define.first;
57 const std::string &value = define.second;
58 if (value == "TRUE") {
59 header << "#define " << name << std::endl;
60 } else if (value == "FALSE") {
61 header << "// #undef " << name << std::endl;
62 } else {
63 header << "#define " << name << " " << value << std::endl;
64 }
65 }
66 header << "#include regen.regen.regen_InstanceID" << std::endl;
67 header << "#include regen.regen.regen_RenderLayer" << std::endl;
68 for (const auto &path: cfg.includes) {
69 header << "#include " << path << std::endl;
70 }
71 std::string headerStr = header.str();
72
73 // load the GLSL code.
74 PreProcessorInput preProcessorInput(
75 headerStr,
76 cfg.unprocessed,
77 cfg.externalFunctions,
78 cfg.specifiedInput);
79 ret = preProcessor->processStages(preProcessorInput);
80}

85void Shader::printLog(
86 uint32_t shader,
87 GLenum shaderType,
88 const char *shaderCode,
89 bool success) {
90 Logging::LogLevel logLevel;
91 if (shaderCode != nullptr) {
92 std::string shaderName = glenum::glslStageName(shaderType);
93
94 if (success) {
95 logLevel = Logging::INFO;
96 //REGEN_LOG(logLevel, shaderName << " Shader compiled successfully!");
97 } else {
98 logLevel = Logging::ERROR;
99 REGEN_LOG(logLevel, shaderName << " Shader failed to compile!");
100 }
101 } else {
102 if (success) {
103 logLevel = Logging::INFO;
104 //REGEN_LOG(logLevel, "Shader linked successfully.");
105 } else {
106 logLevel = Logging::ERROR;
107 REGEN_LOG(logLevel, "Shader failed to link.");
108 }
109 }
110
111 if (!success && shaderCode != nullptr) {
112 std::vector<std::string> codeLines;
113 boost::split(codeLines, shaderCode, boost::is_any_of("\n"));
114 for (uint32_t i = 0; i < codeLines.size(); ++i) {
115 REGEN_LOG(logLevel,
116 std::setw(3) << i << std::setw(0) << " " << codeLines[i]);
117 }
118 }
119
120 static constexpr int maxLogChars = 5000;
121 static char log[maxLogChars];
122 std::string msgPrefix;
123 int actualLogChars = 0;
124 if (shaderType == GL_NONE) {
125 glGetProgramInfoLog(shader, maxLogChars, &actualLogChars, log);
126 msgPrefix = "link info: ";
127 } else {
128 glGetShaderInfoLog(shader, maxLogChars, &actualLogChars, log);
129 msgPrefix = "compile info: ";
130 }
131
132 if (actualLogChars > 0) {
133 // add null terminator
134 log[std::min(actualLogChars, maxLogChars - 1)] = '\0';
135
136 std::vector<std::string> errorLines;
137 boost::split(errorLines, log, boost::is_any_of("\n"));
138 for (const auto &errorLine: errorLines) {
139 if (!errorLine.empty()) REGEN_LOG(logLevel, msgPrefix << errorLine);
140 }
141 }

262bool Shader::compile() {
263 for (auto &shaderCode: shaderCodes_) {
264 const char *source = shaderCode.second.c_str();
265 ref_ptr<uint32_t> shaderStage = ref_ptr<uint32_t>::alloc();
266 uint32_t &stage = *shaderStage.get();
267 stage = glCreateShader(shaderCode.first);
268 int length = -1;
269 int status;
270
271 glShaderSource(stage, 1, &source, &length);
272 glCompileShader(stage);
273
274 glGetShaderiv(stage, GL_COMPILE_STATUS, &status);
275 printLog(stage, shaderCode.first, source, status != 0);
276 if (!status) {
277 glDeleteShader(stage);
278 return false;
279 }
280
281 glAttachShader(id(), stage);
282 shaders_[shaderCode.first] = shaderStage;
283 }
284
285 return true;
286}

288bool Shader::link() {
289 if (!transformFeedback_.empty()) {
290 std::vector<const char *> validNames(transformFeedback_.size());
291 std::set<std::string> validNames_;
292 int validCounter = 0;
293
294 GLenum stage = feedbackStage_;
295 int i = 0;
296 for (; glenum::glslStages()[i] != stage; ++i) {}
297
298 // find next stage
299 std::string nextStagePrefix = "out";
300 for (int j = i + 1; j < glenum::glslStageCount(); ++j) {
301 GLenum nextStage = glenum::glslStages()[j];
302 if (shaders_.count(nextStage) != 0) {
303 nextStagePrefix = glenum::glslStagePrefix(nextStage);
304 break;
305 }
306 }
307
308 for (auto &it: transformFeedback_) {
309 std::string name = IOProcessor::getNameWithoutPrefix(it);
310 if (name == "Position") {
311 name = REGEN_STRING("gl_" << name);
312 } else {
313 name = REGEN_STRING(nextStagePrefix << "_" << name);
314 }
315 if (validNames_.count(name) > 0) { continue; }
316 auto needle = validNames_.insert(name);
317 validNames[validCounter++] = needle.first->c_str();
318 }
319
320 glTransformFeedbackVaryings(id(),
321 validCounter, validNames.data(), feedbackLayout_);
322 }
323
324 glLinkProgram(id());
325 int status;
326 glGetProgramiv(id(), GL_LINK_STATUS, &status);
327 GL_ERROR_LOG();
328 if (status == false) {
329 for (auto &shaderCode: shaderCodes_) {
330 const char *source = shaderCode.second.c_str();
331 printLog(id(), shaderCode.first, source, false);
332 }
333 printLog(id(), GL_NONE, nullptr, false);
334 return false;
335 } else {
336 printLog(id(), GL_NONE, nullptr, true);
337 setupInputLocations();
338 return true;
339 }

359void Shader::setupInputLocations() {
360 int count;
361 int arraySize;
362 GLenum type;
363 char nameC[320];
364
365 samplerLocations_.clear();
366
367 glGetProgramiv(id(), GL_ACTIVE_UNIFORMS, &count);
368 for (int32_t loc_ = 0; loc_ < count; ++loc_) {
369 glGetActiveUniform(id(), loc_, 320, nullptr, &arraySize, &type, nameC);
370 std::string uniformName(nameC);
371 // for arrays..
372 int32_t loc = glGetUniformLocation(id(), nameC);
373 if (loc < 0) { continue; }
374
375 // remember this uniform location
376 std::string attName(nameC);
377 if (hasPrefix(attName, "gl_")) { continue; }
378 if (boost::ends_with(uniformName, "[0]")) {
379 uniformName = uniformName.substr(0, uniformName.size() - 3);
380 }
381 if (hasPrefix(uniformName, "u_")) {
382 uniformName = truncPrefix(uniformName, "u_");
383 } else if (hasPrefix(attName, "in_")) {
384 uniformName = truncPrefix(uniformName, "in_");
385 }
386 if (hasPrefix(uniformName, "instances_")) {
387 uniformName = truncPrefix(uniformName, "instances_");
388 } else if (hasPrefix(uniformName, "tbo_")) {
389 uniformName = truncPrefix(uniformName, "tbo_");
390 }
391 uniformLocations_[std::string(nameC)] = loc;
392 uniformLocations_[uniformName] = loc;
393 uniformLocations_[REGEN_STRING("u_" << uniformName)] = loc;
394 uniformLocations_[REGEN_STRING("in_" << uniformName)] = loc;
395
396 auto needle = inputNames_.find(uniformName);
397 if (needle != inputNames_.end()) {
398 uniforms_.emplace_back(needle->second->in_, loc);
399 }
400
401 // create ShaderInput without data allocated.
402 // still setupInput must be called with this ShaderInput
403 // for the shader to enable it with applyInputs()
404 switch (type) {
405 case GL_FLOAT:
406 case GL_FLOAT_VEC2:
407 case GL_FLOAT_VEC3:
408 case GL_FLOAT_VEC4:
409 case GL_FLOAT_MAT2:
410 case GL_FLOAT_MAT3:
411 case GL_FLOAT_MAT4:
412 case GL_UNSIGNED_INT:
413 case GL_UNSIGNED_INT_VEC2:
414 case GL_UNSIGNED_INT_VEC3:
415 case GL_UNSIGNED_INT_VEC4:
416 case GL_UNSIGNED_INT_ATOMIC_COUNTER:
417 case GL_INT:
418 case GL_INT_VEC2:
419 case GL_INT_VEC3:
420 case GL_INT_VEC4:
421 case GL_BOOL:
422 case GL_BOOL_VEC2:
423 case GL_BOOL_VEC3:
424 case GL_BOOL_VEC4:
425 break;
426
427 case GL_SAMPLER_BUFFER:
428 case GL_SAMPLER_1D:
429 case GL_SAMPLER_2D:
430 case GL_SAMPLER_3D:
431 case GL_SAMPLER_CUBE:
432 case GL_SAMPLER_2D_ARRAY:
433 case GL_SAMPLER_1D_SHADOW:
434 case GL_SAMPLER_2D_SHADOW:
435 case GL_SAMPLER_CUBE_SHADOW:
436 case GL_SAMPLER_2D_ARRAY_SHADOW:
437 case GL_SAMPLER_2D_MULTISAMPLE:
438 case GL_SAMPLER_2D_MULTISAMPLE_ARRAY:
439 case GL_INT_SAMPLER_BUFFER:
440 case GL_INT_SAMPLER_1D:
441 case GL_INT_SAMPLER_2D:
442 case GL_INT_SAMPLER_3D:
443 case GL_INT_SAMPLER_CUBE:
444 case GL_INT_SAMPLER_2D_ARRAY:
445 case GL_INT_SAMPLER_2D_MULTISAMPLE:
446 case GL_INT_SAMPLER_2D_MULTISAMPLE_ARRAY:
447 case GL_UNSIGNED_INT_SAMPLER_1D:
448 case GL_UNSIGNED_INT_SAMPLER_2D:
449 case GL_UNSIGNED_INT_SAMPLER_3D:
450 case GL_UNSIGNED_INT_SAMPLER_CUBE:
451 case GL_UNSIGNED_INT_SAMPLER_2D_ARRAY:
452 case GL_UNSIGNED_INT_SAMPLER_2D_MULTISAMPLE:
453 case GL_UNSIGNED_INT_SAMPLER_2D_MULTISAMPLE_ARRAY:
454 samplerLocations_[uniformName] = loc;
455 break;
456
457 default:
458 REGEN_WARN("unknown shader type 0x"
459 << std::hex << type << std::dec << " for '" << uniformName << "'");
460 break;
461
462 }
463 }
464
465 std::set<int32_t> bindingPoints;
466 // uniform blocks
467 glGetProgramiv(id(), GL_ACTIVE_UNIFORM_BLOCKS, &count);
468 for (int32_t blockIndex = 0; blockIndex < count; ++blockIndex) {
469 // Note: uniforms inside a uniform block do not have individual uniform locations,
470 // but each block has an index within the shader program.
471 glGetActiveUniformBlockName(id(), blockIndex, 320, &arraySize, nameC);
472 std::string blockName(truncPrefix(nameC, "in_"));
473
474 // request a binding point for this block
475 std::uintptr_t bufferRangeID = 0;
476 auto needle = inputNames_.find(blockName);
477 if (needle != inputNames_.end()) {
478 bufferRangeID = reinterpret_cast<std::uintptr_t>(needle->second->in_.get());
479 }
480 int32_t bindingPoint = BindingManager::request(
481 BindingManager::UBO,
482 bufferRangeID,
483 blockName,
484 bindingPoints);
485 bindingPoints.insert(bindingPoint);
486
487 // remember the binding point, and associate a shader block index with this binding point
488 uniformLocations_[blockName] = bindingPoint;
489 uniformLocations_[REGEN_STRING("u_" << blockName)] = bindingPoint;
490 uniformLocations_[REGEN_STRING("in_" << blockName)] = bindingPoint;
491 if (needle != inputNames_.end()) {
492 uniforms_.emplace_back(needle->second->in_, bindingPoint);
493 }
494 glUniformBlockBinding(id(), blockIndex, bindingPoint);
495 }
496
497 // shader storage blocks
498 bindingPoints.clear();
499 glGetProgramInterfaceiv(id(),
500 GL_SHADER_STORAGE_BLOCK,
501 GL_ACTIVE_RESOURCES, &count);
502 for (int32_t blockIndex = 0; blockIndex < count; ++blockIndex) {
503 glGetProgramResourceName(id(),
504 GL_SHADER_STORAGE_BLOCK,
505 blockIndex, 320, &arraySize, nameC);
506 std::string blockName(truncPrefix(nameC, "in_"));
507
508 // request a binding point for this block
509 std::uintptr_t bufferRangeID = 0;
510 auto needle = inputNames_.find(blockName);
511 if (needle != inputNames_.end()) {
512 bufferRangeID = reinterpret_cast<std::uintptr_t>(needle->second->in_.get());
513 }
514 int32_t bindingPoint = BindingManager::request(
515 BindingManager::SSBO,
516 bufferRangeID,
517 blockName,
518 bindingPoints);
519 bindingPoints.insert(bindingPoint);
520
521 // remember the binding point, and associate a shader block index with this binding point
522 uniformLocations_[blockName] = bindingPoint;
523 uniformLocations_[REGEN_STRING("in_" << blockName)] = bindingPoint;
524 if (needle != inputNames_.end()) {
525 uniforms_.emplace_back(needle->second->in_, bindingPoint);
526 }
527 glShaderStorageBlockBinding(id(), blockIndex, bindingPoint);
528 }
529
530 glGetProgramiv(id(), GL_ACTIVE_ATTRIBUTES, &count);
531 for (int32_t loc_ = 0; loc_ < count; ++loc_) {
532 glGetActiveAttrib(id(), loc_, 320, nullptr, &arraySize, &type, nameC);
533 int32_t loc = glGetAttribLocation(id(), nameC);
534
535 // remember this attribute location
536 std::string attName(nameC);
537 if (hasPrefix(attName, "gl_")) { continue; }
538 if (boost::ends_with(attName, "[0]")) {
539 attName = attName.substr(0, attName.size() - 3);
540 }
541 if (hasPrefix(attName, "a_")) {
542 attName = truncPrefix(attName, "a_");
543 } else if (hasPrefix(attName, "vs_")) {
544 attName = truncPrefix(attName, "vs_");
545 } else if (hasPrefix(attName, "in_")) {
546 attName = truncPrefix(attName, "in_");
547 }
548 uniformLocations_[std::string(nameC)] = loc;
549 attributeLocations_[attName] = loc;
550 attributeLocations_[REGEN_STRING("a_" << attName)] = loc;
551 attributeLocations_[REGEN_STRING("in_" << attName)] = loc;
552 attributeLocations_[REGEN_STRING("vs_" << attName)] = loc;
553
554 auto needle = inputNames_.find(attName);
555 if (needle != inputNames_.end()) {
556 attributes_.emplace_back(needle->second->in_, loc);
557 }
558 }

607void Shader::setInput(const ref_ptr<ShaderInput> &in, const std::string &name) {
608 std::string inputName = (name.empty() ? in->name() : name);
609 {
610 auto needle = inputNames_.find(inputName);
611 if (needle == inputNames_.end()) {
612 inputs_.emplace_back(in, inputName);
613 auto it = inputs_.end();
614 --it;
615 inputNames_[inputName] = it;
616 } else {
617 *needle->second = NamedShaderInput(in, inputName);
618 }
619 }
620 if (in->isVertexAttribute()) {
621 if (!in->hasData()) { return; }
622 auto needle = attributeLocations_.find(inputName);
623 if (needle != attributeLocations_.end()) {
624 attributes_.emplace_back(in, needle->second);
625 }
626 } else if (!in->isConstant()) {
627 auto needle = uniformLocations_.find(inputName);
628 if (needle != uniformLocations_.end()) {
629 uniforms_.emplace_back(in, needle->second);
630 }
631 }
