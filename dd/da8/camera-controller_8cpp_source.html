111void CameraController::updateStep(double dt) {
112 step_ = Vec3f::zero();
113 isMoving_ = moveForward_ || moveBackward_ || moveLeft_ || moveRight_ || moveUp_ || moveDown_;
114 const auto orientation = horizontalOrientation_ + meshHorizontalOrientation_;
115 isRotating_ = orientation != lastOrientation_;
116 lastOrientation_ = orientation;
117
118 if (isRotating_) {
119 rot_.setAxisAngle(Vec3f::up(), orientation);
120 Vec3f d = rot_.rotate(Vec3f::front());
121
122 dirXZ_ = Vec3f(d.x, 0.0f, d.z);
123 dirXZ_.normalize();
124 dirSidestep_ = dirXZ_.cross(Vec3f::up());
125 dirSidestep_.normalize();
126 }
127
128 if (moveForward_) {
129 stepForward(moveAmount_ * dt);
130 }
131 else if (moveBackward_) {
132 stepBackward(moveAmount_ * dt);
133 }
134 if (moveLeft_) {
135 stepLeft(moveAmount_ * dt);
136 }
137 else if (moveRight_) {
138 stepRight(moveAmount_ * dt);
139 }
140 else if (moveUp_) {
141 stepUp(moveAmount_ * dt * 0.5);
142 }
143 else if (moveDown_) {
144 stepDown(moveAmount_ * dt * 0.5);
145 }

148void CameraController::updateCameraPose() {
149 if (attachedToTransform_.get()) {
150 // Simple rotation matrix around up vector (0,1,0)
151 float cy = cos(horizontalOrientation_), sy = sin(horizontalOrientation_);
152 matVal_.x[0] = cy;
153 matVal_.x[2] = sy;
154 matVal_.x[8] = -sy;
155 matVal_.x[10] = cy;
156 // Translate to camera position
157 matVal_.x[12] -= pos_.x;
158 matVal_.x[13] -= pos_.y;
159 matVal_.x[14] -= pos_.z;
160 pos_ = Vec3f::zero();
161
162 if(attachedToTransform_->hasModelMat()) {
163 attachedToTransform_->setModelMat(0, matVal_);
164 } else {
165 attachedToTransform_->setModelOffset(0, matVal_.position());
166 }
167
168 camPos_ = meshEyeOffset_;
169 if (attachedToMesh_.get()) {
170 camPos_ += attachedToMesh_->centerPosition();
171 }
172 camPos_ = matVal_.mul_t31(camPos_);
173 } else {
174 camPos_ = pos_;
175 }
176
177 if (isThirdPerson()) {
178 meshPos_ = camPos_;
179 rot_.setAxisAngle(Vec3f::up(), horizontalOrientation_ + meshHorizontalOrientation_);
180 Vec3f dir = rot_.rotate(Vec3f::front());
181 rot_.setAxisAngle(dir.cross(Vec3f::up()), verticalOrientation_);
182 camPos_ -= rot_.rotate(dir * meshDistance_);
183 camDir_ = meshPos_ - camPos_;
184 camDir_.normalize();
185 } else {
186 rot_.setAxisAngle(Vec3f::up(), horizontalOrientation_ + meshHorizontalOrientation_);
187 camDir_ = rot_.rotate(Vec3f::front());
188 rot_.setAxisAngle(dirSidestep_, verticalOrientation_);
189 camDir_ = rot_.rotate(camDir_);
190 }
191}
