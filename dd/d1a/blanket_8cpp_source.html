89void Blanket::updateLifetime(float deltaSeconds) {
90 if (blanketLifetimeMax_ <= 0.0f) {
91 return; // infinite lifetime
92 }
93 float delta = deltaSeconds / blanketLifetimeMax_;
94 for (uint32_t i = 0; i < numBlankets_; ++i) {
95 if (blanketLifetime_[i] <= 0.0f) continue;
96 blanketLifetime_[i] -= delta;
97 if (blanketLifetime_[i] <= 0.0f) {
98 blanketLifetime_[i] = 0.0f;
99 deadBlankets_[numDeadBlankets_++] = i;
100 if (!indexedShapes_->empty()) {
101 // Disable rendering etc. of dead blankets.
102 // - the spatial index will not report any intersections with them.
103 indexedShape(i)->setTraversalMask(0); // disable rendering etc.
104 }
105 }
106 }

109uint32_t Blanket::reviveBlanket() {
110 if (numDeadBlankets_ == 0) {
111 REGEN_WARN("No dead blankets to revive!");
112 return 0; // no dead blankets
113 } else {
114 --numDeadBlankets_;
115 auto idx = deadBlankets_[numDeadBlankets_];
116 blanketLifetime_[idx] = 1.0f;
117
118 const boost::posix_time::ptime &currentTime = systemTime_->p_time;
119 static constexpr float timeScale = 1.0f / 1e+6f;
120 timeOfBirth_->setVertex(idx, currentTime.time_of_day().total_microseconds() * timeScale);
121
122 // reset traversal mask
123 if (!indexedShapes_->empty()) {
124 indexedShape(idx)->setTraversalMask(blanketTraversalMask_);
125 }
