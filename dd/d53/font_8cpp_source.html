52Font::Font(const std::string &fontPath, uint32_t size, uint32_t dpi)
53 : fontPath_(fontPath),
54 size_(size),
55 dpi_(dpi),
56 lineHeight_(1.0f / .83f) {
57 // The object in which Freetype holds information on a given
58 // font is called a "face".
59 FT_Face face;
60
61 if (!boost::filesystem::exists(fontPath)) {
62 throw Error(REGEN_STRING(
63 "Unable to font file at '" << fontPath << "'."));
64 }
65
66 // This is where we load in the font information from the file.
67 // Of all the places where the code might die, this is the most likely,
68 // as FT_New_Face will die if the font file does not exist or is somehow broken.
69 if (FT_New_Face(ftlib_, fontPath.c_str(), 0, &face)) {
70 throw Error(REGEN_STRING(
71 "FT_New_Face failed. " <<
72 "There is probably a problem with the font file at " << fontPath
73 << "."));
74 }
75 // Freetype measures font size in terms of 1/64ths of pixels.
76 // Thus, to make a font h pixels high, we need to request a size of h*64.
77 // (h << 6 is just a prettier way of writing h*64)
78 FT_Set_Char_Size(face, size << 6, size << 6, dpi, dpi);
79
80 // find the bounding box that can hold any glyph of this font,
81 // we need to find this box because all glyph textures must have the same size
82 // for Texture2DArray.
83 int pixels_x = ::FT_MulFix((face->bbox.xMax - face->bbox.xMin), face->size->metrics.x_scale);
84 int pixels_y = ::FT_MulFix((face->bbox.yMax - face->bbox.yMin), face->size->metrics.y_scale);
85 auto textureWidth = (uint32_t) (pixels_x / 64);
86 auto textureHeight = (uint32_t) (pixels_y / 64);
87 // make sure texture dimensions are multiple of 2
88 if (textureWidth % 2 != 0) textureWidth += 1;
89 if (textureHeight % 2 != 0) textureHeight += 1;
90
91 // remembers some geometry infos about glyphs
92 faceData_ = new FaceData[NUMBER_OF_GLYPHS];
93
94 // create a array texture for the glyphs
95 arrayTexture_ = ref_ptr<Texture2DArray>::alloc(GL_TEXTURE_2D_ARRAY, 1);
96 arrayTexture_->set_format(GL_RED);
97 arrayTexture_->set_internalFormat(GL_R8);
98 arrayTexture_->set_pixelType(GL_UNSIGNED_BYTE);
99 arrayTexture_->set_rectangleSize(textureWidth, textureHeight);
100 arrayTexture_->set_depth(NUMBER_OF_GLYPHS);
101 arrayTexture_->allocTexture();
102 // GL expects 4byte aligned rows
103 glPixelStorei(GL_UNPACK_ALIGNMENT, 1);
104 for (unsigned short i = 0; i < NUMBER_OF_GLYPHS; i++) {
105 initGlyph(face, i, textureWidth, textureHeight);
106 }
107 // reset to default
108 glPixelStorei(GL_UNPACK_ALIGNMENT, 4);
109 arrayTexture_->set_wrapping(TextureWrapping::create(GL_CLAMP_TO_BORDER));
110 arrayTexture_->set_filter(TextureFilter::create(GL_LINEAR));
111 arrayTexture_->set_swizzle(TextureSwizzle::create(GL_RED));
112
113 FT_Done_Face(face);
114}

127GLubyte *Font::invertPixmapWithAlpha(
128 const FT_Bitmap &bitmap, uint32_t width, uint32_t height) {
129 const unsigned int arraySize = width * height;
130 auto *inverse = new GLubyte[arraySize];
131 auto *inverse_ptr = inverse;
132
133 memset(inverse, 0x0, sizeof(GLubyte) * arraySize);
134
135 for (auto r = 0u; r < bitmap.rows; ++r) {
136 GLubyte *bitmap_ptr = &bitmap.buffer[bitmap.pitch * r];
137 for (auto p = 0u; p < bitmap.width; ++p) {
138 *inverse_ptr++ = *bitmap_ptr++;
139 }
140 inverse_ptr += (width - bitmap.width);
141 }
142
143 return inverse;

146void Font::initGlyph(FT_Face face, GLushort ch, uint32_t textureWidth, uint32_t textureHeight) {
147 FaceData &glyphData = faceData_[ch];
148 FT_Glyph glyph;
149 GLubyte *inverted;
150
151 // Load the Glyph for our character.
152 if (FT_Load_Glyph(face, FT_Get_Char_Index(face, ch), FT_LOAD_DEFAULT)) {
153 throw Error("FT_Load_Glyph failed");
154 }
155 // Move the face's glyph into a Glyph object.
156 if (FT_Get_Glyph(face->glyph, &glyph)) {
157 throw Error("FT_Get_Glyph failed");
158 }
159 // Convert the glyph to a bitmap.
160 FT_Glyph_To_Bitmap(&glyph, ft_render_mode_normal, 0, 1);
161
162 auto bitmap_glyph = (FT_BitmapGlyph) glyph;
163 // This reference will make accessing the bitmap easier
164 FT_Bitmap &bitmap = bitmap_glyph->bitmap;
165
166 {
167 auto bitmapWith = (float) bitmap.width;
168 auto bitmapHeight = (float) bitmap.rows;
169 float sizeFactor = 1.0f / (float) size();
170
171 glyphData.uvX = bitmapWith / ((float) textureWidth);
172 glyphData.uvY = bitmapHeight / ((float) textureHeight);
173 glyphData.height = bitmapHeight * sizeFactor;
174 glyphData.width = bitmapWith * sizeFactor;
175 glyphData.advanceX = (face->glyph->advance.x >> 6) * sizeFactor;
176 glyphData.left = bitmap_glyph->left * sizeFactor;
177 glyphData.top = bitmap_glyph->top * sizeFactor;
178 }
179
180 {
181 inverted = invertPixmapWithAlpha(bitmap, textureWidth, textureHeight);
182 arrayTexture_->updateSubImage((int) ch, inverted);
183 delete[]inverted;
184 }
185
186 FT_Done_Glyph(glyph);
187}
