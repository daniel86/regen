62Mat4f Astronomy::moonOrientation(const t_aTime &aTime, float latitude, float longitude) {
63 const t_julianDay t(jd(aTime));
64
65 float l, b;
66 Moon::opticalLibrations(t, l, b);
67
68 const Mat4f libLat = Mat4f::rotationMatrix(-_rad(b), 0, 0);
69 const Mat4f libLon = Mat4f::rotationMatrix(0, _rad(l), 0);
70
71 const float a = _rad(Moon::positionAngleOfAxis(t));
72 const float p = _rad(Moon::parallacticAngle(aTime, latitude, longitude));
73
74 const Mat4f zenith = Mat4f::rotationMatrix(0, 0, p - a);
75 const Mat4f R(libLat * libLon * zenith);
76 return R;
77}

79float Astronomy::earthShineIntensity(const t_aTime &aTime, float latitude, float longitude) {
80 auto m = moonPosition(aTime, latitude, longitude, false);
81 auto s = sunPosition(aTime, latitude, longitude, false);
82
83 // ("Multiple Light Scattering" - 1980 - Van de Hulst) and
84 // ("A Physically-Based Night Sky Model" - 2001 - Wann Jensen et al.) -> the 0.19 is the earth full intensity
85 const float ep = acos(s.dot(-m));
86 const float ep2 = ep * ep;
87 const float ep3 = ep * ep2;
88 const float Eem = -0.0061f * ep3 + 0.0289f * ep2 - 0.0105f * sin(ep);
89
90 return Eem;
91}

93Mat4f Astronomy::equToHorTransform(const t_aTime &aTime, float latitude, float longitude) {
94 auto s = static_cast<float>(siderealTime(aTime));
95 return Mat4f::scaleMatrix(Vec3f(-1, 1, 1))
96 * Mat4f::rotationMatrix(_rad(latitude) - _PI_2, 0, 0)
97 * Mat4f::rotationMatrix(0, 0, -_rad(s + longitude));
98}
