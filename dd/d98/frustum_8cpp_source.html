63void Frustum::update(const Vec3f &pos, const Vec3f &dir) {
64 Vec3f d = dir;
65 const float dLength = d.length();
66 if (dLength < 1e-6f) {
67 d = Vec3f::front();
68 REGEN_WARN("Invalid frustum direction " << dir);
69 } else {
70 d /= dLength;
71 }
72 localTransform_.setPosition(pos);
73 direction_->setVertex(0, d);
74 tfOrigin_ = pos;
75
76 if (fov > 0.0) {
77 updatePointsPerspective(pos, d);
78 } else {
79 updatePointsOrthogonal(pos, d);
80 }
81
82 // Top: nTr - nTl - fTl
83 planes[0].set(points[2], points[1], points[5]);
84 // Bottom: nBl - nBr - fBr
85 planes[1].set(points[3], points[0], points[4]);
86 // Left: ntL - nbL - fbL
87 planes[2].set(points[1], points[3], points[7]);
88 // Right: nbR - ntR - fbR
89 planes[3].set(points[0], points[2], points[4]);
90 // Near: nTl - nTr - nBr
91 planes[4].set(points[1], points[2], points[0]);
92 // Far: fTr - fTl - fBl
93 planes[5].set(points[6], points[5], points[7]);

96void Frustum::updatePointsPerspective(const Vec3f &pos, const Vec3f &dir) {
97 auto v = dir.cross(Vec3f::up());
98 const float vLength = v.length();
99 if (vLength < 1e-6f) {
100 // direction is parallel to up vector
101 v = dir.cross(Vec3f::right());
102 v.normalize();
103 } else {
104 v /= vLength;
105 }
106 auto u = v.cross(dir);
107 u.normalize();
108 // near plane points
109 auto nc = pos + dir * near;
110 auto rw = v * nearPlaneHalfSize.x;
111 auto uh = u * nearPlaneHalfSize.y;
112 points[0] = nc - uh + rw; // bottom right
113 points[1] = nc + uh - rw; // top left
114 points[2] = nc + uh + rw; // top right
115 points[3] = nc - uh - rw; // bottom left
116 // far plane points
117 auto fc = pos + dir * far;
118 rw = v * farPlaneHalfSize.x;
119 uh = u * farPlaneHalfSize.y;
120 points[4] = fc - uh + rw; // bottom right
121 points[5] = fc + uh - rw; // top left
122 points[6] = fc + uh + rw; // top right
123 points[7] = fc - uh - rw; // bottom left
124}

126void Frustum::updatePointsOrthogonal(const Vec3f &pos, const Vec3f &dir) {
127 auto v = dir.cross(Vec3f::up());
128 v.normalize();
129 auto u = v.cross(dir);
130 u.normalize();
131 // do not assume that the frustum is centered at the far/near plane centroids!
132 // could be that left/right/top/bottom are not symmetric
133 auto vl = v * orthoBounds.min.x; // left
134 auto vr = v * orthoBounds.max.x; // right
135 auto ub = u * orthoBounds.min.y; // bottom
136 auto ut = u * orthoBounds.max.y; // top
137 // near plane points
138 auto nc = pos + dir * near;
139 points[0] = nc + vr + ub; // bottom right
140 points[1] = nc + vl + ut; // top left
141 points[2] = nc + vr + ut; // top right
142 points[3] = nc + vl + ub; // bottom left
143 // far plane points
144 auto fc = pos + dir * far;
145 points[4] = fc + vr + ub; // bottom right
146 points[5] = fc + vl + ut; // top left
147 points[6] = fc + vr + ut; // top right
148 points[7] = fc + vl + ub; // bottom left
149}

151void Frustum::split(double splitWeight, std::vector<Frustum> &frustumSplit) const {
152 const auto &n = near;
153 const auto &f = far;
154 const auto count = frustumSplit.size();
155 auto ratio = f / n;
156 double si, lastn, currf, currn;
157
158 lastn = n;
159 for (uint32_t i = 1; i < count; ++i) {
160 si = i / (double) count;
161
162 // C_i = \lambda * C_i^{log} + (1-\lambda) * C_i^{uni}
163 currn = splitWeight * (n * (pow(ratio, si))) +
164 (1 - splitWeight) * (n + (f - n) * si);
165 currf = currn * 1.005;
166
167 frustumSplit[i - 1].setPerspective(aspect, fov, lastn, currf);
168
169 lastn = currn;
170 }
171 frustumSplit[count - 1].setPerspective(aspect, fov, lastn, f);
172}

201void Frustum::batchTest_Spheres(BatchedIntersectionCase &td) {
202 // Process numQueuedItems_ nodes from the queue, performing an intersection test with the shape's projection;
203 // and also writing nodeIdx to successor array if the test succeeds.
204 auto *shapeData = static_cast<IntersectionData_Frustum *>(td.shapeData);
205 const auto numQueued = static_cast<int32_t>(td.numQueued);
206
207 auto *batchData = static_cast<const BatchOfSpheres *>(td.batchData);
208 const float *d_spherePosX = batchData->posX().data();
209 const float *d_spherePosY = batchData->posY().data();
210 const float *d_spherePosZ = batchData->posZ().data();
211 const float *d_sphereRadius = batchData->radius().data();
212
213 int32_t queuedIdx = 0;
214 for (; queuedIdx + simd::RegisterWidth <= numQueued; queuedIdx += simd::RegisterWidth) {
215 // Load the sphere data into SIMD registers
216 const BatchOf_float spherePosX = BatchOf_float::loadAligned(d_spherePosX + queuedIdx);
217 const BatchOf_float spherePosY = BatchOf_float::loadAligned(d_spherePosY + queuedIdx);
218 const BatchOf_float spherePosZ = BatchOf_float::loadAligned(d_spherePosZ + queuedIdx);
219 const BatchOf_float sphereRadius = BatchOf_float::loadAligned(d_sphereRadius + queuedIdx);
220 // We'll accumulate a boolean vector as mask; init to true
221 BatchOf_float hasIntersection = BatchOf_float::allOnes();
222
223 for (int p = 0; p < NUM_FRUSTUM_PLANES; ++p) {
224 // tmp = n(dot)p + r
225 const BatchOf_float n_dot_p =
226 (spherePosX * shapeData->planes[p].x) +
227 (spherePosY * shapeData->planes[p].y) +
228 (spherePosZ * shapeData->planes[p].z) +
229 sphereRadius;
230 // (partially) inside if: (n_dot_p + r - w) > 0
231 hasIntersection &= (n_dot_p > shapeData->planes[p].w);
232 // early break if all lanes are dead
233 if (hasIntersection.isAllZero()) break;
234 }
235
236 // Convert survived lanes to bitmask value
237 uint8_t mask = hasIntersection.toBitmask8();
238 while (mask) {
239 int bitIndex = simd::nextBitIndex<uint8_t>(mask);
240 td.hits->push(td.queuedIndices[queuedIdx + bitIndex]);
241 }
242 }
243
244 // Scalar fallback for remaining items
245 for (; queuedIdx < numQueued; queuedIdx++) {
246 bool isOutside = false;
247 for (int p = 0; p < NUM_FRUSTUM_PLANES && !isOutside; ++p) {
248 // tmp = n(dot)p + r
249 float n_dot_p =
250 (d_spherePosX[queuedIdx] * shapeData->planes[p].x) +
251 (d_spherePosY[queuedIdx] * shapeData->planes[p].y) +
252 (d_spherePosZ[queuedIdx] * shapeData->planes[p].z) +
253 d_sphereRadius[queuedIdx];
254 // fully outside if: (n_dot_p + r - w) < 0
255 isOutside = (n_dot_p < shapeData->planes[p].w);
256 }
257 if (!isOutside) {
258 td.hits->push(td.queuedIndices[queuedIdx]);
259 }
260 }
261}

263bool Frustum::hasIntersectionWithAABB(const AABB &box) const {
264 auto &boxBatchData = box.globalBatchData_t();
265 auto boxIdx = box.globalIndex();
266 const float *boxMinX = boxBatchData.minX().data() + boxIdx;
267 const float *boxMinY = boxBatchData.minY().data() + boxIdx;
268 const float *boxMinZ = boxBatchData.minZ().data() + boxIdx;
269 const float *boxMaxX = boxBatchData.maxX().data() + boxIdx;
270 const float *boxMaxY = boxBatchData.maxY().data() + boxIdx;
271 const float *boxMaxZ = boxBatchData.maxZ().data() + boxIdx;
272
273 for (unsigned int i = 0u; i < 6u; ++i) {
274 // Select vertex farthest from the plane in direction of the plane normal.
275 // If this point is behind the plane, the AABB must be outside of the frustum.
276 auto &plane = planes[i];
277 auto &n = plane.normal();
278 const Vec3f p(
279 n.x > 0.0 ? *boxMinX : *boxMaxX,
280 n.y > 0.0 ? *boxMinY : *boxMaxY,
281 n.z > 0.0 ? *boxMinZ : *boxMaxZ);
282 if (n.dot(p) - plane.coefficients.w < 0.0) {
283 // AABB is outside the frustum
284 return false;
285 }
286 }
287 return true;

290void Frustum::batchTest_AABBs(BatchedIntersectionCase &td) {
291 auto *shapeData = static_cast<IntersectionData_Frustum *>(td.shapeData);
292 const auto numQueued = static_cast<int32_t>(td.numQueued);
293
294 auto *batchData = static_cast<const BatchOfAABBs *>(td.batchData);
295 const float *d_aabbMinX = batchData->minX().data();
296 const float *d_aabbMinY = batchData->minY().data();
297 const float *d_aabbMinZ = batchData->minZ().data();
298 const float *d_aabbMaxX = batchData->maxX().data();
299 const float *d_aabbMaxY = batchData->maxY().data();
300 const float *d_aabbMaxZ = batchData->maxZ().data();
301
302 int32_t queuedIdx = 0;
303 for (; queuedIdx + simd::RegisterWidth <= numQueued; queuedIdx += simd::RegisterWidth) {
304 // Load the AABB data into SIMD registers
305 const BatchOf_float aabbMinX = BatchOf_float::loadAligned(d_aabbMinX + queuedIdx);
306 const BatchOf_float aabbMinY = BatchOf_float::loadAligned(d_aabbMinY + queuedIdx);
307 const BatchOf_float aabbMinZ = BatchOf_float::loadAligned(d_aabbMinZ + queuedIdx);
308 const BatchOf_float aabbMaxX = BatchOf_float::loadAligned(d_aabbMaxX + queuedIdx);
309 const BatchOf_float aabbMaxY = BatchOf_float::loadAligned(d_aabbMaxY + queuedIdx);
310 const BatchOf_float aabbMaxZ = BatchOf_float::loadAligned(d_aabbMaxZ + queuedIdx);
311 // We'll accumulate a boolean vector as mask; init to true
312 BatchOf_float hasIntersection = BatchOf_float::allOnes();
313
314 for (int p = 0; p < NUM_FRUSTUM_PLANES; ++p) {
315 // Select vertex farthest from the plane in direction of the plane normal.
316 // If this point is behind the plane, the AABB must be outside the frustum.
317 const BatchOf_float px = shapeData->planes[p].x < 0.0f ? aabbMinX : aabbMaxX;
318 const BatchOf_float py = shapeData->planes[p].y < 0.0f ? aabbMinY : aabbMaxY;
319 const BatchOf_float pz = shapeData->planes[p].z < 0.0f ? aabbMinZ : aabbMaxZ;
320 // tmp = n(dot)p
321 const BatchOf_float n_dot_p =
322 (px * shapeData->planes[p].x) +
323 (py * shapeData->planes[p].y) +
324 (pz * shapeData->planes[p].z);
325 hasIntersection &= (n_dot_p > shapeData->planes[p].w);
326 // early break if all lanes are dead
327 if (hasIntersection.isAllZero()) break;
328 }
329
330 // Convert survived lanes to bitmask value
331 uint8_t mask = hasIntersection.toBitmask8();
332 while (mask) {
333 int bitIndex = simd::nextBitIndex<uint8_t>(mask);
334 td.hits->push(td.queuedIndices[queuedIdx + bitIndex]);
335 }
336 }
337
338 // Scalar fallback for remaining items
339 for (; queuedIdx < numQueued; queuedIdx++) {
340 bool isOutside = false;
341 for (unsigned int p = 0u; p < NUM_FRUSTUM_PLANES && !isOutside; ++p) {
342 // Select vertex farthest from the plane in direction of the plane normal.
343 // If this point is behind the plane, the AABB must be outside of the frustum.
344 const Vec3f farthest(
345 shapeData->planes[p].x < 0.0 ? d_aabbMinX[queuedIdx] : d_aabbMaxX[queuedIdx],
346 shapeData->planes[p].y < 0.0 ? d_aabbMinY[queuedIdx] : d_aabbMaxY[queuedIdx],
347 shapeData->planes[p].z < 0.0 ? d_aabbMinZ[queuedIdx] : d_aabbMaxZ[queuedIdx]);
348 const float n_dot_p = (
349 shapeData->planes[p].x * farthest.x +
350 shapeData->planes[p].y * farthest.y +
351 shapeData->planes[p].z * farthest.z);
352 // Completely outside
353 isOutside = (n_dot_p < shapeData->planes[p].w);
354 }
355 if (!isOutside) {
356 td.hits->push(td.queuedIndices[queuedIdx]);
357 }
358 }

378void Frustum::batchTest_OBBs(BatchedIntersectionCase &td) {
379 auto *shapes = td.indexedShapes->data();
380 auto *frustum = static_cast<IntersectionData_Frustum *>(td.shapeData);
381 const auto numQueued = static_cast<int32_t>(td.numQueued);
382
383 auto *batchData = static_cast<const BatchOfOBBs *>(td.batchData);
384 const float *d_obbCenterX = batchData->centerX().data();
385 const float *d_obbCenterY = batchData->centerY().data();
386 const float *d_obbCenterZ = batchData->centerZ().data();
387 const float *d_obbHalfSizeX = batchData->halfSizeX().data();
388 const float *d_obbHalfSizeY = batchData->halfSizeY().data();
389 const float *d_obbHalfSizeZ = batchData->halfSizeZ().data();
390 auto d_axes = batchData->axes();
391
392 int32_t queuedIdx = 0;
393 for (; queuedIdx + simd::RegisterWidth <= numQueued; queuedIdx += simd::RegisterWidth) {
394 // Load some of the ABB data into SIMD registers, too many registers are needed so
395 // we leave out the axes and rather load them in a loop below.
396 const BatchOf_float obbCenterX = BatchOf_float::loadAligned(d_obbCenterX + queuedIdx);
397 const BatchOf_float obbCenterY = BatchOf_float::loadAligned(d_obbCenterY + queuedIdx);
398 const BatchOf_float obbCenterZ = BatchOf_float::loadAligned(d_obbCenterZ + queuedIdx);
399 const BatchOf_float obbHalfSize[3] = {
400 BatchOf_float::loadAligned(d_obbHalfSizeX + queuedIdx),
401 BatchOf_float::loadAligned(d_obbHalfSizeY + queuedIdx),
402 BatchOf_float::loadAligned(d_obbHalfSizeZ + queuedIdx)};
403 // We'll accumulate a boolean vector as mask; init to true
404 BatchOf_float hasIntersection = BatchOf_float::allOnes();
405
406 for (unsigned int planeIdx = 0u; planeIdx < NUM_FRUSTUM_PLANES; ++planeIdx) {
407 const auto &plane = frustum->planes[planeIdx];
408 // center-to-plane distance + projected radius, radius is initially 0
409 BatchOf_float dr =
410 (obbCenterX * plane.x) +
411 (obbCenterY * plane.y) +
412 (obbCenterZ * plane.z) -
413 BatchOf_float::fromScalar(plane.w);
414 // Accumulate projected radius from each OBB axis
415 for (uint32_t axisIdx=0u; axisIdx < 3u; ++axisIdx) {
416 // Load axis into SIMD registers
417 const auto &axisBatch = d_axes[axisIdx];
418 const BatchOf_float axisX = BatchOf_float::loadAligned(axisBatch.x.data() + queuedIdx);
419 const BatchOf_float axisY = BatchOf_float::loadAligned(axisBatch.y.data() + queuedIdx);
420 const BatchOf_float axisZ = BatchOf_float::loadAligned(axisBatch.z.data() + queuedIdx);
421 // Projected radius
422 dr += obbHalfSize[axisIdx] *
423 (axisX*plane.x + axisY*plane.y + axisZ*plane.z).abs();
424 }
425 hasIntersection &= dr.isPositive();
426 // early break if all lanes are dead
427 if (hasIntersection.isAllZero()) break;
428 }
429
430 // Convert survived lanes to bitmask value
431 uint8_t mask = hasIntersection.toBitmask8();
432 while (mask) {
433 int bitIndex = simd::nextBitIndex<uint8_t>(mask);
434 td.hits->push(td.queuedIndices[queuedIdx + bitIndex]);
435 }
436 }
437
438 // Scalar fallback for remaining items
439 for (; queuedIdx < numQueued; queuedIdx++) {
440 auto itemIdx = td.queuedIndices[queuedIdx];
441
442 const float *axis0X = d_axes[0].x.data() + queuedIdx;
443 const float *axis0Y = d_axes[0].y.data() + queuedIdx;
444 const float *axis0Z = d_axes[0].z.data() + queuedIdx;
445 const float *axis1X = d_axes[1].x.data() + queuedIdx;
446 const float *axis1Y = d_axes[1].y.data() + queuedIdx;
447 const float *axis1Z = d_axes[1].z.data() + queuedIdx;
448 const float *axis2X = d_axes[2].x.data() + queuedIdx;
449 const float *axis2Y = d_axes[2].y.data() + queuedIdx;
450 const float *axis2Z = d_axes[2].z.data() + queuedIdx;
451
452 const OBB &obb = *static_cast<OBB *>(shapes[itemIdx].get());
453 const Vec3f &obbCenter = obb.tfOrigin();
454
455 bool isOutside = false;
456 for (unsigned int planeIdx = 0u; planeIdx < NUM_FRUSTUM_PLANES && !isOutside; ++planeIdx) {
457 const auto &plane = frustum->planes[planeIdx];
458 const Vec3f &n = plane.xyz();
459 // center-to-plane distance + projected radius
460 float dr = n.dot(obbCenter) - plane.w +
461 d_obbHalfSizeX[queuedIdx] * std::abs(n.dot(axis0X, axis0Y, axis0Z)) +
462 d_obbHalfSizeY[queuedIdx] * std::abs(n.dot(axis1X, axis1Y, axis1Z)) +
463 d_obbHalfSizeZ[queuedIdx] * std::abs(n.dot(axis2X, axis2Y, axis2Z));
464 // completely outside
465 isOutside = (dr < 0.0f);
466 }
467 if (!isOutside) {
468 td.hits->push(itemIdx);
469 }
470 }
