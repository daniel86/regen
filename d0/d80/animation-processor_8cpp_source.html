27 auto repeat = input.getValue<std::string>("repeat", "true") == "true";
28 auto meshID = input.getValue("mesh-id");
29 auto friction = input.getValue<float>("friction", 8.0f);
30 auto frequency = input.getValue<float>("frequency", 5.0f);
31 auto compositeMesh = scene->getResource<CompositeMesh>(meshID);
32
33 // configure per-attribute interpolation functions, default is linear
34 auto interpolationsCfg = input.getFirstChild("interpolations");
35 std::list<MeshAnimation::Interpolation> interpolations;
36 if (interpolationsCfg.get() != nullptr) {
37 for (auto it = interpolationsCfg->getChildren().begin();
38 it != interpolationsCfg->getChildren().end(); ++it) {
39 auto &interpolationCfg = *it;
40 auto attributeName = interpolationCfg->getValue("target");
41 auto functionName = interpolationCfg->getValue("function");
42 interpolations.emplace_back(attributeName, "interpolate_" + functionName);
43 }
44 }
45

47 for (const auto &mesh: compositeMesh->meshes()) {
48 // create the animation
49 auto animation = ref_ptr<MeshAnimation>::alloc(mesh, interpolations);
50 animation->setFriction(friction);
51 animation->setFrequency(frequency);
52 if (!input.getName().empty()) {
53 animation->setAnimationName(input.getName());
54 }
55
56 // configure key frames
57 double animationTime = 0.0;
58 auto framesCfg = input.getFirstChild("frames");
59 for (const auto &frameCfg: framesCfg->getChildren()) {
60 auto timeInTicks = frameCfg->getValue<double>("duration", 0.0);
61 auto shape = frameCfg->getValue("shape");
62 if (shape.empty()) {
63 REGEN_WARN("Skipping deformation frame without \"shape\".");
64 continue;
65 }
66 if (shape == "original") {
67 animation->addMeshFrame(timeInTicks);
68 } else if (shape == "sphere") {
69 auto h_radius = frameCfg->getValue<float>("radius", 0.5f);
70 auto v_radius = frameCfg->getValue<float>("radius", 0.5f);
71 auto offset = frameCfg->getValue<Vec3f>("offset", Vec3f(0.0f, 0.0f, 0.0f));
72 animation->addSphereAttributes(h_radius, v_radius, timeInTicks, offset);
73 } else if (shape == "box") {
74 auto width = frameCfg->getValue<float>("width", 1.0f);
75 auto height = frameCfg->getValue<float>("height", 1.0f);
76 auto depth = frameCfg->getValue<float>("depth", 1.0f);
77 auto offset = frameCfg->getValue<Vec3f>("offset", Vec3f(0.0f, 0.0f, 0.0f));
78 animation->addBoxAttributes(width, height, depth, timeInTicks, offset);
79 } else {
80 REGEN_WARN("Ignoring frame with unknown shape '" << shape << "'.");
81 continue;
82 }
83 animationTime += timeInTicks;
84 }
85 Vec2d tickRange = {0.0, animationTime};
86
87 mesh->addAnimation(animation);
88 animation->setTickRange(tickRange);
89 if (repeat) {

91 animation->connect(Animation::ANIMATION_STOPPED, animRepeater);
92 }
93 animations.push_back(animation);
94 }
