65bool StagingBuffer::resizeBuffer(uint32_t segmentSize, uint32_t numRingSegments) {
66 uint32_t numSegments = (flags_.bufferingMode == RING_BUFFER ?
67 numRingSegments :
68 (uint32_t) flags_.bufferingMode);
69 numSegments = std::max(numSegments, 1u); // ensure at least one segment
70
71 // initialize the buffer indices
72 if (numSegments > 1) {
73 writeBufferIndex_ = 1;
74 readBufferIndex_ = 0;
75 }
76 if (bufferSegments_.size() != numSegments) {
77 bufferSegments_.resize(numSegments);
78 }
79
80 if (segmentSize < segmentSize_) {
81 // if size is reduced, keep using the adopted range as long as the size reduction is not too large.
82 auto halfSegmentSize = segmentSize_ / 2;
83 if (segmentSize > halfSegmentSize) {
84 // update the offset for each segment
85 for (uint32_t segmentIdx = 0u; segmentIdx < numSegments; segmentIdx++) {
86 bufferSegments_[segmentIdx].offset = segmentIdx * segmentSize_;
87 }
88 return true; // no need to resize, just keep the existing segments
89 }
90 }
91 segmentSize_ = segmentSize;
92
93 // allocate data array for reading
94 if (storageFlags_ & MAP_READ) {
95 delete[] stagingReadData_;
96 stagingReadData_ = new byte[segmentSize_];
97 }
98 // update the offset for each segment
99 for (uint32_t segmentIdx = 0u; segmentIdx < numSegments; segmentIdx++) {
100 bufferSegments_[segmentIdx].offset = segmentIdx * segmentSize_;
101 }
102
103 // obtain CPU accessible buffer
104 if (flags_.useExplicitStaging()) {
105 // in case of explicit staging, we need obtain separate CPU-accessible storage.
106 // initialize storage for multiple buffers: numSegments*segmentSize_ bytes
107 if (stagingRef_.get()) {
108 BufferObject::orphanBufferRange(stagingRef_.get());
109 }
110#ifdef REGEN_USE_STAGING_ALLOCATOR
111 stagingRef_ = BufferObject::adoptBufferRange(
112 segmentSize_ * numSegments,
113 getStagingAllocator(storageMode_));
114#else
115 stagingRef_ = stagingBO_->adoptBufferRange(
116 segmentSize_ * numSegments);
117#endif
118 if (!stagingRef_->mappedData() && (storageFlags_ & MAP_PERSISTENT)) {
119 REGEN_ERROR("Failed to map buffer " <<
120 " target: " << flags_.target <<
121 " map: " << flags_.mapMode <<
122 " access: " << flags_.accessMode <<
123 " buffering: " << flags_.bufferingMode <<
124 " sync: " << flags_.syncFlags);
125 GL_ERROR_LOG();
126 stagingRef_ = {};
127 return false;
128 }
129 if (clearBufferOnResize_) {
130 // clear the buffer to zero
131 stagingBO_->setBuffersToZero();
132 }
133 stagingCopyRange_.srcBufferID = stagingRef_->bufferID();
134 } else {
135 stagingRef_ = {};
136 }
137 REGEN_DEBUG("Resized staging buffer to "
138 << segmentSize_ / 1024.0 << " KiB per segment, "
139 << numSegments << " segments, total: "
140 << (segmentSize_ * numSegments) / 1024.0 << " KiB "
141 << " swap: " << useSwappingOnAccess_
142 << " clear: " << clearBufferOnResize_
143 << " max segments: " << maxRingSegments_);
144
145 return true;
146}

185void StagingBuffer::pushToFlushQueue(const BufferRange2ui *dirtySegments, uint32_t numDirtySegments) {
186#ifndef REGEN_STAGING_USE_DIRECT_FLUSHING
187 if (flags_.mapMode == BUFFER_MAP_PERSISTENT_FLUSH) {
188 RingSegment &writeSegment = bufferSegments_[writeBufferIndex_];
189 const uint32_t totalDirtySegments = writeSegment.numDirtySegments + numDirtySegments;
190 // ensure the vector has enough space
191 if (totalDirtySegments > writeSegment.dirtySegments.size()) {
192 writeSegment.dirtySegments.resize(totalDirtySegments);
193 }
194 // copy the dirty segments into the vector
195 auto *dataStart = writeSegment.dirtySegments.data() + writeSegment.numDirtySegments;
196 std::memcpy(
197 (byte *) dataStart,
198 (byte *) dirtySegments,
199 numDirtySegments * sizeof(BufferRange2ui));
200 // finally increment the number of dirty segments
201 writeSegment.numDirtySegments = totalDirtySegments;
202 }
203#endif

217void StagingBuffer::setSubData(
218 const ref_ptr<BufferReference> &drawBufferRef,
219 uint32_t localOffset,
220 uint32_t dataSize,
221 const byte *data) {
222 RingSegment &writeSegment = bufferSegments_[writeBufferIndex_];
223 const ref_ptr<BufferReference> &targetRef = (stagingRef_.get() ? stagingRef_ : drawBufferRef);
224
225 if (storageFlags_ & MAP_PERSISTENT) {
226 // in case of persistent mapping, we can write directly to the mapped pointer
227 byte *mappedPtr = getMappedSegment(targetRef, writeSegment.offset);
228 if (mappedPtr) {
229 std::memcpy(mappedPtr + localOffset, data, dataSize);
230 } else {
231 REGEN_ERROR("Failed to write to persistent mapped buffer segment.");
232 GL_ERROR_LOG();
233 }
234 } else if (flags_.isWritable()) {
235 // in case of non-persistent mapping, we need to copy the data into the buffer without mapping.
236 glNamedBufferSubData(
237 targetRef->bufferID(),
238 targetRef->address() + writeSegment.offset + localOffset,
239 dataSize, data);
240 } else {
241 // in case of non-writable buffer, we need to adopt a writable buffer range then do buffer-to-buffer copy.
242 auto tempRef = BufferObject::adoptBufferRange(
243 dataSize,
244 BufferObject::bufferPool(flags_.target, BUFFER_MODE_STATIC_WRITE));
245 glNamedBufferSubData(
246 tempRef->bufferID(),
247 tempRef->address(),
248 dataSize,
249 data);
250 glCopyNamedBufferSubData(
251 tempRef->bufferID(),
252 targetRef->bufferID(),
253 tempRef->address(),
254 targetRef->address() + writeSegment.offset + localOffset,
255 dataSize);
256 }

259byte *StagingBuffer::beginMappedWrite(
260 const ref_ptr<BufferReference> &drawBufferRef,
261 bool isPartialWrite,
262 uint32_t localOffset,
263 uint32_t mappedSize) {
264 RingSegment &writeSegment = bufferSegments_[writeBufferIndex_];
265 const ref_ptr<BufferReference> &targetRef = (stagingRef_.get() ? stagingRef_ : drawBufferRef);
266
267 if (storageFlags_ & MAP_PERSISTENT) {
268 // if we have a persistent mapping, we need to wait for the fence.
269 // the fence marks the point where the segment we want to write to was consumed.
270 // in case of explicit staging, this is the point where staging buffer was copied to the GPU buffer.
271 if (flags_.useSyncFences() &&
272 !writeSegment.fence.wait(flags_.useFrameDropping())) {
273 return nullptr; // drop frame
274 }
275 byte *mappedPtr = getMappedSegment(targetRef, writeSegment.offset);
276 if (mappedPtr) {
277 return mappedPtr + localOffset;
278 } else {
279 REGEN_ERROR("Failed to map persistent buffer segment.");
280 GL_ERROR_LOG();
281 return nullptr; // failed to map the segment
282 }
283 } else {
284 // this case is for non-persistent mapping, where we need to map the buffer segment
285 // for each write operation.
286 GLbitfield mappingFlags = accessFlags_;
287 if (!isPartialWrite) { mappingFlags |= MAP_INVALIDATE_RANGE; }
288 byte *mappedPtr = (byte *) glMapNamedBufferRange(
289 targetRef->bufferID(),
290 targetRef->address() + writeSegment.offset + localOffset,
291 mappedSize,
292 mappingFlags);
293 if (mappedPtr) {
294 return mappedPtr;
295 } else {
296 REGEN_ERROR("Failed to temporary map buffer segment.");
297 GL_ERROR_LOG();
298 return nullptr; // failed to map the segment
299 }
300 }

303void StagingBuffer::endMappedWrite(
304 const ref_ptr<BufferReference> &drawBufferRef,
305 BufferRange &nextDrawBufferRange,
306 uint32_t localOffset) {
307 RingSegment &writeSegment = bufferSegments_[writeBufferIndex_];
308 RingSegment &readSegment = bufferSegments_[readBufferIndex_];
309 writeSegment.hasData = true;
310
311 const bool hasStagingRef = stagingRef_.get() != nullptr;
312 const ref_ptr<BufferReference> &targetRef = (hasStagingRef ? stagingRef_ : drawBufferRef);
313
314 if ((storageFlags_ & MAP_PERSISTENT) == 0) {
315 // non-persistent mapping
316 glUnmapNamedBuffer(targetRef->bufferID());
317 }
318#ifdef REGEN_STAGING_USE_DIRECT_FLUSHING
319 else if (accessFlags_ & MAP_FLUSH_EXPLICIT) {
320 // direct flushing
321 RingSegment &writeSegment = bufferSegments_[writeBufferIndex_];
322 glFlushMappedNamedBufferRange(targetRef->bufferID(),
323 writeSegment.offset + localOffset,
324 drawBufferRef->allocatedSize());
325 }
326#endif
327
328 if (flags_.useExplicitStaging()) {
329#ifndef REGEN_STAGING_USE_DIRECT_FLUSHING
330 // Make sure the last write to current readBuffer is flushed before we copy the data.
331 if (accessFlags_ & MAP_FLUSH_EXPLICIT) {
332 for (uint32_t flushIdx = 0; flushIdx < readSegment.numDirtySegments; ++flushIdx) {
333 // get the segment to flush
334 const BufferRange2ui &flushSegment = readSegment.dirtySegments[flushIdx];
335 glFlushMappedNamedBufferRange(
336 targetRef->bufferID(),
337 readSegment.offset + localOffset + flushSegment.offset,
338 flushSegment.size);
339 }
340 readSegment.numDirtySegments = 0; // reset the dirty segments
341 }
342#endif
343 // NOTE: We delay copy to draw buffer until we reach a read segment that has been written to.
344 // This causes some frames of delay until the upload starts. The draw buffer best has some
345 // meaningful initial value that can be drawn first few frames!
346 if (readSegment.hasData) {
347 // copy data from staging buffer to the draw buffer
348 if (hasStagingRef) {
349 // Let the staging system handle the copy, it may be able
350 // to optimize the copy operation, e.g. by coalescing multiple copies
351 // into a single copy operation.
352 auto &staging = StagingSystem::instance();
353 stagingCopyRange_.dstBufferID = drawBufferRef->bufferID();
354 stagingCopyRange_.srcOffset = targetRef->address() + readSegment.offset + localOffset;
355 stagingCopyRange_.dstOffset = drawBufferRef->address();
356 stagingCopyRange_.size = drawBufferRef->allocatedSize();
357 staging.scheduledCopy(stagingCopyRange_);
358 } else {
359 glCopyNamedBufferSubData(
360 targetRef->bufferID(),
361 drawBufferRef->bufferID(),
362 targetRef->address() + readSegment.offset + localOffset,
363 drawBufferRef->address(),
364 drawBufferRef->allocatedSize());
365 if (storageFlags_ & MAP_PERSISTENT && flags_.useSyncFences()) {
366 // Create a fence just after glCopyNamedBufferSubData -- marking the point where the
367 // written data of this frame has been consumed by the GPU.
368 readSegment.fence.setFencePoint();
369 }
370 }
371 }
372 nextDrawBufferRange.offset_ = drawBufferRef->address();
373 nextDrawBufferRange.segment_ = 0;
374 } else {
375 // NOTE: in case of shared buffers, the draw calls consuming the buffer need to
376 // set the fence point
377 nextDrawBufferRange.offset_ = drawBufferRef->address() + readSegment.offset + localOffset;
378 nextDrawBufferRange.segment_ = readBufferIndex_;
379 }
380
381 // finally swap buffers in case of double/triple buffering
382 if (useSwappingOnAccess_) swapBuffers();
383}

420bool StagingBuffer::readBuffer(
421 const ref_ptr<BufferReference> &drawBufferRef,
422 BufferRange &nextDrawBufferRange,
423 uint32_t localOffset) {
424 RingSegment &nextCopySegment = bufferSegments_[writeBufferIndex_];
425 RingSegment &nextDrawSegment = bufferSegments_[readBufferIndex_];
426 const ref_ptr<BufferReference> &stagingRef = (stagingRef_.get() ? stagingRef_ : drawBufferRef);
427
428 if (flags_.useExplicitStaging()) {
429 // copy data from GPU buffer to staging buffer
430 glCopyNamedBufferSubData(
431 drawBufferRef->bufferID(),
432 stagingRef->bufferID(),
433 drawBufferRef->address(),
434 stagingRef->address() + nextDrawSegment.offset + localOffset,
435 drawBufferRef->allocatedSize());
436 if (storageFlags_ & MAP_PERSISTENT && flags_.useSyncFences()) {
437 // mark the point where copy to staging buffer was done.
438 nextDrawSegment.fence.setFencePoint();
439 }
440 }
441 nextDrawSegment.hasData = true;
442
443 // copy data from staging buffer to CPU memory
444 if (nextCopySegment.hasData) {
445 if (storageFlags_ & MAP_PERSISTENT) {
446 byte *mappedPtr = getMappedSegment(stagingRef, nextCopySegment.offset);
447 if (!mappedPtr) {
448 return false; // ERROR: "Failed to map buffer for reading"
449 }
450 // wait until the last glCopyNamedBufferSubData into this segment is done.
451 if (flags_.useSyncFences() && !nextCopySegment.fence.wait(flags_.useFrameDropping())) {
452 return true; // drop frame
453 }
454 // need to invalidate the range before copying if using persistent mapping
455 // without coherent bit.
456 if (flags_.mapMode == BUFFER_MAP_PERSISTENT_FLUSH) {
457 glInvalidateBufferSubData(
458 stagingRef->bufferID(),
459 stagingRef->address() + nextCopySegment.offset + localOffset,
460 drawBufferRef->allocatedSize());
461 }
462 std::memcpy(
463 stagingReadData_ + localOffset,
464 mappedPtr + localOffset,
465 drawBufferRef->allocatedSize());
466 } else {
467 // temporary mapping case:
468 // map read buffer, copy data to client data, unmap read buffer
469 auto mapped = (byte *) glMapNamedBufferRange(
470 stagingRef->bufferID(),
471 stagingRef->address() + nextCopySegment.offset + localOffset,
472 drawBufferRef->allocatedSize(),
473 GL_MAP_READ_BIT);
474 if (mapped) {
475 std::memcpy(
476 stagingReadData_ + localOffset,
477 mapped,
478 drawBufferRef->allocatedSize());
479 glUnmapNamedBuffer(stagingRef->bufferID());
480 } else {
481 REGEN_ERROR("Failed to map buffer for reading.");
482 GL_ERROR_LOG();
483 return false;
484 }
485 }
486 }
487
488 if (flags_.useExplicitStaging()) {
489 nextDrawBufferRange.offset_ = drawBufferRef->address();
490 nextDrawBufferRange.segment_ = 0;
491 } else {
492 nextDrawBufferRange.offset_ = drawBufferRef->address() + nextCopySegment.offset + localOffset;
493 nextDrawBufferRange.segment_ = readBufferIndex_;
494 }
495
496 // finally swap buffers in case of double/triple buffering
497 if (useSwappingOnAccess_) swapBuffers();
498 return true;
