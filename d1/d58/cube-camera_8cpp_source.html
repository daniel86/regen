59bool CubeCamera::updateView() {
60 auto posStamp = positionStamp();
61 if (posStamp == posStamp_) { return false; }
62 posStamp_ = posStamp;
63
64 const Vec3f *dir = Mat4f::cubeDirections();
65 const Vec3f *up = Mat4f::cubeUpVectors();
66 for (int i = 0; i < 6; ++i) {
67 if (isCubeFaceVisible(i)) {
68 setView(i, Mat4f::lookAtMatrix(position(0), dir[i], up[i]));
69 setViewInverse(i, view(i).lookAtInverse());
70 }
71 }
72 return true;
