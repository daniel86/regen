106void Camera::updateShaderData(float dt) {
107 // Ensure this function is not called while another update is in progress.
108 // Wait until the update flag is set to false.
109 while (isUpdating_.load(std::memory_order_acquire)) {
110 CPU_PAUSE(); // busy wait, we expect very short duration of wait here.
111 }
112 isUpdating_.store(true, std::memory_order_release);
113
114 // Update velocity
115 if (dt > 0.0f) {
116 vel_.xyz() = (position_[0].xyz() - lastPosition_) / dt;
117 }
118 lastPosition_ = position_[0].xyz();
119
120 const bool viewChanged = (lastViewStamp1_ != viewStamp_);
121 const bool projChanged = (lastProjStamp1_ != projStamp_);
122
123 if (viewChanged) {
124 auto m_v = sh_view_->mapClientDataRaw(BUFFER_GPU_WRITE);
125 auto m_v_i = sh_viewInv_->mapClientDataRaw(BUFFER_GPU_WRITE);
126 auto m_vp = sh_viewProj_->mapClientDataRaw(BUFFER_GPU_WRITE);
127 auto m_vp_i = sh_viewProjInv_->mapClientDataRaw(BUFFER_GPU_WRITE);
128
129 std::memcpy(m_v.w, view_.data(), view_.size() * sizeof(Mat4f));
130 std::memcpy(m_v_i.w, viewInv_.data(), viewInv_.size() * sizeof(Mat4f));
131 std::memcpy(m_vp.w, viewProj_.data(), viewProj_.size() * sizeof(Mat4f));
132 std::memcpy(m_vp_i.w, viewProjInv_.data(), viewProjInv_.size() * sizeof(Mat4f));
133
134 if (projChanged) {
135 auto m_pp = sh_projParams_->mapClientDataRaw(BUFFER_GPU_WRITE);
136 auto m_p = sh_proj_->mapClientDataRaw(BUFFER_GPU_WRITE);
137 auto m_p_i = sh_projInv_->mapClientDataRaw(BUFFER_GPU_WRITE);
138
139 std::memcpy(m_pp.w, projParams_.data(), projParams_.size() * sizeof(ProjectionParams));
140 std::memcpy(m_p.w, proj_.data(), proj_.size() * sizeof(Mat4f));
141 std::memcpy(m_p_i.w, projInv_.data(), projInv_.size() * sizeof(Mat4f));
142
143 lastProjStamp1_ = projStamp_;
144 }
145
146 lastViewStamp1_ = viewStamp_;
147 } else if (projChanged) [[unlikely]] {
148 auto m_vp = sh_viewProj_->mapClientDataRaw(BUFFER_GPU_WRITE);
149 auto m_vp_i = sh_viewProjInv_->mapClientDataRaw(BUFFER_GPU_WRITE);
150 auto m_pp = sh_projParams_->mapClientDataRaw(BUFFER_GPU_WRITE);
151 auto m_p = sh_proj_->mapClientDataRaw(BUFFER_GPU_WRITE);
152 auto m_p_i = sh_projInv_->mapClientDataRaw(BUFFER_GPU_WRITE);
153
154 std::memcpy(m_vp.w, viewProj_.data(), viewProj_.size() * sizeof(Mat4f));
155 std::memcpy(m_vp_i.w, viewProjInv_.data(), viewProjInv_.size() * sizeof(Mat4f));
156 std::memcpy(m_pp.w, projParams_.data(), projParams_.size() * sizeof(ProjectionParams));
157 std::memcpy(m_p.w, proj_.data(), proj_.size() * sizeof(Mat4f));
158 std::memcpy(m_p_i.w, projInv_.data(), projInv_.size() * sizeof(Mat4f));
159
160 lastProjStamp1_ = projStamp_;
161 }
162
163 if (lastDirStamp1_ != directionStamp_) {
164 auto m_dir = sh_direction_->mapClientDataRaw(BUFFER_GPU_WRITE);
165 std::memcpy(m_dir.w, direction_.data(), direction_.size() * sizeof(Vec4f));
166 lastDirStamp1_ = directionStamp_;
167 }
168
169 if (lastPosStamp1_ != positionStamp_) {
170 auto m_pos = sh_position_->mapClientDataRaw(BUFFER_GPU_WRITE);
171 auto m_vel = sh_vel_->mapClientDataRaw(BUFFER_GPU_WRITE);
172
173 std::memcpy(m_pos.w, position_.data(), position_.size() * sizeof(Vec4f));
174 std::memcpy(m_vel.w, &vel_.x, sizeof(Vec4f));
175
176 lastPosStamp1_ = positionStamp_;
177 }
178
179 isUpdating_.store(false, std::memory_order_release);
180}

197bool Camera::updateView() {
198 if (positionStamp_ == lastPosStamp_ && directionStamp_ == lastDirStamp_) { return false; }
199 lastPosStamp_ = positionStamp_;
200 lastDirStamp_ = directionStamp_;
201
202 auto numViewLayers = view_.size();
203 for (unsigned int i = 0; i < numViewLayers; ++i) {
204 auto &dir = getClamped(direction_, i);
205 if (std::abs(dir.xyz().dot(Vec3f::up())) > 0.999f) {
206 view_[i] = Mat4f::lookAtMatrix(
207 getClamped(position_, i).xyz(),
208 dir.xyz(), localUp_);
209 viewInv_[i] = view_[i].lookAtInverse();
210 } else {
211 view_[i] = Mat4f::lookAtMatrix(
212 getClamped(position_, i).xyz(),
213 dir.xyz(), Vec3f::up());
214 viewInv_[i] = view_[i].lookAtInverse();
215 }
216 }
217 viewStamp_ += 1u;
218 camStamp_ += 1u;
219
220 return true;
221}

235void Camera::updateViewProjection(unsigned int projectionIndex, unsigned int viewIndex) {
236 auto maxIndex = std::max(projectionIndex, viewIndex);
237 viewProj_[maxIndex] = view_[viewIndex] * proj_[projectionIndex];

352ref_ptr<Camera> Camera::load(LoadingContext &ctx, scene::SceneInputNode &input) {
353 auto cam = createCamera(ctx, input);
354 if (cam.get() == nullptr) {
355 REGEN_WARN("Unable to create Camera for '" << input.getDescription() << "'.");
356 return {};
357 }
358
359 if (input.hasAttribute("fixed-lod")) {
360 cam->setFixedLOD(input.getValue<LODQuality>("fixed-lod", LODQuality::HIGH));
361 }
362
363 for (auto &child: input.getChildren("culling")) {
364 if (child->hasAttribute("index")) {
365 auto spatialIndex = ctx.scene()->getResource<SpatialIndex>(child->getValue("index"));
366 if (spatialIndex.get() == nullptr) {
367 REGEN_WARN("Unable to find SpatialIndex for '" << input.getDescription() << "'.");
368 continue;
369 }
370
371 ref_ptr<Camera> lodCamera;
372 if (input.hasAttribute("lod-camera")) {
373 lodCamera = ctx.scene()->getResource<Camera>(input.getValue("lod-camera"));
374 if (lodCamera.get() == nullptr) {
375 REGEN_WARN("Unable to find LOD Camera for '" << input.getDescription() << "'.");
376 }
377 }
378 if (!lodCamera.get()) {
379 lodCamera = cam;
380 }
381
382 spatialIndex->addCamera(cam, lodCamera,
383 input.getValue<SortMode>("sort", SortMode::FRONT_TO_BACK),
384 input.getValue<Vec4i>("lod-shift", Vec4i::zero()));
385 } else {
386 REGEN_WARN("Ignoring culling node without index for '" << input.getDescription() << "'.");
387 }
388 }
389
390 return cam;
391}

408ref_ptr<Camera> createLightCamera(LoadingContext &ctx, scene::SceneInputNode &input) {
409 ref_ptr<Light> light = ctx.scene()->getResource<Light>(input.getValue("light"));
410 if (light.get() == nullptr) {
411 REGEN_WARN("Unable to find Light for '" << input.getDescription() << "'.");
412 return {};
413 }
414 auto numLayer = input.getValue<uint32_t>("num-layer", 1u);
415 auto splitWeight = input.getValue<double>("split-weight", 0.9);
416 auto cameraType = input.getValue<std::string>("camera-type", "spot");
417 auto near = input.getValue<float>("near", 0.1f);
418 ref_ptr<Camera> lightCamera;
419
420 switch (light->lightType()) {
421 case Light::SPOT: {
422 auto spotCam = ref_ptr<LightCamera_Spot>::alloc(light);
423 spotCam->setLightNear(near);
424 lightCamera = spotCam;
425 break;
426 }
427 case Light::DIRECTIONAL: {
428 auto userCamera = ctx.scene()->getResource<Camera>(input.getValue("camera"));
429 if (userCamera.get() == nullptr) {
430 REGEN_WARN("Unable to find user camera for '" << input.getDescription() << "'.");
431 return {};
432 }
433 auto dirCam = ref_ptr<LightCamera_CSM>::alloc(light, userCamera, numLayer);
434 dirCam->setSplitWeight(splitWeight);
435 if (input.hasAttribute("uniform-z-range")) {
436 dirCam->setUseUniformDepthRange(input.getValue<bool>("uniform-z-range", true));
437 }
438 if (input.hasAttribute("depth-padding")) {
439 dirCam->setDepthPadding(input.getValue<double>("depth-padding", 0.0));
440 }
441 if (input.hasAttribute("ortho-padding")) {
442 dirCam->setOrthoPadding(input.getValue<double>("ortho-padding", 0.0));
443 }
444 lightCamera = dirCam;
445 break;
446 }
447 case Light::POINT: {
448 if (cameraType == "parabolic" || cameraType == "paraboloid") {
449 // parabolic camera
450 auto isDualParabolic = input.getValue<bool>("dual-paraboloid", true);
451 auto parabolic = ref_ptr<LightCamera_Parabolic>::alloc(light, isDualParabolic);
452 if (input.hasAttribute("normal")) {
453 parabolic->setNormal(input.getValue<Vec3f>("normal", Vec3f::down()));
454 }
455 parabolic->setLightNear(near);
456 lightCamera = parabolic;
457 } else {
458 // cube camera
459 auto cube = ref_ptr<LightCamera_Cube>::alloc(light, getHiddenFacesMask(input));
460 cube->setLightNear(near);
461 lightCamera = cube;
462 }
463 break;
464 }
465 }
466 if (lightCamera.get() == nullptr) {
467 REGEN_WARN("Unable to create camera for '" << input.getDescription() << "'.");
468 return {};
469 }
470
471 // make sure initial data is good.
472 lightCamera->updateBuffers();
473 lightCamera->updateCamera();
474 lightCamera->updateShaderData(0.0f);
475 dynamic_cast<LightCamera *>(lightCamera.get())->updateShadowData();
476
477 ctx.scene()->putState(input.getName(), lightCamera);
478
479 return lightCamera;
480}

515ref_ptr<Camera> Camera::createCamera(LoadingContext &ctx, scene::SceneInputNode &input) {
516 auto camType = input.getValue<std::string>("type", "spot");
517 if (input.hasAttribute("reflector") ||
518 input.hasAttribute("reflector-normal") ||
519 input.hasAttribute("reflector-point")) {
520 ref_ptr<Camera> userCamera =
521 ctx.scene()->getResource<Camera>(input.getValue("camera"));
522 if (userCamera.get() == nullptr) {
523 REGEN_WARN("Unable to find Camera for '" << input.getDescription() << "'.");
524 return {};
525 }
526 ref_ptr<ReflectionCamera> cam;
527 bool hasBackFace = input.getValue<bool>("has-back-face", false);
528
529 if (input.hasAttribute("reflector")) {
530 ref_ptr<CompositeMesh> compositeMesh =
531 ctx.scene()->getResource<CompositeMesh>(input.getValue("reflector"));
532 if (compositeMesh.get() == nullptr || compositeMesh->meshes().empty()) {
533 REGEN_WARN("Unable to find Mesh for '" << input.getDescription() << "'.");
534 return {};
535 }
536 const std::vector<ref_ptr<Mesh> > &vec = compositeMesh->meshes();
537 cam = ref_ptr<ReflectionCamera>::alloc(
538 userCamera, vec[0], input.getValue<uint32_t>("vertex-index", 0u), hasBackFace);
539 } else if (input.hasAttribute("reflector-normal")) {
540 auto normal = input.getValue<Vec3f>("reflector-normal", Vec3f(0.0f, 1.0f, 0.0f));
541 auto position = input.getValue<Vec3f>("reflector-point", Vec3f(0.0f, 0.0f, 0.0f));
542 cam = ref_ptr<ReflectionCamera>::alloc(userCamera, normal, position, hasBackFace);
543 }
544 if (cam.get()) {
545 ctx.scene()->putState(input.getName(), cam);
546 }
547 return cam;
548 } else if (input.hasAttribute("light")) {
549 auto cam = createLightCamera(ctx, input);
550 return cam;
551 } else if (camType == "cube") {
552 auto tf = ctx.scene()->getResource<ModelTransformation>(input.getValue("tf"));
553 ref_ptr<CubeCamera> cam = ref_ptr<CubeCamera>::alloc(getHiddenFacesMask(input));
554 // Set perspective parameters
555 cam->setPerspective(
556 1.0f, // aspect ratio is always 1 for cube map faces
557 90.0f, // fov is always 90 for cube map faces
558 input.getValue<float>("near", 0.1f),
559 input.getValue<float>("far", 100.0f));
560 cam->updateBuffers();
561 if (tf.get()) {
562 cam->attachToPosition(tf);
563 }
564 ctx.scene()->putState(input.getName(), cam);
565 return cam;
566 } else if (camType == "parabolic" || camType == "paraboloid") {
567 auto tf = ctx.scene()->getResource<ModelTransformation>(input.getValue("tf"));
568 bool hasBackFace = input.getValue<bool>("dual-paraboloid", true);
569 auto cam = ref_ptr<ParabolicCamera>::alloc(hasBackFace);
570 cam->updateBuffers();
571 if (input.hasAttribute("normal")) {
572 cam->setNormal(input.getValue<Vec3f>("normal", Vec3f::down()));
573 }
574 if (tf.get()) {
575 cam->attachToPosition(tf);
576 } else if (input.hasAttribute("tf")) {
577 REGEN_WARN("Unable to find ModelTransformation for '" << input.getDescription() << "'.");
578 }
579 ctx.scene()->putState(input.getName(), cam);
580 return cam;
581 } else {
582 ref_ptr<Camera> cam = ref_ptr<Camera>::alloc(1);
583 cam->setPosition(0, input.getValue<Vec3f>("position", Vec3f(0.0f, 2.0f, -2.0f)));
584 if (input.hasAttribute("up")) {
585 cam->setLocalUp(input.getValue<Vec3f>("up", Vec3f::right()));
586 }
587
588 auto dir = input.getValue<Vec3f>("direction", Vec3f(0.0f, 0.0f, 1.0f));
589 dir.normalize();
590 cam->setDirection(0, dir);
591
592 if (camType == "ortho" || camType == "orthographic" || camType == "orthogonal") {
593 auto width = input.getValue<float>("width", 10.0f);
594 auto height = input.getValue<float>("height", 10.0f);
595 cam->setOrtho(
596 -width / 2.0f, width / 2.0f,
597 -height / 2.0f, height / 2.0f,
598 input.getValue<float>("near", 0.1f),
599 input.getValue<float>("far", 200.0f));
600 } else {
601 auto viewport = ctx.scene()->screen()->viewport();
