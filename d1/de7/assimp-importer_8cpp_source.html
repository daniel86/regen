117static void setLightRadius(aiLight *aiLight, ref_ptr<Light> &light) {
118 float ax = aiLight->mAttenuationLinear;
119 float ay = aiLight->mAttenuationConstant;
120 float az = aiLight->mAttenuationQuadratic;
121 float z = ay / (2.0f * az);
122
123 float start = 0.01;
124 float stop = 0.99;
125
126 float inner = -z + sqrt(z * z - (ax / start - 1.0f / (start * az)));
127 float outer = -z + sqrt(z * z - (ax / stop - 1.0f / (stop * az)));
128
129 light->setRadius(0, Vec2f(inner, outer));
130}

132vector<ref_ptr<Light> > AssetImporter::loadLights() {
133 vector<ref_ptr<Light> > ret(scene_->mNumLights);
134
135 for (uint32_t i = 0; i < scene_->mNumLights; ++i) {
136 aiLight *assimpLight = scene_->mLights[i];
137 // node could be animated, but for now it is ignored
138 aiNode *node = aiBoneNodes_[string(assimpLight->mName.data)];
139 if (node == nullptr) {
140 REGEN_WARN("Light '" << assimpLight->mName.data << "' has no corresponding node.");
141 continue;
142 }
143 aiVector3D lightPos = node->mTransformation * assimpLight->mPosition;
144
145 ref_ptr<Light> light;
146 switch (assimpLight->mType) {
147 case aiLightSource_DIRECTIONAL: {
148 light = ref_ptr<Light>::alloc(Light::DIRECTIONAL);
149 light->setDirection(0, *((Vec3f *) &lightPos.x));
150 break;
151 }
152 case aiLightSource_POINT: {
153 light = ref_ptr<Light>::alloc(Light::POINT);
154 light->setPosition(0, *((Vec3f *) &lightPos.x));
155 setLightRadius(assimpLight, light);
156 break;
157 }
158 case aiLightSource_SPOT: {
159 light = ref_ptr<Light>::alloc(Light::SPOT);
160 light->setPosition(0, *((Vec3f *) &lightPos.x));
161 light->setDirection(0, *((Vec3f *) &assimpLight->mDirection.x));
162 light->setConeAngles(
163 acosf(assimpLight->mAngleOuterCone) * 360.0f / (2.0f * M_PIf),
164 acosf(assimpLight->mAngleInnerCone) * 360.0f / (2.0f * M_PIf));
165 setLightRadius(assimpLight, light);
166 break;
167 }
168 case aiLightSource_AREA:
169 REGEN_WARN("Area lights are not supported.");
170 break;
171 case aiLightSource_AMBIENT:
172 case aiLightSource_UNDEFINED:
173 case _aiLightSource_Force32Bit:
174 break;
175 }
176 if (light.get() == nullptr) { continue; }
177
178 lightToAiLight_[light.get()] = assimpLight;
179 //light->set_ambient( aiToOgle(&assimpLight->mColorAmbient) );
180 light->setDiffuse(0, aiToOgle(&assimpLight->mColorDiffuse));
181 light->setSpecular(0, aiToOgle(&assimpLight->mColorSpecular));
182
183 ret[i] = light;
184 }
185
186 return ret;
187}

194 aiMaterial *aiMat,

196 uint32_t l, uint32_t k,
197 const string &texturePath) {

199 std::string filePath;
200 int intVal;
201 float floatVal;
202
203 if (aiTexture == nullptr) {
204 if (boost::filesystem::exists(stringVal.data)) {
205 filePath = stringVal.data;
206 } else {
207 std::vector<string> names;
208 filePath = stringVal.data;
209 boost::split(names, filePath, boost::is_any_of("/\\"));
210 filePath = names[names.size() - 1];
211
212 std::string buf = REGEN_STRING(texturePath << "/" << filePath);
213 if (boost::filesystem::exists(buf)) {
214 filePath = buf;
215 } else {
216 REGEN_ERROR("Unable to load texture '" << buf << "'.");
217 return;
218 }
219 }
220
221 try {
222 // try image texture
223 tex = textures::load(filePath);
224 }
225 catch (textures::Error &ie) {
226#ifdef HAS_AV_LIBS
227 // try video texture
228 ref_ptr<VideoTexture> vid = ref_ptr<VideoTexture>::alloc();
229 try {
230 vid->set_file(filePath);
231 tex = vid;
232 vid->startAnimation();
233 }
234 catch (VideoTexture::Error &ve) {
235 REGEN_ERROR("Failed to load texture '" << stringVal.data << "'.");
236 return;
237 }
238#else
239 REGEN_ERROR("Failed to load texture '" << stringVal.data << "'.");
240 return;
241#endif
242 }
243 } else if (aiTexture->mHeight == 0) {
244 // The texture is stored in a "compressed" format such as DDS or PNG
245
246 uint32_t numBytes = aiTexture->mWidth;
247 std::string ext(aiTexture->achFormatHint);
248
249 ILenum type;
250 if (ext == "jpg" || ext == ".j")
251 type = IL_JPG;
252 else if (ext == "png")
253 type = IL_PNG;
254 else if (ext == "tga")
255 type = IL_TGA;
256 else if (ext == "bmp")
257 type = IL_BMP;
258 else if (ext == "dds")
259 type = IL_DDS;
260 else if (ext == "gif")
261 type = IL_GIF;
262 else if (ext == "tif")
263 type = IL_TIF;
264 else {
265 REGEN_ERROR("Unknown texture type '" << ext << "'.");
266 return;
267 }
268
269 try {
270 tex = textures::load(type, numBytes, aiTexture->pcData);
271 }
272 catch (textures::Error &ie) {
273 REGEN_ERROR("Failed to load texture '" << stringVal.data << "'.");
274 return;
275 }
276 } else {
277 // The texture is NOT compressed
278 tex = ref_ptr<Texture2D>::alloc();
279 tex->set_rectangleSize(aiTexture->mWidth, aiTexture->mHeight);
280 tex->set_pixelType(GL_UNSIGNED_BYTE);
281 tex->set_format(GL_RGBA);
282 tex->set_internalFormat(GL_RGBA8);
283 tex->allocTexture();
284 tex->set_filter(TextureFilter::create(GL_LINEAR));
285 tex->set_wrapping(TextureWrapping::create(GL_REPEAT));
286 tex->updateImage((GLubyte *) aiTexture->pcData);
287 }

289 ref_ptr<TextureState> texState = ref_ptr<TextureState>::alloc(tex);
290
291 // Defines miscellaneous flag for the n'th texture on the stack 't'.
292 // This is a bitwise combination of the aiTextureFlags enumerated values.
293 if (AI_SUCCESS == aiMat->Get(AI_MATKEY_TEXFLAGS(textureTypes[l], k - 1), intVal)) {
294 if (intVal & aiTextureFlags_Invert) {
295 texState->set_texelTransfer(TextureState::TEXEL_TRANSFER_INVERT);
296 }
297 if (intVal & aiTextureFlags_UseAlpha) {
298 REGEN_WARN("aiTextureFlags_UseAlpha is not supported.");
299 }
300 if (intVal & aiTextureFlags_IgnoreAlpha) {
301 texState->set_ignoreAlpha(true);
302 }
303 }
304
305 // Defines the height scaling of a bump map (for stuff like Parallax Occlusion Mapping)
306 if (aiMat->Get(AI_MATKEY_BUMPSCALING, floatVal) == AI_SUCCESS) {
307 texState->set_blendFactor(floatVal);
308 }
309
310 // Defines the strength the n'th texture on the stack 't'.
311 // All color components (rgb) are multipled with this factor *before* any further processing is done. -
312 if (AI_SUCCESS == aiMat->Get(AI_MATKEY_TEXBLEND(textureTypes[l], k - 1), floatVal)) {
313 texState->set_blendFactor(floatVal);
314 }
315
316 // One of the aiTextureOp enumerated values. Defines the arithmetic operation to be used
317 // to combine the n'th texture on the stack 't' with the n-1'th.
318 // TEXOP(t,0) refers to the blend operation between the base color
319 // for this stack (e.g. COLOR_DIFFUSE for the diffuse stack) and the first texture.
320 if (AI_SUCCESS == aiMat->Get(AI_MATKEY_TEXOP(textureTypes[l], k - 1), intVal)) {
321 switch (intVal) {
322 case aiTextureOp_Multiply:
323 texState->set_blendMode(BLEND_MODE_MULTIPLY);
324 break;
325 case aiTextureOp_SignedAdd:
326 case aiTextureOp_Add:
327 texState->set_blendMode(BLEND_MODE_ADD);
328 break;
329 case aiTextureOp_Subtract:
330 texState->set_blendMode(BLEND_MODE_SUBTRACT);
331 break;
332 case aiTextureOp_Divide:
333 texState->set_blendMode(BLEND_MODE_DIVIDE);
334 break;
335 case aiTextureOp_SmoothAdd:
336 texState->set_blendMode(BLEND_MODE_SMOOTH_ADD);
337 break;
338 default:
339 REGEN_WARN("Unknown texture operation '" << intVal << "'.");
340 break;
341 }
342 }
343
344 // Defines the base axis to to compute the mapping coordinates for the n'th texture
345 // on the stack 't' from. This is not required for UV-mapped textures.
346 // For instance, if MAPPING(t,n) is aiTextureMapping_SPHERE,
347 // U and V would map to longitude and latitude of a sphere around the given axis.
348 // The axis is given in local mesh space.
349 //TEXMAP_AXIS(t,n) aiVector3D n/a
350
351 // Defines how the input mapping coordinates for sampling the n'th texture on the stack 't'
352 // are computed. Usually explicit UV coordinates are provided, but some model file formats
353 // might also be using basic shapes, such as spheres or cylinders, to project textures onto meshes.
354 if (AI_SUCCESS == aiMat->Get(AI_MATKEY_MAPPING(textureTypes[l], k - 1), intVal)) {
355 switch (intVal) {
356 case aiTextureMapping_UV:
357 // The mapping coordinates are taken from an UV channel.
358 texState->set_mapping(TextureState::MAPPING_TEXCO);
359 break;
360 case aiTextureMapping_SPHERE:
361 texState->set_mapping(TextureState::MAPPING_SPHERE);
362 break;
363 case aiTextureMapping_CYLINDER:
364 texState->set_mapping(TextureState::MAPPING_TUBE);
365 break;
366 case aiTextureMapping_BOX:
367 texState->set_mapping(TextureState::MAPPING_CUBE);
368 break;
369 case aiTextureMapping_PLANE:
370 texState->set_mapping(TextureState::MAPPING_FLAT);
371 break;
372 case aiTextureMapping_OTHER:
373 break;
374 default:
375 REGEN_WARN("Unknown texture mapping '" << intVal << "'.");
376 break;
377 }
378 }
379
380 // Defines the UV channel to be used as input mapping coordinates for sampling the
381 // n'th texture on the stack 't'. All meshes assigned to this material share
382 // the same UV channel setup Presence of this key implies MAPPING(t,n) to be
383 // aiTextureMapping_UV. See How to map UV channels to textures (MATKEY_UVWSRC) for more details.
384 if (AI_SUCCESS == aiMat->Get(AI_MATKEY_UVWSRC(textureTypes[l], k - 1), intVal)) {
385 texState->set_texcoChannel(intVal);
386 }
387
388 TextureWrapping wrapping_(TextureWrapping::create(GL_REPEAT));
389 // Any of the aiTextureMapMode enumerated values. Defines the texture wrapping mode on the
390 // x axis for sampling the n'th texture on the stack 't'.
391 // 'Wrapping' occurs whenever UVs lie outside the 0..1 range.
392 if (AI_SUCCESS == aiMat->Get(AI_MATKEY_MAPPINGMODE_U(textureTypes[l], k - 1), intVal)) {
393 switch (intVal) {
394 case aiTextureMapMode_Wrap:
395 wrapping_.x = GL_REPEAT;
396 break;
397 case aiTextureMapMode_Clamp:
398 wrapping_.x = GL_CLAMP;
399 break;
400 case aiTextureMapMode_Decal:
401 REGEN_WARN("ignoring texture map mode decal.");
402 break;
403 case aiTextureMapMode_Mirror:
404 wrapping_.x = GL_MIRRORED_REPEAT;
405 break;
406 default:
407 REGEN_WARN("Unknown texture map mode '" << intVal << "'.");
408 break;
409 }
410 }
411 // Wrap mode on the v axis. See MAPPINGMODE_U.
412 if (AI_SUCCESS == aiMat->Get(AI_MATKEY_MAPPINGMODE_V(textureTypes[l], k - 1), intVal)) {
413 switch (intVal) {
414 case aiTextureMapMode_Wrap:
415 wrapping_.y = GL_REPEAT;
416 break;
417 case aiTextureMapMode_Clamp:
418 wrapping_.y = GL_CLAMP;
419 break;
420 case aiTextureMapMode_Decal:
421 REGEN_WARN("ignoring texture map mode decal.");
422 break;
423 case aiTextureMapMode_Mirror:
424 wrapping_.y = GL_MIRRORED_REPEAT;
425 break;
426 default:
427 REGEN_WARN("Unknown texture map mode '" << intVal << "'.");
428 break;
429 }
430 }
431#ifdef AI_MATKEY_MAPPINGMODE_W
432 // Wrap mode on the v axis. See MAPPINGMODE_U.
433 if(AI_SUCCESS == aiMat->Get(AI_MATKEY_MAPPINGMODE_W(textureTypes[l],k-1), intVal)) {
434 switch(intVal) {
435 case aiTextureMapMode_Wrap:
436 wrapping_.z = GL_REPEAT;
437 break;
438 case aiTextureMapMode_Clamp:
439 wrapping_.z = GL_CLAMP;
440 break;
441 case aiTextureMapMode_Decal:
442 REGEN_WARN("ignoring texture map mode decal.");
443 break;
444 case aiTextureMapMode_Mirror:
445 wrapping_.z = GL_MIRRORED_REPEAT;
446 break;
447 }
448 }
449#endif
450 tex->set_wrapping(wrapping_);
451
452 switch (textureTypes[l]) {
453 case aiTextureType_DIFFUSE:
454 // The texture is combined with the result of the diffuse lighting equation.
455 texState->set_mapTo(TextureState::MAP_TO_DIFFUSE);
456 break;
457 case aiTextureType_AMBIENT:
458 // The texture is combined with the result of the ambient lighting equation.
459 texState->set_mapTo(TextureState::MAP_TO_AMBIENT);
460 break;
461 case aiTextureType_SPECULAR:
462 // The texture is combined with the result of the specular lighting equation.
463 texState->set_mapTo(TextureState::MAP_TO_SPECULAR);
464 break;
465 case aiTextureType_SHININESS:
466 // The texture defines the glossiness of the material.
467 // The glossiness is in fact the exponent of the specular (phong) lighting equation.
468 // Usually there is a conversion function defined to map the linear color values
469 // in the texture to a suitable exponent. Have fun.
470 texState->set_mapTo(TextureState::MAP_TO_SHININESS);
471 break;
472 case aiTextureType_EMISSIVE:
473 // The texture is added to the result of the lighting calculation.
474 texState->set_mapTo(TextureState::MAP_TO_EMISSION);
475 // enforce multiply blending such that matEmission parameter gives easy control over
476 // amount of material emission.
477 texState->set_blendMode(BLEND_MODE_MULTIPLY);
478 break;
479 case aiTextureType_OPACITY:
480 // The texture defines per-pixel opacity.
481 // Usually 'white' means opaque and 'black' means 'transparency'.
482 // Or quite the opposite. Have fun.
483 texState->set_mapTo(TextureState::MAP_TO_ALPHA);
484 break;
485 case aiTextureType_LIGHTMAP:
486 // Lightmap texture (aka Ambient Occlusion). Both 'Lightmaps' and
487 // dedicated 'ambient occlusion maps' are covered by this material property.
488 // The texture contains a scaling value for the final color value of a pixel.
489 // Its intensity is not affected by incoming light.
490 texState->set_mapTo(TextureState::MAP_TO_LIGHT);
491 break;
492 case aiTextureType_REFLECTION:
493 // Reflection texture. Contains the color of a perfect mirror reflection.
494 // Rarely used, almost never for real-time applications.
495 //texState->setMapTo(MAP_TO_REFLECTION);
496 break;
497 case aiTextureType_DISPLACEMENT:
498 // Displacement texture. The exact purpose and format is application-dependent.
499 // Higher color values stand for higher vertex displacements.
500 texState->set_mapTo(TextureState::MAP_TO_DISPLACEMENT);
501 break;
502 case aiTextureType_HEIGHT:
503 // The texture is a height map. By convention, higher gray-scale values
504 // stand for higher elevations from the base height.
505 texState->set_mapTo(TextureState::MAP_TO_HEIGHT);
506 break;
507 case aiTextureType_NORMALS:
508 // The texture is a (tangent space) normal-map.
509 texState->set_mapTo(TextureState::MAP_TO_NORMAL);
510 break;
511 case aiTextureType_NONE:
512 // Dummy value. No texture, but the value to be used as 'texture semantic'
513 // (aiMaterialProperty::mSemantic) for all material properties *not* related to textures.
514 REGEN_WARN("Unknown texture type 'NONE' in '" << filePath << "'.");
515 break;
516 case aiTextureType_UNKNOWN:
517 // Unknown texture. A texture reference that does not match any of the definitions
518 // above is considered to be 'unknown'. It is still imported, but is excluded
519 // from any further postprocessing.
520 REGEN_WARN("Unknown texture type '" << stringVal.data << "' in '" << filePath << "'.");
521 break;
522 default:
523 REGEN_WARN("Unhandled texture type '" << stringVal.data << "' in '" << filePath << "'.");
524 break;
525 }
526
527 if (texState->isNormalMap()) {
528 // Normal maps should use linear filtering, but no mipmaps.
529 tex->set_filter(TextureFilter(GL_LINEAR, GL_LINEAR));
530 } else {
531 // Other textures should use linear mipmap filtering.
532 // Note: Assimp does not provide a way to specify the filter type.
533 // So we assume that all textures are mipmapped.
534 tex->updateMipmaps();
535 }
536 mat->joinStates(texState);
537 GL_ERROR_LOG();
538}

542std::vector<ref_ptr<Material> > AssetImporter::loadMaterials() {
543 std::vector<ref_ptr<Material> > materials(scene_->mNumMaterials);
544 aiColor4D aiCol;
545 float floatVal;
546 int intVal;
547 aiString stringVal;
548 uint32_t l, k;
549
550 for (uint32_t n = 0; n < scene_->mNumMaterials; ++n) {
551 ref_ptr<Material> mat = ref_ptr<Material>::alloc();
552 materials[n] = mat;
553 aiMaterial *aiMat = scene_->mMaterials[n];
554
555 // load textures
556 for (l = 0; l < numTextureTyps; ++l) {
557 k = 0;
558 while (AI_SUCCESS == aiMat->Get(AI_MATKEY_TEXTURE(textureTypes[l], k), stringVal)) {
559 k += 1;
560 if (stringVal.length < 1) continue;
561
562 aiTexture *aiTex;
563 if (stringVal.data[0] == '*') {
564 char *indexStr = stringVal.data + 1;
565 uint32_t index;
566
567 stringstream ss;
568 ss << indexStr;
569 ss >> index;
570 if (index < scene_->mNumTextures) {
571 aiTex = scene_->mTextures[index];
572 } else {
573 aiTex = nullptr;
574 }
575 } else {
576 aiTex = nullptr;
577 }
578
579 // skip normal map if requested
580 if (textureTypes[l] == aiTextureType_NORMALS && importFlags_ & IGNORE_NORMAL_MAP) {
581 continue;
582 }
583
584 loadTexture(mat, aiTex, aiMat, stringVal, l, k, texturePath_);
585 }
586 }
587
588 float opacity = mat->alpha()->getVertex(0).r;
589 Vec3f diffuseColor = Vec3f::one();
590 if (AI_SUCCESS == aiMat->Get(AI_MATKEY_COLOR_DIFFUSE, aiCol)) {
591 diffuseColor = aiToOgle3f(&aiCol);
592 }
593 // AI_MATKEY_COLOR_TRANSPARENT is a bit confusing, it is meant as the color that
594 // should be fully transparent. e.g. if it is (1,0,0,1) then red parts should
595 // be fully transparent. However, AFAIK this is not supposed to be applied per-pixel,
596 // but rather as an overall material property.
597 if (AI_SUCCESS == aiMat->Get(AI_MATKEY_COLOR_TRANSPARENT, aiCol)) {
598 // Apply color-key transparency
599 if (fabs(diffuseColor.x - aiCol.r) < 0.01f &&
600 fabs(diffuseColor.y - aiCol.g) < 0.01f &&
601 fabs(diffuseColor.z - aiCol.b) < 0.01f) {
602 opacity = 0.0f;
603 }
604 }
605 mat->diffuse()->setVertex(0, diffuseColor);
606
607 if (AI_SUCCESS == aiMat->Get(AI_MATKEY_COLOR_SPECULAR, aiCol)) {
608 mat->specular()->setVertex(0, aiToOgle3f(&aiCol));
609 }
610
611 float emissionStrength = emissionStrength_;
612 if (aiMat->Get(AI_MATKEY_EMISSIVE_INTENSITY, floatVal) == AI_SUCCESS) {
613 if (floatVal > 0.0f && floatVal < 1.0f - 1e-5f) {
614 mat->setEmissionMultiplier(floatVal);
615 }
616 }
617 if (AI_SUCCESS == aiMat->Get(AI_MATKEY_COLOR_EMISSIVE, aiCol)) {
618 mat->set_emission(aiToOgle3f(&aiCol) * emissionStrength);
619 }
620
621 // Defines the base shininess of the material
622 // This is the exponent of the phong shading equation.
623 if (aiMat->Get(AI_MATKEY_SHININESS, floatVal) == AI_SUCCESS) {
624 mat->shininess()->setVertex(0, floatVal);
625 }
626 // Defines the strength of the specular highlight.
627 // This is simply a multiplier to the specular color of a material
628 if (aiMat->Get(AI_MATKEY_SHININESS_STRENGTH, floatVal) == AI_SUCCESS) {
629 mat->setSpecularMultiplier(floatVal);
630 }
631
632 // Defines the base opacity of the material
633 if (aiMat->Get(AI_MATKEY_OPACITY, floatVal) == AI_SUCCESS) {
634 opacity *= floatVal;
635 }
636 mat->alpha()->setVertex(0, opacity);
637
638 // Index of refraction of the material. This is used by some shading models,
639 // e.g. Cook-Torrance. The value is the ratio of the speed of light in a
640 // vacuum to the speed of light in the material (always >= 1.0 in the real world).
641 // Might be of interest for raytracing.
642 if (aiMat->Get(AI_MATKEY_REFRACTI, floatVal) == AI_SUCCESS) {
643 mat->refractionIndex()->setVertex(0, floatVal);
644 }
645
646#ifdef AI_MATKEY_COOK_TORRANCE_PARAM
647 if(aiMat->Get(AI_MATKEY_COOK_TORRANCE_PARAM, floatVal) == AI_SUCCESS) {
648 mat->set_cookTorranceParam( floatVal );
649 }
650#endif
651
652#ifdef AI_MATKEY_ORENNAYAR_ROUGHNESS

709void AssetImporter::loadMeshes(
710 const struct aiNode &node,
711 const Mat4f &transform,
712 const BufferFlags &bufferFlags,
713 const vector<uint32_t> &meshIndices,
714 uint32_t &currentIndex,
715 vector<ref_ptr<Mesh> > &out) {
716 const auto *aiTransform = (const aiMatrix4x4 *) &transform.x;
717
718 // walk through meshes, add primitive set for each mesh
719 for (uint32_t n = 0; n < node.mNumMeshes; ++n) {
720 uint32_t index = 0;
721 for (index = 0; index < meshIndices.size(); ++index) {
722 if (meshIndices[index] == n) break;
723 }
724 if (index == meshIndices.size()) {
725 continue;
726 }
727
728 const struct aiMesh *mesh = scene_->mMeshes[node.mMeshes[n]];
729 if (mesh == nullptr) { continue; }
730
731 aiMatrix4x4 meshTransform = (*aiTransform) * node.mTransformation;
732 ref_ptr<Mesh> meshState = loadMesh(*mesh, *((const Mat4f *) &meshTransform.a1), bufferFlags);
733 // remember mesh material
734 meshMaterials_[meshState.get()] = materials_[mesh->mMaterialIndex];
735 meshToAiMesh_[meshState.get()] = mesh;
736
737 out[index] = meshState;
738 }
739
740 for (uint32_t n = 0; n < node.mNumChildren; ++n) {
741 const struct aiNode *child = node.mChildren[n];
742 if (child == nullptr) { continue; }
743 loadMeshes(*child, transform, bufferFlags, meshIndices, currentIndex, out);
744 }

747ref_ptr<Mesh> AssetImporter::loadMesh(const struct aiMesh &mesh, const Mat4f &transform, const BufferFlags &bufferFlags) {
748 ref_ptr<Mesh> meshState = ref_ptr<Mesh>::alloc(GL_TRIANGLES, bufferFlags.updateHints);
749 if (bufferFlags.accessMode == BUFFER_GPU_ONLY) {
750 meshState->setClientAccessMode(BUFFER_CPU_WRITE);
751 } else if (bufferFlags.accessMode == BUFFER_CPU_READ) {
752 meshState->setClientAccessMode(BUFFER_CPU_READ);
753 } else {
754 meshState->setClientAccessMode(bufferFlags.accessMode);
755 }
756 meshState->setBufferMapMode(bufferFlags.mapMode);
757
758 BufferMemoryLayout layout = (bufferFlags.updateHints.compute == BUFFER_NO_COMPUTE ?
759 BUFFER_MEMORY_PACKED : BUFFER_MEMORY_STD430);
760
761 ref_ptr<ShaderInput3f> pos = ref_ptr<ShaderInput3f>::alloc(ATTRIBUTE_NAME_POS);
762 ref_ptr<ShaderInput3f> nor = ref_ptr<ShaderInput3f>::alloc(ATTRIBUTE_NAME_NOR);
763 ref_ptr<ShaderInput4f> tan = ref_ptr<ShaderInput4f>::alloc(ATTRIBUTE_NAME_TAN);
764 pos->setMemoryLayout(layout);
765 nor->setMemoryLayout(layout);
766 tan->setMemoryLayout(layout);
767
768 const uint32_t numFaceIndices = (mesh.mNumFaces > 0 ? mesh.mFaces[0].mNumIndices : 0);
769 uint32_t numFaces = 0;
770 for (uint32_t t = 0u; t < mesh.mNumFaces; ++t) {
771 const struct aiFace *face = &mesh.mFaces[t];
772 if (face->mNumIndices != numFaceIndices) { continue; }
773 numFaces += 1;
774 }
775 const uint32_t numIndices = numFaceIndices * numFaces;
776
777 switch (numFaceIndices) {
778 case 1:
779 meshState->set_primitive(GL_POINTS);
780 break;
781 case 2:
782 meshState->set_primitive(GL_LINES);
783 break;
784 case 3:
785 meshState->set_primitive(GL_TRIANGLES);
786 break;
787 default:
788 meshState->set_primitive(GL_POLYGON);
789 break;
790 }
791
792 {
793 ref_ptr<ShaderInput> indices = createIndexInput(numIndices, mesh.mNumVertices);
794 auto m_faceIndices = indices->mapClientDataRaw(BUFFER_GPU_WRITE);
795 auto faceIndices = m_faceIndices.w;
796 auto indexType = indices->baseType();
797 uint32_t index = 0, maxIndex = 0;
798 for (uint32_t t = 0u; t < mesh.mNumFaces; ++t) {
799 const struct aiFace *face = &mesh.mFaces[t];
800 if (face->mNumIndices != numFaceIndices) { continue; }
801 for (uint32_t n = 0; n < face->mNumIndices; ++n) {
802 setIndexValue(faceIndices, indexType, index, face->mIndices[n]);
803 if (face->mIndices[n] > maxIndex) { maxIndex = face->mIndices[n]; }
804 index += 1;
805 }
806 }
807 meshState->setIndices(indices, maxIndex);
808 }
809
810 const auto *aiTransform = (const aiMatrix4x4 *) &transform.x;
811 // vertex positions
812 Vec3f min_ = Vec3f::create(999999.9);
813 Vec3f max_ = Vec3f::create(-999999.9);
814 uint32_t numVertices = mesh.mNumVertices;
815 {
816 pos->setVertexData(numVertices);
817 auto m_pos = pos->mapClientData<Vec3f>(BUFFER_GPU_WRITE);
818 auto &v_pos = m_pos.w;
819 for (uint32_t n = 0; n < numVertices; ++n) {
820 aiVector3D aiv = (*aiTransform) * mesh.mVertices[n];
821 const Vec3f p = Vec3f(aiv.x, aiv.y, aiv.z);
822 v_pos[n] = p;
823 min_.setMin(p);
824 max_.setMax(p);
825 }
826 meshState->setInput(pos);
827 }
828 meshState->set_bounds(min_, max_);
829
830 // per vertex normals
831 if (mesh.HasNormals()) {
832 nor->setVertexData(numVertices);
833 auto m_nor = nor->mapClientData<Vec3f>(BUFFER_GPU_WRITE);
834 auto &v_nor = m_nor.w;
835
836 for (uint32_t n = 0; n < numVertices; ++n) {
837 const aiVector3D &an = mesh.mNormals[n];
838 v_nor[n] = Vec3f(an.x, an.y, an.z);
839 }
840 meshState->setInput(nor);
841 }
842
843 // per vertex colors
844 for (uint32_t t = 0; t < AI_MAX_NUMBER_OF_COLOR_SETS; ++t) {
845 if (mesh.mColors[t] == nullptr) continue;
846
847 ref_ptr<ShaderInput4f> col = ref_ptr<ShaderInput4f>::alloc(REGEN_STRING("col" << t));
848 col->setVertexData(numVertices);
849 auto m_col = col->mapClientData<Vec4f>(BUFFER_GPU_WRITE);
850 auto &v_col = m_col.w;
851 for (uint32_t n = 0; n < numVertices; ++n) {
852 v_col[n] = Vec4f(
853 mesh.mColors[t][n].r,
854 mesh.mColors[t][n].g,
855 mesh.mColors[t][n].b,
856 mesh.mColors[t][n].a);
857 }
858 meshState->setInput(col);
859 }
860
861 // load texture coordinates
862 for (uint32_t t = 0; t < AI_MAX_NUMBER_OF_TEXTURECOORDS; ++t) {
863 if (mesh.mTextureCoords[t] == nullptr) { continue; }
864 aiVector3D *aiTexcos = mesh.mTextureCoords[t];
865 uint32_t texcoComponents = mesh.mNumUVComponents[t];
866 string texcoName = REGEN_STRING("texco" << t);
867
868 ref_ptr<ShaderInput> texco;
869 if (texcoComponents == 1) {
870 texco = ref_ptr<ShaderInput1f>::alloc(texcoName);
871 } else if (texcoComponents == 3) {
872 texco = ref_ptr<ShaderInput3f>::alloc(texcoName);
873 } else if (texcoComponents == 4) {
874 texco = ref_ptr<ShaderInput4f>::alloc(texcoName);
875 } else {
876 texco = ref_ptr<ShaderInput2f>::alloc(texcoName);
877 }
878 texco->setMemoryLayout(layout);
879 texco->setVertexData(numVertices);
880
881 if (texcoComponents == 1) {
882 auto m_texco = texco->mapClientData<float>(BUFFER_GPU_WRITE);
883 auto &v_texco = m_texco.w;
884 for (uint32_t n = 0; n < numVertices; ++n) {
885 const aiVector3D &at = aiTexcos[n];
886 v_texco[n] = at.x;
887 }
888 } else if (texcoComponents == 3) {
889 auto m_texco = texco->mapClientData<Vec3f>(BUFFER_GPU_WRITE);
890 auto &v_texco = m_texco.w;
891 for (uint32_t n = 0; n < numVertices; ++n) {
892 const aiVector3D &at = aiTexcos[n];
893 v_texco[n] = Vec3f(at.x, at.y, at.z);
894 }
895 } else if (texcoComponents == 4) {
896 auto m_texco = texco->mapClientData<Vec4f>(BUFFER_GPU_WRITE);
897 auto &v_texco = m_texco.w;
898 for (uint32_t n = 0; n < numVertices; ++n) {
899 const aiVector3D &at = aiTexcos[n];
900 v_texco[n] = Vec4f(at.x, at.y, at.z, 0.0f);
901 }
902 } else {
903 auto m_texco = texco->mapClientData<Vec2f>(BUFFER_GPU_WRITE);
904 auto &v_texco = m_texco.w;
905 for (uint32_t n = 0; n < numVertices; ++n) {
906 const aiVector3D &at = aiTexcos[n];
907 v_texco[n] = Vec2f(at.x, at.y);
908 }
909 }
910 meshState->setInput(texco);
911 }
912
913 // load tangents
914 if (mesh.HasTangentsAndBitangents()) {
915 tan->setVertexData(numVertices);
916 //auto v_tan = (Vec4f*) tan->clientBuffer()->clientData(0);
917 auto m_tan = tan->mapClientData<Vec4f>(BUFFER_GPU_WRITE);
918 auto &v_tan = m_tan.w;
919 for (uint32_t i = 0; i < numVertices; ++i) {
920 const aiVector3D &at = mesh.mTangents[i];
921 const aiVector3D &ab = mesh.mBitangents[i];
922 const aiVector3D &an = mesh.mNormals[i];
923
924 const Vec3f t = Vec3f(at.x, at.y, at.z);
925 const Vec3f b = Vec3f(ab.x, ab.y, ab.z);
926 const Vec3f n = Vec3f(an.x, an.y, an.z);
927 // Calculate the handedness of the local tangent space.
928 float handedness;
929 if (n.cross(t).dot(b) < 0.0) {
930 handedness = -1.0;
931 } else {
932 handedness = 1.0;
933 }
934 v_tan[i] = Vec4f(t.x, t.y, t.z, handedness);
935 }
936 meshState->setInput(tan);
937 }
938
939 // A mesh may have a set of bones in the form of aiBone structures..
940 // Bones are a means to deform a mesh according to the movement of a skeleton.
941 // Each bone has a name and a set of vertices on which it has influence.
942 // Its offset matrix declares the transformation needed to transform from mesh space
943 // to the local space of this bone.
944 if (mesh.HasBones()) {
945 typedef list<pair<float, uint32_t> > WeightList;
946 map<uint32_t, WeightList> vertexToWeights;
947 uint32_t maxNumWeights = 0;
948
949 // collect weights at vertices
950 for (uint32_t boneIndex = 0; boneIndex < mesh.mNumBones; ++boneIndex) {
951 aiBone *assimpBone = mesh.mBones[boneIndex];
952 for (uint32_t t = 0; t < assimpBone->mNumWeights; ++t) {
953 aiVertexWeight &weight = assimpBone->mWeights[t];
954 vertexToWeights[weight.mVertexId].emplace_back(weight.mWeight, boneIndex);
955 maxNumWeights = max(maxNumWeights,
956 (uint32_t) vertexToWeights[weight.mVertexId].size());
957 }
958 }
959 meshState->shaderDefine("NUM_BONE_WEIGHTS", REGEN_STRING(maxNumWeights));
960
961 if (maxNumWeights < 1) {
962 REGEN_ERROR("The model has invalid bone weights number " << maxNumWeights << ".");
963 } else {
964 // create array of bone weights and indices
965 auto boneWeights = ref_ptr<ShaderInput1f>::alloc("boneWeights", maxNumWeights);
966 auto boneIndices = ref_ptr<ShaderInput1ui>::alloc("boneIndices", maxNumWeights);
967 boneWeights->setMemoryLayout(layout);
968 boneIndices->setMemoryLayout(layout);
969 boneWeights->setVertexData(numVertices);
970 boneIndices->setVertexData(numVertices);
971 auto m_weights = boneWeights->mapClientData<float>(BUFFER_GPU_WRITE);
972 auto m_indices = boneIndices->mapClientData<uint32_t>(BUFFER_GPU_WRITE);
973 auto &v_weights = m_weights.w;
974 auto &v_indices = m_indices.w;
975
976 uint32_t offset = 0;
977 for (uint32_t j = 0; j < numVertices; j++) {
978 WeightList &vWeights = vertexToWeights[j];
979
980 uint32_t k = 0;
981 for (auto & vWeight : vWeights) {
982 v_weights[offset+k] = vWeight.first;
983 v_indices[offset+k] = vWeight.second;
984 ++k;
985 }
986 for (; k < maxNumWeights; ++k) {
987 v_weights[offset+k] = 0.0f;
988 v_indices[offset+k] = 0u;
989 }
990
991 offset += maxNumWeights;
992 }
993
994 m_weights.unmap();
995 m_indices.unmap();
996
997 if (maxNumWeights > 1) {
998 meshState->setInput(boneWeights);
999 }
1000 meshState->setInput(boneIndices);
1001 }
1002 }
1003 GL_ERROR_LOG();
1004
1005 meshState->updateVertexData();
1006 meshState->ensureLOD();
1007
1008 return meshState;
1009}

1100void AssetImporter::loadNodeAnimation(const AssimpAnimationConfig &animConfig) {
1101 if (!animConfig.useAnimation) { return; }
1102
1103 std::vector<ref_ptr<BoneNode>> nodes;
1104 loadNodeTree(nodes);
1105 if (nodes.empty()) { return; }
1106
1107 rootNode_ = nodes[0];
1108 ref_ptr<BoneTree> anim = ref_ptr<BoneTree>::alloc(nodes, animConfig.numInstances);
1109
1110 for (auto &node : nodes) {
1111 auto &aiNode = aiBoneNodes_[node->name];
1112 anim->setLocalTransform(node->nodeIdx,
1113 *((Mat4f *) &aiNode->mTransformation.a1));
1114 }
1115
1116 for (uint32_t i = 0; i < scene_->mNumAnimations; ++i) {
1117 aiAnimation *assimpAnim = scene_->mAnimations[i];
1118 // extract ticks per second. Assume default value if not given
1119 double ticksPerSecond = (assimpAnim->mTicksPerSecond != 0.0 ? assimpAnim->mTicksPerSecond : animConfig.ticksPerSecond);
1120 std::string animName(assimpAnim->mName.C_Str());
1121 // Some exporters like blender tend to prepend path into the names like "Armature|Hyenas_A15_Atk".
1122 // We are not interested in the path for now, so let's strip it.
1123 size_t sepPos = animName.find_last_of("|/");
1124 if (sepPos != std::string::npos) {
1125 animName = animName.substr(sepPos + 1);
1126 }
1127 double duration = assimpAnim->mDuration;
1128
1129 REGEN_DEBUG("Loading animation " << animName <<
1130 " with duration " << duration <<
1131 " and ticks per second " << ticksPerSecond <<
1132 " duration in seconds " << duration / ticksPerSecond);
1133
1134 if (assimpAnim->mNumChannels <= 0) continue;
1135
1136 ref_ptr<StaticAnimationData> animData = ref_ptr<StaticAnimationData>::alloc();
1137 animData->channels.resize(assimpAnim->mNumChannels);
1138
1139 for (uint32_t j = 0; j < assimpAnim->mNumChannels; ++j) {
1140 aiNodeAnim *nodeAnim = assimpAnim->mChannels[j];
1141 auto &channel = animData->channels[j];
1142
1143 channel.scalingKeys_.resize(nodeAnim->mNumScalingKeys);
1144 auto &scalingKeys = channel.scalingKeys_;
1145 bool useScale = false;
1146 for (uint32_t k = 0; k < nodeAnim->mNumScalingKeys; ++k) {
1147 auto &key = scalingKeys[k];
1148 key.time = nodeAnim->mScalingKeys[k].mTime;
1149 key.value = *((Vec3f *) &(nodeAnim->mScalingKeys[k].mValue.x));
1150 if (key.time > 0.0001) useScale = true;
1151 }
1152
1153 if (!useScale && !scalingKeys.empty()) {
1154 if (scalingKeys[0].value.isApprox(Vec3f::one(), 1e-6)) {
1155 scalingKeys.resize(0);
1156 } else {
1157 scalingKeys.resize(1, scalingKeys[0]);
1158 }
1159 }
1160
1162 channel.positionKeys_.resize(nodeAnim->mNumPositionKeys);
1163 auto &positionKeys = channel.positionKeys_;
1164 bool usePosition = false;
1165
1166 for (uint32_t k = 0; k < nodeAnim->mNumPositionKeys; ++k) {
1167 auto &key = positionKeys[k];
1168 key.time = nodeAnim->mPositionKeys[k].mTime;
1169 key.value = *((Vec3f *) &(nodeAnim->mPositionKeys[k].mValue.x));
1170 if (key.time > 0.0001) usePosition = true;
1171 }
1172
1173 if (!usePosition && !positionKeys.empty()) {
1174 if (positionKeys[0].value.isApprox(Vec3f::zero(), 1e-6)) {
1175 positionKeys.resize(0);
1176 } else {
1177 positionKeys.resize(1, positionKeys[0]);
1178 }
1179 }
1180
1182 channel.rotationKeys_.resize(nodeAnim->mNumRotationKeys);
1183 auto &rotationKeys = channel.rotationKeys_;
1184 bool useRotation = false;
1185 for (uint32_t k = 0; k < nodeAnim->mNumRotationKeys; ++k) {
1186 auto &key = rotationKeys[k];
1187 key.time = nodeAnim->mRotationKeys[k].mTime;
1188 key.value = *((Quaternion *) &(nodeAnim->mRotationKeys[k].mValue.w));
1189 if (key.time > 0.0001) useRotation = true;
1190 }
1191
1192 if (!useRotation && !rotationKeys.empty()) {
1193 if (rotationKeys[0].value == Quaternion(1, 0, 0, 0)) {
1194 rotationKeys.resize(0);
1195 } else {
1196 rotationKeys.resize(1, rotationKeys[0]);
1197 }
1198 }
1199
1200 channel.nodeName_ = string(nodeAnim->mNodeName.data);
1201 if (animConfig.forceStates) {
1202 channel.postState = animConfig.postState;
1203 channel.preState = animConfig.preState;
1204 } else {
1205 channel.postState = animState(nodeAnim->mPostState);
1206 channel.preState = animState(nodeAnim->mPreState);
1207 }
1208 }
1209
1210 anim->addAnimationTrack(
1211 animName,
1212 animData,
1213 assimpAnim->mDuration,
1214 ticksPerSecond);
1215 }
1216 nodeAnimation_ = anim;
1217}

1219ref_ptr<AssetImporter> AssetImporter::load(LoadingContext &ctx, scene::SceneInputNode &input) {
1220 if (!input.hasAttribute("file")) {
1221 REGEN_WARN("Ignoring Asset '" << input.getDescription() << "' without file.");
1222 return {};
1223 }
1224 const std::string assetPath = resourcePath(input.getValue("file"));
1225
1226 AssimpAnimationConfig animConfig;
1227 animConfig.numInstances =
1228 input.getValue<uint32_t>("animation-instances", 1u);
1229 animConfig.useAnimation = (animConfig.numInstances > 0) &&
1230 input.getValue<bool>("use-animation", true);
