46void WorldModelDebug::drawCircle(const Vec3f &center, float radius, const Vec3f &color) {
47 // draw a circle
48 const int segments = 32;
49 const float angleStep = 2.0f * M_PI / segments;
50 for (int i = 0; i < segments; i++) {
51 float angle = i * angleStep;
52 float nextAngle = (i + 1) * angleStep;
53 Vec3f from = center + Vec3f(std::cos(angle) * radius, 0, std::sin(angle) * radius);
54 Vec3f to = center + Vec3f(std::cos(nextAngle) * radius, 0, std::sin(nextAngle) * radius);
55 drawLine(from, to, color);
56 }
57}

65 Vec3f dirNorm = direction;
