79void Box::generateLODLevel(
80 const Config &cfg,
81 uint32_t sideIndex,
82 uint32_t lodLevel,
83 const std::vector<Tessellation> &tessellations) {
84 static constexpr Vec3f cubeNormals[] = {
85 Vec3f(0.0f, 0.0f, 1.0f), // Front
86 Vec3f(0.0f, 0.0f, -1.0f), // Back
87 Vec3f(0.0f, 1.0f, 0.0f), // Top
88 Vec3f(0.0f, -1.0f, 0.0f), // Bottom
89 Vec3f(1.0f, 0.0f, 0.0f), // Right
90 Vec3f(-1.0f, 0.0f, 0.0f) // Left
91 };
92 static const Mat4f faceRotations[] = {
93 Mat4f::identity(), // Front
94 Mat4f::rotationMatrix(0.0f, M_PI, 0.0f), // Back
95 Mat4f::rotationMatrix(M_PI_2, 0.0f, 0.0f), // Top
96 Mat4f::rotationMatrix(-M_PI_2, 0.0f, 0.0f), // Bottom
97 Mat4f::rotationMatrix(0.0f, -M_PI_2, 0.0f), // Right
98 Mat4f::rotationMatrix(0.0f, M_PI_2, 0.0f) // Left
99 };
100
101 auto &tessellation = tessellations[lodLevel];
102 auto vertexOffset = meshLODs_[lodLevel].d->vertexOffset + sideIndex * tessellation.vertices.size();
103 auto indexOffset = meshLODs_[lodLevel].d->indexOffset + sideIndex * tessellation.outputFaces.size() * 3;
104 const Vec3f &normal = cubeNormals[sideIndex];
105 const Mat4f &faceRotMat = faceRotations[sideIndex];
106 uint32_t nextIndex = indexOffset;
107
108 // map client data for writing
109 auto pos = (Vec3f*) pos_->clientBuffer()->clientData(0);
110 auto nor = (cfg.isNormalRequired ?
111 (Vec3f*) nor_->clientBuffer()->clientData(0) : nullptr);
112 auto tan = (cfg.isTangentRequired ?
113 (Vec4f*) tan_->clientBuffer()->clientData(0) : nullptr);
114 auto indices = (byte*)indices_->clientBuffer()->clientData(0);
115 auto indexType = indices_->baseType();
116
117 for (const auto &tessFace: tessellation.outputFaces) {
118 setIndexValue(indices, indexType, nextIndex++, vertexOffset + tessFace.v1);
119 setIndexValue(indices, indexType, nextIndex++, vertexOffset + tessFace.v2);
120 setIndexValue(indices, indexType, nextIndex++, vertexOffset + tessFace.v3);
121 }
122
123 uint32_t triIndices[3];
124 Vec3f triVertices[3];
125 Vec2f triTexco[3];
126
127 for (uint32_t faceIndex = 0; faceIndex < tessellation.outputFaces.size(); ++faceIndex) {
128 const auto &face = tessellation.outputFaces[faceIndex];
129 uint32_t faceVertIndex = 0;
130
131 for (const auto &tessIndex: {face.v1, face.v2, face.v3}) {
132 const auto &vertex = tessellation.vertices[tessIndex];
133 auto vertexIndex = vertexOffset + tessIndex;
134 triIndices[faceVertIndex] = vertexIndex;
135
136 Vec3f faceVertex = faceRotMat.transformVector(vertex) + normal;
137 Vec3f transformedVertex = cfg.posScale * modelRotation_.transformVector(faceVertex);
138 pos[vertexIndex] = transformedVertex;
139 minPosition_.setMin(transformedVertex);
140 maxPosition_.setMax(transformedVertex);
141 if (cfg.isNormalRequired) {
142 nor[vertexIndex] = normal;
143 }
144 if (texcoMode_ == TEXCO_MODE_CUBE_MAP) {
145 auto texco = (Vec3f*) texco_->clientBuffer()->clientData(0);
146 Vec3f v = faceVertex;
147 v.normalize();
148 texco[vertexIndex] = v;
149 triTexco[faceVertIndex] = Vec2f(vertex.x, vertex.y) * 0.5f + Vec2f::create(0.5f);
150 } else if (texcoMode_ == TEXCO_MODE_UV) {
151 Vec2f uv;
152 switch (sideIndex) {
153 case 0: // Front face
154 uv = Vec2f((faceVertex.x + 1.0f) * 0.5f, (faceVertex.y + 1.0f) * 0.5f);
155 uv.x *= cfg.posScale.x;
156 uv.y *= cfg.posScale.y;
157 break;
158 case 1: // Back face
159 uv = Vec2f((faceVertex.x + 1.0f) * 0.5f, (faceVertex.y + 1.0f) * 0.5f);
160 uv.x *= cfg.posScale.x;
161 uv.y *= cfg.posScale.y;
162 break;
163 case 2: // Top face
164 uv = Vec2f((faceVertex.x + 1.0f) * 0.5f, (faceVertex.z + 1.0f) * 0.5f);
165 uv.x *= cfg.posScale.x;
166 uv.y *= cfg.posScale.z;
167 break;
168 case 3: // Bottom face
169 uv = Vec2f((faceVertex.x + 1.0f) * 0.5f, (faceVertex.z + 1.0f) * 0.5f);
170 uv.x *= cfg.posScale.x;
171 uv.y *= cfg.posScale.z;
172 break;
173 case 4: // Right face
174 uv = Vec2f((faceVertex.z + 1.0f) * 0.5f, (faceVertex.y + 1.0f) * 0.5f);
175 uv.x *= cfg.posScale.z;
176 uv.y *= cfg.posScale.y;
177 break;
178 case 5: // Left face
179 uv = Vec2f((1.0f - faceVertex.z) * 0.5f, (faceVertex.y + 1.0f) * 0.5f);
180 uv.x *= cfg.posScale.z;
181 uv.y *= cfg.posScale.y;
182 break;
183 default:
184 uv = Vec2f::zero();
185 }
186 uv *= cfg.texcoScale;
187 auto texco = (Vec2f*) texco_->clientBuffer()->clientData(0);
188 texco[vertexIndex] = uv;
189 triTexco[faceVertIndex] = uv;
190 }
191 if (cfg.isTangentRequired) {
192 triVertices[faceVertIndex] = pos[vertexIndex];
193 }
194 faceVertIndex += 1;
195 }
196
197 if (cfg.isTangentRequired) {
198 Vec4f tangent = calculateTangent(triVertices, triTexco, normal);
199 for (uint32_t i = 0; i < 3; ++i) {
200 tan[triIndices[i]] = tangent;
201 }
202 }
203 }

206void Box::updateAttributes(const Config &cfg) {
207 std::vector<Tessellation> tessellations;
208 uint32_t numVertices = 0;
209 uint32_t numIndices = 0;
210
211 // Generate base tessellation for a single face (front face)
212 Tessellation baseTess;
213 baseTess.vertices.resize(4);
214 baseTess.vertices[0] = Vec3f(-1.0, -1.0, 0.0);
215 baseTess.vertices[1] = Vec3f(1.0, -1.0, 0.0);
216 baseTess.vertices[2] = Vec3f(1.0, 1.0, 0.0);
217 baseTess.vertices[3] = Vec3f(-1.0, 1.0, 0.0);
218 baseTess.inputFaces.resize(2);
219 baseTess.inputFaces[0] = TessellationFace(0, 1, 3);
220 baseTess.inputFaces[1] = TessellationFace(1, 2, 3);
221
222 // Generate tessellations for each LOD level
223 for (uint32_t lodLevel: cfg.levelOfDetails) {
224 auto &lodTess = tessellations.emplace_back();
225 lodTess.vertices = baseTess.vertices;
226 lodTess.inputFaces = baseTess.inputFaces;
227 tessellate(lodLevel, lodTess);
228
229 auto &x = meshLODs_.emplace_back();
230 x.d->numVertices = lodTess.vertices.size() * 6; // 6 faces
231 x.d->numIndices = lodTess.outputFaces.size() * 3 * 6; // 6 faces
232 x.d->vertexOffset = numVertices;
233 x.d->indexOffset = numIndices;
234 numVertices += x.d->numVertices;
235 numIndices += x.d->numIndices;
236 }
237
238 modelRotation_ = Mat4f::rotationMatrix(cfg.rotation.x, cfg.rotation.y, cfg.rotation.z);
239
240 // allocate attributes
241 pos_->setVertexData(numVertices);
242 if (cfg.isNormalRequired) {
243 nor_->setVertexData(numVertices);
244 }
245 if (cfg.isTangentRequired) {
246 tan_->setVertexData(numVertices);
247 }
248 TexcoMode texcoMode = cfg.texcoMode;
249 if (cfg.isTangentRequired && cfg.texcoMode == TEXCO_MODE_NONE) {
250 texcoMode = TEXCO_MODE_UV;
251 }
252 if (texcoMode == TEXCO_MODE_CUBE_MAP) {
253 texco_ = ref_ptr<ShaderInput3f>::alloc("texco0");
254 texco_->setVertexData(numVertices);
255 } else if (texcoMode == TEXCO_MODE_UV) {
256 texco_ = ref_ptr<ShaderInput2f>::alloc("texco0");
257 texco_->setVertexData(numVertices);
258 }
259 indices_ = createIndexInput(numIndices, numVertices);
260
261 minPosition_ = Vec3f::create(999999.9);
262 maxPosition_ = Vec3f::create(-999999.9);
263 for (auto lodLevel = 0u; lodLevel < tessellations.size(); ++lodLevel) {
264 for (uint32_t sideIndex = 0; sideIndex < 6; ++sideIndex) {
265 generateLODLevel(cfg, sideIndex, lodLevel, tessellations);
266 }
267 }
268
269 auto indexRef = setIndices(indices_, numVertices);
270 setInput(pos_);
271 if (cfg.isNormalRequired)
272 setInput(nor_);
273 if (cfg.texcoMode != TEXCO_MODE_NONE)
274 setInput(texco_);
275 if (cfg.isTangentRequired)
276 setInput(tan_);
277 updateVertexData();
278
279 for (auto &x: meshLODs_) {
280 // add the index buffer offset (in number of bytes)
281 x.d->indexOffset = indexRef->address() + x.d->indexOffset * indices_->dataTypeBytes();
282 }
283 activateLOD(0);
