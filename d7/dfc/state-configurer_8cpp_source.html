111void StateConfigurer::addState(const State *s) {
112 const auto *x1 = dynamic_cast<const FeedbackSpecification *>(s);
113 const auto *x2 = dynamic_cast<const TextureState *>(s);
114 const auto *x3 = dynamic_cast<const StateSequence *>(s);
115 const auto *fboState = dynamic_cast<const FBOState *>(s);
116
117 if (fboState) {
118 // skip this state if it is a FBO and a child node already has a FBO.
119 if (hasFBO_) { return; }
120 }
121
122 {
123 // remember inputs, they will be enabled automatically
124 // when the shader is enabled.
125 for (const auto & it : s->inputs()) {
126 addInput(it.name_, it.in_, it.type_, it.memberSuffix_);
127
128 std::queue<std::pair<const std::string&,ShaderInput*>> queue;
129 queue.emplace(it.in_->name(), it.in_.get());
130
131 while (!queue.empty()) {
132 auto [name, in] = queue.front();
133 const auto &inName = name.empty() ? in->name() : name;
134 queue.pop();
135
136 define(REGEN_STRING("HAS_" << inName), "TRUE");
137 if (in->isVertexAttribute()) {
138 define(REGEN_STRING("HAS_VERTEX_" << inName), "TRUE");
139 }
140 if (in->numArrayElements()>1 || in->forceArray()) {
141 define(REGEN_STRING("IS_ARRAY_" << inName), "TRUE");
142 }
143 if (in->numInstances() > 1) {
144 define("HAS_INSTANCES", "TRUE");
145 cfg_.numInstances_ = in->numInstances();
146 }
147
148 if (in->isStagedBuffer()) {
149 auto bo = dynamic_cast<StagedBuffer*>(in);
150 for (auto& blockUniform : bo->stagedInputs()) {
151 if (!it.memberSuffix_.empty()) {
152 if (blockUniform.name_.empty()) {
153 queue.emplace(
154 REGEN_STRING(blockUniform.in_->name() << it.memberSuffix_),
155 blockUniform.in_.get());
156 } else {
157 queue.emplace(
158 REGEN_STRING(blockUniform.name_ << it.memberSuffix_),
159 blockUniform.in_.get());
160 }
161 } else {
162 queue.emplace(blockUniform.name_, blockUniform.in_.get());
163 }
164 }
165 }
166 }
167 }
168 if (s->numInstances()>1) {
169 define("HAS_INSTANCES", "TRUE");
170 cfg_.numInstances_ = std::max(cfg_.numInstances_, static_cast<uint32_t>(s->numInstances()));
171 }
172 }
173 if (x1) {
174 cfg_.feedbackMode_ = x1->feedbackMode();
175 cfg_.feedbackStage_ = x1->feedbackStage();
176 for (const auto & it : x1->feedbackAttributes()) {
177 cfg_.feedbackAttributes_.push_back(it->name());
178 }
179 }
180 if (x2) {
181 // map for loop index to texture id
182 auto needle = cfg_.textures_.find(x2->name());
183 if (needle == cfg_.textures_.end()) {
184 // add texture to the list
185 auto texIdx = cfg_.textures_.size();
186 auto it = cfg_.textures_.insert({ x2->name(), { x2->texture(), texIdx } });
187 needle = it.first;
188 }
189 auto texIdx = needle->second.second;
190 define(REGEN_STRING("TEX_ID" << texIdx), REGEN_STRING(x2->stateID()));
191 define(REGEN_STRING("TEX_ID_" << x2->name()), REGEN_STRING(x2->stateID()));
192 needle->second.first = x2->texture();
193 addInput(x2->name(), x2->texture(), x2->samplerType());
194 // note: NUM_TEXTURES is rather "num texture states", i.e. one texture can be used
195 // in multiple states with different mapping etc.
196 textureStates_.insert(x2);
197 define("NUM_TEXTURES", REGEN_STRING(textureStates_.size()));
198 }
199
200 setVersion(s->shaderVersion());
201 addDefines(s->shaderDefines());
202 addIncludes(s->shaderIncludes());
203 addFunctions(s->shaderFunctions());
204
205 if (x3) {
206 // add global sequence state
207 addState(x3->globalState().get());
208 // do not add joined states of sequences
209 return;
210 }
211 for (const auto & it : *s->joined().get()) {
212 addState(it.get());
213 }
214}
