37 if constexpr (Mode == WaitMode::LINEAR) {
38 // Linear backoff

40 if (i < 16) { CPU_PAUSE(); }

136 template <class JobType>
137 class JobQueue {
138 std::vector<JobType> buffer;
139 // Producer position, padded to avoid false sharing
140 CachePadded<std::atomic<uint32_t>> head{0};
141 // Consumer position, padded to avoid false sharing
142 CachePadded<std::atomic<uint32_t>> tail{0};
143 public:
144 JobQueue() : buffer(1024) {
145 assert((buffer.size() & mask()) == 0 && "JobQueue size must be power of two");
146 }
147
148 inline uint32_t mask() const {
149 return static_cast<uint32_t>(buffer.size() - 1);
150 }
151
157 void grow() {
158 const size_t oldSize = buffer.size();
159 const size_t newSize = oldSize * 2;
160 const uint32_t t = tail.value.load(std::memory_order_acquire);
161 const uint32_t h = head.value.load(std::memory_order_acquire);
162 const uint32_t count = (h - t) & (oldSize - 1);
163
164 std::vector<JobType> newBuffer(newSize);
165 for (uint32_t i = 0; i < count; ++i) {
166 newBuffer[i] = buffer[(t + i) & (oldSize - 1)];
167 }
168 head.value.store(count, std::memory_order_release);
169 tail.value.store(0, std::memory_order_release);
170 buffer = std::move(newBuffer);
171 }
172
178 bool push(const JobType &j) {
179 const uint32_t mask = this->mask();
180 while (true) {
181 // Load current head and tail
182 uint32_t h = head.value.load(std::memory_order_relaxed);
183 const uint32_t t = tail.value.load(std::memory_order_acquire);
184 if ((h - t) == mask) return false; // full
185 // try to claim slot h by incrementing head -> h+1
186 // on success we "own" index h and can safely write to buffer[h & mask]
187 if (head.value.compare_exchange_weak(h, h+1,
188 std::memory_order_acq_rel,
189 std::memory_order_relaxed)) {
190 buffer[h & mask] = j;
191 return true;
192 }
193 }
194 }
195
201 bool try_pop(JobType &out) {
202 const uint32_t mask = this->mask();
203 while (true) {
204 // Load current tail and head
205 const uint32_t h = head.value.load(std::memory_order_acquire);
206 uint32_t t = tail.value.load(std::memory_order_relaxed);
207 if (t == h) return false; // empty
208 // try to claim slot t by incrementing tail -> t+1
209 // on success we "own" index t and can safely read buffer[t & mask]
210 if (tail.value.compare_exchange_weak(t, t+1,
211 std::memory_order_acq_rel,
212 std::memory_order_relaxed)) {
213 out = buffer[t & mask];
214 return true;
215 }
216 }
217 }
218
223 bool hasUnassignedJobs() const {
224 const uint32_t t = tail.value.load(std::memory_order_acquire);
225 const uint32_t h = head.value.load(std::memory_order_acquire);
226 return t != h;
227 }
228 };

432 template <typename T, size_t SEG_SIZE = 512>
433 class ThreadSafeQueue {
435 struct Segment {
436 // The cursor for writing items into the segment.
437 // Only modified by the producer.
438 // Conceptually the write cursor is "behind" the read cursor,
439 // and stands still when reaching the read cursor.
440 CachePadded<std::atomic<size_t>> writeCursor{0};
441 // The cursor for reading items from the segment.
442 // Only modified by the consumer.
443 CachePadded<std::atomic<size_t>> readCursor{0};
444 // Segments are linked in a singly linked list.
445 // Only modified by the producer.
446 std::atomic<Segment*> next{nullptr};
447 // The items in the segment.
448 T items[SEG_SIZE];
449 };
450
451 // Producer writes to head_
452 std::atomic<Segment*> head_;
453 // Consumer reads from tail_
454 std::atomic<Segment*> tail_;
455
456 // A pool of pre-allocated segments
457 std::vector<Segment*> segmentPool_;
458
459 public:
463 ThreadSafeQueue() {
464 // Initially, create a single empty segment used for both head and tail.
465 auto* s = acquireSegment();
466 head_.store(s, std::memory_order_relaxed);
467 tail_.store(s, std::memory_order_relaxed);
468 }
469
473 ~ThreadSafeQueue() {
474 // Free all remaining segments, for this we traverse the list structure.
475 Segment* seg = tail_.load(std::memory_order_relaxed);
476 while (seg) {
477 Segment* next = seg->next.load(std::memory_order_relaxed);
478 delete seg;
479 seg = next;
480 }
481 // Free segments in the pool
482 for (Segment* s : segmentPool_) {
483 delete s;
484 }
485 }
486
491 void push(const T& item) {
492 // Load the current head segment which we will write to.
493 Segment* seg = head_.load(std::memory_order_acquire);
494 // Load the cursor positions for this segment.
495 const size_t w = seg->writeCursor.value.load(std::memory_order_relaxed);
496 const size_t r = seg->readCursor.value.load(std::memory_order_acquire);
497
498 if (w - r < SEG_SIZE) {
499 // there is enough space in current segment.
500 seg->items[w % SEG_SIZE] = item;
501 seg->writeCursor.value.store(w + 1, std::memory_order_release);
502 return;
503 }
504
505 // Current segment overflow -> allocate a new one.
506 // This means the consumer is at least SEG_SIZE items behind.
507 // In this case, we create a new segment and link it to previous segment.
508 // NOTE: The consumer can release the old segment when it is done with it.
509 auto* newSeg = acquireSegment();
510 newSeg->items[0] = item;
511 newSeg->writeCursor.value.store(1, std::memory_order_release);
512
513 // Link new segment to current head segment, and update head_ to point to new segment.
514 seg->next.store(newSeg, std::memory_order_release);
515 head_.store(newSeg, std::memory_order_release);
516 }
517
522 bool empty() const {
523 const Segment* seg = tail_.load(std::memory_order_acquire);
524 const size_t w = seg->writeCursor.value.load(std::memory_order_acquire);
525 const size_t r = seg->readCursor.value.load(std::memory_order_relaxed);
526 // if cursors are at different positions, queue is not empty
527 if (r != w) return false;
528 // also if there is a next segment, queue is not empty
529 return seg->next.load(std::memory_order_acquire) == nullptr;
530 }
531
537 T& front() {
538 Segment* seg = tail_.load(std::memory_order_acquire);
539 const size_t r = seg->readCursor.value.load(std::memory_order_relaxed);
540 const size_t w = seg->writeCursor.value.load(std::memory_order_acquire);
541
542 if (r == w) {
543 // Advance to next segment if we hit the write cursor.
544 Segment* next = seg->next.load(std::memory_order_acquire);
545 const size_t r_next = next->readCursor.value.load(std::memory_order_relaxed);
546 tail_.store(next, std::memory_order_release);
547 // Old segment can now be released
548 releaseSegment(seg);
549 return next->items[r_next % SEG_SIZE];
550 } else {
551 return seg->items[r % SEG_SIZE];
552 }
553 }
554
559 void pop() {
560 Segment* seg = tail_.load(std::memory_order_acquire);
561 const size_t r = seg->readCursor.value.load(std::memory_order_relaxed) + 1;
562
563 // Advance read cursor
564 seg->readCursor.value.store(r, std::memory_order_release);
565
566 // If segment fully consumed and next exists -> move to next
567 size_t w = seg->writeCursor.value.load(std::memory_order_acquire);
568 if (r == w) {
569 Segment* next = seg->next.load(std::memory_order_acquire);
570 if (next != nullptr) {
571 tail_.store(next, std::memory_order_release);
572 // Old segment can now be released
573 releaseSegment(seg);
574 }
575 }
576 }
577
578 private:
579 AdaptiveLock poolLock_;
580
581 Segment* acquireSegment() {
582 poolLock_.lock();
583 if (!segmentPool_.empty()) {
584 Segment* seg = segmentPool_.back();
585 segmentPool_.pop_back();
586 poolLock_.unlock();
587
588 seg->writeCursor.value.store(0, std::memory_order_relaxed);
589 seg->readCursor.value.store(0, std::memory_order_relaxed);
590 seg->next.store(nullptr, std::memory_order_relaxed);
591
592 return seg;
593 } else {
594 poolLock_.unlock();
595 return new Segment();
596 }
597 }
598
599 void releaseSegment(Segment* seg) {
600 poolLock_.lock();
601 segmentPool_.push_back(seg);
602 poolLock_.unlock();
603 }
604 };
