93void Demuxer::set_file(std::string_view file) {
94 // (re)open file
95 if (formatCtx_) {
96 avformat_close_input(&formatCtx_);
97 formatCtx_ = nullptr;
98 }
99 if (avformat_open_input(&formatCtx_, file.data(), nullptr, nullptr) != 0) {
100 throw Error(REGEN_STRING("Couldn't open file " << file));
101 }
102
103 // Retrieve stream information
104 if (avformat_find_stream_info(formatCtx_, nullptr) < 0) {
105 throw Error("Couldn't find stream information");
106 }
107
108 // Find the first video/audio stream
109 videoStreamIndex_ = -1;
110 audioStreamIndex_ = -1;
111 for (uint32_t i = 0u; i < formatCtx_->nb_streams; ++i) {
112 auto stream = formatCtx_->streams[i];
113 auto codec_type = stream->codecpar->codec_type;
114 if (videoStreamIndex_ == -1 && codec_type == AVMEDIA_TYPE_VIDEO) { videoStreamIndex_ = i; }
115 else if (audioStreamIndex_ == -1 && codec_type == AVMEDIA_TYPE_AUDIO) { audioStreamIndex_ = i; }
116 }
117 if (videoStreamIndex_ != -1) {
118 videoStream_ = ref_ptr<VideoStream>::alloc(
119 formatCtx_->streams[videoStreamIndex_], videoStreamIndex_, 100);
120 }
121 if (audioStreamIndex_ != -1) {
122 audioStream_ = ref_ptr<AudioSource>::alloc(
123 formatCtx_->streams[audioStreamIndex_], audioStreamIndex_, -1);
124 }
125}

166void Demuxer::seekTo(double p) {
167 if (!formatCtx_) { return; }
168 p = std::max(0.0, std::min(1.0, p));
169 seek_.isRequired = true;
170 seek_.flags &= ~AVSEEK_FLAG_BYTE;
171 seek_.rel = 0;
172 seek_.pos = p * formatCtx_->duration;
173 if (formatCtx_->start_time != AV_NOPTS_VALUE) {
174 seek_.pos += formatCtx_->start_time;
175 }
176}

178bool Demuxer::decode() {
179 AVPacket packet;
180
181 SeekPosition seek = seek_;
182 seek_.isRequired = false;
183 if (seek.isRequired) {
184 int64_t seek_min = seek.rel > 0 ? seek.pos - seek.rel + 2 : INT64_MIN;
185 int64_t seek_max = seek.rel < 0 ? seek.pos - seek.rel - 2 : INT64_MAX;
186
187 int ret = avformat_seek_file(formatCtx_,
188 -1, seek_min, seek.pos, seek_max, seek.flags);
189 if (ret < 0) {
190 REGEN_ERROR("seeking failed");
191 } else {
192 clearQueue();
193 seeked_ = true;
194 }
195 } else if (pauseFlag_) {
196 return true;
197 }
198
199 if (av_read_frame(formatCtx_, &packet) < 0) {
200 // end of stream reached
201 if (repeatStream_) {
202 seekTo(0.0);
203 av_packet_unref(&packet);
204 return false;
205 } else {
206 stop();
207 return true;
208 }
209 }
210
211 // Is this a packet from the video stream?
212 if (packet.stream_index == videoStreamIndex_) {
213 videoStream_->decode(&packet);
214 } else if (packet.stream_index == audioStreamIndex_) {
215 audioStream_->decode(&packet);
216 } else {
217 // Free the packet that was allocated by av_read_frame
218 av_packet_unref(&packet);
219 }
220
221 return false;
222}
